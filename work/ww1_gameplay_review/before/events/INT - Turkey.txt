# The assembly abolished the sultanate on 1 November 1922. Mehmed VI did not
# formally abdicate; he left Istanbul on 17 November after losing authority.
country_event = {
	id = INT_turkey.8
	title = INT_turkey.8.t
	desc = INT_turkey.8.d
	picture = GFX_report_event_constituent_assembly
	fire_only_once = yes
	trigger = {
		has_start_date < 1922.11.2
		tag = TUR
		has_start_date < 1923.1.1
		date > 1922.10.31
		date < 1923.1.1
		has_global_flag = armistice_of_mudros
		NOT = { has_global_flag = central_powers_victory }
		has_focus_tree = INT_turkey
		owns_state = 49
		NOT = { has_country_flag = INT_TUR_sultanate_abolished }
	}
	mean_time_to_happen = { days = 1 }
	option = {
		name = INT_turkey.8.a
		if = {
			limit = { NOT = { has_completed_focus = INT_TUR_grand_national_assembly } }
			complete_national_focus = INT_TUR_grand_national_assembly
		}
		INT_TUR_end_sultanate = yes
		if = {
			limit = { NOT = { has_completed_focus = INT_TUR_abolish_sultanate } }
			complete_national_focus = INT_TUR_abolish_sultanate
		}
	}
}
