# Continuous treaty objectives, using the installed native ratio-progress form.
# They never complete and award no initiative, experience or permanent bonuses.
# Their completed collections become empty outside the WW1 scenario/date range,
# so a surviving coalition cannot carry these scaled modifiers into WWII.

faction_manifest_ww1_mutual_defence = {
	name = ww1_mutual_defence_goal_name
	description = ww1_mutual_defence_goal_desc
	is_manifest = yes
	visible = { has_start_date < 1936.1.1 date < 1936.1.1 }
	available = { has_start_date < 1936.1.1 date < 1936.1.1 }
	ratio_progress = {
		total_amount_collection = collection:faction_owned_core_states
		completed_amount_collection = collection:ww1_faction_secured_core_states
		range = { max = 1 }
		scale = {
			war_support_factor = 0.05
		}
	}
}
