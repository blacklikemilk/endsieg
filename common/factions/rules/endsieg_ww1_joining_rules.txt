# Ratified policies govern WW1 membership; diplomacy callbacks also retain
# the existing focus-only entry policy. Scripted historical setup is preserved.
joining_rule_ww1_focus_commitment = {
 type = joining_rules
 visible = { has_start_date < 1936.1.1 date < 1936.1.1 }
 available = { has_start_date < 1936.1.1 date < 1936.1.1 }
 can_remove = { NOT = { AND = { has_start_date < 1936.1.1 date < 1936.1.1 } } }
 ai_will_do = { base = 0 }
 trigger = {
  if = {
   limit = { has_start_date < 1936.1.1 date < 1936.1.1 }
   OR = {
    is_subject_of = FROM
    AND = {
     WW1_faction_diplomacy_allowed = yes
     FROM = {
      ROOT = {
       WW1_focus_authorizes_faction_entry_from_previous = yes
       WW1_country_faction_path_allows_previous = yes
      }
     }
    }
   }
  }
 }
}
