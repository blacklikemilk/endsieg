WWI_levant_campaign = {
	has_start_date < 1919.1.1
	date < 1936.1.1
	NOT = { has_global_flag = player_wants_custom_peacedeals }
	NOT = { has_global_flag = central_powers_victory }
}

WWI_levant_armistice = {
	WWI_levant_campaign = yes
	has_global_flag = armistice_of_mudros
}

# State scope. Only dispose of provinces in Ottoman or Allied administrative hands.
WWI_levant_handover_allowed = {
	OR = { is_owned_by = TUR is_owned_by = ENG is_owned_by = EN3 is_owned_by = FRA }
	OR = { is_controlled_by = TUR is_controlled_by = ENG is_controlled_by = EN3 is_controlled_by = FRA }
}

# State scope in Ottoman Mesopotamia. Wartime Austrian occupation carried no territorial title.
WWI_mesopotamian_handover_allowed = {
	OR = { is_owned_by = TUR is_owned_by = ENG }
	OR = {
		is_controlled_by = TUR is_controlled_by = ENG
		AND = { TUR = { has_country_flag = WWI_mudros_austrian_ally } is_controlled_by = AUH }
	}
}

WWI_mesopotamian_settlement_due = {
	tag = TUR WWI_levant_armistice = yes
	date > 1918.11.3 date < 1932.10.3
	has_country_flag = WWI_mudros_british_participant
	ENG = { exists = yes has_capitulated = no NOT = { has_war_with = TUR } }
	OR = {
		NOT = { country_exists = IRQ }
		IRQ = { is_subject_of = ENG NOT = { has_war_with = TUR } }
	}
	OR = {
		291 = { is_owned_by = TUR WWI_mesopotamian_handover_allowed = yes }
		676 = { is_owned_by = TUR WWI_mesopotamian_handover_allowed = yes }
	}
}
