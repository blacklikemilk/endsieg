# An Entente defeat is distinct from Germany's political or military partition.
WW1_AUH_entente_opponent = {
    OR = {
        has_war_with = FRA
        has_war_with = ENG
        has_war_with = ITA
        has_war_with = USA
        has_war_with = SER
    }
}
WW1_AUH_historical_defeat_context = {

    has_start_date < 1921.1.1
    has_global_flag = welcome_event_WW1_active
    date < 1921.1.1
    is_subject = no
    owns_state = 1051
    NOT = { has_global_flag = armistice_of_villa_giusti }
    NOT = { has_global_flag = WW1_AUH_historical_dissolution_done }
    NOT = { has_global_flag = player_wants_custom_peacedeals }
    NOT = { has_global_flag = central_powers_victory }
    NOT = { has_global_flag = WW1_CP_final_settlement_signed }
    NOT = { has_global_flag = GE1_R_AUH_managed_partition_active }
    NOT = { has_global_flag = GE1_R_AUH_managed_partition_complete }
    NOT = { has_global_flag = WW1_HAB_protocol_pending }

# Historical governments break away after the Balkan front has collapsed.
WW1_AUH_historical_collapse_due = {
    is_ai = yes
    is_historical_focus_on = yes
    NOT = { date < 1918.10.28 }
    OR = {
        has_global_flag = armistice_of_salonica
        has_global_flag = armistice_of_mudros
        has_global_flag = compiegne
    }
    OR = { WW1_AUH_entente_opponent = yes has_country_flag = WW1_AUH_fought_entente }
    NOT = {
        OR = {
            has_country_flag = AUH_ww1_trialism_ratified
            has_country_flag = AUH_ww1_federation_ratified
            has_country_flag = AUH_ww1_danubian_republic
        }
    }
}

# Domestic collapse can precede military surrender despite high war support.
WW1_AUH_revolution_ready = {
    WW1_AUH_historical_defeat_context = yes
    OR = { WW1_AUH_entente_opponent = yes WW1_AUH_historical_collapse_due = yes }
    NOT = { date < 1918.10.31 }
    NOT = { has_country_flag = WW1_AUH_revolution_started }
    NOT = { has_global_flag = aster_revolution }
    OR = {
        WW1_AUH_breakup_pressure = yes
        WW1_AUH_historical_collapse_due = yes
        has_country_flag = WW1_AUH_national_councils_active
    }
}

WW1_AUH_breakup_pressure = {
    OR = {
        NOT = { OR = { has_country_flag = AUH_ww1_trialism_ratified has_country_flag = AUH_ww1_federation_ratified has_country_flag = AUH_ww1_danubian_republic } }
        surrender_progress > 0.30
        AND = { has_stability < 0.25 check_variable = { var = AUH_stab_average value = 80 compare = greater_than } }
    }
    OR = {
        AND = { has_stability < 0.40 check_variable = { var = AUH_stab_average value = 60 compare = greater_than } }
        AND = {
            country_has_war_exhaustion = yes
            check_variable = { war_exhaustion_variable > 1500 }
            OR = {
                has_stability < 0.45
                has_global_flag = armistice_of_salonica
                has_global_flag = armistice_of_mudros
                has_global_flag = compiegne
            }
        }
    }
}

# State scope. A saved crown settlement can only return territories held by
# its original subjects. A conqueror or independent successor keeps its land.
WW1_AUH_state_in_dissolution = {
    OR = {
        is_owned_by = AUH
        AND = {
            has_global_flag = WW1_AUH_repairing_crownland_armistice
            owner = {
                OR = { tag = CZE tag = HUN tag = CRO tag = GLO }
                is_subject_of = AUH
                has_country_flag = WW1_HAB_crown_settlement
            }
        }
    }
}
