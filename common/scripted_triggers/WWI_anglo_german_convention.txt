# Treaty ratios count combat vessels, including all Endsieg WW1 sub-units.
# Convoys, repair ships and support ships are not warships under this convention.
GE1_D_naval_convention_in_force = {
 tag = GER has_start_date < 1936.1.1 date < 1936.1.1
 has_country_flag = GE1_D_naval_convention_enforced
 NOT = { has_country_flag = GE1_D_naval_convention_withdrawn }
 country_exists = ENG NOT = { has_war_with = ENG }
}
WW1_ratified_anglo_german_alliance = {
 has_start_date < 1936.1.1 date < 1936.1.1
 OR = { tag = GER tag = ENG is_subject_of = ENG }
 has_global_flag = GE1_anglo_german_alliance
 GER = { has_country_flag = GE1_D_british_route_active NOT = { has_country_flag = GE1_D_naval_convention_withdrawn } }
 ENG = { has_country_flag = GE1_anglo_german_alliance is_in_faction_with = GER NOT = { has_war_with = GER } NOT = { has_country_flag = EN1_R_neutral_policy } NOT = { has_country_flag = EN1_R_entente_policy } NOT = { has_completed_focus = EN1_stay_neutral } NOT = { has_completed_focus = EN1_back_france } }
}

  AND = { OR = { has_navy_size = { unit = early_hc size > 0 } has_navy_size = { unit = early_heavy_cruiser size > 0 } has_navy_size = { unit = early_light_cruiser size > 0 } has_navy_size = { unit = basic_destroyer size > 0 } has_navy_size = { unit = early_sub size > 0 } has_navy_size = { unit = early_bc size > 0 } has_navy_size = { unit = dreadnought size > 0 } has_navy_size = { unit = heavy_cruiser size > 0 } has_navy_size = { unit = light_cruiser size > 0 } has_navy_size = { unit = destroyer size > 0 } has_navy_size = { unit = submarine size > 0 } has_navy_size = { unit = battle_cruiser size > 0 } has_navy_size = { unit = battleship size > 0 } has_navy_size = { unit = carrier size > 0 } } ENG = { NOT = { OR = { has_navy_size = { unit = early_hc size > 0 } has_navy_size = { unit = early_heavy_cruiser size > 0 } has_navy_size = { unit = early_light_cruiser size > 0 } has_navy_size = { unit = basic_destroyer size > 0 } has_navy_size = { unit = early_sub size > 0 } has_navy_size = { unit = early_bc size > 0 } has_navy_size = { unit = dreadnought size > 0 } has_navy_size = { unit = heavy_cruiser size > 0 } has_navy_size = { unit = light_cruiser size > 0 } has_navy_size = { unit = destroyer size > 0 } has_navy_size = { unit = submarine size > 0 } has_navy_size = { unit = battle_cruiser size > 0 } has_navy_size = { unit = battleship size > 0 } has_navy_size = { unit = carrier size > 0 } } } } }
 }
