# ROOT is the diplomatic initiator; FROM is the recipient for both actions.
WW1_faction_diplomacy_allowed = {
 if = {
  limit = { has_start_date < 1936.1.1 date < 1936.1.1 }
  NOT = { AND = { OR = { tag = ITA is_in_faction_with = ITA } FROM = { WW1_is_balkan_league_member = yes } } }
  NOT = { AND = { WW1_is_balkan_league_member = yes FROM = { OR = { tag = ITA is_in_faction_with = ITA } } } }
  if = {
   limit = { country_exists = GER GER = { OR = { has_country_flag = GE1_ww1_bloc_central has_completed_focus = GE1_D_support_austria } } OR = { tag = GER is_in_faction_with = GER FROM = { OR = { tag = GER is_in_faction_with = GER } } } }
   NOT = { OR = { tag = ENG is_in_faction_with = ENG tag = RIE is_in_faction_with = RIE FROM = { OR = { tag = ENG is_in_faction_with = ENG tag = RIE is_in_faction_with = RIE } } } }
  }
  if = {
   limit = { country_exists = GER GER = { OR = { has_country_flag = GE1_ww1_bloc_british has_completed_focus = GE1_D_limit_naval_rivalry has_global_flag = GE1_anglo_german_alliance } } OR = { tag = GER is_in_faction_with = GER FROM = { OR = { tag = GER is_in_faction_with = GER } } } }
   NOT = { OR = { tag = AUH is_in_faction_with = AUH tag = TUR is_in_faction_with = TUR tag = RIE is_in_faction_with = RIE FROM = { OR = { tag = AUH is_in_faction_with = AUH tag = TUR is_in_faction_with = TUR tag = RIE is_in_faction_with = RIE } } } }
  }
  if = {
   limit = { country_exists = GER GER = { OR = { has_country_flag = GE1_ww1_bloc_russian has_completed_focus = GE1_R_eastern_neutrality has_global_flag = GE1_ww1_russian_bloc_active } } OR = { tag = GER is_in_faction_with = GER FROM = { OR = { tag = GER is_in_faction_with = GER } } } }
   NOT = { OR = { tag = AUH is_in_faction_with = AUH tag = TUR is_in_faction_with = TUR tag = ENG is_in_faction_with = ENG FROM = { OR = { tag = AUH is_in_faction_with = AUH tag = TUR is_in_faction_with = TUR tag = ENG is_in_faction_with = ENG } } } }
  }

  FROM = { ROOT = { WW1_country_faction_path_allows_previous = yes } }
  ROOT = { FROM = { WW1_country_faction_path_allows_previous = yes } }
  # Formal British/Russian offers cannot be skipped through native diplomacy.
  if = {
   limit = { OR = { tag = GER is_in_faction_with = GER FROM = { OR = { tag = GER is_in_faction_with = GER } } } }
   if = { limit = { OR = { tag = ENG is_in_faction_with = ENG FROM = { OR = { tag = ENG is_in_faction_with = ENG } } } } has_global_flag = GE1_anglo_german_alliance }
   if = { limit = { OR = { tag = RIE is_in_faction_with = RIE FROM = { OR = { tag = RIE is_in_faction_with = RIE } } } } has_global_flag = GE1_ww1_russian_bloc_active }
  }

  # Britain may support France diplomatically before entering its alliance.
  if = {
   limit = {
    OR = { AND = { tag = ENG FROM = { OR = { tag = FRA tag = RIE is_in_faction_with = FRA is_in_faction_with = RIE } } } AND = { FROM = { tag = ENG } OR = { tag = FRA tag = RIE is_in_faction_with = FRA is_in_faction_with = RIE } } }
    NOT = { has_global_flag = GE1_anglo_german_alliance }
   }
   ENG = { WW1_british_entente_entry_allowed = yes }
  }
}
}
