INT_TUR_independence_context = {
	tag = TUR
	has_start_date < 1936.1.1
	date < 1936.1.1
	has_global_flag = armistice_of_mudros
	is_subject = no
	has_capitulated = no
	49 = { is_owned_by = TUR is_controlled_by = TUR }
	NOT = {
		OR = {
			has_global_flag = central_powers_victory
			has_global_flag = player_wants_custom_peacedeals
			has_completed_focus = INT_TUR_ottoman_victory
		}
	}
}
