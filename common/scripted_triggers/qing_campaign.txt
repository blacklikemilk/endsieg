# Overview: Qing campaign gates. Every trigger is safe on any country scope.
# They read country flags and variables only, so a non-Qing country gets false rather than an error.

qing_campaign_country = {
	original_tag = QIN
}

qing_campaign_active = {
	has_country_flag = qing_campaign_active
	NOT = { has_country_flag = qing_collapsed }
}

# --- the four survival routes ---
qing_path_constitutional = {
	has_country_flag = qing_route_constitutional
}

qing_path_authoritarian = {
	has_country_flag = qing_route_authoritarian
}

qing_path_yuan = {
	has_country_flag = qing_route_yuan
}

qing_path_federal = {
	has_country_flag = qing_route_federal
}

qing_has_route = {
	OR = {
		qing_path_constitutional = yes
		qing_path_authoritarian = yes
		qing_path_yuan = yes
		qing_path_federal = yes
	}
}

# --- the values as readable bands ---
qing_reform_strong = {
	check_variable = { var = qing_reform value = constant:qing_story.reform_strong compare = greater_than_or_equals }
}

qing_reform_complete = {
	check_variable = { var = qing_reform value = constant:qing_story.reform_complete compare = greater_than_or_equals }
}

qing_legitimacy_weak = {
	check_variable = { var = qing_legitimacy value = constant:qing_story.legitimacy_weak compare = less_than }
}

qing_legitimacy_strong = {
	check_variable = { var = qing_legitimacy value = constant:qing_story.legitimacy_strong compare = greater_than_or_equals }
}

qing_army_loyal = {
	check_variable = { var = qing_army_loyalty value = constant:qing_story.army_strong compare = greater_than_or_equals }
}

qing_army_unreliable = {
	check_variable = { var = qing_army_loyalty value = constant:qing_story.army_weak compare = less_than }
}

qing_provinces_trusting = {
	check_variable = { var = qing_provincial_trust value = constant:qing_story.trust_strong compare = greater_than_or_equals }
}

qing_yuan_dominant = {
	check_variable = { var = qing_yuan_influence value = constant:qing_story.yuan_dominant compare = greater_than_or_equals }
}

# --- the revolution gate ---
# Historical focus is the railroaded timeline: the dynasty falls on the revolution's
# own schedule, so no settlement is offered and no settlement effect can take hold.
qing_historical_collapse = {
	is_historical_focus_on = yes
}

# A dynasty can negotiate its way out of the revolution only when reform,
# army reliability and provincial confidence all hold at the same time.
qing_survival_possible = {
	NOT = { qing_historical_collapse = yes }
	check_variable = { var = qing_reform value = constant:qing_story.survival_reform compare = greater_than_or_equals }
	check_variable = { var = qing_army_loyalty value = constant:qing_story.survival_loyalty compare = greater_than_or_equals }
	check_variable = { var = qing_provincial_trust value = constant:qing_story.survival_trust compare = greater_than_or_equals }
}

# The army can hold Beijing while the court negotiates, even if reform is thin.
qing_can_impose_settlement = {
	NOT = { qing_historical_collapse = yes }
	controls_state = 608
	qing_army_loyal = yes
	OR = {
		qing_reform_strong = yes
		qing_yuan_dominant = yes
	}
}
