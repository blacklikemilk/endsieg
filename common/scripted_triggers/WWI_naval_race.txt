# Country scope: Germany or Britain. A ratified agreement ends the rivalry.
WW1_naval_race_open = {
 has_start_date < 1919.1.1 date < 1919.1.1
 OR = { tag = GER tag = ENG }
 country_exists = GER country_exists = ENG
 GER = { NOT = { has_capitulated = yes } NOT = { has_war_with = ENG } NOT = { is_in_faction_with = ENG } NOT = { GE1_D_naval_convention_in_force = yes } }
 ENG = { NOT = { has_capitulated = yes } }
 NOT = { has_global_flag = GE1_anglo_german_alliance }
 NOT = { has_global_flag = WW1_naval_rivalry_war_seen }
 ENG = { NOT = { has_country_flag = WW1_CP_british_war_concluded } }
}
WW1_naval_arms_race_spirits_active = {
 OR = { WW1_naval_race_open = yes WW1_naval_war_open = yes }
}
WW1_naval_race_focus_ready = {

}
# Compare commissioned battle fleets, including early and modern cruisers.
WW1_naval_race_germany_over_six_tenths = {
 country_exists = GER country_exists = ENG
 set_temp_variable = { WW1_race_ger_count = GER.num_ships_with_type@early_hc }
 add_to_temp_variable = { WW1_race_ger_count = GER.num_ships_with_type@early_bc }
 add_to_temp_variable = { WW1_race_ger_count = GER.num_ships_with_type@dreadnought }
 add_to_temp_variable = { WW1_race_ger_count = GER.num_ships_with_type@battleship }
 add_to_temp_variable = { WW1_race_ger_count = GER.num_ships_with_type@battle_cruiser }
 add_to_temp_variable = { WW1_race_ger_count = GER.num_ships_with_type@early_heavy_cruiser }
 add_to_temp_variable = { WW1_race_ger_count = GER.num_ships_with_type@early_light_cruiser }
 add_to_temp_variable = { WW1_race_ger_count = GER.num_ships_with_type@heavy_cruiser }
 add_to_temp_variable = { WW1_race_ger_count = GER.num_ships_with_type@light_cruiser }
 set_temp_variable = { WW1_race_eng_count = ENG.num_ships_with_type@early_hc }
 add_to_temp_variable = { WW1_race_eng_count = ENG.num_ships_with_type@early_bc }
 add_to_temp_variable = { WW1_race_eng_count = ENG.num_ships_with_type@dreadnought }
 add_to_temp_variable = { WW1_race_eng_count = ENG.num_ships_with_type@battleship }
 add_to_temp_variable = { WW1_race_eng_count = ENG.num_ships_with_type@battle_cruiser }
 add_to_temp_variable = { WW1_race_eng_count = ENG.num_ships_with_type@early_heavy_cruiser }
 add_to_temp_variable = { WW1_race_eng_count = ENG.num_ships_with_type@early_light_cruiser }
 add_to_temp_variable = { WW1_race_eng_count = ENG.num_ships_with_type@heavy_cruiser }
 add_to_temp_variable = { WW1_race_eng_count = ENG.num_ships_with_type@light_cruiser }
 multiply_temp_variable = { WW1_race_ger_count = 5 }
 multiply_temp_variable = { WW1_race_eng_count = 3 }
