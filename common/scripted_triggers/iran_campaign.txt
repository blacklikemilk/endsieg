# Overview: country-scoped validity checks for Iran's chapter campaign.

	original_tag = PER
	has_country_flag = iran_campaign_active
}

	is_subject = no
	country_exists = ENG
	NOT = { has_war_with = ENG }
	ENG = { is_subject = no has_capitulated = no }
}
iran_sponsor_germany_valid = {
	is_subject = no

	NOT = { has_war_with = GER }
	GER = { is_subject = no has_capitulated = no }
}

iran_security_affordable = {
	has_equipment = { infantry_equipment > 249 }
	has_army_experience > 9
}

	has_equipment = { infantry_equipment > 99 }
}
iran_railway_ready = {
	has_country_flag = iran_railway_north
	has_country_flag = iran_railway_south
}
iran_allied_pressure = {
	OR = {
		has_country_flag = SOV_ultimatum_sent_flag
		has_country_flag = PER_has_been_invaded
		has_war_with = ENG
		has_war_with = SOV
		has_war_with = RAJ
		AND = {
			date > 1941.8.24
			country_exists = ENG
			country_exists = SOV
			ENG = { has_war_with = GER }
			SOV = { has_war_with = GER }
			has_country_flag = iran_german_alignment
		}
	}
}

# Can the army actually hold together if the government orders resistance in 1941?
# Officer loyalty is the 1921 contest's long shadow: an officer corps that was
# bought, purged or left unpaid does not dissolve into a resistance movement on
# command. Authority is the state's ability to give the order at all.
iran_army_stands_firm = {
	check_variable = { var = iran_officer_loyalty value = 50 compare = greater_than_or_equals }
	check_variable = { var = iran_authority value = 40 compare = greater_than_or_equals }
	has_war_support > 0.25
}
