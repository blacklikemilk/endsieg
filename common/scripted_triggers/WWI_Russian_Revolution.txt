# A recorded German war survives the source country's annexation. It remains
# pending only while a German treaty is still a coherent diplomatic outcome.
RUS_ww1_brest_pending = {
    country_exists = GER
    GER = { has_capitulated = no }
    NOT = { is_in_faction_with = GER }
    NOT = { has_global_flag = player_wants_custom_peacedeals }
    NOT = { has_global_flag = kaiserschlachtvictory }
    NOT = { has_global_flag = central_powers_victory }
    NOT = { has_global_flag = compiegne }
    OR = { has_global_flag = RUS_ww1_brest_negotiations has_war_with = GER }
}

    exists = yes
    has_start_date < 1918.1.1

    has_global_flag = welcome_event_WW1_active
    NOT = { has_global_flag = treaty_of_brest_litovsk }
    NOT = { has_global_flag = player_wants_custom_peacedeals }
    NOT = { has_global_flag = kaiserschlachtvictory }
    NOT = { has_global_flag = central_powers_victory }
    NOT = { has_global_flag = compiegne }
    country_exists = GER
    GER = { has_capitulated = no }
    NOT = { is_in_faction_with = GER }
    OR = {
        AND = {
            tag = SOV

            has_capitulated = no
            has_global_flag = { flag = october_revolution days > 45 }
            OR = { has_global_flag = RUS_ww1_brest_negotiations has_war_with = GER }

        AND = {
            tag = RIE
            RUS_ww1_historical_revolution_allowed = yes
            has_capitulated = yes
            has_war_with = GER
        }
    }
}
