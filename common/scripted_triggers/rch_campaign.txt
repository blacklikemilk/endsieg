# Overview: Republic of China campaign gates, safe on any country scope.
rch_campaign_country = {
	original_tag = RCH
}

rch_campaign_active = {
	has_country_flag = rch_campaign_active
}

rch_path_parliamentary = {
	has_country_flag = rch_route_parliamentary
}

rch_path_beiyang = {
	has_country_flag = rch_route_beiyang
}

rch_path_kmt = {
	has_country_flag = rch_route_kmt
}

rch_path_provincial = {
	has_country_flag = rch_route_provincial
}

rch_has_route = {
	OR = {
		rch_path_parliamentary = yes
		rch_path_beiyang = yes
		rch_path_kmt = yes
		rch_path_provincial = yes
	}
}

rch_legitimacy_weak = {
	check_variable = { var = rch_legitimacy value = constant:rch_story.legitimacy_weak compare = less_than }
}

rch_legitimacy_strong = {
	check_variable = { var = rch_legitimacy value = constant:rch_story.legitimacy_strong compare = greater_than_or_equals }
}

rch_legitimacy_rescue = {
	check_variable = { var = rch_legitimacy value = constant:rch_story.legitimacy_rescue compare = greater_than_or_equals }
}

rch_military_dominant = {
	check_variable = { var = rch_military_influence value = constant:rch_story.military_dominant compare = greater_than_or_equals }
}

rch_nationalists_strong = {
	check_variable = { var = rch_nationalist_support value = constant:rch_story.nationalist_dominant compare = greater_than_or_equals }
}

# The Beiyang government can hold Beijing only while the army is with it.
rch_beiyang_can_hold = {
	controls_state = 608
	rch_military_dominant = yes
	OR = {
		rch_legitimacy_strong = yes
		has_country_flag = rch_yuan_legacy
	}
}

# Continuity triggers for the later chapters.
rch_path_parliament = {
	OR = {
		has_country_flag = rch_assembly_seated
		has_country_flag = rch_constitution_1923
		has_country_flag = rch_cabinet_government
	}
}

rch_path_army = {
	OR = {
		has_country_flag = rch_beiyang_command
		has_country_flag = rch_officer_pact
		has_country_flag = rch_military_governors
	}
}

rch_path_party = {
	OR = {
		has_country_flag = rch_kmt_reorganised
		has_country_flag = rch_whampoa
		has_country_flag = rch_northern_expedition
	}
}

rch_path_provinces = {
	OR = {
		has_country_flag = rch_provincial_compact
		has_country_flag = rch_tuchun_recognised
		has_country_flag = rch_federal_council
	}
}

rch_gate_constitution = {
	has_country_flag = rch_gate_constitution
}

rch_gate_officer_rotation = {
	has_country_flag = rch_gate_officer_rotation
}

rch_gate_civil_governors = {
	has_country_flag = rch_gate_civil_governors
}

rch_gate_treasury = {
	has_country_flag = rch_gate_treasury
}

rch_gate_throne_refused = {
	has_country_flag = rch_gate_throne_refused
}

# The four reforms plus the throne answer, as the settlement counts them.
rch_gate_settled_count = {
	check_variable = { var = rch_reform_settled value = constant:rch_story.succession_tier0_settled compare = greater_than_or_equals }
}

rch_succession_governors_below_three = {
	check_variable = { var = rch_governor_count value = constant:rch_story.pressure_governor_cap compare = less_than }
}

rch_succession_collapse = {
	OR = {
		check_variable = { var = rch_military_influence value = constant:rch_story.succession_collapse_military compare = greater_than_or_equals }
		check_variable = { var = rch_legitimacy value = constant:rch_story.succession_collapse_legitimacy compare = less_than }
	}
}

# A war against another Chinese government. Manchukuo is excluded because a war with
# it is a war with Japan, which the seat condition does not count as a Chinese war.
rch_chinese_war = {
	OR = {
		has_war_with = CHI
		has_war_with = GDC
		has_war_with = GXC
		has_war_with = PRC
		has_war_with = QIN
		has_war_with = SHX
		has_war_with = SIK
		has_war_with = SND
		has_war_with = XSM
		has_war_with = YUN
	}
}

# Tier 3 is an override: no score and no settlement can buy out of it.
rch_succession_tier_0 = {
	NOT = { rch_succession_collapse = yes }
	check_variable = { var = rch_succession_score value = constant:rch_story.succession_tier0_score compare = greater_than_or_equals }
	rch_gate_settled_count = yes
	rch_gate_throne_refused = yes
	rch_succession_governors_below_three = yes
	controls_state = 608
	NOT = { rch_chinese_war = yes }
}

rch_succession_tier_1 = {
	NOT = { rch_succession_collapse = yes }
	check_variable = { var = rch_succession_score value = constant:rch_story.succession_tier1_score compare = greater_than_or_equals }
}
