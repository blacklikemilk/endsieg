
AST_taog = { # Trigger to check if playing as Australia while having TAOG
	original_tag = AST
	has_dlc = "Thunder at Our Gates"
}

AST_historical_ai = {
	OR = {
		AND = {
			is_historical_focus_on = yes
			has_game_rule = {
				rule = AST_AI_BEHAVIOR
				option = DEFAULT
			}
		}
		has_country_flag = AST_AI_RANDOM_TAOG_HISTORICAL
		has_game_rule = {
			rule = AST_AI_BEHAVIOR
			option = HISTORICAL
		}
	}
}

AST_conscription_level_limited_conscription = {
	if = {
		limit = { NOT = { has_idea = limited_conscription } }
		custom_trigger_tooltip = {
			tooltip = AST_conscription_level_allowed_tt
			check_variable = { AST_conscription_level_allowed > 0 }
		}
	}
}
