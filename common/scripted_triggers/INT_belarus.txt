# Soviet scope. Once displaced, each government may return only in the next historical phase.
INT_BLR_minsk_government_can_form = {
	INT_eastern_historical_soviet_campaign = yes
	has_country_flag = INT_historical_blr_war
	NOT = { has_global_flag = INT_USSR_founded }
	NOT = { country_exists = BLR }
	NOT = { any_subject_country = { original_tag = BLR } }
	206 = { is_owned_by = SOV is_fully_controlled_by = SOV }
	OR = {
		AND = { date > 1918.12.31 date < 1919.8.9 NOT = { has_global_flag = INT_BLR_first_republic_established } }
		AND = { date > 1920.7.30 NOT = { has_global_flag = INT_BLR_second_republic_established } }
	}
}
