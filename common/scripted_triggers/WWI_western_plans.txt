WW1_western_plan_can_continue = {
	tag = GER
	has_start_date < 1919.1.1
	date > 1914.6.1
	date < 1936.1.1
	has_war_with = FRA
	has_capitulated = no
	NOT = {
		OR = {
			has_global_flag = WW1_french_armistice_signed
			has_global_flag = compiegne
			has_global_flag = versailles
			has_global_flag = WW1_CP_final_settlement_signed
		}
	}
}

WW1_western_plan_ui_available = {
	tag = GER
	has_start_date < 1919.1.1
	date < 1936.1.1
	OR = {
		WW1_western_frontline_available = yes
		has_country_flag = WW1_western_plan_started
		WW1_western_plan_ui_schlieffen = yes
		WW1_western_plan_ui_rupprecht = yes
		WW1_western_plan_ui_kronprinz = yes
	}
}
