# Overview: predicates for the Spanish campaign of 1910-1945.
# Chapters, decisions, events and focus branches read these instead of repeating flag and variable checks.
# Variable thresholds come from constant:spain_campaign so tuning lives in one place.

# ---------------------------------------------------------------- chapters

spain_chapter_open = {
	OR = {
		has_country_flag = spain_chapter_restoration
		has_country_flag = spain_chapter_rivera
		has_country_flag = spain_chapter_republic
	}
}

spain_in_restoration_chapter = { has_country_flag = spain_chapter_restoration }
spain_in_rivera_chapter = { has_country_flag = spain_chapter_rivera }
spain_in_republic_chapter = { has_country_flag = spain_chapter_republic }
spain_in_civil_war_chapter = { has_country_flag = spain_chapter_civil_war }
spain_in_postwar_chapter = { has_country_flag = spain_chapter_postwar }
spain_in_republican_postwar_chapter = { has_country_flag = spain_chapter_republican_postwar }

spain_pre_civil_war = {
	NOT = { has_global_flag = spanish_civil_war }
}

spain_civil_war_active = {
	has_global_flag = spanish_civil_war
	NOT = { has_global_flag = scw_over }
}

spain_post_civil_war = {
	OR = {
		has_country_flag = spain_chapter_postwar
		has_country_flag = spain_chapter_republican_postwar
	}
}

# ---------------------------------------------------------------- officer corps

spain_loyalty_low = {
	check_variable = { var = spain_military_loyalty value = constant:spain_campaign.loyalty_wavering compare = less_than }
}

spain_loyalty_high = {
	check_variable = { var = spain_military_loyalty value = constant:spain_campaign.loyalty_loyal compare = greater_than_or_equals }
}

spain_coup_risk = {
	spain_chapter_open = yes
	spain_pre_civil_war = yes
	spain_loyalty_low = yes
	check_variable = { var = spain_coup_pressure value = constant:spain_campaign.coup_attempt_floor compare = greater_than_or_equals }
}

# ---------------------------------------------------------------- Crown against Cortes

spain_crown_dominant = {
	is_power_balance_in_range = { id = SPR_crown_and_cortes range = SPR_crown_absolute }
}

spain_crown_ascendant = {
	OR = {
		is_power_balance_in_range = { id = SPR_crown_and_cortes range = SPR_crown_absolute }
		is_power_balance_in_range = { id = SPR_crown_and_cortes range = SPR_crown_ascendant }
	}
}

spain_cortes_dominant = {
	is_power_balance_in_range = { id = SPR_crown_and_cortes range = SPR_cortes_sovereign }
}

spain_cortes_ascendant = {
	OR = {
		is_power_balance_in_range = { id = SPR_crown_and_cortes range = SPR_cortes_sovereign }
		is_power_balance_in_range = { id = SPR_crown_and_cortes range = SPR_cortes_ascendant }
	}
}

spain_political_deadlock = {
	is_power_balance_in_range = { id = SPR_crown_and_cortes range = SPR_crown_cortes_deadlock }
}
