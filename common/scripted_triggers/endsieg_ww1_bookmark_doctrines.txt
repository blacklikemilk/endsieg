	has_start_date > 1915.12.31
	OR = {
		has_war_with = GER
		has_war_with = AUH
		has_war_with = FRA
		has_war_with = RIE
		has_war_with = TUR
		AND = { is_in_faction_with = GER has_war_with = ENG }
	}
}
