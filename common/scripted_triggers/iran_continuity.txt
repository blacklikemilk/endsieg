# Iran 1910-1945 - cross-era continuity.
#
# The 1910-18 chapter sets 85 outcome flags. These triggers turn those scattered
# flags into the handful of named paths the later chapters actually test, so a
# decision taken in 1913 can open, close or reshape content in 1925 and 1941.
#
# Every trigger here is safe to call on any country scope: each one is a plain OR
# over country flags and simply returns false when none are set.

# --- who governed: the three domestic poles of the constitutional era ---
iran_path_majlis = {
	OR = {
		has_country_flag = iran_majlis_seated
		has_country_flag = iran_anjomans
		has_country_flag = iran_anjoman_elections
		has_country_flag = iran_budget_authority
		has_country_flag = iran_free_press
		has_country_flag = iran_constitution_secured
		has_country_flag = iran_defended_constitution
		has_country_flag = iran_constitutionalist
	}
}

# --- how the war ended for Persia ---
iran_path_humiliated = {
	OR = {
		has_country_flag = iran_ottoman_frontier
		has_country_flag = iran_refugee_administration
		has_country_flag = iran_grain_requisition
	}
}

iran_path_intact = {
	NOT = { has_country_flag = iran_ottoman_frontier }
	NOT = { has_country_flag = iran_refugee_administration }
	iran_path_armed_neutrality = yes
}

# How many of the great-power paths were taken, used for flavour and AI weighting.
iran_path_foreign_count = {
	check_variable = { var = iran_foreign_paths value > 1 }
}
