# Iran 1910-1945 - cross-era continuity.
#
# The 1910-18 chapter sets 85 outcome flags. These triggers turn those scattered
# flags into the handful of named paths the later chapters actually test, so a
# decision taken in 1913 can open, close or reshape content in 1925 and 1941.
#
# Every trigger here is safe to call on any country scope: each one is a plain OR
# over country flags and simply returns false when none are set.
#
# The 1919-25 and 1936-45 chapters write outcome flags of their own. A path is not
# only what the constitutional era chose: it is also what the settlement, the coup,
# the war and the great powers did to the country afterwards, so those outcomes are
# folded into the same named paths. The boundary then reads how the previous
# chapter actually ended instead of only how it began.

# --- who governed: the three domestic poles of the constitutional era ---
iran_path_majlis = {
	OR = {
		has_country_flag = iran_majlis_seated
		has_country_flag = iran_anjomans
		has_country_flag = iran_anjoman_elections
		has_country_flag = iran_budget_authority
		has_country_flag = iran_free_press
		has_country_flag = iran_constitution_secured
		has_country_flag = iran_defended_constitution
		has_country_flag = iran_constitutionalist
		has_country_flag = iran_majlis_restored
		has_country_flag = iran_cabinet_responsibility
		has_country_flag = iran_electoral_law
		# iran_campaign.17.a / .18.a / .50.b / .29.a: the chamber was the body the
		# Shah governed with, the assembly endorsed him, the coup was resisted and
		# the peace conference was petitioned rather than boycotted.
		has_country_flag = iran_shah_with_parliament
		has_country_flag = iran_assembly_endorsed
		has_country_flag = iran_coup_resisted
		has_country_flag = iran_conference_petition
	}
}

iran_path_court = {
	OR = {
		has_country_flag = iran_royal_prerogative
		has_country_flag = iran_court_cabinet
		has_country_flag = iran_palace_guard
		has_country_flag = iran_royal_household
		has_country_flag = iran_shah_majority
		has_country_flag = iran_regency_council
		has_country_flag = iran_court_compromise
		# iran_campaign.17.b / .18.b: the Shah governed through the household and
		# the assembly was managed rather than consulted.
		has_country_flag = iran_shah_with_court
		has_country_flag = iran_assembly_managed
	}
}

iran_path_army = {
	OR = {
		has_country_flag = PER_cossack_brigade
		has_country_flag = iran_cossack_tradition
		has_country_flag = iran_cossack_paid
		has_country_flag = iran_professional_officers
		has_country_flag = iran_officer_loyalty
		has_country_flag = iran_royal_army
		has_country_flag = iran_constitutional_army
		has_country_flag = iran_reserve_established
		# iran_campaign.50.a / .54.a: the 1921 coup was settled by the officers,
		# and the tribes were disarmed by them.
		has_country_flag = iran_coup_settled
		has_country_flag = iran_tribes_disarmed
	}
}

iran_path_gendarmerie = {
	OR = {
		has_country_flag = PER_swedish_gendarmerie
		has_country_flag = iran_national_gendarmerie
		has_country_flag = iran_gendarmerie_tradition
		has_country_flag = iran_gendarmerie_recruits
		has_country_flag = iran_gendarme_resistance
	}
}

iran_path_tribal = {
	OR = {
		has_country_flag = iran_bakhtiari_pact
		has_country_flag = iran_tribal_levies
		has_country_flag = iran_tribal_frontier
		has_country_flag = iran_azerbaijan_command

		# bargained with instead of disarmed.
		has_country_flag = iran_congress_declared
		has_country_flag = iran_tribes_negotiated
	}
}

iran_path_oil_concession = {
	OR = {
		has_country_flag = iran_apoc_concession
		has_country_flag = iran_british_oil_interest
		has_country_flag = iran_concession_guarantees

iran_path_oil_national = {
	OR = {
		has_country_flag = iran_oil_programme
		has_country_flag = iran_royalty_renegotiation

		has_country_flag = iran_oil_dispute
	}
}

# --- foreign alignment ---
iran_path_entente = {
	OR = {
		has_country_flag = iran_british_credit
		has_country_flag = iran_russian_understanding
		has_country_flag = iran_hague_appeal
		has_country_flag = iran_british_oil_interest
		# iran_campaign.61.a / .64.a: London guaranteed the withdrawal and Iran
		# accepted British arbitration over the Gulf.
		has_country_flag = iran_withdrawal_guaranteed
		has_country_flag = iran_gulf_arbitration_accepted
	}
}

iran_path_germany = {
	OR = {
		has_country_flag = iran_third_power
		has_country_flag = iran_niedermayer
		# iran_campaign.70.a / .70.c / .74.c: the mission was accepted and the
		# railway was handed to Berlin rather than to the Allies.
		has_country_flag = iran_german_mission_accepted
		has_country_flag = iran_agreement_transit_axis
	}
}

iran_path_armed_neutrality = {
	OR = {
		has_country_flag = iran_armed_neutrality
		has_country_flag = iran_neutrality_declared
		has_country_flag = PER_neutral_1914
		has_country_flag = iran_mobilized
		# iran_campaign.27.a: neutrality was proclaimed in public, not only held.
		has_country_flag = iran_neutrality_proclaimed
	}
}

# --- how the war ended for Persia ---
iran_path_humiliated = {
	OR = {
		has_country_flag = iran_ottoman_frontier
		has_country_flag = iran_refugee_administration
		has_country_flag = iran_grain_requisition
		# iran_campaign.28.b: the armistice terms were signed as presented.
		has_country_flag = iran_armistice_accepted
	}
}

# "Intact" is a statement about the war, not about neutrality. A Persia that held
# its frontier and administered no refugees came through 1914-18 whole whether it
# declared neutrality, permitted transit or simply was not reached. Requiring
# iran_path_armed_neutrality here classified the transit and allied routes as
# neither intact nor humiliated, so the boundary could not tell them apart.
iran_path_intact = {
	NOT = { iran_path_humiliated = yes }
}

# How many of the great-power openings were taken. Only IR1_entente_understanding
# and IR1_german_opening increment iran_foreign_paths, so this stays for the 1914-18
# owner: no interwar or wartime reader can see a value that stops moving in 1918.
iran_path_foreign_count = {
	check_variable = { var = iran_foreign_paths value > 1 }
}
