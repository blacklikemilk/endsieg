# In state scope. A founding government must actually control its seat.
INT_USSR_delegation_in_soviet_hands = {
	owner = {
		controls_state = PREV
		OR = { tag = SOV AND = { has_government = communism is_subject_of = SOV } }
	}
}

INT_USSR_can_form = {
	tag = SOV has_government = communism is_subject = no has_capitulated = no
	has_start_date < 1922.12.30 date > 1922.12.29
	NOT = { has_global_flag = INT_USSR_founded }
	owns_state = 219 controls_state = 219
	221 = { INT_USSR_delegation_in_soviet_hands = yes owner = { OR = { tag = SOV original_tag = UKR } } }
	206 = { INT_USSR_delegation_in_soviet_hands = yes owner = { OR = { tag = SOV original_tag = BLR } } }
	1016 = { INT_USSR_delegation_in_soviet_hands = yes owner = { OR = { tag = SOV original_tag = AZR } } }

	231 = { INT_USSR_delegation_in_soviet_hands = yes owner = { OR = { tag = SOV original_tag = GEO } } }
}
