# Legacy signature: standard armistice accepted, but neither preparation stage
# ever began before the old automatic deadline. Actual refusal routes have no
# standard-path receipt; completed, alternative and newly confirmed defaults stay.
WW1_GER_missed_preparation_failure = {
	tag = GER
	has_global_flag = welcome_event_WW1_active
	date < 1921.1.1
	has_global_flag = compiegne
	has_global_flag = ww1_compiegne_armistice_failed
	has_country_flag = GE1_compiegne_resolution_started
	has_country_flag = GE1_compiegne_standard_path
	has_government = democratic
	is_subject = no
	has_capitulated = no
	FRA = { exists = yes has_capitulated = no }
	ENG = { exists = yes has_capitulated = no }
	NOT = { has_active_mission = ww1_establish_weimar_republic }
	NOT = { has_active_mission = ww1_treaty_of_versailles }
	NOT = {
		OR = {
			has_global_flag = versailles
			has_global_flag = central_powers_victory
			has_global_flag = player_wants_custom_peacedeals
			has_country_flag = ww1_weimar_republic_established
			has_country_flag = WW1_GER_weimar_countdown_announced
			has_country_flag = WW1_GER_versailles_countdown_announced
			has_country_flag = WW1_GER_postwar_deadline_reconciled_v3
			has_country_flag = WW1_GER_treaty_default_confirmed_v3
			has_country_flag = WW1_GER_missed_preparation_recovered_v3
		}
	}
}
