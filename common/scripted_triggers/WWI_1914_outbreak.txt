# The accepted July Crisis, rather than the calendar alone, authorizes this clock.
WW1_1914_historical_sequence = {
 tag = GER
 is_historical_focus_on = yes
 WW1_crisis_german_escalation = yes
 NOT = { has_global_flag = kronprinz_plan }
 NOT = { has_global_flag = rupprecht_plan }
 NOT = { has_completed_focus = GE1_kronprinz_plan }
 NOT = { has_completed_focus = GE1_rupprecht_plan }
}

# Explicit player escalation during an existing continental war remains available.
# A failed historical-AI precondition alone never authorizes the fallback.
WW1_1914_schlieffen_action_authorized = {
 OR = {
  WW1_1914_historical_sequence = yes
  NOT = { is_historical_focus_on = yes }
  AND = { is_ai = no has_global_flag = schlieffen_plan OR = { has_war_with = FRA has_war_with = RIE } }
 }
}
