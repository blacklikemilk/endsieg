on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { has_start_date < 1919.1.1 country_exists = GER }
				GER = { WW1_update_naval_arms_race_spirits = yes }
			}
		}
	}
	on_weekly_GER = {
		effect = {
			if = {
				limit = { has_start_date < 1919.1.1 date < 1936.1.1 }
				WW1_update_naval_arms_race_spirits = yes
			}
		}
	}
	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					has_start_date < 1919.1.1
					date < 1936.1.1
					OR = {
						AND = { tag = GER FROM = { tag = ENG } }
						AND = { tag = ENG FROM = { tag = GER } }
					}
				}
				GER = { WW1_update_naval_arms_race_spirits = yes }
			}
		}
	}
	on_peace = {
		effect = {
			if = {
				limit = {
					OR = { tag = GER tag = ENG }
					has_start_date < 1919.1.1
					date < 1936.1.1
					has_global_flag = WW1_naval_rivalry_war_seen
					country_exists = GER
				}
				GER = { WW1_update_naval_arms_race_spirits = yes }
			}
		}
	}
}
