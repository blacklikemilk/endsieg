# Country histories grant technologies without invoking every research callback.
# The monthly country pulse covers old saves and countries created afterward.
on_actions = {
	on_startup = {
		effect = {
			every_country = {
				endsieg_interwar_finalize_unit_offsets = yes
				set_country_flag = endsieg_interwar_support_offsets_initialized_v1
			}
		}
	}
	on_monthly = {
		effect = {
			if = {
				limit = { NOT = { has_country_flag = endsieg_interwar_support_offsets_initialized_v1 } }
				endsieg_interwar_finalize_unit_offsets = yes
				set_country_flag = endsieg_interwar_support_offsets_initialized_v1
			}
		}
	}
}
