on_actions = {
    on_startup = {
        effect = {
            # Country histories finish before startup effects. Standardise every
            # initialized country's political power after its scenario data loads.
            every_country = {
                limit = { any_owned_state = { is_capital = yes } }
                set_political_power = 50
            }

            # Establish the Russian Civil War only after both sides are initialized.
            if = {
                limit = {
                    date > 1917.12.31
                    date < 1919.1.1
                    country_exists = SOV
                    country_exists = WHR
                }
                SOV = {
                    if = {
                        limit = { NOT = { has_war_with = WHR } }
                        declare_war_on = { target = WHR type = civil_war }
                        add_civil_war_target = WHR
                    }
                }
            }

            # Country histories are evaluated before every country is fully initialized.
            # Establish the United States' WWI wars after initialization instead.
            if = {
                limit = {
                    date > 1917.4.6
                    date < 1919.1.1
                    country_exists = USC
                }
                if = {
                    limit = {
                        country_exists = FRA
                        FRA = { is_in_faction = yes }
                        USC = { is_in_faction = no }
                    }
                    FRA = { add_to_faction = USC }
                }
                if = {
                    limit = { country_exists = GER }
                    GER = {
                        if = {
                            limit = { NOT = { has_war_with = USC } }
                            declare_war_on = { target = USC type = annex_everything }
                        }
                    }
                }
                if = {
                    limit = { country_exists = AUH }
                    AUH = {
                        if = {
                            limit = { NOT = { has_war_with = USC } }
                            declare_war_on = { target = USC type = annex_everything }
                        }
                    }
                }
                if = {
                    limit = { country_exists = BUL }
                    BUL = {
                        if = {
                            limit = { NOT = { has_war_with = USC } }
                            declare_war_on = { target = USC type = annex_everything }
                        }
                    }
                }
                if = {
                    limit = { country_exists = TUR }
                    TUR = {
                        if = {
                            limit = { NOT = { has_war_with = USC } }
                            declare_war_on = { target = USC type = annex_everything }
                        }
                    }
                }
            }

            # WW2 starts explicitly use the installed vanilla focus trees.
            if = {
                limit = { date > 1936.1.1 }
                every_country = {
                    # Country tags without a history/capital are dormant definitions.
                    # Loading a focus tree for them during scenario setup can dereference
                    # an uninitialised country; only initialise countries on this map.
                    limit = { any_owned_state = { is_capital = yes } }
                    if = {
                        limit = {
                            NOT = {
                                OR = {
                                    tag = GER tag = FRA tag = ITA tag = TUR tag = ENG tag = SOV tag = JAP
                                    tag = HOL tag = EST tag = ROM tag = FIN tag = HUN tag = CAN tag = MEX
                                    tag = NZL tag = NOR tag = URG tag = USA tag = USC
                                }
                            }
                        }
                        load_focus_tree = { tree = generic_focus keep_completed = yes }
                    }
                    if = {
                        limit = { tag = GER }
                        load_focus_tree = { tree = german_focus keep_completed = yes }
                    }
                    if = {
                        limit = { tag = FRA }
                        load_focus_tree = { tree = french_focus keep_completed = yes }
                    }
                    if = {
                        limit = { tag = ITA }
                        load_focus_tree = { tree = italian_focus keep_completed = yes }
                    }
                    if = {
                        limit = { tag = TUR }
                        load_focus_tree = { tree = turkish_focus keep_completed = yes }
                    }
                    if = {
                        limit = { tag = ENG }
                        load_focus_tree = { tree = british_focus keep_completed = yes }
                    }
                    if = {
                        limit = { tag = SOV }
                        load_focus_tree = { tree = soviet_focus keep_completed = yes }
                    }
                    if = {
                        limit = { tag = JAP }
                        load_focus_tree = { tree = japan_wtt_focus keep_completed = yes }
                    }
                    if = {
                        limit = { tag = HOL }
                        load_focus_tree = { tree = netherlands_focus keep_completed = yes }
                    }
                    if = {
                        limit = { tag = EST }
                        load_focus_tree = { tree = estonia_tree keep_completed = yes }
                    }
                    if = {
                        limit = { tag = ROM }
                        load_focus_tree = { tree = romanian_focus keep_completed = yes }
                    }
                    if = {
                        limit = { tag = FIN }
                        load_focus_tree = { tree = finnish_focus keep_completed = yes }
                    }
                    if = {
                        limit = { tag = HUN }
                        load_focus_tree = { tree = hungarian_focus keep_completed = yes }
                    }
                    if = {
                        limit = { tag = CAN }
                        load_focus_tree = { tree = canadian_focus keep_completed = yes }
                    }
                    if = {
                        limit = { tag = MEX }
                        load_focus_tree = { tree = mexico_focus_tree keep_completed = yes }
                    }
                    if = {
                        limit = { tag = NZL }
                        load_focus_tree = { tree = new_zealander_focus keep_completed = yes }
                    }
                    if = {
                        limit = { tag = NOR }
                        load_focus_tree = { tree = norwegian_focus keep_completed = yes }
                    }
                    if = {
                        limit = { tag = URG }
                        load_focus_tree = { tree = uruguay_focus keep_completed = yes }
                    }
                    if = {
                        limit = { OR = { tag = USA tag = USC } }
                        load_focus_tree = { tree = usa_focus keep_completed = yes }
                    }
                }

                # Ensure the WW2 Allies faction exists even when a scenario skips
                # the country-history faction initialization step.
                if = {
                    limit = { ENG = { is_in_faction = no } }
                    ENG = {
                        create_faction_from_template = faction_template_allies
                        add_to_faction = FRA
                        add_to_faction = CAN
                        add_to_faction = AST
                        add_to_faction = NZL
                        add_to_faction = SAF
                        add_to_faction = RAJ
                        add_to_faction = MAL
                    }
                }

                # Later scenario histories normally add these members in the
                # country history blocks.  Keep the roster intact when a
                # scenario starts after those blocks or skips their ordering.
                if = {
                    limit = {
                        date > 1941.1.1
                        ENG = { is_in_faction = yes }
                    }
                    ENG = {
                        if = { limit = { USA = { is_in_faction = no } } add_to_faction = USA }
                        if = { limit = { FRA = { is_in_faction = no } } add_to_faction = FRA }
                        if = { limit = { CAN = { is_in_faction = no } } add_to_faction = CAN }
                        if = { limit = { AST = { is_in_faction = no } } add_to_faction = AST }
                        if = { limit = { NZL = { is_in_faction = no } } add_to_faction = NZL }
                        if = { limit = { RAJ = { is_in_faction = no } } add_to_faction = RAJ }
                        if = { limit = { MAL = { is_in_faction = no } } add_to_faction = MAL }
                        if = { limit = { ANA = { is_in_faction = no } } add_to_faction = ANA }
                        if = { limit = { SAF = { is_in_faction = no } } add_to_faction = SAF }
                        if = { limit = { PER = { is_in_faction = no } } add_to_faction = PER }
                        if = { limit = { BRA = { is_in_faction = no } } add_to_faction = BRA }
                        if = { limit = { IRQ = { is_in_faction = no } } add_to_faction = IRQ }
                        if = { limit = { OMA = { is_in_faction = no } } add_to_faction = OMA }
                        if = { limit = { YEM = { is_in_faction = no } } add_to_faction = YEM }
                        if = { limit = { BEL = { is_in_faction = no } } add_to_faction = BEL }
                        if = { limit = { HOL = { is_in_faction = no } } add_to_faction = HOL }
                        if = { limit = { INS = { is_in_faction = no } } add_to_faction = INS }
                        if = { limit = { ETH = { is_in_faction = no } } add_to_faction = ETH }
                        if = { limit = { HAI = { is_in_faction = no } } add_to_faction = HAI }
                        if = { limit = { DOM = { is_in_faction = no } } add_to_faction = DOM }
                        if = { limit = { BHU = { is_in_faction = no } } add_to_faction = BHU }
                        if = { limit = { NEP = { is_in_faction = no } } add_to_faction = NEP }
                    }
                }

                # The 1944/45 Italian successor is an Allied member in the
                # corresponding Endsieg scenarios.
                if = {
                    limit = {
                        date > 1943.1.1
                        ENG = { is_in_faction = yes }
                        KIT = { is_in_faction = no }
                    }
                    ENG = { add_to_faction = KIT }
                }
            }

            # Resource effects cannot run from country history.  The matching
            # history entries set these temporary flags; complete the original
            # vanilla focuses now, after history initialization has finished.
            HOL = {
                if = {
                    limit = { has_country_flag = endsieg_defer_HOL_expand_curacao_oil_refineries }
                    complete_national_focus = HOL_expand_curacao_oil_refineries
                    clr_country_flag = endsieg_defer_HOL_expand_curacao_oil_refineries
                }
            }
            EST = {
                if = {
                    limit = { has_country_flag = endsieg_defer_EST_natural_resources }
                    complete_national_focus = BALTIC_natural_resources
                    clr_country_flag = endsieg_defer_EST_natural_resources
                }
            }
            ITA = {
                if = {
                    limit = { has_country_flag = endsieg_defer_ITA_steel_industry_in_terni }
                    complete_national_focus = ITA_steel_industry_in_terni
                    clr_country_flag = endsieg_defer_ITA_steel_industry_in_terni
                }
            }
            ROM = {
                if = {
                    limit = { has_country_flag = endsieg_defer_ROM_hunedoara_steel_works }
                    complete_national_focus = ROM_hunedoara_steel_works
                    clr_country_flag = endsieg_defer_ROM_hunedoara_steel_works
                }
            }
            TUR = {
				if = {
					limit = { has_country_flag = endsieg_defer_TUR_reconfigure_foreign_policy }
					complete_national_focus = TUR_reconfigure_our_foreign_policy
					clr_country_flag = endsieg_defer_TUR_reconfigure_foreign_policy
				}
                if = {
                    limit = { has_country_flag = endsieg_defer_TUR_continue_sumerbank }
                    complete_national_focus = TUR_continue_the_sumerbank_industrialization_scheme
                    clr_country_flag = endsieg_defer_TUR_continue_sumerbank
                }
            }
			JAP = {
				if = {
					limit = { has_country_flag = endsieg_defer_JAP_sea_guide_the_zaibatsus }
					complete_national_focus = JAP_sea_guide_the_zaibatsus
					clr_country_flag = endsieg_defer_JAP_sea_guide_the_zaibatsus
				}
			}
            FIN = {
                if = {
                    limit = { has_country_flag = endsieg_defer_FIN_outokumpu_for_defence_industry }
                    complete_national_focus = FIN_outokumpu_for_defence_industry
                    clr_country_flag = endsieg_defer_FIN_outokumpu_for_defence_industry
                }
                if = {
                    limit = { has_country_flag = endsieg_defer_FIN_industrial_upgrade_in_harjavalta }
                    complete_national_focus = FIN_industrial_upgrade_in_harjavalta
                    clr_country_flag = endsieg_defer_FIN_industrial_upgrade_in_harjavalta
                }
                if = {
                    limit = { has_country_flag = endsieg_defer_FIN_janiskoski_power_plant }
                    complete_national_focus = FIN_janiskoski_power_plant
                    clr_country_flag = endsieg_defer_FIN_janiskoski_power_plant
                }
            }
            HUN = {
                if = {
                    limit = { has_country_flag = endsieg_defer_HUN_joint_aluminum_mining_company }
                    complete_national_focus = HUN_joint_aluminum_mining_company
                    clr_country_flag = endsieg_defer_HUN_joint_aluminum_mining_company
                }
            }
            CAN = {
                if = {
                    limit = { has_country_flag = endsieg_defer_CAN_alberta_coal_towns }
                    complete_national_focus = CAN_alberta_coal_towns
                    clr_country_flag = endsieg_defer_CAN_alberta_coal_towns
                }
            }
            MEX = {
                if = {
                    limit = { has_country_flag = endsieg_defer_MEX_focus_oil_field_expansion }
                    complete_national_focus = MEX_focus_oil_field_expansion
                    clr_country_flag = endsieg_defer_MEX_focus_oil_field_expansion
                }
            }
            NZL = {
                if = {
                    limit = { has_country_flag = endsieg_defer_NZL_new_zealand_steel }
                    complete_national_focus = NZL_new_zealand_steel
                    clr_country_flag = endsieg_defer_NZL_new_zealand_steel
                }
                if = {
                    limit = { has_country_flag = endsieg_defer_NZL_taranaki_oil }
                    complete_national_focus = NZL_taranaki_oil
                    clr_country_flag = endsieg_defer_NZL_taranaki_oil
                }
            }
            URG = {
                if = {
                    limit = { has_country_flag = endsieg_defer_URG_la_teja_refinery }
                    complete_national_focus = URG_la_teja_refinery
                    clr_country_flag = endsieg_defer_URG_la_teja_refinery
                }
            }
        }
    }

    on_peaceconference_ended = {
        effect = {
            if = {
                limit = {
                    ROOT = { TAG = SOV }
                    FROM = { TAG = UKR }
                    SOV = { is_ai = yes }
                }
                SOV = { annex_country = { target = UKR transfer_troops = no } }
            }
            if = {
                limit = {
                    ROOT = { TAG = ITA }
                    FROM = { TAG = SEN }
                    ITA = { is_ai = yes }
                }
                ITA = { annex_country = { target = SEN transfer_troops = no } }
            }
            if = {
                limit = {
                    ROOT = { TAG = MEX }
                    FROM = { TAG = VIL }
                    MEX = { is_ai = yes }
                }
                MEX = { annex_country = { target = VIL transfer_troops = no } }
            }
            if = {
                limit = {
                    ROOT = { TAG = GER }
                    FROM = { TAG = BAV }
                    GER = { is_ai = yes }
                }
                GER = { annex_country = { target = BAV transfer_troops = yes } }
            }
        }
    }
    on_puppet = {
        effect = {
            if = {
                limit = {
                    original_tag = EGY
                    FROM = { original_tag = ENG }
                }
                set_cosmetic_tag = EGY_ENG
            }

            if = {
                limit = {
                    original_tag = PRM
                    FROM = { original_tag = HUN }
                }
                set_cosmetic_tag = PRM_HUN
            }
        }
    }

    on_subject_free = {
        effect = {
            if = {
                limit = {
                    original_tag = EGY
                }
                EGY = { drop_cosmetic_tag = yes }
            }

            if = {
                limit = {
                    original_tag = PRM
                    FROM = { original_tag = HUN }
                }
                PRM = { drop_cosmetic_tag = yes }
            }
        }
    }

    on_release_as_puppet = {
        effect = {
            if = {
                limit = {
                    original_tag = EGY
                }
                set_cosmetic_tag = EGY_ENG
            }

            if = {
                limit = {
                    original_tag = PRM
                }
                set_cosmetic_tag = PRM_HUN
            }
        }
    }
}
