                remove_ideas = { IT1_ww1_general_staff IT1_ww1_alpini IT1_ww1_assault_corps IT1_ww1_artillery_survey IT1_ww1_logistics_service IT1_ww1_aviation_service IT1_ww1_caproni_contracts IT1_ww1_adriatic_patrols IT1_ww1_fleet_command IT1_ww1_army_transition IT1_ww1_modernized_army IT1_ww1_industrial_mobilization_spirit IT1_ww1_industrial_labor IT1_ww1_state_arsenals IT1_ww1_private_consortia_spirit IT1_ww1_transition_economy IT1_ww1_saver_protection IT1_ww1_domestic_credit IT1_ww1_trade_credit IT1_ww1_consular_network IT1_ww1_colonial_trade_board IT1_ww1_armed_neutrality_spirit IT1_ww1_mediterranean_council IT1_ww1_parliamentary_government IT1_ww1_catholic_coalition IT1_ww1_liberal_cabinet_spirit IT1_ww1_royal_cabinet_spirit IT1_ww1_professional_civil_service IT1_ww1_republican_guarantees IT1_ww1_factory_councils }
                if = { limit = { has_focus_tree = italian_focus_ww1 } load_focus_tree = { tree = italian_focus keep_completed = yes } }

    # ROOT is the capitulated country; FROM is the winner. Defer annexation
    # until the capitulation callbacks have finished, even in a wider war.
    on_capitulation = {
        effect = {
            if = {
                limit = {
                    date < 1936.1.1
                    has_global_flag = russian_civil_war
                    NOT = { has_global_flag = RUS_CW_Over }
                    ROOT = { original_tag = WHR }
                    FROM = { tag = SOV }
                }
                SOV = {
                    set_country_flag = INT_SOV_white_government_defeated
                    country_event = { id = interwar_soviet.63 hours = 1 }
                }
            }
            if = {
                limit = {
                    date > 1917.12.31
                    date < 1936.1.1
                    ROOT = { tag = UKR }
                    FROM = { tag = SOV }
                    country_exists = UKR
                }
                ROOT = { set_country_flag = INT_ukraine_defeated_by_soviets }
                SOV = {
                    set_country_flag = INT_ukraine_integrated
                    country_event = { id = INT_eastern.4 hours = 1 }
                }
            }
            if = {
                limit = {
                    date < 1936.1.1
                    ROOT = { original_tag = DER }
                    country_exists = ENG
                }
                set_global_flag = endsieg_dervish_capitulated
                ENG = { country_event = { id = ww1_britain.47 hours = 1 } }
            }
            if = {
                limit = {
                    date < 1936.1.1
                    ROOT = { original_tag = GER }
                    country_exists = AST
                }
                set_global_flag = endsieg_germany_ww1_capitulated
                AST = { country_event = { id = ww1_germany.71 hours = 1 } }
            }
        }
    }

    # ROOT is the annexing country; FROM is the country being annexed.
    on_annex = {
        effect = {
            if = {
                limit = {
                    date < 1936.1.1
                    has_global_flag = russian_civil_war
                    NOT = { has_global_flag = RUS_CW_Over }
                    ROOT = { tag = SOV }
                    FROM = { original_tag = WHR }
                }
                SOV = {
                    set_country_flag = INT_SOV_white_government_defeated
                    country_event = { id = interwar_soviet.63 hours = 1 }
                }
            }
            if = {
                limit = {
                    date > 1917.12.31
                    date < 1936.1.1
                    ROOT = { tag = SOV }
                    FROM = { original_tag = UKR }
                }
                ROOT = {
                    set_country_flag = INT_ukraine_integrated
                    country_event = { id = INT_eastern.4 hours = 1 }
                }
            }
        }
    }

    on_peaceconference_ended = {
        effect = {
            # This callback scopes ROOT to a participant returning to peace,
            # not to a winner with the defeated country in FROM.
            if = {
                limit = {
                    date > 1917.12.31
                    date < 1936.1.1
                    tag = SOV
                    has_country_flag = INT_ukraine_integrated
                }
                country_event = { id = INT_eastern.4 hours = 1 }
            }
            # A Finnish Civil War victory must reunify Finland even if the
            # conference would otherwise leave the defeated side as a puppet.
            if = {
                limit = {
                    ROOT = { tag = FNR }
                    FROM = { tag = FNN }
                    country_exists = FNN
                }
                FNN = { every_owned_state = { transfer_state_to = FNR } }
                FNR = { annex_country = { target = FNN transfer_troops = no } }
            }
            if = {
                limit = {
                    ROOT = { tag = FNN }
                    FROM = { tag = FNR }
                    country_exists = FNR
                }
                FNR = { every_owned_state = { transfer_state_to = FNN } }
                FNN = { annex_country = { target = FNR transfer_troops = no } }
            }
            # Complete only a recorded Soviet victory; this callback does not
            # provide a defeated country in FROM.
            if = {
                limit = {
                    date < 1936.1.1
                    has_global_flag = russian_civil_war
                    tag = SOV
                    has_country_flag = INT_SOV_white_government_defeated
                    NOT = { has_global_flag = RUS_CW_Over }
                }
                country_event = { id = interwar_soviet.63 hours = 1 }
            }
            # Soviet victories in the civil-war period absorb every remaining
            # Belarusian holding, including a puppet left by the conference.
            if = {
                limit = {
                    date > 1917.12.31
                    date < 1936.1.1
                    ROOT = { tag = SOV }
                    FROM = { tag = BLR }
                    country_exists = BLR
                }
                SOV = { annex_country = { target = BLR transfer_troops = no } }
            }
            if = {
                limit = {
                    ROOT = { TAG = SOV }
                    FROM = { TAG = UKR }
                    country_exists = UKR
                    OR = {
                        SOV = { is_ai = yes }
                        AND = { date > 1918.11.11 date < 1936.1.1 }
                    }
                }
                SOV = { annex_country = { target = UKR transfer_troops = no } }
            }
            if = {
                limit = {
                    date < 1936.1.1
                    ROOT = { TAG = ITA }
                    FROM = { TAG = SES }
                    ITA = { is_ai = yes }
                }
                ITA = { annex_country = { target = SES transfer_troops = no } }
            }
            if = {
                limit = {
                    ROOT = { TAG = MEX }
                    FROM = { TAG = VIL }
                    MEX = { is_ai = yes }
                }
                MEX = { annex_country = { target = VIL transfer_troops = no } }
            }
            if = {
                limit = {
                    ROOT = { TAG = GER }
                    FROM = { TAG = BAV }
                    GER = { is_ai = yes }
                }
                GER = { annex_country = { target = BAV transfer_troops = yes } }
            }
        }
    }
    on_puppet = {
        effect = {

            if = {
                limit = {
                    date > 1917.12.31
                    date < 1936.1.1

                    FROM = { original_tag = HUN }

            }
        }
    }
