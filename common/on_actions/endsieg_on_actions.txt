    # If Germany declares the French war during a 1945 start, create the
    # Dunkerque garrison immediately after that declaration.  The startup
    # branch above covers scenarios that already begin with this war active.
    # Argentina declared war on Germany on 27 March 1945, after the 20 March bookmark.
    # Keep it date-gated so it is not already at war on the bookmark itself.
    on_daily = {
        effect = {
            # The civil Reichskommissariat replaced the military administration
            # of Belgium and Northern France on 13 July 1944. Do not create it
            # for bookmarks after its December 1944 replacement.
            if = {
                limit = {
                    tag = GER
                    date > 1944.7.12
                    date < 1944.12.15
                    NOT = { has_country_flag = endsieg_RKB_formed }
                    6 = { is_owned_by = GER is_controlled_by = GER }
                    29 = { is_owned_by = GER is_controlled_by = GER }
                    34 = { is_owned_by = GER is_controlled_by = GER }
                }
                6 = { transfer_state_to = RKB }
                29 = { transfer_state_to = RKB }
                34 = { transfer_state_to = RKB }
                set_autonomy = { target = RKB autonomy_state = autonomy_integrated_puppet }
                add_to_faction = RKB
                set_country_flag = endsieg_RKB_formed
            }

            # On 15 December 1944 the Reichskommissariat was replaced by the
            # Reichsgaue. Restore liberated Belgian and French territory and
            # remove the temporary subject.
            if = {
                limit = {
                    tag = GER
                    date > 1944.12.14
                    NOT = { has_country_flag = endsieg_RKB_dissolved }
                }
                if = {
                    limit = {
                        RKB = {
                            exists = yes
                            is_subject_of = GER
                        }
                    }
                    6 = { transfer_state_to = BEL }
                    34 = { transfer_state_to = BEL }
                    29 = { transfer_state_to = FRA }
                    annex_country = { target = RKB transfer_troops = yes }
                }
                set_country_flag = endsieg_RKB_dissolved
            }

            if = {
                limit = {
                    tag = ARG
                    date > 1945.3.26
                    country_exists = GER
                    NOT = { has_country_flag = endsieg_ARG_declared_war_on_GER }
                }
                if = {
                    limit = { NOT = { has_war_with = GER } }
                    declare_war_on = { target = GER type = annex_everything }
                }
                set_country_flag = endsieg_ARG_declared_war_on_GER
            }
        }
    }
