on_actions = {
    on_startup = {
        effect = {
            # WW2 starts explicitly use the installed vanilla focus trees.
            if = {
                limit = { date > 1935.12.30 }
                every_country = {
                    load_focus_tree = { tree = generic_focus keep_completed = yes }
                }
                GER = { load_focus_tree = { tree = german_focus keep_completed = yes } }
                FRA = { load_focus_tree = { tree = french_focus keep_completed = yes } }
                ITA = { load_focus_tree = { tree = italian_focus keep_completed = yes } }
                TUR = { load_focus_tree = { tree = turkish_focus keep_completed = yes } }
                ENG = { load_focus_tree = { tree = british_focus keep_completed = yes } }
                SOV = { load_focus_tree = { tree = soviet_focus keep_completed = yes } }
                USA = { load_focus_tree = { tree = usa_focus keep_completed = yes } }
                USC = { load_focus_tree = { tree = usa_focus keep_completed = yes } }
            }
        }
    }

    on_peaceconference_ended = {
        effect = {
            if = {
                limit = {
                    ROOT = { TAG = SOV }
                    FROM = { TAG = UKR }
                    SOV = { is_ai = yes }
                }
                SOV = { annex_country = { target = UKR transfer_troops = no } }
            }
            if = {
                limit = {
                    ROOT = { TAG = ITA }
                    FROM = { TAG = SEN }
                    ITA = { is_ai = yes }
                }
                ITA = { annex_country = { target = SEN transfer_troops = no } }
            }
            if = {
                limit = {
                    ROOT = { TAG = MEX }
                    FROM = { TAG = VIL }
                    MEX = { is_ai = yes }
                }
                MEX = { annex_country = { target = VIL transfer_troops = no } }
            }
            if = {
                limit = {
                    ROOT = { TAG = GER }
                    FROM = { TAG = BAV }
                    GER = { is_ai = yes }
                }
                GER = { annex_country = { target = BAV transfer_troops = yes } }
            }
        }
    }
    on_puppet = {
        effect = {
            if = {
                limit = {
                    original_tag = EGY
                    FROM = { original_tag = ENG }
                }
                set_cosmetic_tag = EGY_ENG
            }

            if = {
                limit = {
                    original_tag = PRM
                    FROM = { original_tag = HUN }
                }
                set_cosmetic_tag = PRM_HUN
            }
        }
    }

    on_subject_free = {
        effect = {
            if = {
                limit = {
                    original_tag = EGY
                }
                EGY = { drop_cosmetic_tag = yes }
            }

            if = {
                limit = {
                    original_tag = PRM
                    FROM = { original_tag = HUN }
                }
                PRM = { drop_cosmetic_tag = yes }
            }
        }
    }

    on_release_as_puppet = {
        effect = {
            if = {
                limit = {
                    original_tag = EGY
                }
                set_cosmetic_tag = EGY_ENG
            }

            if = {
                limit = {
                    original_tag = PRM
                }
                set_cosmetic_tag = PRM_HUN
            }
        }
    }
}
