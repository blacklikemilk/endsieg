on_actions = {
	# Existing saves can contain the older, larger interwar offsets. Reconcile
	# them on the next monthly country tick; new receipts are marked as v2.
	on_monthly = {
		effect = {
			if = {
				limit = {
					OR = {
						AND = { has_country_flag = endsieg_interwar_offset_wood_flooring NOT = { has_country_flag = endsieg_interwar_offset_wood_flooring_v2 } }
						AND = { has_country_flag = endsieg_interwar_offset_better_trench_conditions NOT = { has_country_flag = endsieg_interwar_offset_better_trench_conditions_v2 } }
					}
				}
				endsieg_interwar_migrate_trench_recovery_offsets = yes
			}
		}
	}
}
