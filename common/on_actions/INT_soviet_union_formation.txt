on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { date > 1917.11.6 date < 1922.12.30 SOV = { exists = yes has_government = communism is_subject = no } }
				SOV = { INT_USSR_prepare_republics = yes }
			}
		}
	}
	on_daily_SOV = {
		effect = {
			if = {
				limit = {
					has_start_date < 1922.12.30 date > 1917.11.6
					has_government = communism is_subject = no NOT = { has_global_flag = INT_USSR_founded }
				}
				INT_USSR_prepare_republics = yes
				if = {
					limit = {
						INT_USSR_can_form = yes
						OR = {
							has_completed_focus = SO2_treaty_of_creation_ussr
							AND = { is_ai = yes is_historical_focus_on = yes NOT = { has_global_flag = player_wants_custom_peacedeals } }
						}
					}
					country_event = { id = interwar_soviet.28 }
				}
			}
			else_if = {
				limit = { OR = { has_cosmetic_tag = INT_russian_sfsr has_cosmetic_tag = INT_russian_soviet_republic } }
				drop_cosmetic_tag = yes
			}
		}
	}
	on_weekly = {
		effect = {
			if = {
				limit = {
					has_start_date < 1922.12.30 date > 1917.11.6
					OR = {
						has_cosmetic_tag = INT_BLR_soviet_republic has_cosmetic_tag = INT_BLR_litbel
					has_cosmetic_tag = INT_BLR_workers_government has_cosmetic_tag = INT_GEO_soviet_republic
						has_cosmetic_tag = INT_ARM_soviet_republic has_cosmetic_tag = INT_UKR_red_banner
						has_cosmetic_tag = INT_UKR_socialist_republic has_cosmetic_tag = AZR_revkom
						has_cosmetic_tag = AZR_1921_constitution
					}
					NOT = { AND = { is_subject_of = SOV has_government = communism SOV = { has_government = communism } } }
				}
				drop_cosmetic_tag = yes
			}
		}
	}
}
