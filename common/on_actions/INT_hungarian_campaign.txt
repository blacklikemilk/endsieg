on_actions = {
	# ROOT is the capitulating country; FROM is its military victor.
	# Run synchronously, before an ordinary conference creates several Hungarys.
	on_capitulation_immediate = {
		effect = {
			if = {
				limit = {
					tag = HUN
					FROM = { OR = { tag = ROM tag = CZE } }
					is_ai = yes
					has_country_flag = INT_HUN_1919_kun_installed
					has_country_flag = INT_HUN_1919_romanian_front
				}
				set_country_flag = INT_HUN_1919_capitulation_intercepted
				if = { limit = { INT_HUN_1919_ai_military_defeat_ready = yes } INT_HUN_1919_finish_romanian_campaign = yes }
				clr_country_flag = INT_HUN_1919_capitulation_intercepted
			}
		}
	}
	on_daily_HUN = {
		effect = {
			HUN = {
				if = {
					limit = { OR = { INT_HUN_1919_ai_military_defeat_ready = yes INT_HUN_1919_partition_recovery_ready = yes } }
					INT_HUN_1919_finish_romanian_campaign = yes
				}
				INT_HUN_1919_update_postwar_cabinets = yes
				INT_HUN_1919_end_budapest_occupation = yes
			}
		}
	}
}
