on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = { exists = yes }
				endsieg_rations_initialize = yes
			}
		}
	}

	on_daily = {
		effect = {
			if = {
				limit = {
					exists = yes
					NOT = { has_country_flag = endsieg_rations_disabled }
				}
				endsieg_rations_daily_tick = yes
			}
		}
	}
}
