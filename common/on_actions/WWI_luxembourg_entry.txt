on_actions = {
	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					has_start_date < 1919.1.1
					date < 1919.1.1
					OR = {
						AND = { tag = GER FROM = { tag = LUX } }
						AND = { tag = LUX FROM = { tag = GER } }
					}
				}
				LUX = { WW1_LUX_join_entente_on_invasion = yes }
			}
		}
	}
	on_daily_LUX = {
		effect = {
			if = {
				limit = {
					has_start_date < 1919.1.1
					date < 1919.1.1
					has_war_with = GER
					NOT = { has_country_flag = WW1_LUX_entente_war_linked }
				}
				WW1_LUX_join_entente_on_invasion = yes
			}
		}
	}
}
