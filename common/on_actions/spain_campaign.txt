# Overview: Spain-only campaign clock.
# Uses a tag-scoped daily hook rather than a world-iterating on_action.
# The civil-war revolter (SPA) has no daily hook of its own, so spain_campaign_tick drives it
# from the original country every week instead of adding a second periodic world loop.

on_actions = {
	on_daily_SPR = {
		effect = { spain_campaign_tick = yes }
	}
	on_monthly_SPR = {
		effect = { spain_party_monthly = yes }
	}
	# a finished focus can move the chamber through the flags it sets, so the seats and the
	# hemicycle are recounted the moment it completes rather than at the next month
	on_national_focus_complete = {
		effect = {
			if = {
				limit = {
					original_tag = SPR
					has_country_flag = spain_campaign_initialized
				}
				spain_cortes_recount = yes
			}
		}
	}
}
