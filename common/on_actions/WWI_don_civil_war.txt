				limit = {

				}
				country_event = { id = INT_don_settlement.2 }

			# The Don Republic is KUB in the Russian Civil War scenario.
			KUB = {
				if = {
					limit = {
						exists = yes
						is_ai = yes
						has_start_date < 1921.1.1
						date > 1917.12.31
						date < 1922.1.1
						has_global_flag = welcome_event_WW1_active

							has_global_flag = russian_civil_war
							AND = {
								NOT = { has_start_date < 1918.1.1 }

							}
						}

						has_capitulated = no

						NOT = { has_war_with = SOV }

						targeted_alliance = WHR
						enemy = SOV

			}
