on_actions = {
 on_daily_ITA = {
  effect = {
   if = {
    limit = { NOT = { has_country_flag = WW1_border_libya_reported } WW1_border_libya = yes }
    set_country_flag = WW1_border_libya_reported
    add_named_threat = { threat = 1 name = WW1_border_threat_libya }
   }
  }
 }
 on_daily_SER = {
  effect = {
   if = {
    limit = { NOT = { has_global_flag = WW1_border_first_balkan_reported } WW1_border_first_balkan = yes }
    set_global_flag = WW1_border_first_balkan_reported
    add_named_threat = { threat = 2 name = WW1_border_threat_first_balkan }
   }
  }
 }
 on_daily_BUL = {
  effect = {
   if = {
    limit = {
     NOT = { has_country_flag = WW1_border_second_balkan_reported }
     NOT = { has_global_flag = second_balkan_war }
     WW1_border_second_balkan = yes
     OR = {
      WW1_border_can_cover = { TARGET = SER }
      WW1_border_can_cover = { TARGET = GRE }
     }
    }
    set_country_flag = WW1_border_second_balkan_reported
    add_named_threat = { threat = 2 name = WW1_border_threat_second_balkan }
   }
  }
 }
}
