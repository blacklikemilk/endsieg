on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = {
					has_start_date > 1943.12.31
					country_exists = ITA
					country_exists = KIT
					ITA = { has_government = fascism }
				}
				ITA = {
					if = {
						limit = { has_character = ITA_giovanni_messe }
						ITA_giovanni_messe = { set_nationality = KIT }
					}
					if = {
						limit = { has_character = ITA_alessandro_pirzio_biroli }
						ITA_alessandro_pirzio_biroli = { set_nationality = KIT }
					}
					if = {
						limit = { has_character = ITA_vittorio_ambrosio }
						ITA_vittorio_ambrosio = { set_nationality = KIT }
					}
				}
			}
		}
	}

	on_daily_SOV = {
		effect = {
			if = {
				limit = {
					has_start_date > 1935.12.31
					date < 1946.1.1
				}
				if = {
					limit = {
						has_start_date < 1941.12.19
						NOT = { date < 1941.12.19 }
						has_war_with = GER
						has_character = SOV_lev_dovator
						SOV_lev_dovator = { endsieg_character_is_available = yes }
					}
					country_event = { id = endsieg_soviet_commanders.1 }
				}
				if = {
					limit = {
						has_start_date < 1942.5.14
						NOT = { date < 1942.5.14 }
						has_character = SOV_boris_shaposhnikov
						SOV_boris_shaposhnikov = { endsieg_character_is_available = yes }
					}
					country_event = { id = endsieg_soviet_commanders.2 }
				}
				if = {
					limit = {
						has_start_date < 1944.4.15
						NOT = { date < 1944.4.15 }
						has_war_with = GER
						has_character = SOV_nikolay_vatutin
						SOV_nikolay_vatutin = { endsieg_character_is_available = yes }
					}
					country_event = { id = endsieg_soviet_commanders.3 }
				}
				if = {
					limit = {
						has_start_date < 1945.2.18
						NOT = { date < 1945.2.18 }
						has_war_with = GER
						has_character = SOV_ivan_chernyakhovsky
						SOV_ivan_chernyakhovsky = { endsieg_character_is_available = yes }
					}
					country_event = { id = endsieg_soviet_commanders.4 }
				}
				if = {
					limit = {
						has_start_date < 1945.3.26
						NOT = { date < 1945.3.26 }
						has_character = SOV_boris_shaposhnikov
					}
					country_event = { id = endsieg_soviet_commanders.5 }
				}
			}
		}
	}
}
