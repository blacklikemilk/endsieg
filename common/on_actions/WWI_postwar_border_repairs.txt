on_actions = {
    on_daily_GER = {
        effect = {
            if = {
                limit = {
                    has_start_date < 1922.1.1
                    date < 1936.1.1
                    has_global_flag = compiegne
                    NOT = { has_global_flag = central_powers_victory }
                    NOT = { has_global_flag = player_wants_custom_peacedeals }
                    NOT = { has_country_flag = WW1_African_settlement_applied }
                    FRA = { exists = yes NOT = { has_war_with = GER } }
                    ENG = { exists = yes NOT = { has_war_with = GER } }
                }
                WW1_settle_german_african_possessions = yes
                set_country_flag = WW1_African_settlement_applied
            }
        }
    }
    on_daily_TRC = {
        effect = {
            if = {
                limit = { has_global_flag = compiegne is_subject_of = GER has_country_flag = RUS_ww1_transcaucasian_protectorate_set }
                GER = { set_autonomy = { target = TRC autonomy_state = autonomy_free end_wars = no end_civil_wars = no } }
            }
            else_if = {
                limit = { NOT = { has_global_flag = compiegne } }
                RUS_ww1_establish_transcaucasian_protectorate = yes
            }
            WW1_TRC_finish_caucasus_breakup = yes
        }
    }
}
