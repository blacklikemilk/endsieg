on_actions = {
	on_daily_TUR = {
		effect = {
			# Recover WWI saves already renamed by the ruling-party on-action.
			if = {
				limit = {
					OT1_P_active = yes
					NOT = { has_country_flag = INT_TUR_ankara_assembly }
					NOT = { has_country_flag = INT_TUR_republic_proclaimed }
					OR = {
						has_cosmetic_tag = TUR_neutrality
						has_cosmetic_tag = TUR_democratic
						has_cosmetic_tag = TUR_fascism
						has_cosmetic_tag = TUR_communism
					}
				}
				set_cosmetic_tag = TUR_ottoman_empire
			}
			if = {
				limit = { WWI_levant_armistice = yes date > 1918.11.12 }
				WWI_update_anatolian_occupation = yes
				if = {
					limit = {
						has_country_flag = INT_TUR_ankara_assembly has_government = neutrality
						NOT = { has_country_flag = INT_TUR_republic_proclaimed }
						NOT = { has_cosmetic_tag = INT_TUR_ankara_assembly_government }
					}
					set_cosmetic_tag = INT_TUR_ankara_assembly_government
				}
			}
			if = {
				limit = { date > 1919.5.18 date < 1924.1.1 }
				INT_TUR_update_independence_war = yes
			}
		}
	}
}
