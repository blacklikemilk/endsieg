on_actions = {
	on_daily_CRI = {
		effect = {
			# Existing saves already have the leader created by INT_eastern.15.
			if = {
				limit = {
					has_cosmetic_tag = INT_CRI_1919_soviet
					NOT = { has_country_flag = INT_CRI_ulyanov_portrait_applied }
					has_country_leader = { name = "Dmitry Ulyanov" ruling_only = yes }
				}
				set_country_leader_portrait = { portrait = GFX_portrait_CRI_dmitry_ulyanov }
				set_country_flag = INT_CRI_ulyanov_portrait_applied
			}
		}
	}
}
