		effect = {
			if = {
				limit = { is_historical_focus_on = yes WW1_balkan_focus_window = yes }

						limit = { NOT = { has_completed_focus = shared_GEN_political_effort } }
						complete_national_focus = shared_GEN_political_effort
					}
					complete_national_focus = shared_GEN_balkan_league

					limit = { WW1_is_balkan_league_member = yes NOT = { has_global_flag = treaty_of_london_1913 } }

						is_ai = yes

						NOT = { has_completed_focus = shared_GEN_first_balkan_war }

				}
			}
		}
	}

			if = {

					WW1_balkan_can_start_second_war = yes

	}
