on_actions = {
	on_declare_war = {
		effect = {
			if = {
				limit = {
					tag = ENG
					FROM = { tag = USA }
					has_start_date < 1919.1.1
					date < 1924.1.1
					has_completed_focus = EN1_R_american_ultimatum
					NOT = { has_country_flag = WW1_alt_atlantic_accepted }
					NOT = { has_country_flag = WW1_alt_mexican_offer_sent }
					MEX = { WW1_alt_mexico_can_accept_british_offer = yes }
				}
				set_country_flag = WW1_alt_mexican_offer_sent
				MEX = { country_event = { id = ww1_anglo_american_war.1 days = 2 } }
			}
		}
	}
}
