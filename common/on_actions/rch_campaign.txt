# Overview: Republic of China chapter clock, without a world-iterating periodic hook.
on_actions = {
	on_daily_RCH = {
		effect = {
			if = {
				limit = {
					has_start_date < 1912.1.1
					date < 1914.1.1
					NOT = { has_country_flag = WW1_RCH_revolution_reserves_raised }
				}
				WW1_RCH_raise_revolution_reserves = yes
			}
			if = {
				limit = {
					has_start_date < 1912.1.1
					date < 1914.1.1
					NOT = { has_country_flag = WW1_RCH_xinhai_front_reinforced }
				}
				WW1_RCH_reinforce_xinhai_front = yes
			}
			if = {
				limit = {
					has_start_date < 1912.1.1
					date < 1914.1.1
					NOT = { has_country_flag = WW1_RCH_revolution_rations_issued }
				}
				WW1_RCH_issue_revolution_rations = yes
			}
			if = {
				limit = { has_start_date < 1912.1.1 date < 1912.3.11 }
				WW1_RCH_sync_revolution_leadership = yes
			}
			WW1_RCH_sync_revolution_capitals = yes
			rch_campaign_tick = yes
			# Reunification is tested from the Republic's own clock, which already exists and
			# is bound to RCH. The test is guarded by the model's own presence, walks a
			# bounded list of Chinese campaign countries and stops at the first match, so a
			# campaign that has not reunited pays a few variable checks a day and no world
			# sweep. It is deliberately not a world-iterating on_daily.
			china_test_reunification = yes

			# One guarded release hook for the pre-1936 warlord roster. The wrapper is
			# idempotent, but the flag is checked here as well so the call is skipped outright
			# once Shanxi exists. rch_campaign_tick above is the Republic's own clock and is not
			# part of this hook.
			if = {
				limit = { date > 1916.6.5 NOT = { has_country_flag = china_released_shx } }
				china_release_warlord_shx = yes
				if = {
					limit = { has_country_flag = china_released_shx }
					SHX = { country_event = { id = shx_campaign.1 days = 1 } }
				}
			}
		}
	}
}
