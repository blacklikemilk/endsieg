# Overview: Republic of China chapter clock, without a world-iterating periodic hook.
on_actions = {
	on_daily_RCH = {
		effect = {
			# Announce either the abdication settlement or a military defeat once.
			# Control is sufficient during an unfinished peace conference; do not
			# change ownership, end wars, or assume that other enemies have surrendered.
			if = {
				limit = {
					has_start_date < 1912.1.1
					date < 1936.1.1
					has_global_flag = xinhai_revolution
					NOT = { has_global_flag = WW1_xinhai_victory_news_sent }
					OR = {
						AND = { has_global_flag = end_of_qing NOT = { country_exists = QIN } }
						AND = { NOT = { country_exists = QIN } owns_state = 608 }
						AND = {
							country_exists = QIN
							has_war_with = QIN
							QIN = { has_capitulated = yes }
							608 = { is_controlled_by = RCH }
						}
					}
				}
				set_global_flag = WW1_xinhai_victory_news_sent
				if = {
					limit = { country_exists = QIN }
					set_global_flag = WW1_xinhai_news_military_victory
				}
				news_event = { id = ww1_xinhai_news.1 hours = 1 }
			}
			WW1_RCH_repair_spawn_scope = yes
			if = {
				limit = {
					has_start_date < 1912.1.1
					date < 1914.1.1
					NOT = { has_country_flag = WW1_RCH_revolution_reserves_raised }
				}
				WW1_RCH_raise_revolution_reserves = yes
			}
			if = {
				limit = {
					has_start_date < 1912.1.1
					date < 1914.1.1
					NOT = { has_country_flag = WW1_RCH_xinhai_front_expanded }
				}
				WW1_RCH_reinforce_xinhai_front = yes
			}
			WW1_RCH_mobilize_xinhai_provinces = yes
			if = {
				limit = {
					has_start_date < 1912.1.1
					date < 1914.1.1
					NOT = { has_country_flag = WW1_RCH_revolution_rations_issued }
				}
				WW1_RCH_issue_revolution_rations = yes
			}
			if = {
				limit = { has_start_date < 1912.1.1 date < 1912.3.11 }
				WW1_RCH_sync_revolution_leadership = yes
			}
			WW1_RCH_sync_revolution_capitals = yes
			# Historical AI succession; a human or an alternative government keeps its chosen leader.
			if = {
				limit = { is_ai = yes is_historical_focus_on = yes date > 1917.8.5 date < 1918.10.10 has_government = democratic has_character = RCH_feng_guozhang has_country_leader = { name = "Li Yuanhong" ruling_only = yes } NOT = { country_exists = QIN } NOT = { has_country_flag = WW1_RCH_feng_succession_done } }
				promote_character = { character = RCH_feng_guozhang ideology = conservatism }
				set_country_flag = WW1_RCH_feng_succession_done
			}
			rch_campaign_tick = yes
			# Reunification is tested from the Republic's own clock, which already exists and
			# is bound to RCH. The test is guarded by the model's own presence, walks a
			# bounded list of Chinese campaign countries and stops at the first match, so a
			# campaign that has not reunited pays a few variable checks a day and no world
			# sweep. It is deliberately not a world-iterating on_daily.
			china_test_reunification = yes

			# One guarded release hook for the pre-1936 warlord roster. The wrapper is
			# idempotent, but the flag is checked here as well so the call is skipped outright
			# once Shanxi exists. rch_campaign_tick above is the Republic's own clock and is not
			# part of this hook.
			if = {
				limit = { date > 1916.6.5 NOT = { has_country_flag = china_released_shx } }
				china_release_warlord_shx = yes
				if = {
					limit = { has_country_flag = china_released_shx }
					SHX = { country_event = { id = shx_campaign.1 days = 1 } }
				}
			}
		}
	}
}
