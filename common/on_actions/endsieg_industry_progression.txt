on_actions = {
	on_startup = {
		effect = {
			every_country = {
				endsieg_industry_reconcile = yes
			}
		}
	}
	# One-time initialization for existing saves and newly created countries.
	# Researched technologies maintain the modifier through their own callbacks.
	on_monthly = {
		effect = {
			if = {
				limit = {
					OR = {
						NOT = { has_country_flag = endsieg_industry_progression_v1 }
						NOT = { has_dynamic_modifier = endsieg_industry_development }
					}
				}
				endsieg_industry_reconcile = yes
			}
		}
	}
}
