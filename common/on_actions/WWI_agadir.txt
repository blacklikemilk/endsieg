on_actions = {
	on_daily_GER = {
		effect = {
			if = {
				limit = { has_global_flag = GE1_agadir_crisis_pending }
				WW1_agadir_advance_negotiations = yes
			}
		}
	}
	on_daily_FRA = {
		effect = {
			if = {
				limit = {
					has_global_flag = WW1_agadir_morocco_congo_agreement_signed
					NOT = { has_country_flag = WW1_agadir_fez_conference_queued }
					date > 1912.3.29
					date < 1914.7.1
					WW1_agadir_fez_available = yes
				}
				set_country_flag = WW1_agadir_fez_conference_queued
				country_event = { id = ww1_agadir.2 }
			}
			if = {
				limit = {
					has_global_flag = WW1_agadir_treaty_of_fez_signed
					NOT = { has_global_flag = WW1_agadir_rif_convention_queued }
					date > 1912.11.26
					date < 1914.7.1
					country_exists = SPR
					NOT = { has_war_with = SPR }
					1000 = { is_owned_and_controlled_by = FRA }
				}
				set_global_flag = WW1_agadir_rif_convention_queued
				SPR = { country_event = { id = ww1_agadir.3 } }
			}
		}
	}
}
