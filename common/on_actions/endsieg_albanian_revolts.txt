on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { date < 1914.1.1 country_exists = ALB }
				ALB = {
					WW1_ALB_restore_1911_revolt = yes
					WW1_ALB_refresh_active_revolt = yes
				}
			}
		}
	}
	on_daily = {
		effect = {
			if = {
				limit = {
					tag = ALB
					date < 1914.1.1
					has_war_with = TUR
					NOT = { has_country_flag = WW1_ALB_revolt_setup_current }
				}
				WW1_ALB_restore_1911_revolt = yes
				WW1_ALB_refresh_active_revolt = yes
			}
		}
	}
}
