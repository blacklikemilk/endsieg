on_actions = {
 on_declare_war = {
  effect = {
   if = { limit = { tag = GER FROM = { tag = BEL } has_start_date < 1936.1.1 date < 1936.1.1 } set_global_flag = WW1_belgian_neutrality_violated  ENG = { WW1_british_queue_war_entry = yes } }
  }
 }
 on_daily = {
  effect = {
   WW1_initialize_bookmark_event_history = yes
   if = {
    limit = { tag = ENG has_focus_tree = british_focus_ww1 has_start_date < 1936.1.1 date < 1936.1.1 }
    WW1_british_prepare_war_entry = yes
    WW1_british_queue_war_entry = yes
   }

   if = {
    limit = { tag = GER }
    WW1_reconcile_alternative_ai_routes = yes
    WW1_1914_outbreak_tick = yes
    GE1_baghdad_railway_tick = yes
		if = {
			limit = {
				NOT = { has_country_flag = WW1_western_plan_closed }
				OR = {
					has_global_flag = versailles
					has_global_flag = WW1_CP_final_settlement_signed
				}
			}
			WW1_western_plan_close_after_peace = yes
		}
   }

 if = { limit = { OR = { has_country_flag = WW1_focus_faction_central_pending has_country_flag = WW1_focus_faction_entente_pending } } WW1_complete_focus_faction_entry = yes }

 WW1_reconcile_crisis_focus_completion = yes
 WW1_finish_completed_focus_actions = yes
}
 }

 on_daily_GER = {
  effect = {
   GER = {
    WW1_western_plan_begin = yes
    WW1_western_frontline_tick = yes
    WW1_western_front_update_strength = yes
   }
  }
 }

 on_daily_RIE = {
  effect = {
   WW1_1914_migrate_russian_mobilization = yes
   RUS_ww1_repair_revolution_chain = yes
  }
 }
 on_daily_SOV = { effect = { RUS_ww1_repair_revolution_chain = yes } }

 on_daily_TIB = { effect = { WW1_TIB_restore_uprising_leadership = yes } }

 on_startup = {
  effect = {
   every_country = { WW1_initialize_bookmark_event_history = yes }
   every_country = { limit = { OR = { tag = RIE tag = SOV } } RUS_ww1_repair_revolution_chain = yes }

   GER = {
    WW1_initialize_alternative_ai_routes = yes
    WW1_western_plan_begin = yes
    WW1_western_front_update_strength = yes
   }
   ENG = { WW1_british_prepare_war_entry = yes WW1_british_queue_war_entry = yes }
}
 }
	on_daily_ITA = {
		effect = {
			if = {
				limit = { IT1_ww1_libyan_transport_missing = yes }
				country_event = { id = ww1_italy.221 }
			}
		}
	}
	on_daily_MEX = {
		effect = {
			if = {
				limit = { WW1_MEX_maderista_identity_repair_due = yes }
				country_event = { id = endsieg_maderista_identity.1 }
			}
		}
	}
	on_daily_MOR = {
		effect = {
			if = {
				limit = { has_government = neutrality has_start_date < 1936.1.1 date < 1936.1.1 }
				if = {
					limit = {
						date < 1912.8.13
						has_country_leader = { name = "Mohammed V" ruling_only = yes }
					}
					MOR_install_sultan = { sultan = MOR_abd_al_hafid }
				}
				else_if = {
					limit = {
						NOT = { date < 1912.8.13 }
						date < 1927.11.18
						OR = {
							has_country_leader = { name = "Abd al Hafid" ruling_only = yes }
							has_country_leader = { name = "Mohammed V" ruling_only = yes }
						}
					}
					MOR_install_sultan = { sultan = MOR_moulay_youssef }
				}
				else_if = {
					limit = {
						NOT = { date < 1927.11.18 }
						has_country_leader = { name = "Moulay Youssef" ruling_only = yes }
					}
					MOR_install_sultan = { sultan = MOR_mohammed_v }
				}
			}
		}
	}
}
