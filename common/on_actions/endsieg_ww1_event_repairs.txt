on_actions = {
 on_declare_war = {
  effect = {
   if = { limit = { tag = GER FROM = { tag = BEL } has_start_date < 1936.1.1 date < 1936.1.1 } set_global_flag = WW1_belgian_neutrality_violated  ENG = { if = { limit = { WW1_british_historical_belgian_response = yes } EN1_R_join_entente_war = yes } } }
  }
 }
 on_daily = {
  effect = {
   if = {
    limit = { WW1_british_historical_belgian_response = yes OR = { NOT = { is_in_faction_with = FRA } NOT = { has_war_with = GER } } }
    EN1_R_join_entente_war = yes
   }

   if = { limit = { OR = { tag = RIE tag = SOV } } RUS_ww1_repair_revolution_chain = yes }

   if = { limit = { tag = GER } WW1_reconcile_alternative_ai_routes = yes }

 if = { limit = { OR = { has_country_flag = WW1_focus_faction_central_pending has_country_flag = WW1_focus_faction_entente_pending } } WW1_complete_focus_faction_entry = yes }

 WW1_finish_completed_focus_actions = yes
}
 }

 on_startup = {
  effect = {
   every_country = { limit = { OR = { tag = RIE tag = SOV } } RUS_ww1_repair_revolution_chain = yes }

   GER = { WW1_initialize_alternative_ai_routes = yes }
}
 }
}
