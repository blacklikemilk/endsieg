on_actions = {
 on_declare_war = {
  effect = {
   if = { limit = { tag = GER FROM = { tag = BEL } has_start_date < 1936.1.1 date < 1936.1.1 } set_global_flag = WW1_belgian_neutrality_violated  ENG = { WW1_british_queue_war_entry = yes } }
  }
 }
 on_daily = {
  effect = {
   if = {
    limit = { tag = ENG has_focus_tree = british_focus_ww1 has_start_date < 1936.1.1 date < 1936.1.1 }
    WW1_british_prepare_war_entry = yes
    WW1_british_queue_war_entry = yes
   }

   if = { limit = { OR = { tag = RIE tag = SOV } } RUS_ww1_repair_revolution_chain = yes }

   if = { limit = { tag = GER } WW1_reconcile_alternative_ai_routes = yes WW1_1914_outbreak_tick = yes }

 if = { limit = { OR = { has_country_flag = WW1_focus_faction_central_pending has_country_flag = WW1_focus_faction_entente_pending } } WW1_complete_focus_faction_entry = yes }

 WW1_reconcile_crisis_focus_completion = yes
 WW1_finish_completed_focus_actions = yes
}
 }

 on_startup = {
  effect = {
   every_country = { limit = { OR = { tag = RIE tag = SOV } } RUS_ww1_repair_revolution_chain = yes }

   GER = { WW1_initialize_alternative_ai_routes = yes }
   ENG = { WW1_british_prepare_war_entry = yes WW1_british_queue_war_entry = yes }
}
 }
}
