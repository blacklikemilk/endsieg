on_actions = {
	on_daily_SOV = {
		effect = {
			SOV = {
				if = {
					limit = {
						date > 1917.12.31 date < 1921.8.14 has_start_date < 1921.8.14
						is_historical_focus_on = yes is_ai = yes has_government = communism has_capitulated = no
						NOT = { has_global_flag = player_wants_custom_peacedeals }
						NOT = { has_global_flag = central_powers_victory }
						329 = { is_core_of = TAN }
					}
					# Existing saves retain the old 1910 release core.
					if = {
						limit = {
							TAN = {
								exists = yes is_ai = yes owns_state = 329 controls_state = 329
								has_government = communism is_neighbor_of = SOV
								check_variable = { num_owned_states = 1 }
								OR = { is_subject_of = SOV AND = { is_subject = no is_in_faction = no has_war = no } }
							}
						}
						annex_country = { target = TAN transfer_troops = no }
					}
					if = {
						limit = { NOT = { country_exists = TAN } 329 = { owner = { is_ai = yes OR = { tag = SOV tag = WHR } } } }
						329 = { remove_core_of = TAN }
					}
				}
				if = { limit = { is_ai = yes is_historical_focus_on = yes INT_TAN_khural_can_meet = yes } country_event = { id = INT_tuvan_khural.1 } }
				if = { limit = { date > 1918.9.30 DON = { exists = yes has_country_flag = INT_DON_1918_rostov_government } } country_event = { id = INT_don_settlement.4 } }
				if = {
					limit = { INT_DON_historical_soviet_settlement = yes any_other_country = { INT_DON_red_remnant = yes } }
					country_event = { id = INT_don_settlement.1 }
				}
			}
		}
	}
	on_capitulation_immediate = {
		effect = {
			if = {
				limit = {
					original_tag = KUB is_ai = yes FROM = { tag = SOV }
					INT_DON_historical_soviet_settlement = yes
					has_war_with = SOV
					OR = { is_subject = no is_subject_of = WHR }
				}
				SOV = { annex_country = { target = PREV transfer_troops = no } }
			}
		}
	}
	on_daily_TAN = {
		effect = {
			TAN = {
				if = { limit = { has_country_flag = INT_TAN_historical_government has_government = communism date < 1936.1.1 } INT_TAN_update_government = yes }
			}
		}
	}
}
