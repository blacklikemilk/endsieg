# Install the conditional pressure modifier without changing the mission timer.
# Startup handles new campaigns and loads; the daily guard also covers older saves.
on_actions = {
	on_startup = {
		effect = {
			GER = {
				if = {
					limit = {
						exists = yes
						NOT = { has_dynamic_modifier = endsieg_ww1_peace_treaty_pressure }
					}
					add_dynamic_modifier = { modifier = endsieg_ww1_peace_treaty_pressure }
				}
			}
		}
	}
	on_daily_GER = {
		effect = {
			if = {
				limit = {
					NOT = { has_dynamic_modifier = endsieg_ww1_peace_treaty_pressure }
				}
				add_dynamic_modifier = { modifier = endsieg_ww1_peace_treaty_pressure }
			}
		}
	}
}
