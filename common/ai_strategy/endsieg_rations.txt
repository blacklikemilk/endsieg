# Keep enough ration factories for the deployed army while its reserve is low.
# The production flag is held from 90 to 180 days of supply so lines have time
# to gain efficiency instead of being removed as soon as the gauge fills.
endsieg_ai_rations_small = {
	enable = {
		is_ai = yes
		has_country_flag = endsieg_rations_ai_produce
		NOT = { has_country_flag = endsieg_rations_disabled }
		num_of_military_factories > 0
		check_variable = { endsieg_ration_daily_use > 0 }
		check_variable = { endsieg_ration_daily_use < 20 }
		OR = {
			has_tech = early_infantry_weapons
			has_tech = rations
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = equipment_production_min_factories_archetype
		id = rations
		value = 1
	}
}

# Arms Against Tyranny's market uses the same equipment archetype IDs as
# production. Rations are not required by division templates, so add an
# explicit shortage reserve for countries that can use the market. Market
# access is still checked by the engine when it evaluates the purchase.
endsieg_ai_rations_market_buy_small = {
	allowed = { has_dlc = "Arms Against Tyranny" }
	enable = {
		is_ai = yes
		has_country_flag = endsieg_rations_ai_produce
		NOT = { has_country_flag = endsieg_rations_disabled }
		num_of_civilian_factories > 0
		check_variable = { endsieg_ration_daily_use > 0 }
		check_variable = { endsieg_ration_daily_use < 20 }
		check_variable = { endsieg_ration_days < 90 }
		OR = {
			has_tech = early_infantry_weapons
			has_tech = rations
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = equipment_market_buying_threshold
		id = rations
		value = 1800
	}
}

# Current vanilla sale strategies target equipment TYPE IDs such as infantry,
# support, armor, and train. A custom ration archetype has no dedicated type;
# targeting support would also sell ordinary support equipment, so no ration
# seller strategy is installed. The engine can still buy rations by archetype.
