italy_complete_libya_campaign = {
	allowed = {
		original_tag = ITA
	}
	enable = {
		IT1_ww1_libyan_campaign_active = yes
		is_ai = yes
	}
	abort_when_not_enabled = yes

	# Keep Italy's available army in North Africa rather than encouraging
	# broad naval invasions of Anatolia.
	ai_strategy = {
		type = area_priority
		id = north_africa
		value = 200
	}

	ai_strategy = {
		type = front_unit_request
		area = north_africa
		value = 100
	}

	ai_strategy = {
		type = front_control
		tag = TUR
		state_trigger = {
			OR = {
				state = 448 # Tripoli
				state = 449 # Libyan Coast
				state = 450 # Benghasi
				state = 451 # Derna
				state = 661 # Tripolitania
				state = 662 # Sirte
				state = 663 # Cyrenaica
			}
		}
		ratio = 0.9
		priority = 100
		ordertype = front
		execution_type = balanced
		execute_order = yes
	}

	# Supply the beachheads and make escort missions worthwhile during the war.
	ai_strategy = {
		type = unit_ratio
		id = convoy
		value = 30
	}
	ai_strategy = {
		type = naval_mission_threshold
		id = MISSION_CONVOY_ESCORT
		value = -50
	}
}

# The countrywide push ends as soon as Italy has secured the Libyan ports.
italy_libyan_naval_landings = {
	allowed = { original_tag = ITA }
	enable = {
		IT1_ww1_libyan_campaign_active = yes
		is_ai = yes
		has_tech = ww1_transport
		TUR = {
			OR = { controls_province = 1149 controls_province = 11954 controls_province = 1130 }
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = invade id = TUR value = 250 }
	ai_strategy = { type = naval_invasion_focus value = 180 }
	ai_strategy = { type = naval_invasion_support_priority id = 29 value = 220 } # Central Mediterranean
	ai_strategy = { type = naval_invasion_support_priority id = 69 value = 200 } # Eastern Mediterranean, off Derna
	ai_strategy = { type = naval_invasion_support_priority id = 169 value = 140 } # Tyrrhenian Sea, Italian approach
}
