# Native diplomatic AI restraint only; the dated declaration helpers remain authoritative.

WW1_historical_hold_GER_SER = {
 allowed = { original_tag = GER }
 enable = {
  is_ai = yes is_historical_focus_on = yes
  has_global_flag = welcome_event_WW1_active has_start_date < 1936.1.1
  date < 1914.8.6 NOT = { has_war_with = SER }
  NOT = { has_global_flag = compiegne } NOT = { has_global_flag = central_powers_victory }
  has_war_with = RIE
 }
 abort_when_not_enabled = yes
 ai_strategy = { type = dont_join_wars_with id = AUH target_country = SER value = 99999 }
 ai_strategy = { type = dont_join_wars_with id = TUR target_country = SER value = 99999 }
}
