# The Russian army covers its own theatres instead of reinforcing western allies.
RIE_WW1_keep_armies_on_russian_fronts = {
 allowed = { original_tag = RIE }
 enable = {
  tag = RIE
  is_ai = yes
  has_start_date < 1936.1.1
  date < 1936.1.1
  has_global_flag = welcome_event_WW1_active
  OR = {
   is_in_faction_with = FRA
   is_in_faction_with = ENG
   is_in_faction_with = BEL
   is_in_faction_with = HOL
   is_in_faction_with = LUX
  }
 }
 abort_when_not_enabled = yes

 ai_strategy = { type = dont_defend_ally_borders id = FRA value = 100 }
 ai_strategy = { type = dont_defend_ally_borders id = ENG value = 100 }
 ai_strategy = { type = dont_defend_ally_borders id = BEL value = 100 }
 ai_strategy = { type = dont_defend_ally_borders id = HOL value = 100 }
 ai_strategy = { type = dont_defend_ally_borders id = LUX value = 100 }

 # State targets also cover allied territory temporarily occupied by the enemy.
 ai_strategy = {
  type = front_unit_request
  state_trigger = {
   NOT = { is_core_of = RIE }
   OR = {
    AND = { is_core_of = FRA FRA = { is_in_faction_with = RIE } }
    AND = { is_core_of = ENG ENG = { is_in_faction_with = RIE } }
    AND = { is_core_of = BEL BEL = { is_in_faction_with = RIE } }
    AND = { is_core_of = HOL HOL = { is_in_faction_with = RIE } }
    AND = { is_core_of = LUX LUX = { is_in_faction_with = RIE } }
   }
  }
  value = -100
 }
 ai_strategy = {
  type = invasion_unit_request
  state_trigger = {
   NOT = { is_core_of = RIE }
   OR = {
    AND = { is_core_of = FRA FRA = { is_in_faction_with = RIE } }
    AND = { is_core_of = ENG ENG = { is_in_faction_with = RIE } }
    AND = { is_core_of = BEL BEL = { is_in_faction_with = RIE } }
    AND = { is_core_of = HOL HOL = { is_in_faction_with = RIE } }
    AND = { is_core_of = LUX LUX = { is_in_faction_with = RIE } }
   }
  }
  value = -100
 }
}
