# Faction manifests are evaluated in the faction leader's country scope.
# Keep the denominator unfiltered: empty completion after 1935 gives zero
# progress and therefore zero scaled modifiers, even in a surviving faction.

ww1_faction_secured_core_states = {
	input = collection:faction_owned_core_states
	operators = {
		limit = {
			ROOT = { has_start_date < 1936.1.1 date < 1936.1.1 }
			is_controlled_by_ROOT_or_ally = yes
		}
	}
	name = WW1_COLLECTION_SECURED_CORE_STATES
}

ww1_faction_members = {
	input = collection:faction_members
	name = WW1_COLLECTION_FACTION_MEMBERS
}

ww1_faction_members_at_peace = {
	input = collection:faction_members
	operators = {
		limit = {
			ROOT = { has_start_date < 1936.1.1 date < 1936.1.1 }
			has_war = no
		}
	}
	name = WW1_COLLECTION_MEMBERS_AT_PEACE
}
