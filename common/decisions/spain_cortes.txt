# Overview: bills before the Spanish Cortes.
# A bill is no longer a political-power spend gated only on the government's majority. Each bill has a
# supporting bloc of party ids in common/scripted_effects/spain_cortes_bills.txt, which counts the
# seats that would carry it (spain_bill_votes_<bill>). The bill is offered only while the chamber can
# still carry it: either the supporting bloc holds a clear majority of the 400 seats, or the government
# is within constant:spain_campaign.bill_negotiation_reach seats of one.
# Taking the bill pays its political-power cost and opens the negotiation event spain_bills.<n>, where
# the government passes it cleanly, buys the missing votes with a concession, waters it down, or
# withdraws. Passage is once per campaign; a bill that fails can be brought again after its re-enable.
# Documented in docs/spain_cortes_bills.md.

SPR_cortes = {

	# Budget. Backed by the government, the Liberals and the Conservatives.
	SPR_bill_budget = {
		icon = GFX_decision_category_generic_economy
		cost = 50
		days_re_enable = 120
		visible = {
			hidden_trigger = {
				has_country_flag = spain_campaign_initialized
				NOT = { has_country_flag = spain_bill_budget_passed }
				OR = {
					has_country_flag = spain_chapter_restoration
					has_country_flag = spain_chapter_rivera
					has_country_flag = spain_chapter_republic
				}
			}
		}
		available = {
			custom_trigger_tooltip = {
				tooltip = spain_bill_dissolved_tt
				NOT = { has_country_flag = spain_cortes_dissolved }
			}
			custom_trigger_tooltip = {
				tooltip = spain_bill_cooldown_tt
				NOT = { has_country_flag = spain_bill_budget_cooldown }
			}
			custom_trigger_tooltip = {
				tooltip = spain_bill_reach_budget_tt
				OR = {
					NOT = { has_variable = spain_bill_votes_budget }
					check_variable = { var = spain_bill_votes_budget value = spain_bill_floor compare = greater_than_or_equals }
					check_variable = { var = spain_government_seats value = spain_bill_floor compare = greater_than_or_equals }
				}
			}
		}
		complete_effect = {
			country_event = { id = spain_bills.1 }
		}
		ai_will_do = {
			factor = 60
			modifier = {
				factor = 2
				check_variable = { var = spain_bill_votes_budget value = spain_bill_needed compare = greater_than_or_equals }
			}
			modifier = {
				factor = 1.5
				has_stability < constant:spain_campaign.election_unrest_threshold
			}
			modifier = {
				factor = 0.6
				check_variable = { var = spain_coup_pressure value = constant:spain_campaign.coup_attempt_floor compare = greater_than_or_equals }
			}
		}
	}
