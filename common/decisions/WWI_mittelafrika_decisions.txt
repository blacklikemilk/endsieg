ww1_mittelafrika_category = {
	ww1_establish_mittelafrika = {
		icon = ger_reichskommissariats
		priority = 100
		fire_only_once = yes
		cost = 100
		days_remove = 180

		allowed = {
			original_tag = GER
		}

		visible = {
			has_country_flag = ww1_fate_of_colonial_empires_complete
			date < 1936.1.1
			RKA = { exists = no }
		}

		available = {
			OR = {
				owns_state = 546 # Tanganyika
				owns_state = 547 # East Africa, ceded in Fate of the Colonial Empires
			}
		}

		remove_effect = {
			custom_effect_tooltip = ww1_establish_mittelafrika_effect_tt
			hidden_effect = {
				# Create Mittelafrika from an African state before transferring the rest.
				if = {
					limit = { owns_state = 546 }
					546 = { transfer_state_to = RKA }
				}
				else = {
					547 = { transfer_state_to = RKA }
				}

				# The custom map marks Madagascar and the other African islands as Africa.
				every_owned_state = {
					limit = { is_on_continent = africa }
					transfer_state_to = RKA
				}

				puppet = RKA
				RKA = {
					inherit_technology = GER
					drop_cosmetic_tag = yes
					set_cosmetic_tag = RKA_WWI_MITTELAFRIKA
					if = {
						limit = { owns_state = 546 }
						set_capital = { state = 546 }
					}
					else = {
						set_capital = { state = 547 }
					}
					# Germany relinquishes its claims to the states assigned to Mittelafrika.
					every_owned_state = {
						GER = { remove_state_claim = PREV }
					}
				}
			}
		}

		ai_will_do = { factor = 100 }
	}
