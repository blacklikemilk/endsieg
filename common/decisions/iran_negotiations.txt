# Overview: Iran's renegotiation decisions - the way out of a concession once its terms have run.
# A demand that was accepted, compromised or signed by decree binds for three years (iran_<id>_binding,
# set by the demand events). When that flag lapses the concession can be reopened: the price is political
# power, a patron that will not forget it, and - if the Majlis is strong - a chamber that has to be carried.
# The two older arrangements, the Anglo-Persian Agreement and the Abadan deal, are reopened by date.

iran_great_powers = {

	iran_renegotiate_russian_railway = {
		priority = 20
		icon = generic_break_treaty
		allowed = { original_tag = PER }
		visible = {
			has_country_flag = iran_campaign_active
			has_country_flag = iran_demand_russian_railway_accepted
			NOT = { has_country_flag = iran_demand_russian_railway_binding }
			NOT = { has_country_flag = iran_demand_russian_railway_renegotiated }
		}
		available = {
			political_power > 99
			country_exists = SOV
			OR = {
				NOT = { has_power_balance = { id = PER_crown_and_majlis } }
				NOT = { power_balance_value = { id = PER_crown_and_majlis value > constant:iran_story.negotiation_majlis_block } }
				has_country_flag = PER_bop_crown_side
			}
		}
		cost = 100
		fire_only_once = yes
		complete_effect = {
			iran_reclaim_northern_large = yes
			add_to_variable = { var = iran_northern_standing value = constant:iran_story.negotiation_standing_abandon }
			clamp_variable = { var = iran_northern_standing min = 0 max = 100 }
			if = { limit = { has_idea = iran_northern_railway_concession } remove_ideas = iran_northern_railway_concession }
			clr_country_flag = iran_obligation_russian_railway
			clr_country_flag = iran_demand_russian_railway_accepted
			set_country_flag = iran_demand_russian_railway_renegotiated
			add_war_support = constant:iran_story.war_support_gain
			if = {
				limit = { has_country_flag = PER_bop_majlis_side }
				PER_bop_push_majlis_small = yes
			}
			else = { PER_bop_push_crown_small = yes }
			iran_refresh_mechanics_hidden = yes
		}
		ai_will_do = {
			factor = 2
			modifier = { factor = 3 has_country_flag = iran_nationalist_front }
			modifier = { factor = 2 has_country_flag = PER_bop_majlis_side }
			modifier = { factor = 0.5 has_country_flag = PER_bop_crown_side }
			modifier = { factor = 0.5 is_historical_focus_on = yes }
		}
	}
