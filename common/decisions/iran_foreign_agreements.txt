# Overview: Great Game 2.0 agreements and the sovereignty programme.
#
# Every agreement buys something real and costs freedom of action: influence on
# one patron, leverage overall, and sovereignty. The four bottom decisions are
# the long way back - they need authority, a reformed customs service and time,
# and they are the only way to buy independence down.
iran_foreign_agreements = {

	# ------------------------------------------------------------------ loans
	iran_accept_british_loan = {
		icon = generic_fundraising
		allowed = { original_tag = PER }
		visible = {
			has_country_flag = iran_campaign_active
			date < 1936.1.1
			country_exists = ENG
		}
		available = {
			has_political_power > 99
			NOT = { has_country_flag = iran_agreement_british_loan }
					OR = {
				has_country_flag = PER_bop_crown_side
				power_balance_value = { id = PER_crown_and_majlis value > -0.10 }
			}
}
		cost = 100
		fire_only_once = yes
		complete_effect = {
			if = {
				limit = { has_country_flag = PER_bop_crown_side }
				add_stability = -0.05
				subtract_from_variable = { iran_sovereignty = 2 }
			}

			set_country_flag = iran_agreement_british_loan
			add_to_variable = { iran_british_influence = constant:iran_story.influence_per_agreement }
			add_to_variable = { iran_foreign_leverage = 10 }
			subtract_from_variable = { iran_sovereignty = constant:iran_story.sovereignty_agreement_cost }
			add_timed_idea = { idea = iran_british_financial_mission days = 1825 }
			add_stability = constant:iran_story.stability_cost
			PER_bop_sign_agreement = yes
			iran_refresh_mechanics_hidden = yes
		}
		ai_will_do = { factor = 2 }
	}

	# --------------------------------------------------------------- railways
	iran_accept_northern_railway = {
		icon = generic_construction
		allowed = { original_tag = PER }
		visible = {
			has_country_flag = iran_campaign_active
			date < 1936.1.1
			OR = { country_exists = RIE country_exists = SOV }
		}
		available = {
			has_political_power > 99
			NOT = { has_country_flag = iran_agreement_northern_railway }
					OR = {
				has_country_flag = PER_bop_crown_side
				power_balance_value = { id = PER_crown_and_majlis value > -0.10 }
			}
}
		cost = 100
		fire_only_once = yes
		complete_effect = {
			if = {
				limit = { has_country_flag = PER_bop_crown_side }
				add_stability = -0.05
				subtract_from_variable = { iran_sovereignty = 2 }
			}

			set_country_flag = iran_agreement_northern_railway
			add_to_variable = { iran_northern_influence = constant:iran_story.influence_per_agreement }
			add_to_variable = { iran_foreign_leverage = 10 }
			subtract_from_variable = { iran_sovereignty = constant:iran_story.sovereignty_agreement_cost }
			417 = { add_building_construction = { type = railway level = 2 instant_build = yes } }
			add_timed_idea = { idea = iran_northern_railway_concession days = 1825 }
			add_stability = constant:iran_story.stability_cost
			PER_bop_sign_agreement = yes
			iran_refresh_mechanics_hidden = yes
		}
		ai_will_do = { factor = 2 }
	}

	# ---------------------------------------------------------------- missions
	iran_accept_military_mission = {
		icon = generic_military
		allowed = { original_tag = PER }
		visible = {
			has_country_flag = iran_campaign_active
			NOT = { has_country_flag = iran_agreement_military_mission }
		}
		available = {
			has_political_power > 74
			OR = {
				check_variable = { var = iran_british_influence value = 40 compare = greater_than_or_equals }
				check_variable = { var = iran_northern_influence value = 40 compare = greater_than_or_equals }
			}
					OR = {
				has_country_flag = PER_bop_crown_side
				power_balance_value = { id = PER_crown_and_majlis value > -0.10 }
			}
}
		cost = 75
		fire_only_once = yes
		complete_effect = {
			if = {
				limit = { has_country_flag = PER_bop_crown_side }
				add_stability = -0.05
				subtract_from_variable = { iran_sovereignty = 2 }
			}

			set_country_flag = iran_agreement_military_mission
			add_army_experience = 25
			add_timed_idea = { idea = iran_foreign_military_mission days = 1460 }
			subtract_from_variable = { iran_sovereignty = constant:iran_story.sovereignty_agreement_cost }
			add_to_variable = { iran_foreign_leverage = 10 }
			# the patron with the ear of the court sends the mission
			if = {
				limit = {
					check_variable = { var = iran_british_influence value = iran_northern_influence compare = greater_than_or_equals }
				}
				add_to_variable = { iran_british_influence = 10 }
			}
			else = {
				add_to_variable = { iran_northern_influence = 10 }
			}
			PER_bop_sign_agreement = yes
			iran_refresh_mechanics_hidden = yes
		}
		ai_will_do = { factor = 3 }
	}

			}

			set_country_flag = iran_agreement_foreign_customs
			add_to_variable = { iran_foreign_leverage = 15 }

		icon = generic_political_discourse

		}

			has_political_power > 149

	# ---------------------------------------------------------- renegotiation
	iran_renegotiate_concessions = {
		icon = generic_break_treaty
		allowed = { original_tag = PER }
		visible = {
			has_country_flag = iran_campaign_active
			OR = {
				has_country_flag = iran_agreement_british_loan
				has_country_flag = iran_agreement_northern_railway
			}
		}
		available = {
			has_political_power > 199
			check_variable = { var = iran_sovereignty value = constant:iran_story.sovereignty_renegotiate_threshold compare = greater_than_or_equals }
		}
		cost = 200
		days_re_enable = 730
		complete_effect = {
			if = {
				limit = { has_country_flag = iran_agreement_british_loan }
				clr_country_flag = iran_agreement_british_loan
				remove_ideas = iran_british_financial_mission
				subtract_from_variable = { iran_british_influence = constant:iran_story.influence_per_agreement }
			}
			if = {
				limit = { has_country_flag = iran_agreement_northern_railway }
				clr_country_flag = iran_agreement_northern_railway
				remove_ideas = iran_northern_railway_concession
				subtract_from_variable = { iran_northern_influence = constant:iran_story.influence_per_agreement }
			}

			subtract_from_variable = { iran_foreign_leverage = 10 }

		}
