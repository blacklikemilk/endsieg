# Iran 1936-1945 - the political question.
#
# These decisions exist for the fallback 1936 chapter (used when the Graveyard of
# Empires DLC is absent, or when it is present but neither its Persian tree nor
# iran_ww2 is the loaded one). With the DLC tree active Iran follows that tree's
# own political paths, so these are hidden there to avoid two competing
# political systems.
#
# They live inside the iran_political_question category defined in
# common/decisions/categories/iran_political_question.txt: a decisions file maps
# one top-level block to one category, so the previous layout - five decision
# bodies exposed at top level - had no category to attach to and would not load.

iran_political_question = {

	iran_political_question_qajar = {
		icon = GFX_decision_generic_speech
		allowed = { original_tag = PER }
		visible = {
			has_focus_tree = iran_ww2
			NOT = { has_country_flag = iran_qajar_restoration }
			date > 1935.12.31
		}
		available = {
			political_power > 100
			check_variable = { var = iran_authority value = 40 compare = less_than }
		}
		cost = 100
		fire_only_once = yes
		complete_effect = { iran_shift_to_qajar = yes }
