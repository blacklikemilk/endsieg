WW1_alt_take_danish_islands = {
	if = { limit = { 101 = { is_owned_by = DEN is_controlled_by = DEN } } transfer_state = 101 }
	if = {
		limit = {
			100 = { OR = { is_owned_by = DEN AND = { is_owned_by = ICE ICE = { is_subject_of = DEN } } } controller = { OR = { tag = DEN is_subject_of = DEN } } }
		}
		transfer_state = 100
	}
}

WW1_alt_danish_refusal = {
	set_country_flag = WW1_alt_copenhagen_refused
	clr_country_flag = WW1_alt_copenhagen_pending
	if = { limit = { 101 = { is_owned_by = DEN } } add_state_claim = 101 }
	if = { limit = { 100 = { OR = { is_owned_by = DEN owner = { is_subject_of = DEN } } } } add_state_claim = 100 }
	create_wargoal = { type = puppet_wargoal_focus target = DEN }
	add_named_threat = { threat = 4 name = WW1_alt_northern_threat }
	country_event = { id = ww1_alt_wars.64 hours = 1 }
}
