# Read reserves once when each new set of orders is issued.
WW1_western_plan_prepare_phase = {
	set_variable = { WW1_western_plan_phase_days = WW1_western_plan_phase_base_days }

	set_variable = { WW1_western_plan_rifle_coverage = 1 }
	if = {
		limit = { check_variable = { num_target_equipment_in_armies@infantry_equipment > 0 } }
		set_variable = { WW1_western_plan_rifle_coverage = num_equipment@infantry_equipment }
		clamp_variable = { var = WW1_western_plan_rifle_coverage min = 0 }
		set_temp_variable = { WW1_western_plan_rifle_need = num_target_equipment_in_armies@infantry_equipment }
		multiply_temp_variable = { WW1_western_plan_rifle_need = 0.1 }
		clamp_temp_variable = { var = WW1_western_plan_rifle_need min = 1 }
		divide_variable = { WW1_western_plan_rifle_coverage = WW1_western_plan_rifle_need }
		clamp_variable = { var = WW1_western_plan_rifle_coverage min = 0 max = 1 }
	}

	set_variable = { WW1_western_plan_artillery_coverage = 1 }
	if = {
		limit = { check_variable = { num_target_equipment_in_armies@artillery_equipment > 0 } }
		set_variable = { WW1_western_plan_artillery_coverage = num_equipment@artillery_equipment }
		clamp_variable = { var = WW1_western_plan_artillery_coverage min = 0 }
		set_temp_variable = { WW1_western_plan_artillery_need = num_target_equipment_in_armies@artillery_equipment }
		multiply_temp_variable = { WW1_western_plan_artillery_need = 0.1 }
		clamp_temp_variable = { var = WW1_western_plan_artillery_need min = 1 }
		divide_variable = { WW1_western_plan_artillery_coverage = WW1_western_plan_artillery_need }
		clamp_variable = { var = WW1_western_plan_artillery_coverage min = 0 max = 1 }
	}

	set_variable = { WW1_western_plan_ration_coverage = 1 }
	if = {
		limit = { NOT = { has_country_flag = endsieg_rations_disabled } }
		endsieg_rations_measure_demand = yes
		if = {
			limit = { check_variable = { endsieg_ration_daily_use > 0 } }
			set_variable = { WW1_western_plan_ration_coverage = num_equipment@rations_1 }
			clamp_variable = { var = WW1_western_plan_ration_coverage min = 0 }
			set_temp_variable = { WW1_western_plan_ration_need = endsieg_ration_daily_use }
			multiply_temp_variable = { WW1_western_plan_ration_need = WW1_western_plan_phase_base_days }
			divide_variable = { WW1_western_plan_ration_coverage = WW1_western_plan_ration_need }
			clamp_variable = { var = WW1_western_plan_ration_coverage min = 0 max = 1 }
		}
	}

	set_variable = { WW1_western_plan_reserve_coverage = WW1_western_plan_rifle_coverage }
	if = {
		limit = { check_variable = { WW1_western_plan_artillery_coverage < WW1_western_plan_reserve_coverage } }
		set_variable = { WW1_western_plan_reserve_coverage = WW1_western_plan_artillery_coverage }
	}
	if = {
		limit = { check_variable = { WW1_western_plan_ration_coverage < WW1_western_plan_reserve_coverage } }
		set_variable = { WW1_western_plan_reserve_coverage = WW1_western_plan_ration_coverage }
	}
	set_variable = { WW1_western_plan_timetable_factor = WW1_western_plan_reserve_coverage }
	multiply_variable = { WW1_western_plan_timetable_factor = 0.5 }
	add_to_variable = { WW1_western_plan_timetable_factor = 0.75 }
	multiply_variable = { WW1_western_plan_phase_days = WW1_western_plan_timetable_factor }
	round_variable = WW1_western_plan_phase_days
	clamp_variable = { var = WW1_western_plan_phase_days min = 1 }
}

WW1_western_plan_initialize_christmas_deadline = {
	if = {
		limit = { NOT = { has_variable = WW1_western_plan_bonus_deadline_year } }
		set_variable = { WW1_western_plan_bonus_deadline_year = global.year }
		if = {
			limit = { has_global_flag = { flag = WW1_world_tension_outbreak_recorded days > 180 } }
			# Recover the missing timetable without awarding a new opening offensive years later.
			set_variable = { WW1_western_plan_bonus_deadline_year = 0 }
		}
	}
}

WW1_western_plan_begin = {
	WW1_western_frontline_check_transition = yes
	if = {
		limit = {
			WW1_western_plan_can_continue = yes
			NOT = { has_country_flag = WW1_western_plan_started }
			NOT = { has_country_flag = WW1_western_frontline_active }
			NOT = { WW1_western_plan_controls_paris = yes }
		}
		WW1_western_plan_open_selected_offensive = yes
	}
}

WW1_western_plan_open_selected_offensive = {
	if = {
		limit = {
			has_completed_focus = GE1_schlieffen_plan
			has_global_flag = schlieffen_plan
			OR = {
				has_war_with = BEL
				has_global_flag = WW1_belgian_neutrality_violated
				WW1_western_plan_controls_brussels = yes
			}
		}
		set_country_flag = WW1_western_plan_started
		set_country_flag = WW1_western_plan_in_progress
		WW1_western_plan_start_schlieffen_meuse = yes
		country_event = { id = ww1_western_plans.1 }
	}
	else_if = {
		limit = {
			has_completed_focus = GE1_rupprecht_plan
			has_global_flag = rupprecht_plan
			OR = { has_war_with = SWI WW1_western_plan_controls_bern = yes }
		}
		set_country_flag = WW1_western_plan_started
		set_country_flag = WW1_western_plan_in_progress
		WW1_western_plan_start_rupprecht_switzerland = yes
		country_event = { id = ww1_western_plans.2 }
	}
	else_if = {
		limit = { has_completed_focus = GE1_kronprinz_plan has_global_flag = kronprinz_plan }
		set_country_flag = WW1_western_plan_started
		set_country_flag = WW1_western_plan_in_progress
		if = {
			limit = { OR = { NOT = { OR = { has_war_with = RIE has_war_with = SOV } } WW1_western_plan_controls_warsaw = yes WW1_western_plan_controls_lodz = yes } }
			WW1_western_plan_start_kronprinz_lorraine = yes
		}
		else = { WW1_western_plan_start_kronprinz_warsaw = yes }
		country_event = { id = ww1_western_plans.3 }
	}
}

	clr_country_flag = WW1_western_plan_in_progress

		limit = { WW1_western_plan_can_continue = yes }

}

# Close the timetable when the final peace settlement ends the Western Front. This also removes missions and temporary bonuses
# from saves that still had an active timetable when the treaty was signed.
WW1_western_plan_close_after_peace = {
	set_country_flag = WW1_western_plan_closed
	WW1_western_frontline_clear_timetable = yes
	clr_country_flag = WW1_western_plan_in_progress
	clr_country_flag = WW1_western_frontline_active
	clr_country_flag = WW1_western_frontline_review_done
	clr_country_flag = WW1_western_frontline_objective_held
	clr_country_flag = WW1_WF_secondary_objective
	clear_variable = WW1_western_frontline_days
	clear_variable = WW1_western_frontline_sector
	clear_variable = WW1_western_plan_phase_days
	clear_variable = WW1_western_plan_phase_base_days
	clear_variable = WW1_western_plan_bonus_deadline_year
	clear_variable = WW1_western_plan_rifle_coverage
	clear_variable = WW1_western_plan_artillery_coverage
	clear_variable = WW1_western_plan_ration_coverage
	clear_variable = WW1_western_plan_reserve_coverage
	clear_variable = WW1_western_plan_timetable_factor
	remove_ideas = {
		WW1_WF_replacement_drafts_idea WW1_WF_assault_training_idea
		WW1_WF_siegfried_positions_idea WW1_WF_artillery_preparation_idea
	}
	every_state = {
		limit = { has_dynamic_modifier = { modifier = WW1_western_frontline_supply scope = GER } }
		remove_dynamic_modifier = { modifier = WW1_western_frontline_supply scope = GER }
	}
}
