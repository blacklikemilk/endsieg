# Called after the eastern settlement, once the successor states own land.
# The 1919 bookmark has dated OOBs; only campaigns begun before 1919 raise
# an additional reserve brigade. Country flags prevent repeated stockpiles.
INT_eastern_raise_reserve_brigade = {
	division_template = {
		name = "Postwar Volunteer Brigade"
		regiments = {
			infantry = { x = 0 y = 0 }
			infantry = { x = 0 y = 1 }
			infantry = { x = 1 y = 0 }
			infantry = { x = 1 y = 1 }
		}
	}
	capital_scope = {
		create_unit = {
			division = "division_template = \"Postwar Volunteer Brigade\" start_experience_factor = 0.2 start_equipment_factor = 0.8"
			owner = PREV
			count = 1
			allow_spawning_on_enemy_provs = no
		}
	}
}
