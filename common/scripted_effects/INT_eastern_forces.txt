# Four infantry battalions per volunteer brigade.
INT_eastern_prepare_brigade_template = {
	if = {
		limit = { NOT = { has_template = "Postwar Volunteer Brigade" } }
		division_template = {
			name = "Postwar Volunteer Brigade"
			regiments = {
				infantry = { x = 0 y = 0 }
				infantry = { x = 0 y = 1 }
				infantry = { x = 1 y = 0 }
				infantry = { x = 1 y = 1 }
			}
		}
	}
}
