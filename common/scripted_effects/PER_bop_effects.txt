PER_bop_push_crown = {

		id = PER_crown_and_majlis
		value = -0.20
	}
}

PER_bop_push_crown_small = {
	add_power_balance_value = {
		id = PER_crown_and_majlis
		value = -0.10
	}
}

	add_power_balance_value = {
		id = PER_crown_and_majlis
		value = 0.20
	}
}

PER_bop_push_majlis_small = {
	add_power_balance_value = {
		id = PER_crown_and_majlis
		value = 0.10
	}
}

# Pulls whichever institution is ahead back towards the constitutional centre.
PER_bop_pull_to_centre = {
	if = {
		limit = { power_balance_value = { id = PER_crown_and_majlis value > 0 } }
		add_power_balance_value = { id = PER_crown_and_majlis value = -0.25 }
	}
	else = {
		add_power_balance_value = { id = PER_crown_and_majlis value = 0.25 }
	}
}
