INT_FRA_fund_public_works = {
	# A new appropriation renews the project for two years.
	if = { limit = { has_idea = INT_FRA_public_works } remove_ideas = INT_FRA_public_works }
	add_timed_idea = { idea = INT_FRA_public_works days = 730 }
	set_country_flag = INT_FRA_public_works_funded
}
