# Overview: negotiated government bills in the Spanish Cortes.
# A bill no longer passes on a political-power spend alone. Each bill has a supporting bloc of party
# ids, spain_bill_count_votes turns that bloc into spain_bill_votes out of the 400 seats, and the
# negotiation in events/spain_cortes_bills.txt decides whether the bill passes cleanly, passes after a
# concession to a swing party, passes watered down, or is withdrawn.
#
# Chain: seats (spain_cortes_effects.txt) -> spain_bill_count_votes -> spain_bill_votes_<bill>
#        -> decision availability (common/decisions/spain_cortes.txt)
#        -> spain_bills.<n> negotiation -> outcome effect -> spain_cortes_recount.
#
# Seat totals are cached per bill because a trigger cannot add variables, so the decision buttons read
# the cache and the hidden clerk event spain_bills.90 refreshes it roughly every month.
# Documented in docs/spain_cortes_bills.md.

# days = does not accept a script constant, so the cooldown length lives here as a file constant.
# Keep it equal to the cooldown documented in docs/spain_cortes_bills.md.
@bill_cooldown_days = 60

# =====================================================================================
# Counting the chamber
# =====================================================================================

# Common start of every count. spain_bill_needed is a clean majority of the chamber;
# spain_bill_floor is that majority less constant:spain_campaign.bill_negotiation_reach, the seats a
# government may still be short of the bar and still bring the bill to the floor.
spain_bill_prepare_count = {
	set_variable = { var = spain_bill_votes value = 0 }
	set_variable = { var = spain_bill_swing_votes value = 0 }
	set_variable = { var = spain_bill_needed value = constant:spain_campaign.cortes_majority }
	set_variable = { var = spain_bill_floor value = spain_bill_needed }
	subtract_from_variable = { var = spain_bill_floor value = constant:spain_campaign.bill_negotiation_reach }
}

# The government's own benches always vote for a bill the government brings forward.
# spain_government_seats already holds the ruling party and its coalition partners, so every
# supporter helper below only adds a party that sits outside those three: that is what keeps
# spain_bill_votes free of double counting.
spain_bill_add_government = {
	add_to_variable = { var = spain_bill_votes value = spain_government_seats }
}

spain_bill_add_monarchists = {

# spain_bill_current selects the bill for spain_bill_count_votes, so a single call site can count
# whichever bill the negotiation event is about.
spain_bill_count_votes = {
	spain_bill_prepare_count = yes
	if = { limit = { check_variable = { var = spain_bill_current value = 1 } } spain_bill_count_budget = yes }
	else_if = { limit = { check_variable = { var = spain_bill_current value = 2 } } spain_bill_count_agrarian = yes }
	else_if = { limit = { check_variable = { var = spain_bill_current value = 3 } } spain_bill_count_labour = yes }
	else_if = { limit = { check_variable = { var = spain_bill_current value = 4 } } spain_bill_count_army = yes }
	else_if = { limit = { check_variable = { var = spain_bill_current value = 5 } } spain_bill_count_statute = yes }
	else_if = { limit = { check_variable = { var = spain_bill_current value = 6 } } spain_bill_count_electoral = yes }
	else = {
		clamp_variable = { var = spain_bill_votes min = 0 max = constant:spain_campaign.cortes_total }
	}
}

# Budget: the Treasury bill is carried by the government, the Liberals and the Conservatives.
# The Liberals are also the benches whose votes can be bought, so spain_bill_swing_votes records
# the total the bill would reach with them.
spain_bill_count_budget = {
	spain_bill_prepare_count = yes
	spain_bill_add_government = yes
	spain_bill_add_liberals = yes
	spain_bill_add_conservatives = yes
	clamp_variable = { var = spain_bill_votes min = 0 max = constant:spain_campaign.cortes_total }
	set_variable = { var = spain_bill_votes_budget value = spain_bill_votes }
	set_variable = { var = spain_bill_swing_votes value = spain_bill_votes }
	if = {
		limit = {
			check_variable = { var = spain_government_party value = 2 compare = not_equals }
			check_variable = { var = spain_coalition_1 value = 2 compare = not_equals }
			check_variable = { var = spain_coalition_2 value = 2 compare = not_equals }
		}
		add_to_variable = { var = spain_bill_swing_votes value = spain_cortes_seats_liberals }
	}
}

# One hidden clerk event chain keeps the cached counts current. It is started by the first
# negotiation and stops when the chamber is dissolved.
spain_bills_start_ticker = {
	if = {
		limit = {
			NOT = { has_country_flag = spain_bills_ticker }
			has_country_flag = spain_campaign_initialized
		}
		set_country_flag = spain_bills_ticker
		country_event = { id = spain_bills.90 days = 1 }
	}
}

# A bill that passes gives its sponsor the credit.
spain_bill_credit_government = {
	if = { limit = { check_variable = { var = spain_government_party value = 1 } } add_to_variable = { var = spain_party_shift_monarchists value = constant:spain_campaign.bill_concession_gain } }
	if = { limit = { check_variable = { var = spain_government_party value = 2 } } add_to_variable = { var = spain_party_shift_liberals value = constant:spain_campaign.bill_concession_gain } }
	if = { limit = { check_variable = { var = spain_government_party value = 3 } } add_to_variable = { var = spain_party_shift_conservatives value = constant:spain_campaign.bill_concession_gain } }
	if = { limit = { check_variable = { var = spain_government_party value = 4 } } add_to_variable = { var = spain_party_shift_republicans value = constant:spain_campaign.bill_concession_gain } }

# A bill that fails or is withdrawn costs the government that brought it: the chamber loses
# confidence, the army reads the failure, and the governing party loses support.
spain_bill_penalise_government = {
	add_stability = constant:spain_campaign.stability_cost
	spain_government_strain = yes
	if = { limit = { check_variable = { var = spain_government_party value = 1 } } add_to_variable = { var = spain_party_shift_monarchists value = constant:spain_campaign.bill_defeat_shift } }
	if = { limit = { check_variable = { var = spain_government_party value = 2 } } add_to_variable = { var = spain_party_shift_liberals value = constant:spain_campaign.bill_defeat_shift } }
	if = { limit = { check_variable = { var = spain_government_party value = 3 } } add_to_variable = { var = spain_party_shift_conservatives value = constant:spain_campaign.bill_defeat_shift } }
	if = { limit = { check_variable = { var = spain_government_party value = 4 } } add_to_variable = { var = spain_party_shift_republicans value = constant:spain_campaign.bill_defeat_shift } }
	if = { limit = { check_variable = { var = spain_government_party value = 5 } } add_to_variable = { var = spain_party_shift_socialists value = constant:spain_campaign.bill_defeat_shift } }
	if = { limit = { check_variable = { var = spain_government_party value = 6 } } add_to_variable = { var = spain_party_shift_carlists value = constant:spain_campaign.bill_defeat_shift } }

	spain_cortes_recount = yes
}

spain_bill_pass_statute = {
	add_to_variable = { var = spain_catalan_autonomy value = constant:spain_campaign.autonomy_gain }
	add_to_variable = { var = spain_basque_autonomy value = constant:spain_campaign.autonomy_gain }
	add_stability = constant:spain_campaign.stability_cost
	spain_refresh_campaign_ideas = yes
	set_country_flag = spain_bill_statute_passed
	spain_bill_credit_government = yes
	spain_cortes_recount = yes
}

# The army's price: the Captain-General is reassured that the statute costs him nothing, and the
# officer corps is told the regions are a political matter. The loyalty lost is permanent.
spain_bill_concede_statute = {
	add_to_variable = { var = spain_military_loyalty value = constant:spain_campaign.loyalty_loss }
	add_to_variable = { var = spain_catalan_autonomy value = constant:spain_campaign.autonomy_gain }
	add_to_variable = { var = spain_basque_autonomy value = constant:spain_campaign.autonomy_gain }
	add_stability = constant:spain_campaign.stability_cost
	spain_refresh_campaign_ideas = yes
	set_country_flag = spain_bill_statute_passed
	spain_cortes_recount = yes
}

spain_bill_water_statute = {
	add_to_variable = { var = spain_catalan_autonomy value = constant:spain_campaign.bill_autonomy_half }
	add_to_variable = { var = spain_basque_autonomy value = constant:spain_campaign.bill_autonomy_half }

spain_bill_fail_statute = {

	spain_bill_cooldown_statute = yes
}

# =====================================================================================
# Electoral law
# =====================================================================================

# A new franchise and a new register. The chamber that votes it dissolves into the election it has
# just made possible, so the bill's real cost is the vote that follows.
spain_bill_pass_electoral = {
	spain_push_cortes_large = yes
	add_stability = constant:spain_campaign.stability_small
	set_country_flag = spain_electoral_reform
	spain_bill_credit_government = yes
	spain_cortes_recount = yes
	spain_hold_election = yes
}

# The register is reformed but the chambers keep their veto, and the government that spent its
# credit on the reform gets nothing back from it.
spain_bill_water_electoral = {
	add_stability = constant:spain_campaign.stability_cost
	set_country_flag = spain_electoral_reform
	spain_cortes_recount = yes
	spain_hold_election = yes
}

spain_bill_fail_electoral = {
	spain_bill_penalise_government = yes
	spain_bill_cooldown_electoral = yes
}
