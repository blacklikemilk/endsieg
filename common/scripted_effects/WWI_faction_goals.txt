WW1_FG_register_central = {
 if = {
  limit = { is_faction_leader = yes WW1_FG_central = yes NOT = { has_country_flag = WW1_FG_central_registered } }
  # A leader change must not overwrite choices made in an upgraded faction.
  if = {
   limit = { NOT = { OR = { has_faction_goal = faction_goal_ww1_bulgarian_alliance has_faction_goal = faction_goal_ww1_constantinople_corridor has_faction_goal = faction_goal_ww1_eastern_peace has_faction_goal = faction_goal_ww1_polish_regency has_faction_goal = faction_goal_ww1_western_settlement } } }
   set_faction_manifest = faction_manifest_ww1_central_historical
   if = { limit = { has_faction_goal = faction_goal_an_infantry_army NOT = { has_completed_faction_goal = faction_goal_an_infantry_army } } remove_faction_goal = faction_goal_an_infantry_army }
   if = { limit = { has_faction_goal = faction_goal_convoy_escort_force NOT = { has_completed_faction_goal = faction_goal_convoy_escort_force } } remove_faction_goal = faction_goal_convoy_escort_force }

   if = { limit = { NOT = { has_faction_goal = faction_goal_ww1_bulgarian_alliance } } add_faction_goal = faction_goal_ww1_bulgarian_alliance }
   if = { limit = { NOT = { has_faction_goal = faction_goal_ww1_constantinople_corridor } } add_faction_goal = faction_goal_ww1_constantinople_corridor }
   if = { limit = { NOT = { has_faction_goal = faction_goal_ww1_eastern_peace } } add_faction_goal = faction_goal_ww1_eastern_peace }
