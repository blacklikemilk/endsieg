# Country scope. Campaign start dates distinguish past bookmark history from
# new focus choices made while playing an earlier campaign.
WW1_initialize_bookmark_event_history = {
    if = {
        limit = {
            has_start_date < 1936.1.1
            date < 1936.1.1
            NOT = { has_country_flag = WW1_bookmark_event_history_initialized }
        }
        if = {
            limit = { tag = GER }
            if = {
                limit = { NOT = { has_start_date < 1914.8.4 } has_completed_focus = GE1_schlieffen_plan }
                set_country_flag = WW1_focus_action_ww1_germany_21_seen
                clr_country_flag = WW1_focus_action_ww1_germany_21_queued
            }
            if = {
                limit = { NOT = { has_start_date < 1915.2.18 } has_completed_focus = GE1_unrestricted_sub }
                set_country_flag = WW1_focus_action_ww1_germany_39_seen
            }
            if = {
                limit = { NOT = { has_start_date < 1917.1.31 } has_completed_focus = GE1_continue_unrestricted_sub }
                set_country_flag = WW1_focus_action_ww1_germany_57_seen
            }
            if = {
                limit = { NOT = { has_start_date < 1917.3.1 } has_completed_focus = GE1_zimmerman }
                set_country_flag = WW1_focus_action_ww1_germany_13_seen
            }
            if = {
                limit = { NOT = { has_start_date < 1915.10.14 } is_in_faction_with = BUL }
                set_country_flag = WW1_focus_action_ww1_germany_279_seen
                set_country_flag = GE1_RP_279_resolved
            }
            # The pre-war diplomatic exchange is already part of later starts.
            # Unfinished construction remains available through the railway chain.
            if = {
                limit = { NOT = { has_start_date < 1914.1.1 } has_completed_focus = GE1_berlin_baghdad_1 }
                set_country_flag = GE1_baghdad_railway_announced
                set_country_flag = GE1_baghdad_railway_british_response_received
                set_country_flag = GE1_baghdad_railway_british_response_notified
                clr_country_flag = GE1_baghdad_railway_british_response_pending
                clr_country_flag = GE1_baghdad_railway_british_reply_pending
            }
        }
        if = {
            limit = { tag = ENG NOT = { has_start_date < 1914.8.4 } has_completed_focus = EN1_join_war }
            set_country_flag = WW1_ENG_war_entry_presented
            set_country_flag = WW1_ENG_war_entry_resolved
            set_country_flag = WW1_ENG_war_entry_accepted
            clr_country_flag = WW1_ENG_war_entry_pending
        }
        if = {
            limit = { tag = JAP NOT = { has_start_date < 1914.8.23 } }
            set_country_flag = WW1_focus_action_ww1_japan_1_seen
            set_country_flag = JP1_R_ultimatum_sent
            clr_country_flag = WW1_focus_action_ww1_japan_1_queued
        }
        if = {
            limit = { tag = TUR NOT = { has_start_date < 1914.10.29 } is_in_faction_with = GER }
            set_country_flag = WW1_focus_action_ww1_turkey_2_seen
            set_country_flag = WW1_ottoman_black_sea_raid_accepted
            set_country_flag = OT1_R_german_agreement
            set_country_flag = OT1_R_central_war_authorized
            clr_country_flag = WW1_focus_action_ww1_turkey_2_queued
        }
        if = {
            limit = { tag = ITA NOT = { has_start_date < 1915.5.23 } is_in_faction_with = FRA }
            set_country_flag = WW1_focus_action_ww1_italy_9_seen
            set_country_flag = IT1_ww1_entente_offer_accepted
            clr_country_flag = WW1_focus_action_ww1_italy_9_queued
        }
        set_country_flag = WW1_bookmark_event_history_initialized
    }

    # War relations are installed after some startup callbacks. Seed the
    # existing blockade once those relations are available, also for old saves.
    if = {
        limit = {
            tag = ENG
            has_focus_tree = british_focus_ww1
            NOT = { has_start_date < 1914.8.4 }
            has_start_date < 1918.11.11
            date < 1936.1.1
            has_war_with = GER
            NOT = { is_in_faction_with = GER }
            NOT = { has_global_flag = compiegne }
            NOT = { has_global_flag = central_powers_victory }
            NOT = { has_country_flag = WW1_bookmark_blockade_initialized }
        }
        set_country_flag = WW1_bookmark_blockade_initialized
        set_global_flag = WW1_blockade_authorized
        if = { limit = { NOT = { has_completed_focus = EN1_north_sea_blockade } } unlock_national_focus = EN1_north_sea_blockade }
        GER = { WW1_update_north_sea_blockade = yes }
    }
}
