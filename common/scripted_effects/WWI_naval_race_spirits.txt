# Runs in Germany's scope for the two participants.
WW1_update_naval_arms_race_spirits = {
	if = {
		limit = { country_exists = ENG has_war_with = ENG }
		set_global_flag = WW1_naval_rivalry_war_seen
	}
	if = {
		limit = { WW1_naval_arms_race_spirits_active = yes }
		if = {
			limit = { NOT = { has_idea = WW1_GER_naval_arms_race } }
			add_ideas = WW1_GER_naval_arms_race
		}
		ENG = {
			if = {
				limit = { NOT = { has_idea = WW1_ENG_naval_arms_race } }
				add_ideas = WW1_ENG_naval_arms_race
			}
		}
	}
	else = {
		if = {
			limit = { has_idea = WW1_GER_naval_arms_race }
			remove_ideas = WW1_GER_naval_arms_race
		}
		if = {
			limit = { country_exists = ENG ENG = { has_idea = WW1_ENG_naval_arms_race } }
			ENG = { remove_ideas = WW1_ENG_naval_arms_race }
		}
	}
}
