# The two input slots use the save format: 1 is blank, 2..102 are real
# readings. Convert them to a one-based slope frame and a screen position.
INT_GER_measure_mark_segment = {
	if = {
		limit = { check_variable = { mark_older > 1 } check_variable = { mark_newer > 1 } }
		set_temp_variable = { mark_slope = mark_newer }
		subtract_from_temp_variable = { mark_slope = mark_older }
		add_to_temp_variable = { mark_slope = 102 }
		set_temp_variable = { mark_y = mark_newer }
		if = {
			limit = { check_variable = { mark_older > mark_newer } }
			set_temp_variable = { mark_y = mark_older }
		}
		# Zero is y=287. The line begins six pixels inside the texture.
		multiply_temp_variable = { mark_y = -2 }
		add_to_temp_variable = { mark_y = 285 }
	}
	else = {
		set_temp_variable = { mark_slope = 1 }
		set_temp_variable = { mark_y = 0 }
	}
}
