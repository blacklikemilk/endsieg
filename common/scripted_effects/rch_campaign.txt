# Overview: Republic of China chapter routing, 1912-1936.
# Country scope throughout. The Republic is never given the Qing tree and never keeps the generic one.
# rch_route_chapter is the only authority that loads a focus tree: one date band decides
# one chapter per day, so no day can load two opposing trees or pay a chapter reward twice.
rch_refresh_mechanics = {
	clamp_variable = { var = rch_legitimacy min = constant:rch_story.legitimacy_floor max = constant:rch_story.legitimacy_cap }
	clamp_variable = { var = rch_military_influence min = constant:rch_story.legitimacy_floor max = constant:rch_story.legitimacy_cap }
	clamp_variable = { var = rch_nationalist_support min = constant:rch_story.legitimacy_floor max = constant:rch_story.legitimacy_cap }

	remove_ideas = {
		rch_beiyang_republic rch_parliamentary_republic rch_nationalist_government rch_provincial_union
		rch_warlord_armies rch_new_army rch_party_army
		rch_empty_treasury rch_foreign_tutelage rch_japanese_pressure
		rch_succession_constitutional rch_succession_contested rch_succession_garrison rch_succession_dissolved
	}
	if = {
		limit = { rch_path_kmt = yes }
		add_ideas = rch_nationalist_government
	}
	else_if = {
		limit = { rch_path_beiyang = yes }
		add_ideas = rch_beiyang_republic
	}
	else_if = {
		limit = { rch_path_provincial = yes }
		add_ideas = rch_provincial_union
	}
	else_if = {
		limit = { rch_path_parliamentary = yes }
		add_ideas = rch_parliamentary_republic
	}
	else = {
		add_ideas = rch_beiyang_republic
	}

	if = {
		limit = { rch_path_kmt = yes }
		add_ideas = rch_party_army
	}
	else_if = {
		limit = { rch_military_dominant = yes }
		add_ideas = rch_warlord_armies
	}
	else = {
		add_ideas = rch_new_army
	}

	if = {
		limit = { has_country_flag = rch_japan_war }
		add_ideas = rch_japanese_pressure
	}
	else_if = {
		limit = { rch_legitimacy_weak = yes }
		add_ideas = rch_empty_treasury
	}

	# The seat spirit is restored from the frozen line flag, so a routine refresh of the
	# other families cannot erase the succession result, and the display is republished
	# immediately after any settlement that moved the dial.
	rch_succession_seat_spirit = yes
	rch_beiyang_refresh = yes
}

rch_initialize_campaign = {
	if = {
		limit = {
			rch_campaign_country = yes
			any_owned_state = { always = yes }
			NOT = { has_country_flag = rch_campaign_active }
		}
		set_country_flag = rch_campaign_active
		set_variable = { rch_legitimacy = constant:rch_story.legitimacy_opening }
		set_variable = { rch_military_influence = constant:rch_story.military_opening }
		set_variable = { rch_nationalist_support = constant:rch_story.nationalist_opening }
		set_variable = { rch_governor_count = 0 }
		set_variable = { rch_reform_settled = 0 }
		set_variable = { rch_bop_band = -1 }
		if = {
			limit = { date > 1927.12.31 }
			set_variable = { rch_legitimacy = constant:rch_story.legitimacy_nanjing }
			set_variable = { rch_military_influence = constant:rch_story.military_nanjing }
			set_variable = { rch_nationalist_support = constant:rch_story.nationalist_nanjing }
			set_country_flag = rch_route_kmt
			set_country_flag = rch_northern_expedition
		}
		else_if = {
			limit = { date > 1915.12.31 }
			set_country_flag = rch_route_beiyang
			set_country_flag = rch_yuan_legacy
		}
		if = {
			limit = { date < 1919.1.1 }
			rch_open_chapter_one = yes
		}
		else_if = {
			limit = { date < 1928.1.1 }
			rch_open_chapter_two = yes
		}
		else = {
			rch_open_chapter_three = yes
		}
		rch_refresh_mechanics = yes
		rch_succession_endow = yes
		rch_beiyang_bootstrap = yes
		if = {
			limit = { NOT = { has_country_flag = rch_beiyang_intro_seen } }
			set_country_flag = rch_beiyang_intro_seen
			country_event = { id = rch_campaign.2 days = 5 }
		}
	}
}

# Each opener leaves exactly one chapter flag set, so a chapter can never be
# recorded twice and a skipped chapter cannot survive into the next band.
rch_open_chapter_one = {
	set_country_flag = rch_chapter_one
	clr_country_flag = rch_chapter_two
	clr_country_flag = rch_chapter_three
	load_focus_tree = { tree = rch_republic keep_completed = yes }
	mark_focus_tree_layout_dirty = yes
	country_event = { id = rch_campaign.1 days = 2 }
}

# The one date-to-chapter decision. A band whose chapter flag is missing performs the
# transition through the matching opener, which sets the flag, loads the tree and fires
# the chapter event once. A band whose flag is already set repairs only a replaced tree
# and any stale sibling flag, so no day pays a second chapter reward.
rch_route_chapter = {
	# The 1936 settlement runs exactly once, before any band decides on a tree.
	if = {
		limit = { date > 1935.12.31 NOT = { has_country_flag = rch_1936_settled } }
		rch_settle_1936 = yes
	}

	if = {
		limit = { date < 1919.1.1 }
		if = {
			limit = { NOT = { has_country_flag = rch_chapter_one } }
			rch_open_chapter_one = yes
		}
		else = {
			clr_country_flag = rch_chapter_two
			clr_country_flag = rch_chapter_three
			if = {
				limit = { NOT = { has_focus_tree = rch_republic } }
				load_focus_tree = { tree = rch_republic keep_completed = yes }
				mark_focus_tree_layout_dirty = yes
			}
		}
	}
	else_if = {
		limit = { date < 1928.1.1 }
		if = {
			limit = { NOT = { has_country_flag = rch_chapter_two } }
			rch_open_chapter_two = yes
		}
		else = {
			clr_country_flag = rch_chapter_one
			clr_country_flag = rch_chapter_three
			if = {
				limit = { NOT = { has_focus_tree = rch_warlord } }
				load_focus_tree = { tree = rch_warlord keep_completed = yes }
				mark_focus_tree_layout_dirty = yes
			}
		}
	}
	else_if = {
		limit = { date < 1936.1.1 }
		if = {
			limit = { NOT = { has_country_flag = rch_chapter_three } }
			rch_open_chapter_three = yes
		}
		else = {
			clr_country_flag = rch_chapter_one
			clr_country_flag = rch_chapter_two
			if = {
				limit = { NOT = { has_focus_tree = rch_nanjing } }
				load_focus_tree = { tree = rch_nanjing keep_completed = yes }
				mark_focus_tree_layout_dirty = yes
			}
		}
	}
	else = {
		# The handover records the Nanjing chapter it came through and drops any stale
		# sibling flag an older save may still carry, without a reward or a tree load.
		if = {
			limit = { NOT = { has_country_flag = rch_chapter_three } }
			set_country_flag = rch_chapter_three
		}
		clr_country_flag = rch_chapter_one
		clr_country_flag = rch_chapter_two
		# The saved Republic keeps playing as RCH and takes the Chinese tree the
		# installed DLC provides.
		if = {
			limit = { has_country_flag = rch_republic_saved }
			rch_load_handover_tree = yes
		}
		# The ordinary settlement has moved the player to CHI, which still carries the
		# generic WWI tree from the 1910 setup, so CHI takes the same DLC tree once.
		else_if = {
			limit = { country_exists = CHI CHI = { NOT = { has_country_flag = rch_1936_handover_tree } } }
			CHI = {
				rch_load_handover_tree = yes
				set_country_flag = rch_1936_handover_tree
			}
		}
	}
}

# Loads the 1936 Chinese tree that the installed DLC provides, and only when the
# receiving country does not already hold it. Call it in the country that should get
# the tree; the caller scopes ROOT for the saved Republic and CHI for the handover.
rch_load_handover_tree = {
	if = {
		limit = {
			has_dlc = "No Compromise, No Surrender"
			NOT = { has_focus_tree = china_nationalist_sea_focus }
		}
		load_focus_tree = { tree = china_nationalist_sea_focus keep_completed = yes }
		mark_focus_tree_layout_dirty = yes
	}
	else_if = {
		limit = {
			NOT = { has_dlc = "No Compromise, No Surrender" }
			has_dlc = "Waking the Tiger"
			NOT = { has_focus_tree = china_nationalist_focus }
		}
		load_focus_tree = { tree = china_nationalist_focus keep_completed = yes }
		mark_focus_tree_layout_dirty = yes
	}
	else_if = {
		limit = {
			NOT = { has_dlc = "No Compromise, No Surrender" }
			NOT = { has_dlc = "Waking the Tiger" }
			NOT = { has_focus_tree = generic_focus }
		}
		load_focus_tree = { tree = generic_focus keep_completed = yes }
		mark_focus_tree_layout_dirty = yes
	}
}

rch_campaign_tick = {
	if = {
		limit = { rch_campaign_active = yes }
		rch_route_chapter = yes

		if = {
			limit = { date > 1913.7.1 NOT = { has_country_flag = rch_second_revolution_seen } }
			set_country_flag = rch_second_revolution_seen
			country_event = { id = rch_campaign.3 days = 1 }
		}
		if = {
			limit = { date > 1915.1.18 NOT = { has_country_flag = rch_21_demands_seen } }
			set_country_flag = rch_21_demands_seen
			country_event = { id = rch_campaign.4 days = 1 }
		}
		# The settlement is taken once on 12 December 1915, before the throne is answered,
		# so the answer cannot move the reform count.
		if = {
			limit = { date > 1915.12.11 }
			rch_settlement_snapshot = yes
		}
		if = {
			limit = {
				date > 1915.12.11
				date < 1916.1.1
				NOT = { has_country_flag = rch_throne_choice_seen }
			}
			set_country_flag = rch_throne_choice_seen
			country_event = { id = rch_campaign.42 days = 1 }
		}
		# The tick owns the succession on the first day after 5 June 1916: it resolves, then
		# queues the popup. A deferred, closed or lost popup cannot change the outcome.
		if = {
			limit = { date > 1916.6.5 NOT = { has_country_flag = rch_succession_resolved } }
			rch_succession_resolve = yes
		}
		if = {
			limit = { date > 1917.7.1 NOT = { has_country_flag = rch_restoration_seen } }
			set_country_flag = rch_restoration_seen
			country_event = { id = rch_campaign.6 days = 1 }
		}
		if = {
			limit = { date > 1918.11.11 NOT = { has_country_flag = rch_armistice_seen } }
			set_country_flag = rch_armistice_seen
			country_event = { id = rch_campaign.7 days = 1 }
		}
		if = {
			limit = { date > 1919.5.4 NOT = { has_country_flag = rch_may_fourth_seen } }
			set_country_flag = rch_may_fourth_seen
			country_event = { id = rch_campaign.8 days = 1 }
		}
		if = {
			limit = { date > 1920.7.14 NOT = { has_country_flag = rch_zhili_anhui_seen } }
			set_country_flag = rch_zhili_anhui_seen
			country_event = { id = rch_campaign.21 days = 1 }
		}
		if = {
			limit = { date > 1922.4.28 NOT = { has_country_flag = rch_zhili_fengtian_seen } }
			set_country_flag = rch_zhili_fengtian_seen
			country_event = { id = rch_campaign.22 days = 1 }
		}
		if = {
			limit = { date > 1924.1.20 NOT = { has_country_flag = rch_kmt_reorganisation_seen } }
			set_country_flag = rch_kmt_reorganisation_seen
			country_event = { id = rch_campaign.23 days = 1 }
		}
		if = {
			limit = { date > 1926.7.9 NOT = { has_country_flag = rch_expedition_seen } }
			set_country_flag = rch_expedition_seen
			country_event = { id = rch_campaign.24 days = 1 }
		}
		if = {
			limit = { date > 1928.6.8 NOT = { has_country_flag = rch_nanjing_seen } }
			set_country_flag = rch_nanjing_seen
			country_event = { id = rch_campaign.31 days = 1 }
		}
		if = {
			limit = { date > 1931.9.18 NOT = { has_country_flag = rch_manchuria_seen } }
			set_country_flag = rch_manchuria_seen
			country_event = { id = rch_campaign.33 days = 1 }
		}
		# The 1936 handover notice pays out once. Its previous test asked for the Waking
		# the Tiger tree alone, which the DLC branch no longer guarantees, so use a flag.
		if = {
			limit = { date > 1935.12.31 NOT = { has_country_flag = rch_1936_handover_seen } }
			set_country_flag = rch_1936_handover_seen
			country_event = { id = rch_campaign.35 days = 1 }
		}
		# The display is republished daily from the dial, so every existing writer of
		# military influence is mirrored within a day without touching those writers.
		rch_beiyang_refresh = yes
	}
}

# The 1915 settlement, the 1916 succession and the Beiyang pressure cards.
# Country scope throughout. Values live in constant:rch_story.*; the literals
# below exist only where an engine field rejects a constant, and each one is
# asserted against its constant by .tools/tests/test_rch_succession.py.

# One snapshot, taken on 12 December 1915 and never retaken. A reform counts only
# when its flag stood longer than the maturity window at that date, so nothing
# bought afterwards can enlarge the count.
rch_settlement_snapshot = {
	if = {
		limit = { NOT = { has_country_flag = rch_settlement_taken } }
		set_country_flag = rch_settlement_taken
		set_variable = { rch_reform_settled = 0 }
		if = {
			limit = { has_country_flag = { flag = rch_gate_constitution days > 180 } }
			add_to_variable = { rch_reform_settled = 1 }
		}
		if = {
			limit = { has_country_flag = { flag = rch_gate_officer_rotation days > 180 } }
			add_to_variable = { rch_reform_settled = 1 }
		}
		if = {
			limit = { has_country_flag = { flag = rch_gate_civil_governors days > 180 } }
			add_to_variable = { rch_reform_settled = 1 }
		}
		if = {
			limit = { has_country_flag = { flag = rch_gate_treasury days > 180 } }
			add_to_variable = { rch_reform_settled = 1 }
		}
	}
}

# The score as a pure function of the three dials, the frozen settlement count, the
# throne answer, the governor counter and a Chinese war. Assigned, never accumulated.
rch_succession_refresh = {
	set_variable = { var = rch_succession_base value = var:rch_legitimacy }
	add_to_variable = { var = rch_succession_base value = constant:rch_story.legitimacy_cap }
	subtract_from_variable = { var = rch_succession_base value = var:rch_military_influence }
	set_variable = { var = rch_succession_half_nat value = var:rch_nationalist_support }
	divide_variable = { var = rch_succession_half_nat value = constant:rch_story.succession_half_divisor }
	round_variable = rch_succession_half_nat
	add_to_variable = { var = rch_succession_base value = var:rch_succession_half_nat }
	divide_variable = { var = rch_succession_base value = constant:rch_story.succession_divisor }
	round_variable = rch_succession_base

	set_variable = { var = rch_succession_gate_term value = var:rch_reform_settled }
	multiply_variable = { var = rch_succession_gate_term value = constant:rch_story.succession_gate_weight }
	add_to_variable = { var = rch_succession_base value = var:rch_succession_gate_term }
	if = {
		limit = { rch_gate_throne_refused = yes }
		add_to_variable = { var = rch_succession_base value = constant:rch_story.succession_gate_weight }
	}

	set_variable = { var = rch_succession_governor_term value = var:rch_governor_count }
	multiply_variable = { var = rch_succession_governor_term value = constant:rch_story.succession_governor_penalty }
	subtract_from_variable = { var = rch_succession_base value = var:rch_succession_governor_term }
	if = {
		limit = { rch_chinese_war = yes }
		subtract_from_variable = { var = rch_succession_base value = constant:rch_story.succession_war_penalty }
	}

	set_variable = { var = rch_succession_score value = var:rch_succession_base }
	clamp_variable = { var = rch_succession_score min = constant:rch_story.succession_score_floor max = constant:rch_story.succession_score_cap }
}

# One exclusive precedence: collapse first, then the two earned tiers, historical as
# the floor. The tier is a variable, not a chain of flags, so nothing reads it twice.
rch_succession_evaluate = {
	if = {
		limit = { rch_succession_collapse = yes }
		set_variable = { rch_succession_tier = 3 }
	}
	else_if = {
		limit = { rch_succession_tier_0 = yes }
		set_variable = { rch_succession_tier = 0 }
	}
	else_if = {
		limit = { rch_succession_tier_1 = yes }
		set_variable = { rch_succession_tier = 1 }
	}
	else = {
		set_variable = { rch_succession_tier = 2 }
	}
}

# The one seat spirit for the frozen tier. Removing the family before adding one keeps
# exactly one seat spirit even when a refresh runs after the crisis.
rch_succession_seat_spirit = {
	remove_ideas = {
		rch_succession_constitutional rch_succession_contested rch_succession_garrison rch_succession_dissolved
	}
	if = {
		limit = { has_country_flag = rch_line_unified }
		add_ideas = rch_succession_constitutional
	}
	else_if = {
		limit = { has_country_flag = rch_line_contested }
		add_ideas = rch_succession_contested
	}
	else_if = {
		limit = { has_country_flag = rch_line_garrison }
		add_ideas = rch_succession_garrison
	}
	else_if = {
		limit = { has_country_flag = rch_line_dissolved }
		add_ideas = rch_succession_dissolved
	}
}

# The succession itself, without a popup. The tick owns this on 6 June 1916 and the
# initialisation of a post-crisis start uses it silently, so no popup can change it.
rch_succession_settle = {
	if = {
		limit = { NOT = { has_country_flag = rch_succession_resolved } }
		set_country_flag = rch_succession_resolved
		set_country_flag = rch_yuan_death_seen
		set_country_flag = rch_yuan_legacy
		rch_succession_refresh = yes
		rch_succession_evaluate = yes
		set_variable = { rch_succession_final_score = var:rch_succession_score }
		set_variable = { rch_succession_final_tier = var:rch_succession_tier }
		clr_country_flag = rch_hongxian
		if = {
			limit = { check_variable = { var = rch_succession_final_tier value = 0 compare = equals } }
			set_country_flag = rch_line_unified
		}
		else_if = {
			limit = { check_variable = { var = rch_succession_final_tier value = 1 compare = equals } }
			set_country_flag = rch_line_contested
		}
		else_if = {
			limit = { check_variable = { var = rch_succession_final_tier value = 2 compare = equals } }
			set_country_flag = rch_line_garrison
		}
		else = {
			set_country_flag = rch_line_dissolved
		}
		rch_succession_seat_spirit = yes
		rch_refresh_mechanics = yes
	}
}

# Settle, then announce. The popup only presents the frozen tier.
rch_succession_resolve = {
	rch_succession_settle = yes
	if = {
		limit = { NOT = { has_country_flag = rch_succession_popup_queued } }
		set_country_flag = rch_succession_popup_queued
		country_event = { id = rch_campaign.5 days = 1 }
	}
}

# What a start inherits from the years it did not play. A Republic opened at or after
# 1 January 1916 has already lived through the proclamation, so it receives the
# Hongxian state and a sealed zero snapshot rather than retroactive maturity.
rch_succession_endow = {
	if = {
		limit = { date > 1915.12.31 }
		set_country_flag = rch_hongxian
		set_country_flag = rch_throne_choice_seen
		if = {
			limit = { NOT = { has_country_flag = rch_settlement_taken } }
			set_country_flag = rch_settlement_taken
			set_variable = { rch_reform_settled = 0 }
		}
	}
	if = {
		limit = { date > 1916.6.5 }
		rch_succession_settle = yes
	}
}

# The panel is created once. Everything below is DLC-guarded, and the eight cards and
# the succession run identically without Arms Against Tyranny.
rch_beiyang_bootstrap = {
	if = {
		limit = {
			has_dlc = "Arms Against Tyranny"
			NOT = { has_power_balance = { id = RCH_beiyang_order } }
		}
		set_power_balance = { id = RCH_beiyang_order set_value = 0.0 }
		set_variable = { rch_bop_band = -1 }
	}
}

# The display mirrors military influence and nothing else, read from the dial itself, so
# every existing focus, decision and event writer is reflected within a day. A band
# change removes the published offset and applies the new one exactly once; the panel is
# never re-initialised from a daily call.
rch_beiyang_refresh = {
	if = {
		limit = { has_dlc = "Arms Against Tyranny" }
		rch_beiyang_bootstrap = yes
		if = {
			limit = { has_power_balance = { id = RCH_beiyang_order } }
			if = {
				limit = { check_variable = { var = rch_military_influence value = constant:rch_story.succession_bop_high compare = greater_than } }
				set_variable = { rch_bop_band_new = 4 }
			}
			else_if = {
				limit = { check_variable = { var = rch_military_influence value = constant:rch_story.succession_bop_upper compare = greater_than } }
				set_variable = { rch_bop_band_new = 3 }
			}
			else_if = {
				limit = { check_variable = { var = rch_military_influence value = constant:rch_story.succession_bop_mid compare = greater_than } }
				set_variable = { rch_bop_band_new = 2 }
			}
			else_if = {
				limit = { check_variable = { var = rch_military_influence value = constant:rch_story.succession_bop_low compare = greater_than } }
				set_variable = { rch_bop_band_new = 1 }
			}
			else = {
				set_variable = { rch_bop_band_new = 0 }
			}
			if = {
				limit = { NOT = { check_variable = { var = rch_bop_band_new value = var:rch_bop_band compare = equals } } }
				# Retire the band that is published, in the five literals the dial maps to.
				if = {
					limit = { check_variable = { var = rch_bop_band value = 4 compare = equals } }
					add_power_balance_value = { id = RCH_beiyang_order value = -0.85 }
				}
				else_if = {
					limit = { check_variable = { var = rch_bop_band value = 3 compare = equals } }
					add_power_balance_value = { id = RCH_beiyang_order value = -0.25 }
				}
				else_if = {
					limit = { check_variable = { var = rch_bop_band value = 1 compare = equals } }
					add_power_balance_value = { id = RCH_beiyang_order value = 0.25 }
				}
				else_if = {
					limit = { check_variable = { var = rch_bop_band value = 0 compare = equals } }
					add_power_balance_value = { id = RCH_beiyang_order value = 0.85 }
				}
				# Publish the new band.
				if = {
					limit = { check_variable = { var = rch_bop_band_new value = 4 compare = equals } }
					add_power_balance_value = { id = RCH_beiyang_order value = 0.85 }
				}
				else_if = {
					limit = { check_variable = { var = rch_bop_band_new value = 3 compare = equals } }
					add_power_balance_value = { id = RCH_beiyang_order value = 0.25 }
				}
				else_if = {
					limit = { check_variable = { var = rch_bop_band_new value = 1 compare = equals } }
					add_power_balance_value = { id = RCH_beiyang_order value = -0.25 }
				}
				else_if = {
					limit = { check_variable = { var = rch_bop_band_new value = 0 compare = equals } }
					add_power_balance_value = { id = RCH_beiyang_order value = -0.85 }
				}
				set_variable = { rch_bop_band = var:rch_bop_band_new }
			}
		}
	}
}
