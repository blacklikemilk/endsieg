WW1_replace_modern_trains = {
	if = {
		limit = { has_tech = basic_train }
		set_technology = { basic_train = 0 early_basic_train = 1 popup = no }
	}
	if = {
		limit = { has_tech = wartime_train }
		set_technology = { wartime_train = 0 early_wartime_train = 1 popup = no }
	}
	if = {
		limit = { has_tech = armored_train }
		set_technology = { armored_train = 0 early_armored_train = 1 popup = no }
	}

	if = {
		limit = { has_equipment = { train_equipment > 0 } }
		set_temp_variable = { ww1_trains_to_replace = num_equipment@train_equipment_1 }
		if = {
			limit = { check_variable = { ww1_trains_to_replace > 0 } }
			set_temp_variable = { ww1_trains_to_remove = ww1_trains_to_replace }
			multiply_temp_variable = { ww1_trains_to_remove = -1 }
			add_equipment_to_stockpile = { type = train_equipment_1 amount = var:ww1_trains_to_remove }
			add_equipment_to_stockpile = { type = early_train_equipment_1 amount = var:ww1_trains_to_replace }
		}

		set_temp_variable = { ww1_trains_to_replace = num_equipment@train_equipment_2 }
		if = {
			limit = { check_variable = { ww1_trains_to_replace > 0 } }
			set_temp_variable = { ww1_trains_to_remove = ww1_trains_to_replace }
			multiply_temp_variable = { ww1_trains_to_remove = -1 }
			add_equipment_to_stockpile = { type = train_equipment_2 amount = var:ww1_trains_to_remove }
			add_equipment_to_stockpile = { type = early_train_equipment_2 amount = var:ww1_trains_to_replace }
		}

		set_temp_variable = { ww1_trains_to_replace = num_equipment@train_equipment_3 }
		if = {
			limit = { check_variable = { ww1_trains_to_replace > 0 } }
			set_temp_variable = { ww1_trains_to_remove = ww1_trains_to_replace }
			multiply_temp_variable = { ww1_trains_to_remove = -1 }
			add_equipment_to_stockpile = { type = train_equipment_3 amount = var:ww1_trains_to_remove }
			add_equipment_to_stockpile = { type = early_train_equipment_3 amount = var:ww1_trains_to_replace }
		}
	}
}
