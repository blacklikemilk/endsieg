# A completed focus may await its promised date or a valid host. It never
# assigns an unfinished focus or undoes an incompatible completed commitment.
WW1_leave_balkan_league_for_focus = {
 if = {
  limit = { WW1_is_balkan_league_member = yes }
  if = { limit = { is_faction_leader = yes } random_other_country = { limit = { is_in_faction_with = PREV NOT = { tag = PREV } is_subject = no } set_faction_leader = yes } }
  if = { limit = { is_faction_leader = yes } dismantle_faction = yes }
  else = { leave_faction = yes }
 }
}
WW1_complete_focus_faction_entry = {
 if = {
  limit = { has_start_date < 1936.1.1 date < 1936.1.1 has_focus_tree = generic_focus_ww1 is_subject = no }
  if = { limit = { has_country_flag = WW1_focus_faction_central_pending has_completed_focus = shared_GEN_join_cp is_in_faction_with = GER } clr_country_flag = WW1_focus_faction_central_pending }
  if = { limit = { has_country_flag = WW1_focus_faction_entente_pending has_completed_focus = shared_GEN_join_entente is_in_faction_with = FRA } clr_country_flag = WW1_focus_faction_entente_pending }

  if = {
   limit = { has_country_flag = WW1_focus_faction_central_pending has_completed_focus = shared_GEN_join_cp WW1_generic_central_entry_date = yes NOT = { tag = BUL }
    OR = { is_in_faction = no WW1_is_balkan_league_member = yes } NOT = { has_war_with = GER }
    GER = { exists = yes is_faction_leader = yes GE1_ww1_may_choose_central_bloc = yes }
    GER = { ROOT = { WW1_country_faction_path_allows_previous = yes } }
   }
   WW1_leave_balkan_league_for_focus = yes
   GER = { add_to_faction = ROOT }
   if = { limit = { is_in_faction_with = GER } clr_country_flag = WW1_focus_faction_central_pending }
  }
  if = {
   limit = { has_country_flag = WW1_focus_faction_entente_pending has_completed_focus = shared_GEN_join_entente WW1_generic_entente_entry_date = yes
    OR = { is_in_faction = no WW1_is_balkan_league_member = yes } NOT = { has_war_with = FRA }
    FRA = { exists = yes is_faction_leader = yes FR3_R_may_enter_entente = yes }
    FRA = { ROOT = { WW1_country_faction_path_allows_previous = yes } }
   }
   WW1_leave_balkan_league_for_focus = yes
   FRA = { add_to_faction = ROOT }
   if = { limit = { is_in_faction_with = FRA } clr_country_flag = WW1_focus_faction_entente_pending }
  }
 }
}
