endsieg_ww1_anti_huerta_coalition = {
	if = {
		limit = { CNT = { has_war_with = MEX }  CNT = { has_completed_focus = shared_GEN_revolutionary_cooperation } }
		CNT = {
			if = { limit = { is_in_faction = no } if = { limit = { has_start_date < 1936.1.1 date < 1936.1.1 } create_faction_from_template = { template = faction_template_ww1_anti_huerta name = ANTI_HUERTA_FORCES } }
else = { create_faction = ANTI_HUERTA_FORCES } }
			if = {
				limit = { is_faction_leader = yes }
				if = { limit = { ZAP = { has_war_with = MEX is_in_faction = no } } add_to_faction = ZAP }
				if = { limit = { VIL = { has_war_with = MEX is_in_faction = no } } add_to_faction = VIL }
			}
		}
	}
}
endsieg_ww1_start_maderista_revolution = {
	if = {
		limit = { tag = MEX NOT = { has_global_flag = maderista_revolution } NOT = { has_global_flag = maderista_victory } }
		set_global_flag = maderista_revolution
		news_event = { hours = 6 id = endsieg_news.202 }
		USA = { diplomatic_relation = { country = MEX relation = guarantee active = no } }
		MDS = {
			transfer_state = 482
			set_cosmetic_tag = MDS
			set_capital = { state = 482 }
			set_politics = { ruling_party = democratic elections_allowed = no }
			promote_character = { character = MDS_francisco_madero ideology = conservatism }
			declare_war_on = { target = MEX type = civil_war }
			load_oob = Maderistas_1911
			add_equipment_to_stockpile = { type = early_infantry_equipment_2 amount = 15000 producer = USA }
		}
		country_event = { id = ww1_mexico.8 days = 45 }
	}
}

endsieg_ww1_absorb_maderistas = {
	if = {
		limit = { country_exists = MDS  }
		MDS = {
			if = { limit = { is_faction_leader = yes } dismantle_faction = yes }
			else = { leave_faction = yes }
		}
		# Native reunification order: preserve the human player before removing their tag.
		if = { limit = { is_ai = yes MDS = { is_ai = no } } change_tag_from = MDS }
		annex_country = { target = MDS transfer_troops = yes }
	}
	if = {
		limit = { country_exists = MAD MAD = { has_war_with = MEX owns_state = 482 } }
		MAD = {
			if = { limit = { is_faction_leader = yes } dismantle_faction = yes }
			else = { leave_faction = yes }
		}
		# Native reunification order: preserve the human player before removing their tag.
		if = { limit = { is_ai = yes MAD = { is_ai = no } } change_tag_from = MAD }
		annex_country = { target = MAD transfer_troops = yes }
	}
}
