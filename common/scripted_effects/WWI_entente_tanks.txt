  limit = { WW1_entente_tank_programme_active = yes NOT = { has_country_flag = WW1_entente_tank_research_issued } }

   limit = { tag = ENG }

    limit = { has_dlc = "No Step Back" }

     limit = { NOT = { has_tech = early_tank } }

     limit = { NOT = { has_tech = eg_early_tank } }

     limit = { NOT = { has_tech = early_tank } }

  }

      light_armor = { x = 0 y = 0 }
      infantry = { x = 1 y = 0 }
     }
    }

      progress = 0
      efficiency = 0.10
     }

     add_equipment_production = {

      requested_factories = 6

    if = {
     limit = { has_dlc = "No Step Back" }

      equipment = { type = early_tank_chassis_0 creator = FRA version_name = "Char d'assaut experimental" }

      equipment = { type = early_tank_equipment_0 creator = FRA }
