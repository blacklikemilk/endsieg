# Only seed experience in the kinds of wings actually deployed in this bookmark.
endsieg_initialize_ww1_air_doctrine = {
	set_grand_doctrine = endsieg_great_war_air
	if = {
		limit = {
			OR = {
				has_deployed_air_force_size = { size > 0 type = fighter }
				has_deployed_air_force_size = { size > 0 type = interceptor }
				has_deployed_air_force_size = { size > 0 type = scout_plane }
			}
		}
		if = {
			limit = {
				OR = {
					has_tech = flying_circus
					AND = { tag = GER has_start_date > 1915.12.31 }
				}
			}
			set_sub_doctrine = endsieg_ww1_pursuit
		}
		else = { set_sub_doctrine = endsieg_ww1_observers }
		if = {
			limit = { endsieg_ww1_bookmark_service_veteran = yes }
			if = {
				limit = { endsieg_ww1_bookmark_long_service = yes }
				add_mastery = { amount = 300 track = endsieg_ww1_patrols }
			}
			else = { add_mastery = { amount = 125 track = endsieg_ww1_patrols } }
		}
	}
	if = {
		limit = {
			endsieg_ww1_bookmark_service_veteran = yes
			OR = {
				has_deployed_air_force_size = { size > 0 type = cas }
				has_deployed_air_force_size = { size > 0 type = tactical_bomber }
				has_deployed_air_force_size = { size > 0 type = strategic_bomber }
			}
		}
		if = {
			limit = {
				OR = {
					has_tech = str_concept
					AND = { tag = ENG has_start_date > 1918.3.20 }
				}
			}
			set_sub_doctrine = endsieg_ww1_independent_bombing
		}
		else = { set_sub_doctrine = endsieg_ww1_army_support }
		if = {
			limit = { endsieg_ww1_bookmark_long_service = yes }
			add_mastery = { amount = 225 track = endsieg_ww1_bombing }
		}
		else = { add_mastery = { amount = 100 track = endsieg_ww1_bombing } }
	}
}
