# Overview: Iran's foreign acts for the Great Game, 1910-1945. Country scope, Iran only.
# One effect per patron, direction and tier, shared by focuses, decisions and events, so every foreign deal
# moves the patron's influence, sovereignty and the foreign leverage that depends on them in one place.
# concede: the patron gains influence, Iran loses sovereignty and takes on leverage (dependence).
# reclaim: the inverse. Patrons: british, northern (Russian Empire, then Soviet), german.
# The *_leading_* variants act on whichever patron currently holds the most influence (ties resolve british,
# then northern); domestic reforms use them. None of them writes dependence: each republishes it from the
# influence it just moved, so one concession can no longer be booked twice.
# Every patron has its own explicit effect because the engine's parameter substitution inside variable names is unproven.
# Tooltips are the *_tt localisation keys; the arithmetic is hidden. Values live in constant:iran_story.*.

iran_concede_british_small = {
	custom_effect_tooltip = iran_concede_british_small_tt
	hidden_effect = {
		add_to_variable = { iran_british_influence = constant:iran_story.concede_small_influence }
		clamp_variable = { var = iran_british_influence min = 0 max = 100 }
		subtract_from_variable = { iran_sovereignty = constant:iran_story.concede_small_sovereignty }
		clamp_variable = { var = iran_sovereignty min = 0 max = constant:iran_story.sovereignty_cap }
		iran_recompute_leverage = yes
	}
}

iran_concede_leading_small = {
	custom_effect_tooltip = iran_concede_leading_small_tt
	hidden_effect = {
		if = {
			limit = {
				check_variable = { var = iran_british_influence value = iran_northern_influence compare = greater_than_or_equals }
				check_variable = { var = iran_british_influence value = iran_german_influence compare = greater_than_or_equals }
			}
			iran_concede_british_small = yes
		}
		else_if = {
			limit = { check_variable = { var = iran_northern_influence value = iran_german_influence compare = greater_than_or_equals } }
			iran_concede_northern_small = yes
		}
		else = { iran_concede_german_small = yes }
	}
}
