# Undo only the two British holdings taken by the obsolete 1925 release event.
# The event marker distinguishes this from the original WWI Senussi state.
INT_ENG_undo_postwar_senussi_release = {
	if = {
		limit = {
			tag = ENG
			date > 1924.12.31
			date < 1936.1.1
			has_global_flag = senussi_uprising_2
		}
		if = { limit = { 975 = { is_owned_by = SES } } transfer_state = 975 }
		if = { limit = { 1059 = { is_owned_by = SES } } transfer_state = 1059 }
		clr_global_flag = senussi_uprising_2
	}
}
