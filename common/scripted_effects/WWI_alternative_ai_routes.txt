# Daily observation only resolves a British wait; it never assigns a fresh random route.
WW1_reconcile_alternative_ai_routes = {
 if = {
  limit = { has_start_date < 1936.1.1 date < 1936.1.1 NOT = { is_historical_focus_on = yes } }
  ENG = {
   if = {
    limit = { is_ai = yes has_country_flag = WW1_AI_ENG_berlin NOT = { has_global_flag = GE1_anglo_german_alliance }
     OR = {
      date > 1914.6.30
      has_global_flag = GE1_D_anglo_german_refused
      has_country_flag = GE1_D_anglo_german_refused
      has_completed_focus = EN1_back_france
      has_completed_focus = EN1_stay_neutral
      has_country_flag = EN1_R_entente_policy
      has_country_flag = EN1_R_neutral_policy
      GER = { NOT = { GE1_ww1_may_choose_british_bloc = yes } }
     }
    }
    clr_country_flag = WW1_AI_ENG_berlin
    if = { limit = { NOT = { has_completed_focus = EN1_stay_neutral } NOT = { has_country_flag = EN1_R_neutral_policy } NOT = { has_completed_focus = EN1_R_ratify_berlin } } set_country_flag = WW1_AI_ENG_entente }
   }
  }
   RIE = {
   if = {
    limit = { is_ai = yes has_country_flag = WW1_AI_RIE_german NOT = { has_country_flag = RUS_ww1_german_agreement }
     OR = { date > 1914.6.30 has_global_flag = GE1_RP_russian_offer_refused GER = { has_country_flag = GE1_RP_278_resolved } GER = { NOT = { GE1_ww1_may_choose_russian_bloc = yes } } }
    }
    clr_country_flag = WW1_AI_RIE_german
    if = { limit = { NOT = { has_completed_focus = RUS_ww1_armed_neutrality } NOT = { has_country_flag = RUS_ww1_armed_neutrality } } set_country_flag = WW1_AI_RIE_responsible }
   }
  }
}
