# Historical English name for the city and its separate urban state.
# Do not repeatedly reapply names: formable countries may choose their own.
endsieg_update_constantinople_name = {
	if = {
		limit = {
			date < 1930.3.28
			NOT = { has_global_flag = endsieg_constantinople_scope_repaired }
			797 = { is_owned_by = TUR }
			TUR = { NOT = { has_country_flag = TUR_ott_sultanate_flag } }
		}
		set_province_name = { id = 9833 name = ENDSIEG_CONSTANTINOPLE }
		797 = { set_state_name = ENDSIEG_CONSTANTINOPLE }
		set_global_flag = endsieg_constantinople_name_initialized
		set_global_flag = endsieg_constantinople_scope_repaired
		set_global_flag = endsieg_constantinople_historical_name_active
	}
	if = {
		limit = {
			date > 1930.3.27
			has_global_flag = endsieg_constantinople_historical_name_active
			797 = { is_owned_by = TUR is_controlled_by = TUR }
			TUR = {
				OR = {
					has_country_flag = INT_TUR_republic_proclaimed
					has_completed_focus = INT_TUR_republic
				}
				NOT = { has_country_flag = TUR_ott_sultanate_flag }
				NOT = { has_completed_focus = INT_TUR_ottoman_victory }
			}
		}
		# Native defaults supply Istanbul in each supported game language.
		reset_province_name = 9833
		797 = { reset_state_name = yes }
		clr_global_flag = endsieg_constantinople_historical_name_active
		set_global_flag = endsieg_istanbul_name_adopted
	}
}
