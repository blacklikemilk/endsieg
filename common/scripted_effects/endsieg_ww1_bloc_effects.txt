# Detach the starting Central partners only after the accepted alternative alliance exists.
GE1_ww1_detach_central_partners = {
 if = {
  limit = { tag = GER has_start_date < 1936.1.1 date < 1936.1.1 OR = { has_war = no has_country_flag = GE1_R_AUH_wartime_truce_accepted } is_faction_leader = yes  GE1_ww1_alternative_alliance_formed = yes }
  every_country = {
   limit = { is_in_faction_with = GER OR = { tag = AUH tag = TUR is_subject_of = AUH is_subject_of = TUR } }
   leave_faction = yes
  }
 }
}
GE1_ww1_form_russian_bloc = {
 if = {
  limit = {
   tag = GER has_start_date < 1936.1.1 date < 1936.1.1
   GE1_ww1_may_choose_russian_bloc = yes
   has_completed_focus = GE1_R_renewed_reinsurance
   has_global_flag = GE1_RP_russian_neutrality_accepted
   has_war = no is_subject = no
   OR = { is_in_faction = no is_faction_leader = yes }
   RIE = { exists = yes has_war = no is_subject = no has_country_flag = RUS_ww1_german_agreement has_country_flag = RUS_ww1_bloc_consent NOT = { has_completed_focus = RUS_entente_diplomacy } NOT = { has_capitulated = yes } }
  }
  set_country_flag = GE1_ww1_bloc_russian
  # Central partners remain until Russia is actually admitted.
  every_country = { limit = { is_in_faction_with = GER OR = { tag = ENG tag = FRA is_subject_of = ENG is_subject_of = FRA } } leave_faction = yes }
  if = { limit = { is_in_faction = no } set_rule = { can_create_factions = yes } if = { limit = { has_start_date < 1936.1.1 date < 1936.1.1 } create_faction_from_template = { template = faction_template_ww1_german_russian name = GE1_ww1_russian_faction } }
else = { create_faction = GE1_ww1_russian_faction } }
  if = { limit = { RIE = { is_in_faction = yes NOT = { is_in_faction_with = GER } is_faction_leader = yes } }
   RIE = { random_other_country = { limit = { is_in_faction_with = RIE NOT = { tag = RIE } is_subject = no } set_faction_leader = yes } }
  }
  if = { limit = { RIE = { is_in_faction = yes NOT = { is_in_faction_with = GER } is_faction_leader = yes } } RIE = { dismantle_faction = yes } }
  else_if = { limit = { RIE = { is_in_faction = yes NOT = { is_in_faction_with = GER } } } RIE = { leave_faction = yes } }
  add_to_faction = RIE
  set_faction_name = GE1_ww1_russian_faction
  if = { limit = { RIE = { is_in_faction_with = GER } } clr_country_flag = GE1_ww1_russian_bloc_pending set_global_flag = GE1_ww1_russian_bloc_active WW1_configure_german_russian_faction = yes RIE = { set_country_flag = RUS_ww1_german_bloc }  GE1_ww1_detach_central_partners = yes }
 }
}

# A neutrality focus completed during a war takes effect diplomatically after
# peace. Do not repeat the announcement or detach a ratified Russian bloc.
RUS_ww1_finish_neutral_withdrawal = {
 if = {
  limit = {
   tag = RIE has_start_date < 1936.1.1 date < 1936.1.1 has_war = no is_in_faction = yes
   OR = { has_country_flag = RUS_ww1_armed_neutrality has_completed_focus = RUS_ww1_armed_neutrality }
   NOT = { has_country_flag = RUS_ww1_german_bloc }
   NOT = { AND = { has_country_flag = RUS_ww1_german_agreement is_in_faction_with = GER } }
  }
  if = { limit = { is_faction_leader = yes } random_other_country = { limit = { is_in_faction_with = RIE NOT = { tag = RIE } is_subject = no } set_faction_leader = yes } }
  if = { limit = { is_faction_leader = yes } dismantle_faction = yes }
  else_if = { limit = { is_in_faction = yes } leave_faction = yes }
 }
}

GE1_ww1_request_russian_bloc = {
 if = {
  limit = {
   tag = GER OR = { is_in_faction = no is_faction_leader = yes } has_start_date < 1936.1.1 date < 1936.1.1 has_war = no is_subject = no
   GE1_ww1_may_choose_russian_bloc = yes has_completed_focus = GE1_R_renewed_reinsurance
   has_global_flag = GE1_RP_russian_neutrality_accepted
   NOT = { has_country_flag = GE1_ww1_russian_bloc_pending }
   NOT = { has_country_flag = GE1_ww1_russian_bloc_refused }
   NOT = { has_global_flag = GE1_ww1_russian_bloc_active }
   RIE = { exists = yes NOT = { has_capitulated = yes } has_war = no is_subject = no has_country_flag = RUS_ww1_german_agreement NOT = { has_completed_focus = RUS_entente_diplomacy } }
  }
  set_country_flag = GE1_ww1_russian_bloc_pending
  RIE = { country_event = { id = endsieg_ww1_bloc.1 days = 1 } }
 }
}
