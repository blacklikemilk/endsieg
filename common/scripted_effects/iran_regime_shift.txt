# Iran 1936-1945 - changing the regime after the tree has loaded.
#
# The 1936 chapter only knows the Pahlavi line, because a 1936 bookmark sets
# iran_pahlavi during initialisation and never plays the 1919-35 settlements.
# These effects let the player leave that line inside the chapter itself: the
# old regime flags are cleared and the new one set, which switches allow_branch
# on the political continuations without reloading the tree.

iran_clear_regime_flags = {
	clr_country_flag = iran_pahlavi
	clr_country_flag = iran_qajar_restoration
	clr_country_flag = iran_republic
	clr_country_flag = iran_constitutional_monarchy
	clr_country_flag = iran_socialist
	clr_country_flag = iran_nationalist_front
}

iran_shift_to_qajar = {
	iran_clear_regime_flags = yes
	set_country_flag = iran_qajar_restoration
	set_country_flag = iran_regime_settled
	clr_country_flag = PER_pahlavi_dynasty
	set_politics = { ruling_party = neutrality elections_allowed = no }
	add_to_variable = { iran_authority = -10 }
	add_stability = 0.05
	mark_focus_tree_layout_dirty = yes
}

iran_shift_to_republic = {
	iran_clear_regime_flags = yes
	set_country_flag = iran_republic
	set_country_flag = PER_republic
	set_country_flag = iran_regime_settled
	clr_country_flag = PER_pahlavi_dynasty
	set_politics = { ruling_party = democratic elections_allowed = yes }
	add_to_variable = { iran_authority = -15 }
	add_stability = 0.05
	mark_focus_tree_layout_dirty = yes
}

iran_shift_to_constitutional = {
	iran_clear_regime_flags = yes
	set_country_flag = iran_constitutional_monarchy
	set_country_flag = iran_constitutionalist
	set_country_flag = iran_regime_settled
	set_politics = { ruling_party = neutrality elections_allowed = yes }
	add_stability = 0.10
	mark_focus_tree_layout_dirty = yes
}

iran_shift_to_socialist = {
	iran_clear_regime_flags = yes
	set_country_flag = iran_socialist
	set_country_flag = iran_regime_settled
	clr_country_flag = PER_pahlavi_dynasty
	set_politics = { ruling_party = communist elections_allowed = no }
	add_to_variable = { iran_authority = -20 }
	add_stability = -0.05
	mark_focus_tree_layout_dirty = yes
}

iran_shift_to_nationalist = {
	iran_clear_regime_flags = yes
	set_country_flag = iran_nationalist_front
	set_country_flag = iran_regime_settled
	clr_country_flag = PER_pahlavi_dynasty
	set_politics = { ruling_party = neutrality elections_allowed = yes }
	add_war_support = 0.05
	mark_focus_tree_layout_dirty = yes
}
