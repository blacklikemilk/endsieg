# Applied before the temporary wartime factory ledger is initialized.
WW1_rebalance_starting_military_industry = {
	set_variable = { WW1_starting_military_reduction = 0 }
	every_owned_state = {
		limit = { is_core_of = PREV is_fully_controlled_by = PREV }
		add_to_variable = { PREV.WW1_starting_military_reduction = arms_factory_level }
	}
	multiply_variable = { WW1_starting_military_reduction = 0.18 }
	round_variable = WW1_starting_military_reduction
	while_loop_effect = {
		limit = {
			check_variable = { WW1_starting_military_reduction > 0 }
			any_owned_state = {
				is_core_of = PREV
				is_fully_controlled_by = PREV
				arms_factory > 0
			}
		}
		random_owned_controlled_state = {
			limit = { is_core_of = PREV arms_factory > 0 }
			remove_building = { type = arms_factory level = 1 }
		}
		subtract_from_variable = { WW1_starting_military_reduction = 1 }
	}
	clear_variable = WW1_starting_military_reduction
}

# One additional civilian factory and military factory for each smaller Central power.
WW1_strengthen_starting_central_industry = {
	if = {
		limit = { tag = TUR }
		49 = {
			if = {
				limit = { is_owned_by = PREV is_fully_controlled_by = PREV }
				add_extra_state_shared_building_slots = 1
				add_building_construction = { type = arms_factory level = 1 instant_build = yes }
			}
		}
		347 = {
			if = {
				limit = { is_owned_by = PREV is_fully_controlled_by = PREV }
				add_extra_state_shared_building_slots = 1
				add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
			}
		}
		add_equipment_to_stockpile = { type = early_infantry_equipment_0 amount = 2000 producer = TUR }
		add_equipment_to_stockpile = { type = ww1_artillery_0 amount = 60 producer = TUR }
	}
	else_if = {
		limit = { tag = BUL }
		212 = {
			if = {
				limit = { is_owned_by = PREV is_fully_controlled_by = PREV }
				add_extra_state_shared_building_slots = 1
				add_building_construction = { type = arms_factory level = 1 instant_build = yes }
			}
		}
		48 = {
			if = {
				limit = { is_owned_by = PREV is_fully_controlled_by = PREV }
				add_extra_state_shared_building_slots = 1
				add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
			}
		}
		add_equipment_to_stockpile = { type = early_infantry_equipment_0 amount = 3000 producer = BUL }
		add_equipment_to_stockpile = { type = ww1_artillery_0 amount = 90 producer = BUL }
	}
}

WW1_initialize_starting_balance = {
	if = {
		limit = {
			has_start_date < 1919.1.1
			date < 1919.1.1
			NOT = { has_global_flag = WW1_starting_balance_initialized }
		}
		every_country = {
			limit = { OR = { tag = ENG tag = FRA tag = RIE tag = ITA tag = USA tag = JAP } }
			WW1_rebalance_starting_military_industry = yes
		}
		TUR = { WW1_strengthen_starting_central_industry = yes }
		BUL = { WW1_strengthen_starting_central_industry = yes }
		set_global_flag = WW1_starting_balance_initialized
	}
}
