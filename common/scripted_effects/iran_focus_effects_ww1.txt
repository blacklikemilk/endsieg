# Overview: support effects for Iran's 1910-1918 focus tree. Country scope, Iran only.
# The Gulf rewards cannot name a single state: Bushehr is in Fars (412), which is the only coastal state the
# chapter can reach, and Britain occupies Fars from 1915 until the chapter hands over to the interwar tree.
# Fars is therefore used while the government holds it and Khuzestan (413), which stays Persian, otherwise.
# Each body is hidden so the focus tooltip shows one sentence instead of the placement arithmetic.

iran_ww1_gulf_naval_station = {
	custom_effect_tooltip = iran_ww1_gulf_naval_station_tt
	hidden_effect = {
		if = {
			limit = { controls_state = 412 }
			412 = { add_building_construction = { type = naval_base level = 1 instant_build = yes } }
		}
		else = { 413 = { add_building_construction = { type = naval_base level = 1 instant_build = yes } } }
	}
}

iran_ww1_gulf_dockyard = {
	custom_effect_tooltip = iran_ww1_gulf_dockyard_tt
	hidden_effect = {
		if = {
			limit = { controls_state = 412 }
			412 = {
				add_extra_state_shared_building_slots = constant:iran_story.project_slots
				add_building_construction = { type = dockyard level = 1 instant_build = yes }
			}
		}
		else = {
			413 = {
				add_extra_state_shared_building_slots = constant:iran_story.project_slots
				add_building_construction = { type = dockyard level = 1 instant_build = yes }
			}
		}
	}
}

# The foreign military mission is a state spirit, so the idea itself is added without a line of its own.
iran_ww1_foreign_military_mission = {
	custom_effect_tooltip = iran_ww1_foreign_military_mission_tt
	hidden_effect = {
		if = {
			limit = { NOT = { has_idea = iran_foreign_military_mission } }
			add_ideas = iran_foreign_military_mission
		}
	}
}
