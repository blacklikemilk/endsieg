# Romanian historical AI follows its own Entente focuses before intervening.
# Run only against an ongoing Great War and never undo a conflicting path.
WW1_ROM_historical_entry = {
 if = {
  limit = {
   tag = ROM is_ai = yes is_historical_focus_on = yes
   has_focus_tree = generic_focus_ww1 has_start_date < 1919.1.1
   NOT = { date < 1916.8.27 } date < 1919.1.1
   NOT = { WW1_ROM_observing_armistice = yes }
   has_global_flag = WW1_starts
   NOT = { has_global_flag = compiegne }
   NOT = { has_global_flag = versailles }
   NOT = { has_global_flag = central_powers_victory }
   NOT = { has_country_flag = WW1_ROM_historical_war_entry }
   is_subject = no has_capitulated = no
   country_exists = FRA country_exists = AUH country_exists = GER
   FRA = { is_faction_leader = yes FR3_R_may_enter_entente = yes has_war_with = AUH has_war_with = GER }
   OR = { is_in_faction = no WW1_is_balkan_league_member = yes is_in_faction_with = FRA }
   NOT = { has_war_with = FRA }
   NOT = { is_in_faction_with = AUH }
   NOT = { is_in_faction_with = GER }
   NOT = { has_non_aggression_pact_with = AUH }
   NOT = { has_completed_focus = shared_GEN_support_central_powers }
   NOT = { has_completed_focus = shared_GEN_join_cp }
   NOT = { has_completed_focus = shared_GEN_send_volunteers_2 }
   NOT = { has_completed_focus = shared_GEN_neutrality_focus }
   NOT = { has_idea = neutrality_idea }
   FRA = { ROOT = { WW1_country_faction_path_allows_previous = yes } }
  }
  if = { limit = { NOT = { has_completed_focus = shared_GEN_industrial_effort } } complete_national_focus = shared_GEN_industrial_effort }
  if = { limit = { NOT = { has_completed_focus = shared_GEN_political_effort } } complete_national_focus = shared_GEN_political_effort }
  if = { limit = { NOT = { has_completed_focus = shared_GEN_take_position } } complete_national_focus = shared_GEN_take_position }
  if = { limit = { NOT = { has_completed_focus = shared_GEN_support_entente } } complete_national_focus = shared_GEN_support_entente }
  if = { limit = { NOT = { has_completed_focus = shared_GEN_state_claims_2 } } complete_national_focus = shared_GEN_state_claims_2 }
  if = { limit = { NOT = { has_completed_focus = shared_GEN_send_aid_2 } } complete_national_focus = shared_GEN_send_aid_2 }
  if = { limit = { NOT = { has_completed_focus = shared_GEN_join_entente } } complete_national_focus = shared_GEN_join_entente }
  # Focus rewards can be evaluated before completion is registered; retry the
  # existing faction effect explicitly now and on subsequent days if needed.
  if = { limit = { has_completed_focus = shared_GEN_join_entente NOT = { is_in_faction_with = FRA } } set_country_flag = WW1_focus_faction_entente_pending WW1_complete_focus_faction_entry = yes }
  if = {
   limit = { is_in_faction_with = FRA NOT = { has_war_with = AUH } }
   declare_war_on = { target = AUH type = annex_everything }
  }
  if = {
   limit = { is_in_faction_with = FRA has_war_with = AUH }
   set_country_flag = WW1_ROM_historical_war_entry
   if = { limit = { NOT = { has_global_flag = WW1_ROM_war_entry_announced } } set_global_flag = WW1_ROM_war_entry_announced news_event = { id = endsieg_news.191 hours = 6 } }
  }
 }
}

# Peacetime controllers revert to the owner. Track military administration
# separately from treaty cessions and keep the original Romanian cores.
WW1_ROM_start_bookmark_occupation = {
    if = {
        limit = {
            has_start_date > 1918.3.20 has_start_date < 1918.3.22
            date < 1918.11.11
            NOT = { has_country_flag = WW1_ROM_bookmark_occupation_registered }
            OR = { has_country_flag = WW1_ROM_focsani_armistice has_global_flag = bucharestreaty }
            NOT = { has_global_flag = compiegne }
            NOT = { has_global_flag = central_powers_victory }
            NOT = { has_global_flag = bucharestreatyreversed }
            owns_state = 79 controls_state = 79
            NOT = { has_war_with = GER }
            NOT = { has_war_with = AUH }
            NOT = { has_war_with = BUL }
        }
        # The bookmark already gives the occupier ownership so peace cannot reset
        # its control. Older saves can still have Romania as the owner.
        if = {
            limit = { GER = { exists = yes has_capitulated = no } }
            every_state = {
                limit = {
                    OR = {
                        state = 46 state = 81
                        AND = {
                            OR = { state = 1015 state = 1030 }
                            ROM = { has_country_flag = WW1_ROM_focsani_armistice }
                            NOT = { has_global_flag = bucharestreaty }
                            NOT = { has_global_flag = armistice_of_salonica }
                            NOT = { has_global_flag = armistice_of_villa_giusti }
                        }
                    }
                    OR = { is_owned_by = ROM is_owned_by = GER }
                    NOT = { has_state_flag = WW1_ROM_bucharest_occupation }
                    OR = { is_fully_controlled_by = ROM is_fully_controlled_by = GER }
                }
                set_state_flag = WW1_ROM_bucharest_occupation
                if = { limit = { is_owned_by = ROM } transfer_state_to = GER }
                else_if = { limit = { is_fully_controlled_by = ROM } set_state_controller_to = GER }
            }
        }
        if = {
            limit = {
                has_country_flag = WW1_ROM_focsani_armistice
                NOT = { has_global_flag = bucharestreaty }
                NOT = { has_global_flag = armistice_of_salonica }
                NOT = { has_global_flag = armistice_of_villa_giusti }
            }
            77 = {
                if = {
                    limit = {
                        BUL = { exists = yes has_capitulated = no }
                        OR = { is_owned_by = ROM is_owned_by = BUL }
                        NOT = { has_state_flag = WW1_ROM_bucharest_occupation }
                        OR = { is_fully_controlled_by = ROM is_fully_controlled_by = BUL }
                    }
                    set_state_flag = WW1_ROM_bucharest_occupation
                    if = { limit = { is_owned_by = ROM } transfer_state_to = BUL }
                    else_if = { limit = { is_fully_controlled_by = ROM } set_state_controller_to = BUL }
                }
            }
            1018 = {
                if = {
                    limit = {
                        AUH = { exists = yes has_capitulated = no }
                        OR = { is_owned_by = ROM is_owned_by = AUH }
                        NOT = { has_state_flag = WW1_ROM_bucharest_occupation }
                        OR = { is_fully_controlled_by = ROM is_fully_controlled_by = AUH }
                    }
                    set_state_flag = WW1_ROM_bucharest_occupation
                    if = { limit = { is_owned_by = ROM } transfer_state_to = AUH }
                    else_if = { limit = { is_fully_controlled_by = ROM } set_state_controller_to = AUH }
                }
            }
        }
        if = {
            limit = { any_state = { has_state_flag = WW1_ROM_bucharest_occupation } }
            set_country_flag = WW1_ROM_military_occupation_pending
            if = { limit = { 46 = { has_state_flag = WW1_ROM_bucharest_occupation } NOT = { 79 = { is_capital = yes } } } set_capital = { state = 79 } }
        }
        if = {
            limit = {
                46 = { has_state_flag = WW1_ROM_bucharest_occupation }
                81 = { has_state_flag = WW1_ROM_bucharest_occupation }
                OR = {
                    has_global_flag = bucharestreaty
                    has_global_flag = armistice_of_salonica
                    has_global_flag = armistice_of_villa_giusti
                    AND = {
                        77 = { has_state_flag = WW1_ROM_bucharest_occupation }
                        1015 = { has_state_flag = WW1_ROM_bucharest_occupation }
                        1030 = { has_state_flag = WW1_ROM_bucharest_occupation }
                        1018 = { has_state_flag = WW1_ROM_bucharest_occupation }
                    }
                }
            }
            set_country_flag = WW1_ROM_bookmark_occupation_registered
        }
    }
}
WW1_ROM_end_military_occupation = {
    every_state = {
        limit = {
            has_state_flag = WW1_ROM_bucharest_occupation
            OR = { is_owned_by = GER is_owned_by = AUH is_owned_by = AUS is_owned_by = BUL is_owned_by = ROM }
            OR = { is_fully_controlled_by = GER is_fully_controlled_by = AUH is_fully_controlled_by = AUS is_fully_controlled_by = BUL is_fully_controlled_by = ROM }
        }
        clr_state_flag = WW1_ROM_bucharest_occupation
        if = { limit = { is_owned_by = ROM } set_state_controller_to = ROM }
        else = { transfer_state_to = ROM }
    }
    if = {
        limit = { NOT = { any_state = { has_state_flag = WW1_ROM_bucharest_occupation } } }
        clr_country_flag = WW1_ROM_military_occupation_pending
    }
}

# Focsani suspends fighting on the Romanian front. Preserve the occupation across white peace.
WW1_ROM_armistice_with_central_powers = {
    set_country_flag = WW1_ROM_military_occupation_pending
    every_owned_state = {
        limit = {
            OR = { state = 46 state = 81 state = 77 state = 1015 state = 1030 state = 1018 }
            OR = { is_fully_controlled_by = GER is_fully_controlled_by = AUH is_fully_controlled_by = BUL }
        }
        set_state_flag = WW1_ROM_bucharest_occupation
        set_state_flag = WW1_ROM_preserve_controller
        if = { limit = { is_fully_controlled_by = AUH } set_state_flag = WW1_ROM_occupied_by_AUH }
        else_if = { limit = { is_fully_controlled_by = BUL } set_state_flag = WW1_ROM_occupied_by_BUL }
    }
    if = { limit = { has_faction_template = faction_template_ww1_entente } leave_faction = yes }
    every_enemy_country = {
        limit = { OR = { tag = GER tag = AUH tag = BUL tag = TUR is_in_faction_with = GER } }
        white_peace = ROM
    }
    every_owned_state = {
        limit = { has_state_flag = WW1_ROM_preserve_controller }
        clr_state_flag = WW1_ROM_preserve_controller
        if = { limit = { has_state_flag = WW1_ROM_occupied_by_AUH } transfer_state_to = AUH clr_state_flag = WW1_ROM_occupied_by_AUH }
        else_if = { limit = { has_state_flag = WW1_ROM_occupied_by_BUL } transfer_state_to = BUL clr_state_flag = WW1_ROM_occupied_by_BUL }
        else = { transfer_state_to = GER }
    }
}
