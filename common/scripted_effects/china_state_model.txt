# Overview: the single writers of the global China state model. One writer per symbol —

# Readers live in common/scripted_triggers/china_state_model.txt and never write.

# The clamp bounds are the tuning constants themselves, read directly: a `constant:` operand is

# and vanilla assigns one to a variable (common/decisions/SIA.txt:419

# China proper, built once per campaign and never rebuilt.
# Rule: a state belongs to China proper when its effective history/states entry (the mod's file
# where one exists, otherwise the game's) carries add_core_of = CHI inside its history block,
# outside every dated block. The set is the Nationalist government's permanent core set, not
# current ownership: it is identical at every bookmark, because no state in either tree adds or
# removes a CHI core inside a dated block (measured: 0 such blocks). It therefore includes the
# treaty ports and leased territories held by other powers in 1936 — Hong Kong 326, Macau 729,
# Guangzhouwan 728, Taiwan 524, Qingdao 743, Dalian 745, Weihaiwei 1040 — and the Manchurian and
# frontier states. 62 states, listed literally so the set cannot change without a source edit.
# The test re-derives this list from history/states by the same rule and fails when the two
# disagree. The array holds state scopes (THIS inside the state's own scope), because the

# The published tier: a copy of the frozen Republic tier. Called once, from the freeze site in
# rch_succession_settle, so the published value is the frozen one and never a live score.
china_publish_fragmentation_tier = {
	if = {
		limit = { has_variable = rch_succession_final_tier }
		set_temp_variable = { china_published_tier = var:rch_succession_final_tier }
		clamp_temp_variable = {
			var = china_published_tier
			min = constant:china_story.tier_floor
			max = constant:china_story.tier_ceiling
		}
		set_variable = { global.china_fragmentation_tier = var:china_published_tier }
	}
}

# The seat holder. Replaced on every transfer, so the previous holder never survives a
# transfer. The holder is passed explicitly, which keeps the effect correct from any scope and
# keeps a reader from ever becoming a writer. The short assignment form is vanilla's for a global
# country variable: set_variable = { global.POR_iberian_summit_target = GER }
# (common/decisions/POR.txt:2137), set_variable = { global.AH_originator_country = ROOT }
# (common/national_focus/austria.txt:4734).
china_set_seat_holder = {
	set_variable = { global.china_seat_holder = $COUNTRY$ }
}

# The first reunifier is recorded permanently and takes the seat. Every later write is refused,
# so a second unifier cannot take the seat from the first.
china_record_reunifier = {
	if = {
		limit = { NOT = { has_variable = global.china_reunified_by } }
		set_variable = { global.china_reunified_by = $COUNTRY$ }
		china_set_seat_holder = { COUNTRY = $COUNTRY$ }
	}
}

# Reunification test, called from the existing on_daily_RCH. Cheapest conditions first: the
# array must exist and no reunifier may be recorded, so a campaign that has resolved this pays
# two variable checks a day. It then walks the bounded list of Chinese campaign countries — never
# every_country and never a world sweep — and each candidate costs one pass over the 62-state
# array. A tag outside this list cannot reunify; see docs/systems/china_state_model.md.
china_test_reunification = {
	if = {
		limit = {
			check_variable = { global.china_proper_states^num > 0 }
			NOT = { has_variable = global.china_reunified_by }
