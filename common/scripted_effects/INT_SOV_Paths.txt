# Share the settlement between a military victory and the historical fallback.
# Country annexation includes every custom state without resetting research.
INT_SOV_finish_russian_civil_war = {
	if = {
		limit = {
			tag = SOV
			date < 1936.1.1
			has_global_flag = russian_civil_war
			NOT = { has_global_flag = RUS_CW_Over }
			has_capitulated = no
			is_subject = no
		}
		set_global_flag = RUS_CW_Over
		remove_ideas = INT_SOV_red_army_mobilization
		# Integrate domestic White allies before annexing their overlord.
		every_country = {
			limit = {
				exists = yes
				OR = { original_tag = KUB original_tag = TSS original_tag = DON }
				OR = { has_war_with = SOV is_subject_of = WHR is_subject_of = SOV }
			}
			SOV = { annex_country = { target = PREV transfer_troops = no } }
		}
		if = {
			limit = {
				country_exists = WHR
				OR = {
					has_war_with = WHR
					has_country_flag = INT_SOV_white_government_defeated
					WHR = { is_subject_of = SOV }
				}
			}
			annex_country = { target = WHR transfer_troops = no }
		}
		clr_country_flag = INT_SOV_rcw_fallback_queued
		clr_country_flag = INT_SOV_white_government_defeated
		news_event = { id = INT_russian_news.14 hours = 6 }
	}
}

# The fall of Vladivostok on 25 October 1922 provides a deterministic end date.
# The open lower bound also repairs older campaigns whose war is still running.
INT_SOV_schedule_civil_war_fallback = {
	if = {
		limit = {
			tag = SOV
			date > 1922.10.24
			date < 1936.1.1
			has_global_flag = russian_civil_war
			NOT = { has_global_flag = RUS_CW_Over }
			NOT = { has_country_flag = INT_SOV_rcw_fallback_queued }
			NOT = { has_country_flag = INT_SOV_white_government_defeated }
			has_government = communism
			has_capitulated = no
			is_subject = no
			has_war_with = WHR
		}
		set_country_flag = INT_SOV_rcw_fallback_queued
		country_event = { id = interwar_soviet.62 days = 0 }
	}
}

# The Red Army mobilizes once against the White government. This also covers
# existing saves and scenario starts that bypass the civil-war setup event.
INT_SOV_civil_war_support = {
	if = {
		limit = {
			tag = SOV
			date < 1936.1.1
			has_war_with = WHR
			NOT = { has_global_flag = RUS_CW_Over }
		}
		# The October transition can carry the obsolete Imperial army penalties.
		remove_ideas = { russia_sucks russia_sucks_less russia_meh }
		if = {
			limit = { NOT = { has_idea = INT_SOV_red_army_mobilization } }
			add_ideas = INT_SOV_red_army_mobilization
		}
		if = {
			limit = { NOT = { has_country_flag = INT_SOV_civil_war_support_received } }
			set_country_flag = INT_SOV_civil_war_support_received
			add_manpower = 100000
			add_equipment_to_stockpile = { type = early_infantry_equipment_3 amount = 15000 }
			add_equipment_to_stockpile = { type = support_equipment_1 amount = 750 }
			add_equipment_to_stockpile = { type = ww1_artillery_3 amount = 500 }
		}
	}
	else_if = {
		limit = { has_idea = INT_SOV_red_army_mobilization }
		remove_ideas = INT_SOV_red_army_mobilization
	}
}
