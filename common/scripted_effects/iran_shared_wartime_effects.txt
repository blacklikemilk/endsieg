# Overview: effects shared between an Iran decision and the Iran wartime events.
# One body per act, so the decision (common/decisions/iran_great_powers.txt) and the
# event option (events/iran_focus_ww2.txt) cannot drift apart. Country scope, Iran only.
# Values live in constant:iran_story.*; the arithmetic is hidden.

# The 1941 rupture with the German technical mission: the agreement is cancelled, the
# mission leaves, German influence and the leverage it bought are written off, and both
# the British and the northern patron read the withdrawal as a concession to them.
# Called by iran_expel_the_german_mission (complete_effect) and iran_f3.2 (option a).
iran_expel_german_mission_effects = {
	clr_country_flag = iran_agreement_german_mission
	remove_ideas = iran_german_technical_mission
	subtract_from_variable = { iran_german_influence = constant:iran_story.influence_per_agreement }
	add_to_variable = { iran_sovereignty = 6 }
	add_to_variable = { iran_british_standing = constant:iran_story.great_game_court_gain }
	add_to_variable = { iran_northern_standing = constant:iran_story.great_game_court_gain }
	set_country_flag = iran_german_mission_withdrawn
	iran_refresh_mechanics_hidden = yes
}
