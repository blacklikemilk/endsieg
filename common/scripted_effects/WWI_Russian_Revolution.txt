# This is selected once at the October transition. Lenin must actually have
# returned; otherwise Trotsky leads until an actual return event or succession.
RUS_ww1_select_october_leader = {
    if = {
        limit = { tag = SOV exists = yes NOT = { has_country_flag = RUS_ww1_october_leader_selected } }
        if = {
            limit = { RUS_ww1_lenin_can_lead = yes }
            if = {
                limit = { NOT = { endsieg_character_active_sov_vladimir_lenin_c95c867e = yes } }
                endsieg_character_activate_sov_vladimir_lenin_c95c867e = yes
            }
            promote_character = { character = SOV_Vladimir_Lenin ideology = leninism }
        }
        else = {
            if = {
                limit = { endsieg_character_active_sov_vladimir_lenin_c95c867e = yes }
                endsieg_character_retire_sov_vladimir_lenin_c95c867e = yes
            }
            if = {
                limit = { has_character = SOV_lev_trotsky }
                endsieg_character_activate_sov_lev_trotsky_f0fa5e52 = yes
                add_country_leader_role = {
                    character = SOV_lev_trotsky
                    country_leader = { ideology = leninism traits = { } expire = "1953.3.1.1" }
                    promote_leader = yes
                }
                set_country_flag = RUS_ww1_trotsky_october_caretaker
            }
        }
        set_country_flag = RUS_ww1_october_leader_selected
    }
}

RUS_ww1_execute_october_transition = {
    if = {
        limit = {
            RUS_ww1_october_context = yes
            NOT = { country_exists = SOV }
            any_owned_state = { is_capital = yes }
        }
        # Mark the handoff before releasing the capital. Re-enter from SOV's
        # own daily scope: the old RIE event scope can become invalid on release.
        if = { limit = { has_war_with = GER } set_global_flag = RUS_ww1_brest_negotiations }
        set_global_flag = RUS_ww1_october_takeover_pending
        capital_scope = { add_core_of = SOV }
        # Transfer only owned core states; enemy province controllers stay put.
        release_autonomy = {
            target = SOV
            autonomy_state = autonomy_puppet
            release_non_owned_controlled = no
            force_change_controller_for_non_ally_controlled = no
        }
        if = {
            limit = { country_exists = SOV }
            set_autonomy = { target = SOV autonomy_state = autonomy_free end_wars = no end_civil_wars = no }
            # Join synchronously, before a daily tick can invalidate occupation.
            SOV = { RUS_ww1_preserve_october_central_wars = yes }
        }
    }
}

# Called from SOV's own country scope, never from an RIE event that the
# annexation destroys. The second condition recovers the previously broken
# October handoff in existing pre-civil-war saves.
RUS_ww1_finish_october_takeover = {
    if = {
        limit = {
            tag = SOV exists = yes has_start_date < 1918.1.1 date < 1936.1.1
            NOT = { has_global_flag = russian_civil_war }
            NOT = { has_global_flag = RUS_CW_Over }
            OR = {
                has_global_flag = RUS_ww1_october_takeover_pending
                AND = { has_global_flag = october_revolution has_global_flag = RUS_ww1_brest_negotiations has_country_flag = RUS_ww1_october_initialized country_exists = RIE RIE = { NOT = { has_war_with = SOV } } OR = { is_ai = yes RIE = { is_ai = yes } } }
            }
        }
        if = {
            limit = { country_exists = RIE RIE = { NOT = { has_war_with = SOV } } OR = { is_ai = yes RIE = { is_ai = yes } } }
            if = { limit = { is_subject_of = RIE } RIE = { set_autonomy = { target = SOV autonomy_state = autonomy_free end_wars = no end_civil_wars = no } } }
            # Retry safely if the handoff resumed from an older pending save.
            RUS_ww1_preserve_october_central_wars = yes
            WW1_CP_anchor_russian_major = yes
            inherit_technology = RIE
            if = { limit = { is_ai = yes RIE = { is_ai = no } } change_tag_from = RIE }
            annex_country = { target = RIE transfer_troops = yes }
        }
        if = {
            limit = { NOT = { country_exists = RIE } }
            clr_global_flag = RUS_ww1_october_takeover_pending
            RUS_ww1_initialize_october_government = yes
        }
    }
}
