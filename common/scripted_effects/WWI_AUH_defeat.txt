WW1_AUH_declare_secession_war = {
    if = {
        limit = {
            exists = yes
            is_subject = no
            is_in_faction = no
            has_capitulated = no
            AUH = { exists = yes has_capitulated = no }
            NOT = { has_country_flag = WW1_AUH_secession_war_started }
        }
        if = {
            limit = { NOT = { has_war_with = AUH } }
            save_event_target_as = WW1_AUH_seceding_country
            # Imperial troops already stationed in the new state must withdraw before hostilities.
            if = {
                limit = { AUH = { controls_state = 1051 } }
                every_owned_state = {
                    limit = { is_fully_controlled_by = PREV }
                    teleport_armies = { to_state = 1051 limit = { tag = AUH } }
                }
            }
            if = {
                limit = { has_army_size = { size < 1 } capital_scope = { is_controlled_by = PREV } }
                division_template = {
                    name = "National Council Guard"
                    regiments = {
                        infantry = { x = 0 y = 0 }
                        infantry = { x = 0 y = 1 }
                        infantry = { x = 0 y = 2 }
                        infantry = { x = 1 y = 0 }
                    }
                }
                capital_scope = {
                    create_unit = { owner = PREV division = "name = \"1. National Council Guard\" division_template = \"National Council Guard\" start_equipment_factor = 0.75" }
                    create_unit = { owner = PREV division = "name = \"2. National Council Guard\" division_template = \"National Council Guard\" start_equipment_factor = 0.75" }
                }
            }
            # Join the existing front against Vienna to avoid a separate peace conference.
            AUH = {
                random_enemy_country = {
                    limit = {
                        OR = { tag = ITA tag = FRA tag = ENG tag = USA }
                        has_capitulated = no
                    }
                    event_target:WW1_AUH_seceding_country = {
                        add_to_war = {
                            targeted_alliance = PREV
                            enemy = AUH
                            hostility_reason = asked_to_join
                            single_target_only = yes
                        }
                    }
                }
            }
            if = {
                limit = { NOT = { has_war_with = AUH } }
                declare_war_on = { target = AUH type = annex_everything }
            }
        }
        if = {
            limit = { has_war_with = AUH }
            set_country_flag = WW1_AUH_secession_war_started
        }
    }
}
