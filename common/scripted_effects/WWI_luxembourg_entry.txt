WW1_LUX_join_entente_on_invasion = {
	if = {
		limit = {
			tag = LUX
			has_start_date < 1919.1.1
			date < 1919.1.1
			has_war_with = GER
			is_subject = no
			OR = { is_in_faction = no is_in_faction_with = FRA }
			NOT = { has_war_with = FRA }
			FRA = { exists = yes WW1_FG_entente = yes }
		}
		if = {
			limit = {
				has_focus_tree = generic_focus_ww1
				NOT = { has_completed_focus = shared_GEN_join_entente }
			}
			unlock_national_focus = shared_GEN_join_entente
		}
		if = {
			limit = { NOT = { is_in_faction_with = FRA } }
			FRA = { add_to_faction = LUX }
		}
		if = {
			limit = {
				NOT = { has_country_flag = WW1_LUX_entente_war_linked }
				is_in_faction_with = FRA
			}
			if = {
				limit = { FRA = { has_war_with = GER } }
				add_to_war = { targeted_alliance = FRA enemy = GER hostility_reason = asked_to_join }
			}
			else_if = {
				limit = { RIE = { exists = yes is_in_faction_with = FRA has_war_with = GER } }
				add_to_war = { targeted_alliance = RIE enemy = GER hostility_reason = asked_to_join }
			}
			if = {
				limit = {
					OR = {
						AND = { FRA = { has_war_with = GER } has_war_together_with = FRA }
						AND = { RIE = { has_war_with = GER is_in_faction_with = FRA } has_war_together_with = RIE }
					}
				}
				set_country_flag = WW1_LUX_entente_war_linked
			}

		}
	}
}
