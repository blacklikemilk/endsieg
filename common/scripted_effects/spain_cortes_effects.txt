# Overview: the Spanish Cortes as a chamber of seat blocs.
# Raw support is written to spain_cortes_raw_* and normalised to exactly 400 seats in
# spain_cortes_seats_*, so a majority is always 201 however support drifts.
# A decision category (SPR_cortes) mounts the chamber panel; its bills are ordinary decisions
# gated on the bloc that has to carry them.

# =====================================================================================
# Initialisation
# =====================================================================================

spain_cortes_initialize = {
	spain_cortes_recount = yes
}

# =====================================================================================
# Seats
# =====================================================================================

spain_cortes_recount = {
	# --- raw support, restoration and dictatorship
	set_variable = { var = spain_cortes_raw_monarchists value = 40 }
	set_variable = { var = spain_cortes_raw_liberals value = 30 }
	set_variable = { var = spain_cortes_raw_conservatives value = 25 }
	set_variable = { var = spain_cortes_raw_republicans value = 15 }
	set_variable = { var = spain_cortes_raw_socialists value = 8 }
	set_variable = { var = spain_cortes_raw_carlists value = 5 }
	set_variable = { var = spain_cortes_raw_falange value = 1 }

	if = { limit = { has_country_flag = spain_electoral_reform } subtract_from_variable = { var = spain_cortes_raw_monarchists value = 15 } add_to_variable = { var = spain_cortes_raw_republicans value = 14 } }
	if = { limit = { has_country_flag = spain_elections_managed } add_to_variable = { var = spain_cortes_raw_monarchists value = 12 } subtract_from_variable = { var = spain_cortes_raw_republicans value = 6 } }
	if = { limit = { has_country_flag = spain_press_freedom } add_to_variable = { var = spain_cortes_raw_republicans value = 6 } }
	if = { limit = { has_country_flag = spain_labour_conciliation } add_to_variable = { var = spain_cortes_raw_socialists value = 14 } }
	if = { limit = { has_country_flag = spain_labour_repression } subtract_from_variable = { var = spain_cortes_raw_socialists value = 6 } add_to_variable = { var = spain_cortes_raw_carlists value = 5 } }
	if = { limit = { has_country_flag = spain_liberal_governments } add_to_variable = { var = spain_cortes_raw_liberals value = 9 } }
	if = { limit = { has_country_flag = spain_conservative_governments } add_to_variable = { var = spain_cortes_raw_conservatives value = 9 } }
	if = { limit = { has_country_flag = spain_mancomunitat } add_to_variable = { var = spain_cortes_raw_republicans value = 6 } }
	if = { limit = { has_country_flag = spain_mancomunitat_suppressed } add_to_variable = { var = spain_cortes_raw_carlists value = 6 } subtract_from_variable = { var = spain_cortes_raw_republicans value = 4 } }

	# --- raw support once the Republic exists
	if = {
		limit = {
			NOT = { has_country_flag = spain_chapter_restoration }
			NOT = { has_country_flag = spain_chapter_rivera }
		}
		set_variable = { var = spain_cortes_raw_monarchists value = 12 }
		set_variable = { var = spain_cortes_raw_liberals value = 22 }
		set_variable = { var = spain_cortes_raw_conservatives value = 60 }
		set_variable = { var = spain_cortes_raw_republicans value = 95 }
		set_variable = { var = spain_cortes_raw_socialists value = 80 }
		set_variable = { var = spain_cortes_raw_carlists value = 18 }
		set_variable = { var = spain_cortes_raw_falange value = 5 }
		if = { limit = { has_country_flag = spain_ceda_government } add_to_variable = { var = spain_cortes_raw_conservatives value = 30 } subtract_from_variable = { var = spain_cortes_raw_republicans value = 20 } }
		if = { limit = { has_country_flag = spain_falange_street } add_to_variable = { var = spain_cortes_raw_falange value = 12 } }
		if = { limit = { has_country_flag = spain_cnt_split } add_to_variable = { var = spain_cortes_raw_socialists value = 8 } }
		if = { limit = { has_country_flag = spain_generalitat } add_to_variable = { var = spain_cortes_raw_republicans value = 5 } }
	}

	# --- raw total, never zero
	set_variable = { var = spain_cortes_raw_total value = 0 }
	add_to_variable = { var = spain_cortes_raw_total value = spain_cortes_raw_monarchists }
	add_to_variable = { var = spain_cortes_raw_total value = spain_cortes_raw_liberals }
	add_to_variable = { var = spain_cortes_raw_total value = spain_cortes_raw_conservatives }
	add_to_variable = { var = spain_cortes_raw_total value = spain_cortes_raw_republicans }
	add_to_variable = { var = spain_cortes_raw_total value = spain_cortes_raw_socialists }
	add_to_variable = { var = spain_cortes_raw_total value = spain_cortes_raw_carlists }
	add_to_variable = { var = spain_cortes_raw_total value = spain_cortes_raw_falange }
	clamp_variable = { var = spain_cortes_raw_total min = 1 max = 10000 }

	# --- normalised seats: round(raw * 400 / raw_total)
	set_variable = { var = spain_cortes_seats_monarchists value = spain_cortes_raw_monarchists }
	multiply_variable = { var = spain_cortes_seats_monarchists value = 400 }
	divide_variable = { var = spain_cortes_seats_monarchists value = spain_cortes_raw_total }
	round_variable = spain_cortes_seats_monarchists

	set_variable = { var = spain_cortes_seats_liberals value = spain_cortes_raw_liberals }
	multiply_variable = { var = spain_cortes_seats_liberals value = 400 }
	divide_variable = { var = spain_cortes_seats_liberals value = spain_cortes_raw_total }
	round_variable = spain_cortes_seats_liberals

	set_variable = { var = spain_cortes_seats_conservatives value = spain_cortes_raw_conservatives }
	multiply_variable = { var = spain_cortes_seats_conservatives value = 400 }
	divide_variable = { var = spain_cortes_seats_conservatives value = spain_cortes_raw_total }
	round_variable = spain_cortes_seats_conservatives

	set_variable = { var = spain_cortes_seats_republicans value = spain_cortes_raw_republicans }
	multiply_variable = { var = spain_cortes_seats_republicans value = 400 }
	divide_variable = { var = spain_cortes_seats_republicans value = spain_cortes_raw_total }
	round_variable = spain_cortes_seats_republicans

	set_variable = { var = spain_cortes_seats_socialists value = spain_cortes_raw_socialists }
	multiply_variable = { var = spain_cortes_seats_socialists value = 400 }
	divide_variable = { var = spain_cortes_seats_socialists value = spain_cortes_raw_total }
	round_variable = spain_cortes_seats_socialists

	set_variable = { var = spain_cortes_seats_carlists value = spain_cortes_raw_carlists }
	multiply_variable = { var = spain_cortes_seats_carlists value = 400 }
	divide_variable = { var = spain_cortes_seats_carlists value = spain_cortes_raw_total }
	round_variable = spain_cortes_seats_carlists

	set_variable = { var = spain_cortes_seats_falange value = spain_cortes_raw_falange }
	multiply_variable = { var = spain_cortes_seats_falange value = 400 }
	divide_variable = { var = spain_cortes_seats_falange value = spain_cortes_raw_total }
	round_variable = spain_cortes_seats_falange

	set_variable = { var = spain_cortes_total value = 0 }
	add_to_variable = { var = spain_cortes_total value = spain_cortes_seats_monarchists }
	add_to_variable = { var = spain_cortes_total value = spain_cortes_seats_liberals }
	add_to_variable = { var = spain_cortes_total value = spain_cortes_seats_conservatives }
	add_to_variable = { var = spain_cortes_total value = spain_cortes_seats_republicans }
	add_to_variable = { var = spain_cortes_total value = spain_cortes_seats_socialists }
	add_to_variable = { var = spain_cortes_total value = spain_cortes_seats_carlists }
	add_to_variable = { var = spain_cortes_total value = spain_cortes_seats_falange }

	spain_cortes_refresh_display = yes
}

spain_cortes_refresh_display = {
	set_variable = { var = spain_cortes_right_bloc value = 0 }
	add_to_variable = { var = spain_cortes_right_bloc value = spain_cortes_seats_monarchists }
	add_to_variable = { var = spain_cortes_right_bloc value = spain_cortes_seats_liberals }
	add_to_variable = { var = spain_cortes_right_bloc value = spain_cortes_seats_conservatives }
	add_to_variable = { var = spain_cortes_right_bloc value = spain_cortes_seats_carlists }
	add_to_variable = { var = spain_cortes_right_bloc value = spain_cortes_seats_falange }

	set_variable = { var = spain_cortes_left_bloc value = 0 }
	add_to_variable = { var = spain_cortes_left_bloc value = spain_cortes_seats_republicans }
	add_to_variable = { var = spain_cortes_left_bloc value = spain_cortes_seats_socialists }

	set_variable = { var = spain_cortes_bar_frame value = spain_cortes_left_bloc }
	multiply_variable = { var = spain_cortes_bar_frame value = 100 }
	divide_variable = { var = spain_cortes_bar_frame value = 400 }
	round_variable = spain_cortes_bar_frame
	clamp_variable = { var = spain_cortes_bar_frame min = 0 max = 100 }
}

spain_hold_election = {
	set_country_flag = spain_election_held
	add_political_power = constant:spain_campaign.pp_medium
	add_stability = constant:spain_campaign.stability_small
	spain_cortes_recount = yes
}
