# Use the same Directory settlement for the event and the dated bookmark.
INT_UKR_install_directory = {
	INT_UKR_end_german_coalition_wars = yes
	if = {
		limit = { is_subject_of = GER }
		GER = { set_autonomy = { target = UKR autonomy_state = autonomy_free end_wars = no end_civil_wars = no } }
	}
	if = {
		limit = {
			is_in_faction = yes
			OR = { is_in_faction_with = GER has_country_flag = skoropadskytakesover }
		}
		leave_faction = yes
	}
	clr_country_flag = skoropadskytakesover
	set_cosmetic_tag = INT_UKR_people_republic
	set_politics = { ruling_party = democratic elections_allowed = no }
	set_popularities = { democratic = 60 neutrality = 25 communism = 15 fascism = 0 }
	set_party_name = { ideology = democratic name = INT_UKR_directory_party long_name = INT_UKR_directory_party_long }
	if = {
		limit = { NOT = { has_country_leader = { name = "Volodymyr Vynnychenko" ruling_only = yes } } }
		create_country_leader = {
			name = "Volodymyr Vynnychenko"
			picture = "gfx/leaders/UKR/Portrait_UKR_Volodymyr_Vynnychenko.dds"
			ideology = socialism
			expire = "1953.3.1"
		}
	}
	set_country_flag = INT_UKR_directory_installed
	set_global_flag = hetmanate_falls
}

# Called only for a historical Polish client or the West Ukrainian government
# freed by this settlement. Never give Polish-owned land back to a client.
INT_UKR_settle_galician_client = {
	if = {
		limit = { date < 1919.7.17 }
		if = {
			limit = { tag = WUR is_subject_of = POL }
			POL = { set_autonomy = { target = PREV autonomy_state = autonomy_free end_wars = no end_civil_wars = no } }
			set_country_flag = INT_UKR_galician_client_freed
		}
		else_if = {
			limit = {
				NOT = { tag = WUR }
				OR = { NOT = { country_exists = WUR } WUR = { is_ai = yes is_subject = no is_in_faction = no } }
			}
			every_owned_state = {
				limit = { OR = { state = 89 state = 91 } OR = { is_fully_controlled_by = PREV is_fully_controlled_by = POL } }
				add_core_of = WUR
				transfer_state_to = WUR
				WUR = { set_country_flag = INT_UKR_galician_client_freed }
			}
		}
	}
	else = {
		every_owned_state = {
			limit = {
				OR = { state = 89 state = 91 }
				OR = { is_fully_controlled_by = PREV is_fully_controlled_by = POL }
			}
			add_core_of = POL
			transfer_state_to = POL
		}
	}
}

# combining the Polish and Ukrainian wars, so Riga can end Poland's war only.

# An occupied Kharkiv can host the rival Soviet government while the Directory still fights.
INT_UKR_settle_soviet_government = {
	if = {
		limit = { INT_UKR_soviet_settlement_context = yes }
		# Older saves can contain a Ukrainian copy holding Kharkiv with the
		# Directory's party and no ministry. Identify it by territory, never D01.
		if = {
			limit = {
				NOT = { has_global_flag = INT_USSR_founded }
				221 = { owner = {
					original_tag = UKR NOT = { tag = UKR } is_ai = yes
					OR = {
						has_country_flag = INT_UKR_kharkiv_cabinet_pending
						AND = {
							has_global_flag = INT_UKR_kharkiv_government_established
							NOT = { has_country_flag = INT_UKR_rakovsky_ministry }
							date < 1923.7.15
						}
					}
					OR = { is_subject_of = SOV AND = { is_subject = no is_in_faction = no NOT = { has_war_with = SOV } } }
				} }
			}
			221 = { owner = { INT_UKR_institute_kharkiv_government = yes } }
		}
		if = {
			limit = { INT_UKR_kharkiv_government_ready = yes }
			country_event = { id = ww1_ukraine.19 }
		}
		if = {
			limit = {
				NOT = { has_global_flag = INT_USSR_founded } NOT = { country_exists = UKR }
				has_country_flag = INT_ukraine_integrated
				NOT = { any_subject_country = { original_tag = UKR } }
				221 = { is_owned_and_controlled_by = SOV }
			}
			221 = { transfer_state_to = UKR }
			UKR = { set_politics = { ruling_party = communism elections_allowed = no } }
			puppet = UKR
		}
		# A conference can puppet the former Directory after Kharkiv already exists.
		# Keep one Soviet Ukrainian administration and preserve its acquired states.
		if = {
			limit = {
				UKR = { INT_UKR_soviet_subject = yes }
				any_subject_country = { INT_UKR_soviet_subject = yes NOT = { tag = UKR } owns_state = 221 }
			}
			random_subject_country = {
				limit = { INT_UKR_soviet_subject = yes NOT = { tag = UKR } owns_state = 221 }
				annex_country = { target = UKR transfer_troops = yes }
			}
			set_country_flag = INT_ukraine_integrated
		}
		every_subject_country = {
			limit = { INT_UKR_soviet_subject = yes }
			if = {
				limit = {
					has_global_flag = INT_USSR_founded
					has_capitulated = no
					OR = { owns_state = 221 owns_state = 202 }
				}
				SOV = {
					annex_country = { target = PREV transfer_troops = yes }
					set_country_flag = INT_UKR_union_accession
				}
			}
			else_if = {
				limit = { NOT = { has_global_flag = INT_USSR_founded } }
				INT_UKR_administer_soviet_republic = yes
			}
		}
	}
}

INT_UKR_administer_soviet_republic = {
	if = {
		limit = {
			date < 1923.7.15
			NOT = { has_country_leader = { name = "Khristian Rakovsky" ruling_only = yes } }
		}
		set_country_flag = INT_UKR_rakovsky_ministry
		set_party_name = { ideology = communism name = INT_UKR_bolshevik_party long_name = INT_UKR_bolshevik_party_long }
		set_politics = { ruling_party = communism elections_allowed = no }
		if = {
			limit = { has_character = UKR_khristian_rakovsky }
			promote_character = { character = UKR_khristian_rakovsky ideology = marxism }
		}
		else = {
			# The breakaway Soviet republic can have a different country tag.
			create_country_leader = {
				name = "Khristian Rakovsky"
				picture = "gfx/leaders/UKR/Portrait_Khristian_Rakovsky.dds"
				ideology = marxism
				expire = "1923.7.15"
			}
		}
	}
	if = { limit = { date < 1919.3.10 NOT = { has_cosmetic_tag = INT_UKR_red_banner } } set_cosmetic_tag = INT_UKR_red_banner }
	else_if = { limit = { date > 1919.3.9 NOT = { has_cosmetic_tag = INT_UKR_socialist_republic } } set_cosmetic_tag = INT_UKR_socialist_republic }

	save_event_target_as = INT_UKR_soviet_ministry
	SOV = {
		every_owned_state = {
			limit = {
				is_core_of = UKR is_fully_controlled_by = SOV
				OR = {
					state = 192 state = 193 state = 196 state = 197 state = 198 state = 199
					state = 201 state = 202 state = 203 state = 221 state = 225 state = 226
					state = 200 state = 227 state = 228 state = 259 state = 834 state = 989
				}
			}
			transfer_state_to = event_target:INT_UKR_soviet_ministry
		}
	}
	if = {
		limit = { owns_state = 221 controls_province = 418 NOT = { capital_scope = { state = 221 } } }
		set_capital = { state = 221 remember_old_capital = no }
	}
	if = {
		limit = { INT_UKR_regional_war_policy = yes }
		# Settle each inherited opponent once. Keep the recipient country itself
		# so a conference-created Ukrainian tag has its own settlement history.
		every_enemy_country = {
			limit = {
				is_ai = yes OR = { tag = EST tag = LAT tag = LIT tag = BLR }
				NOT = {
					OR = {
						has_wargoal_against = event_target:INT_UKR_soviet_ministry
						has_war_with_wargoal_against = { target = event_target:INT_UKR_soviet_ministry }
					}
				}
				event_target:INT_UKR_soviet_ministry = {
					NOT = {
						OR = {
							is_in_array = { array = INT_UKR_settled_fronts value = PREV }
							has_wargoal_against = PREV
							has_war_with_wargoal_against = { target = PREV }
						}
					}
				}
			}
			event_target:INT_UKR_soviet_ministry = {
				white_peace = { tag = PREV }
				if = {
					limit = { NOT = { has_war_with = PREV } }
					add_to_array = { INT_UKR_settled_fronts = PREV }
				}
			}
		}

		# Normal calls group Soviet wars together. Join only the local enemy
		# when those calls are blocked by a simultaneous Baltic campaign.
		if = {
			limit = { SOV = { INT_UKR_ally_has_remote_front = yes } }
			SOV = {
				every_enemy_country = {
					limit = {
						OR = {
							tag = WHR
							AND = { tag = UKR INT_UKR_directory_active = yes }
							AND = { tag = POL NOT = { has_global_flag = sov_pol_ceasefire } }
						}
						NOT = {
							OR = {
								has_war_with = event_target:INT_UKR_soviet_ministry
								has_truce_with = event_target:INT_UKR_soviet_ministry
								has_non_aggression_pact_with = event_target:INT_UKR_soviet_ministry
							}
						}
					}
					event_target:INT_UKR_soviet_ministry = {
						add_to_war = { targeted_alliance = SOV enemy = PREV hostility_reason = asked_to_join single_target_only = yes }
					}
				}
			}
		}
	}
	if = {
		limit = { has_global_flag = sov_pol_ceasefire POL = { is_ai = yes NOT = { has_war_with = SOV } } }
		if = { limit = { has_war_with = POL } white_peace = POL }
		93 = { if = { limit = { is_owned_by = event_target:INT_UKR_soviet_ministry OR = { is_fully_controlled_by = event_target:INT_UKR_soviet_ministry is_fully_controlled_by = POL } } transfer_state_to = POL } }
	}
}
