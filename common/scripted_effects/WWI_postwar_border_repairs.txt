# Preserve transfers already made by the war or a peace conference.
WW1_settle_german_african_possessions = {
    if = {
        limit = { SAF = { exists = yes NOT = { has_war_with = GER } } }
        GER = {
            every_owned_state = {
                limit = { OR = { state = 541 state = 893 state = 894 state = 895 } }
                transfer_state_to = SAF
            }
        }
    }
    if = {
        limit = { ENG = { exists = yes NOT = { has_war_with = GER } } }
        GER = {
            every_owned_state = {
                limit = { OR = { state = 546 state = 974 state = 1035 } }
                transfer_state_to = ENG
            }
        }
    }
    if = {
        limit = { FRA = { exists = yes NOT = { has_war_with = GER } } }
        GER = {
            every_owned_state = {
                limit = { OR = { state = 773 state = 777 state = 1082 state = 1083 state = 1084 state = 774 } }
                transfer_state_to = FRA
            }
        }
    }
    # Belgian administration follows the British wartime handover in the bookmarks.
    if = {
        limit = { BEL = { exists = yes NOT = { has_war_with = GER } } }
        768 = { if = { limit = { OR = { is_owned_by = GER is_owned_by = ENG } } transfer_state_to = BEL } }
        769 = { if = { limit = { OR = { is_owned_by = GER is_owned_by = ENG } } transfer_state_to = BEL } }
    }
}

WW1_TRC_finish_caucasus_breakup = {
    if = {
        limit = {
            has_start_date < 1920.1.1
            NOT = { date < 1919.2.6 }
            date < 1922.1.1
            has_global_flag = treaty_of_brest_litovsk
            exists = yes
            NOT = { owns_state = 231 }
            NOT = { owns_state = 229 }
            NOT = { owns_state = 230 }
            NOT = { owns_state = 990 }
            NOT = { owns_state = 1016 }
            owns_state = 999
            country_exists = GEO
            999 = { is_controlled_by = TRC }
        }
        if = {
            limit = {
                country_exists = WHR
                WHR = { has_capitulated = no }
                999 = { any_neighbor_state = { is_controlled_by = WHR } }
            }
            999 = { transfer_state_to = WHR }
        }
        else_if = {
            limit = { country_exists = SOV }
            999 = { transfer_state_to = SOV }
        }
    }
}
