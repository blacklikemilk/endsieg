# Compatibility cleanup for retired Italian WWI food, social-tension and debt systems.
# Public-service and diplomatic decisions use ordinary country effects.
IT1_ww1_initialize_administration = {
	hidden_effect = {
		clear_variable = IT1_ww1_food_security
		clear_variable = IT1_ww1_food_monthly_change
		clear_variable = IT1_ww1_social_tension
		clear_variable = IT1_ww1_tension_monthly_change
		clear_variable = IT1_ww1_debt
		clear_variable = IT1_ww1_debt_monthly_change
		clr_country_flag = IT1_ww1_administration_initialized
		clr_country_flag = IT1_ww1_treasury_programme_active
		if = { limit = { has_idea = IT1_ww1_food_shortage } remove_ideas = IT1_ww1_food_shortage }
		if = { limit = { has_idea = IT1_ww1_food_crisis } remove_ideas = IT1_ww1_food_crisis }
		if = { limit = { has_idea = IT1_ww1_social_unrest } remove_ideas = IT1_ww1_social_unrest }
		if = { limit = { has_idea = IT1_ww1_social_crisis } remove_ideas = IT1_ww1_social_crisis }
	}
}

# Expire remaining WWI public-service programmes at the 1936 handoff.
IT1_ww1_end_administration = {
	hidden_effect = {
		IT1_ww1_initialize_administration = yes
		if = { limit = { has_idea = IT1_ww1_rationing } remove_ideas = IT1_ww1_rationing }
		if = { limit = { has_idea = IT1_ww1_strike_agreement } remove_ideas = IT1_ww1_strike_agreement }
		if = { limit = { has_idea = IT1_ww1_colonial_constabulary } remove_ideas = IT1_ww1_colonial_constabulary }
		if = { limit = { has_idea = IT1_ww1_colonial_investment } remove_ideas = IT1_ww1_colonial_investment }
		if = { limit = { has_idea = IT1_ww1_neutral_trade } remove_ideas = IT1_ww1_neutral_trade }
		if = { limit = { has_idea = IT1_ww1_mediation_mission } remove_ideas = IT1_ww1_mediation_mission }
		clr_country_flag = IT1_ww1_home_front_programme_active
		clr_country_flag = IT1_ww1_colonial_programme_active
		clr_country_flag = IT1_ww1_diplomatic_programme_active
	}
}

# The declaration represents the initial October 1911 landings together. Only
# the two harbour provinces change wartime controller; Ottoman ownership and
# the inland campaign remain intact. The legacy event and declaration share
# one flag, so retries or repeated event calls cannot duplicate the expedition.
IT1_ww1_deploy_libyan_expedition = {
	if = {
		limit = {
			tag = ITA
			date < 1936.1.1
			has_country_flag = IT1_ww1_libyan_campaign_started
			has_war_with = TUR
			NOT = { has_country_flag = IT1_ww1_italian_libya_oob_loaded }
			NOT = { has_country_flag = IT1_ww1_ottoman_accommodation }
			NOT = { has_global_flag = treaty_of_ouchy }
			448 = { OR = { is_owned_by = TUR is_owned_by = ITA } }
			450 = { OR = { is_owned_by = TUR is_owned_by = ITA } }
			OR = { controls_province = 1149 TUR = { controls_province = 1149 } }
			OR = { controls_province = 11954 TUR = { controls_province = 11954 } }
		}
		set_country_flag = IT1_ww1_italian_libya_oob_loaded
		TUR = {
			if = {
				limit = { NOT = { has_country_flag = IT1_ww1_ottoman_libya_oob_loaded } }
				load_oob = TUR_Libya
				set_country_flag = IT1_ww1_ottoman_libya_oob_loaded
			}
		}
		set_province_controller = 1149 # Tripoli harbour, state 448
		set_province_controller = 11954 # Benghazi harbour, state 450
		load_oob = ITA_Libya
	}
}
