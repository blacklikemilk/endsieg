# Starting hulls and guns were separate research grants in many early histories.
WW1_sync_baseline_naval_armament = {
	if = {
		limit = { has_start_date < 1936.1.1 date < 1936.1.1 }
		if = {
			limit = {
				NOT = { has_tech = ww1_basic_battery }
				OR = {
					has_tech = ww1_early_ship_hull_light
					has_tech = ww1_basic_ship_hull_light
					has_tech = ww1_improved_ship_hull_light
					has_tech = ww1_advanced_ship_hull_light
					has_tech = ww1_early_ship_hull_cruiser
					has_tech = ww1_basic_ship_hull_cruiser
					has_tech = ww1_improved_ship_hull_cruiser
					has_tech = ww1_advanced_ship_hull_cruiser
					has_tech = ww1_early_ship_hull_heavy
					has_tech = ww1_basic_ship_hull_heavy
					has_tech = ww1_improved_ship_hull_heavy
					has_tech = ww1_advanced_ship_hull_heavy
				}
			}
			set_technology = { ww1_basic_battery = 1 popup = no }
		}
		if = {
			limit = {
				NOT = { has_tech = ww1_basic_torpedo }
				OR = {
					has_tech = ww1_early_ship_hull_light
					has_tech = ww1_basic_ship_hull_light
					has_tech = ww1_improved_ship_hull_light
					has_tech = ww1_advanced_ship_hull_light
				}
			}
			set_technology = { ww1_basic_torpedo = 1 popup = no }
		}
	}
}
