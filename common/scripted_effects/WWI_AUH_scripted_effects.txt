# Compatibility entry points: food is retained; the former stability tiers,
# three decision categories and monthly stability drip have been replaced.
AUH_ww1_clamp_administration = { clamp_variable = { var = AUH_ww1_food_security min = 0 max = 100 } }
AUH_ww1_initialize_administration = {
 if = { limit = { AUH_stab_active = yes }
  if = { limit = { NOT = { has_variable = AUH_ww1_food_security } }
   if = { limit = { date > 1917.12.31 } set_variable = { AUH_ww1_food_security = 20 } }
   else_if = { limit = { date > 1915.12.31 } set_variable = { AUH_ww1_food_security = 35 } }
   else_if = { limit = { date > 1913.12.31 } set_variable = { AUH_ww1_food_security = 60 } }
   else = { set_variable = { AUH_ww1_food_security = 75 } }
  }
  if = { limit = { has_completed_focus = AUH_ww1_constitutional_commission } set_country_flag = AUH_ww1_constitutional_process }
  set_country_flag = AUH_ww1_administration_initialized
  AUH_stab_initialize = yes
 }
 else = { AUH_ww1_end_administration = yes }
}
AUH_ww1_refresh_administration = {
 if = { limit = { AUH_stab_active = yes }
  AUH_stab_refresh = yes
 }
 else = { AUH_ww1_end_administration = yes }
}
AUH_ww1_monthly_administration = {
 if = { limit = { AUH_stab_active = yes }
  AUH_ww1_initialize_administration = yes
  set_variable = { AUH_ww1_food_monthly_change = 1 }
  if = { limit = { has_war = yes }
   set_variable = { AUH_ww1_food_monthly_change = -2 }
   if = { limit = { has_completed_focus = AUH_ww1_hungarian_grain_agreement } set_variable = { AUH_ww1_food_monthly_change = -1 } }
  }
  add_to_variable = { AUH_ww1_food_security = AUH_ww1_food_monthly_change }
  AUH_ww1_clamp_administration = yes
  AUH_stab_monthly = yes
  AUH_ww1_refresh_administration = yes
 }
 else = { AUH_ww1_end_administration = yes }
}
AUH_ww1_end_administration = {
 AUH_stab_suspend = yes
 AUH_stab_retire_old_system = yes
 remove_ideas = AUH_ww1_food_strained remove_ideas = AUH_ww1_food_crisis
 clr_country_flag = AUH_ww1_administration_initialized
}
