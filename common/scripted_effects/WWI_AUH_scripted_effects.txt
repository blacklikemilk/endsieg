# AUH uses HOI4's native stability and a single 0..100 food-security index.
# The stability variable below is a monthly delta, never a second stability score.
AUH_ww1_clamp_administration = {
	clamp_variable = { var = AUH_ww1_food_security min = 0 max = 100 }
}

AUH_ww1_initialize_administration = {
	if = {
		limit = { tag = AUH date < 1936.1.1 }
		# Remove obsolete save variables without reading or converting their values.
		clear_variable = AUH_ww1_debt
		clear_variable = AUH_ww1_cohesion
		clear_variable = AUH_ww1_debt_monthly_change
		clear_variable = AUH_ww1_cohesion_monthly_change
		clr_country_flag = AUH_ww1_treasury_programme_active
		# History completes focuses without executing their rewards. Seed the convention
		# mandate for late bookmarks so their regional-negotiation focuses can fire.
		if = {
			limit = { has_completed_focus = AUH_ww1_constitutional_commission }
			set_country_flag = AUH_ww1_constitutional_process
		}
		if = {
			limit = { NOT = { has_country_flag = AUH_ww1_administration_initialized } }
			if = {
				limit = { NOT = { has_variable = AUH_ww1_food_security } }
				if = {
					limit = { date > 1917.12.31 }
					set_variable = { AUH_ww1_food_security = 20 }
				}
				else_if = {
					limit = { date > 1915.12.31 }
					set_variable = { AUH_ww1_food_security = 35 }
				}
				else_if = {
					limit = { date > 1913.12.31 }
					set_variable = { AUH_ww1_food_security = 60 }
				}
				else = { set_variable = { AUH_ww1_food_security = 75 } }
			}
			set_country_flag = AUH_ww1_administration_initialized
		}
	}
	AUH_ww1_refresh_administration = yes
}

AUH_ww1_refresh_administration = {
	if = {
		limit = { tag = AUH date < 1936.1.1 has_country_flag = AUH_ww1_administration_initialized }
		AUH_ww1_clamp_administration = yes
		# Native stability includes the game's existing modifiers and wartime penalty.
		# These tiers must not themselves modify stability_factor.
		if = {
			limit = { NOT = { has_stability < 0.60 } }
			remove_ideas = AUH_ww1_strained_institutions
			remove_ideas = AUH_ww1_fragile_institutions
			remove_ideas = AUH_ww1_fragmented_institutions
			if = {
				limit = { NOT = { has_idea = AUH_ww1_stable_institutions } }
				add_ideas = AUH_ww1_stable_institutions
			}
		}
		else_if = {
			limit = { NOT = { has_stability < 0.40 } }
			remove_ideas = AUH_ww1_stable_institutions
			remove_ideas = AUH_ww1_fragile_institutions
			remove_ideas = AUH_ww1_fragmented_institutions
			if = {
				limit = { NOT = { has_idea = AUH_ww1_strained_institutions } }
				add_ideas = AUH_ww1_strained_institutions
			}
		}
		else_if = {
			limit = { NOT = { has_stability < 0.25 } }
			remove_ideas = AUH_ww1_stable_institutions
			remove_ideas = AUH_ww1_strained_institutions
			remove_ideas = AUH_ww1_fragmented_institutions
			if = {
				limit = { NOT = { has_idea = AUH_ww1_fragile_institutions } }
				add_ideas = AUH_ww1_fragile_institutions
			}
		}
		else = {
			remove_ideas = AUH_ww1_stable_institutions
			remove_ideas = AUH_ww1_strained_institutions
			remove_ideas = AUH_ww1_fragile_institutions
			if = {
				limit = { NOT = { has_idea = AUH_ww1_fragmented_institutions } }
				add_ideas = AUH_ww1_fragmented_institutions
			}
		}
		if = {
			limit = { check_variable = { AUH_ww1_food_security = 20 compare = less_than } }
			remove_ideas = AUH_ww1_food_strained
			if = {
				limit = { NOT = { has_idea = AUH_ww1_food_crisis } }
				add_ideas = AUH_ww1_food_crisis
			}
		}
		else_if = {
			limit = { check_variable = { AUH_ww1_food_security = 40 compare = less_than } }
			remove_ideas = AUH_ww1_food_crisis
			if = {
				limit = { NOT = { has_idea = AUH_ww1_food_strained } }
				add_ideas = AUH_ww1_food_strained
			}
		}
		else = {
			remove_ideas = AUH_ww1_food_strained
			remove_ideas = AUH_ww1_food_crisis
		}
		set_variable = { AUH_ww1_food_monthly_change = 1 }
		set_variable = { AUH_ww1_stability_monthly_change = 0 }
		if = {
			limit = { has_war = yes }
			set_variable = { AUH_ww1_food_monthly_change = -2 }
			set_variable = { AUH_ww1_stability_monthly_change = -0.002 }
			if = {
				limit = { has_completed_focus = AUH_ww1_hungarian_grain_agreement }
				set_variable = { AUH_ww1_food_monthly_change = -1 }
			}
		}
		else_if = {
			limit = { has_stability < 0.50 }
			set_variable = { AUH_ww1_stability_monthly_change = 0.002 }
		}
		if = {
			limit = { check_variable = { AUH_ww1_food_security = 20 compare = less_than } }
			add_to_variable = { AUH_ww1_stability_monthly_change = -0.003 }
		}
		if = {
			limit = { has_country_flag = AUH_ww1_january_strikes }
			add_to_variable = { AUH_ww1_stability_monthly_change = -0.002 }
		}
		if = {
			limit = { has_country_flag = AUH_ww1_cattaro_unrest }
			add_to_variable = { AUH_ww1_stability_monthly_change = -0.002 }
		}
	}
	else = { AUH_ww1_end_administration = yes }
}

AUH_ww1_monthly_administration = {
	if = {
		limit = { tag = AUH date < 1936.1.1 }
		AUH_ww1_initialize_administration = yes
		add_to_variable = { AUH_ww1_food_security = AUH_ww1_food_monthly_change }
		add_stability = var:AUH_ww1_stability_monthly_change
		AUH_ww1_refresh_administration = yes
	}
	else = { AUH_ww1_end_administration = yes }
}
