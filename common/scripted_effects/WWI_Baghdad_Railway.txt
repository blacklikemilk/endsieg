GE1_baghdad_railway_announce = {
	if = {
		limit = {
			has_start_date < 1936.1.1
			date < 1936.1.1
			has_completed_focus = GE1_berlin_baghdad_1
			has_capitulated = no
			TUR = { exists = yes has_capitulated = no }
			NOT = { has_war_with = TUR }
			NOT = { has_country_flag = GE1_baghdad_railway_announced }
		}
		set_country_flag = GE1_baghdad_railway_announced
		GE1_baghdad_railway_reconcile_british_response = yes
	}
}

# A queued event is not an answered proposal. Keep a dated pending receipt
# until Britain answers, then deliver that answer to Germany separately.
GE1_baghdad_railway_reconcile_british_response = {
	if = {
		limit = {
			tag = GER
			has_start_date < 1936.1.1
			date < 1936.1.1
			has_country_flag = GE1_baghdad_railway_announced
			ENG = { exists = yes }
		}
		if = {
			limit = {
				NOT = { has_country_flag = GE1_baghdad_railway_british_response_received }
				OR = {
					is_in_array = { array = endsieg_focus_opinion_baghdad_railway_gulf_safeguards value = ENG }
					ENG = { is_in_array = { array = endsieg_focus_opinion_baghdad_railway_gulf_safeguards value = GER } }
					has_opinion_modifier = baghdad_railway_gulf_safeguards
					ENG = { has_opinion_modifier = baghdad_railway_gulf_safeguards }
				}
			}
			set_country_flag = GE1_baghdad_railway_british_response_received
			set_country_flag = GE1_baghdad_railway_british_accommodation
		}
		if = {
			limit = {
				NOT = { has_country_flag = GE1_baghdad_railway_british_response_received }
				has_country_flag = GE1_baghdad_railway_british_protest
			}
			set_country_flag = GE1_baghdad_railway_british_response_received
			set_country_flag = GE1_baghdad_railway_british_war_response
		}
		if = {
			limit = {
				NOT = { has_country_flag = GE1_baghdad_railway_british_response_received }
				ENG = { has_capitulated = no }
				OR = {
					NOT = { has_country_flag = GE1_baghdad_railway_british_response_pending }
					has_country_flag = { flag = GE1_baghdad_railway_british_response_pending days > 30 }
				}
			}
			clr_country_flag = GE1_baghdad_railway_british_response_pending
			set_country_flag = GE1_baghdad_railway_british_response_pending
			set_country_flag = GE1_baghdad_railway_reaction_attempted
			ENG = { country_event = { id = ww1_baghdad.1 days = 1 } }
		}
		if = {
			limit = {
				has_country_flag = GE1_baghdad_railway_british_response_received
				NOT = { has_country_flag = GE1_baghdad_railway_british_response_notified }
				OR = {
					NOT = { has_country_flag = GE1_baghdad_railway_british_reply_pending }
					has_country_flag = { flag = GE1_baghdad_railway_british_reply_pending days > 30 }
				}
			}
			clr_country_flag = GE1_baghdad_railway_british_reply_pending
			set_country_flag = GE1_baghdad_railway_british_reply_pending
			country_event = { id = ww1_baghdad.2 hours = 1 }
		}
	}
}

GE1_baghdad_railway_advance_construction = {
	if = {
		limit = {
			tag = GER
			has_country_flag = GE1_baghdad_railway_announced

			OR = {
				has_country_flag = GE1_baghdad_railway_british_response_notified
				ENG = { OR = { exists = no has_capitulated = yes } }
			}
		}
		if = {
			limit = { NOT = { has_country_flag = GE1_baghdad_railway_anatolian_approaches_done } }
			if = {
				limit = { has_country_flag = GE1_baghdad_railway_anatolian_approaches_building }
				if = {
					limit = { GE1_baghdad_railway_anatolian_approaches_secured = no }
					clr_country_flag = GE1_baghdad_railway_anatolian_approaches_building
					remove_ideas = GE1_baghdad_railway_works_1
				}
				else_if = {
					limit = { has_country_flag = { flag = GE1_baghdad_railway_anatolian_approaches_building days > 89 } }

# Country scope: Germany. Also called by the central daily WW1 reconciler.
# Each transition records its receipt before queuing a popup or spending PP.
GE1_baghdad_railway_tick = {
 if = {
  limit = { tag = GER has_start_date < 1936.1.1 date < 1936.1.1 has_completed_focus = GE1_berlin_baghdad_1 }
  if = { limit = { NOT = { has_country_flag = GE1_baghdad_railway_announced } } GE1_baghdad_railway_announce = yes }
  GE1_baghdad_railway_reconcile_british_response = yes
  # An unanswered construction popup must not strand an existing save forever.
  # A second stale popup cannot charge twice or restart an active works timer.
  if = {
   limit = { has_country_flag = { flag = GE1_baghdad_railway_event_pending days > 30 } }
   clr_country_flag = GE1_baghdad_railway_event_pending
  }
  GE1_baghdad_railway_advance_construction = yes
 }
}
