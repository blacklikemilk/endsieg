# A small industrial base adjustment for WWI starts and existing campaigns.
# Each factory is granted once and only where the empire still controls a core.
WW1_TUR_initialize_industry = {
 if = {
  limit = { tag = TUR has_focus_tree = turkish_focus_ww1 has_start_date < 1919.1.1 date < 1919.1.1 NOT = { has_country_flag = WW1_TUR_civilian_industry_revised } 340 = { is_owned_by = ROOT is_fully_controlled_by = ROOT is_core_of = ROOT } }
  340 = { add_extra_state_shared_building_slots = 1 add_building_construction = { type = industrial_complex level = 1 instant_build = yes } }
  set_country_flag = WW1_TUR_civilian_industry_revised
 }
 if = {
  limit = { tag = TUR has_focus_tree = turkish_focus_ww1 has_start_date < 1919.1.1 date < 1919.1.1 NOT = { has_country_flag = WW1_TUR_military_industry_revised } 797 = { is_owned_by = ROOT is_fully_controlled_by = ROOT is_core_of = ROOT } }
  797 = { add_extra_state_shared_building_slots = 1 add_building_construction = { type = arms_factory level = 1 instant_build = yes } }
  set_country_flag = WW1_TUR_military_industry_revised
 }
}
