# Country scope. The outbreak is recorded once; small final rounding is below 0.2 percentage points.
WW1_raise_outbreak_tension = {
 if = {
  limit = { has_start_date < 1919.1.1 date < 1919.1.1 NOT = { has_global_flag = WW1_world_tension_outbreak_recorded } }
  set_global_flag = WW1_world_tension_outbreak_recorded
  if = { limit = { threat < 0.5 } add_named_threat = { threat = 50 name = WW1_tension_great_war } }
  if = { limit = { threat < 0.75 } add_named_threat = { threat = 25 name = WW1_tension_great_war } }
  if = { limit = { threat < 0.875 } add_named_threat = { threat = 12.5 name = WW1_tension_great_war } }
  if = { limit = { threat < 0.9375 } add_named_threat = { threat = 6.25 name = WW1_tension_great_war } }
  if = { limit = { threat < 0.96875 } add_named_threat = { threat = 3.125 name = WW1_tension_great_war } }
  if = { limit = { threat < 0.984375 } add_named_threat = { threat = 1.5625 name = WW1_tension_great_war } }
  if = { limit = { threat < 0.9921875 } add_named_threat = { threat = 0.78125 name = WW1_tension_great_war } }
  if = { limit = { threat < 0.99609375 } add_named_threat = { threat = 0.390625 name = WW1_tension_great_war } }
  if = { limit = { threat < 0.998046875 } add_named_threat = { threat = 0.1953125 name = WW1_tension_great_war } }
  if = { limit = { threat < 1 } add_named_threat = { threat = 0.2 name = WW1_tension_great_war } }
 }
}

# Save/bookmark reconciliation applies only to a war already underway.
WW1_reconcile_outbreak_tension = {
 if = {
  limit = { tag = GER has_start_date < 1919.1.1 date < 1919.1.1 NOT = { has_global_flag = WW1_world_tension_outbreak_recorded } NOT = { has_global_flag = compiegne } NOT = { has_global_flag = central_powers_victory } OR = { AND = { has_global_flag = WW1_starts country_exists = AUH country_exists = SER AUH = { has_war_with = SER } } AND = { country_exists = FRA has_war_with = FRA country_exists = RIE has_war_with = RIE } } }
  WW1_raise_outbreak_tension = yes
 }
}

# Policy receipts keep these small pre-war entries independent of the outbreak.
WW1_record_french_three_year_law_tension = {
 if = {
  limit = {
   tag = FRA has_start_date < 1913.8.7 date > 1913.8.6 date < 1914.7.28
   has_global_flag = welcome_event_WW1_active
   NOT = { has_global_flag = WW1_world_tension_outbreak_recorded }
   NOT = { has_global_flag = compiegne }
   NOT = { has_global_flag = versailles }
   NOT = { has_global_flag = central_powers_victory }
   NOT = { has_country_flag = WW1_tension_three_year_law_recorded }
   has_completed_focus = FR3_three_year_law
   has_country_flag = FR3_R_three_year_law
   is_subject = no has_capitulated = no
   country_exists = GER
   NOT = { has_war_with = GER }
   NOT = { is_in_faction_with = GER }
   GER = { has_capitulated = no }
  }
  set_country_flag = WW1_tension_three_year_law_recorded
  add_named_threat = { threat = 1 name = WW1_tension_three_year_law }
  # France's enactment event already broadcasts ww1_france.223.
 }
}
