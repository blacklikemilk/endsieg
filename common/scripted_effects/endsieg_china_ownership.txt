# Overview: repairs ownership of the Chinese states when the dated scenario history
# hands them to another tag while a continuous campaign is running.
# ROOT is the country that should hold China. Called from the Qing and Republic ticks.

endsieg_china_reclaim_states = {
	every_state = {
		limit = {
			OR = {
				state = 609
				state = 743
				AND = {
					is_on_continent = asia
					OR = {
						is_owned_by = CHI
						is_owned_by = RCH
						is_owned_by = YUN
						is_owned_by = GXC
						is_owned_by = XSM
						is_owned_by = SIK
						is_owned_by = SHX
						is_owned_by = MAN
						is_owned_by = PRC
						is_owned_by = SND
					}
				}
			}
		}
		set_state_owner_to = ROOT
		set_state_controller_to = ROOT
		add_core_of = ROOT
	}
}
