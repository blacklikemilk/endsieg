# The campaign command's wartime presence accrued no title to Persian territory.
WWI_end_persian_occupation = {
	if = {
		limit = { country_exists = PER NOT = { PER = { has_war_with = ENG } } }
		every_state = {
			limit = {
				OR = { state = 410 state = 412 state = 413 state = 1058 }
				OR = { is_owned_by = ENG is_owned_by = EN3 }
				OR = { is_controlled_by = ENG is_controlled_by = EN3 is_controlled_by = PER }
			}
			remove_core_of = EN3
			transfer_state_to = PER
		}
	}
}

# State scope. Mark each temporary military administration so evacuation never
# changes an unrelated conquest. Keep Turkish cores throughout.
WWI_open_occupation_district = {
	if = {
		limit = {
			is_owned_by = TUR
			WWI_occupation_district_available = { ally = $ally$ }
		}
		set_state_flag = WWI_anatolian_occupation_district
		transfer_state_to = $ally$
	}
}

WWI_return_occupation_district = {
	if = {
		limit = { has_state_flag = WWI_anatolian_occupation_district is_owned_by = $ally$ WWI_occupation_district_available = { ally = $ally$ } }
		transfer_state_to = TUR
		clr_state_flag = WWI_anatolian_occupation_district
	}
}

# Reconcile earlier saves whose initial events recorded just the port garrisons.
WWI_update_anatolian_occupation = {
	if = {
		limit = {
			has_country_flag = WWI_constantinople_allied_garrison
			ENG = { exists = yes has_capitulated = no NOT = { has_war_with = TUR } }
		}
		if = {
			limit = { date < 1923.10.2 }
			797 = { WWI_open_occupation_district = { ally = ENG } }
			# Both shores are the map's Dardanelles strait provinces.
			if = { limit = { owns_state = 341 controls_province = 849 } ENG = { set_province_controller = 849 } }
			if = { limit = { owns_state = 340 controls_province = 6864 } ENG = { set_province_controller = 6864 } }
		}
	}
	if = {
		limit = {
			has_country_flag = WWI_italian_antalya_landing date < 1921.7.5
			ITA = { exists = yes has_capitulated = no NOT = { has_war_with = TUR } }
		}
		342 = { WWI_open_occupation_district = { ally = ITA } }
	}
	if = {
		limit = {
			FRA = { exists = yes has_capitulated = no NOT = { has_war_with = TUR } }
			OR = { has_country_flag = WWI_cilician_garrisons has_country_flag = INT_TUR_ankara_accord_signed }
		}
		if = {
			limit = { date < 1922.1.5 has_country_flag = WWI_mudros_french_participant }
			344 = { WWI_open_occupation_district = { ally = FRA } }
			345 = { WWI_open_occupation_district = { ally = FRA } }
			if = {
				limit = { 344 = { is_owned_by = FRA has_state_flag = WWI_anatolian_occupation_district } }
				# Malatya remained outside the occupation. Maraş expelled its garrison in February 1920.
				if = { limit = { FRA = { controls_province = 7620 } } set_province_controller = 7620 }
				if = { limit = { OR = { date < 1919.2.22 date > 1920.2.11 } FRA = { controls_province = 6813 } } set_province_controller = 6813 }
				if = { limit = { date > 1921.12.24 FRA = { controls_province = 9893 } } set_province_controller = 9893 }
				if = {
					limit = { has_country_flag = WWI_mudros_british_participant ENG = { exists = yes has_capitulated = no NOT = { has_war_with = TUR } NOT = { has_war_with = FRA } } }
					if = {
						limit = { date < 1919.10.29 }
						if = { limit = { date > 1918.12.16 FRA = { controls_province = 9893 } } ENG = { set_province_controller = 9893 } }
						if = { limit = { date > 1919.2.21 OR = { controls_province = 6813 FRA = { controls_province = 6813 } } } ENG = { set_province_controller = 6813 } }
					}
					else = {
						if = { limit = { ENG = { controls_province = 9893 } } FRA = { set_province_controller = 9893 } }
						if = { limit = { ENG = { controls_province = 6813 } } FRA = { set_province_controller = 6813 } }
					}
				}
			}
		}
		else = {
			344 = { WWI_return_occupation_district = { ally = FRA } }
			345 = { WWI_return_occupation_district = { ally = FRA } }
			if = { limit = { has_country_flag = WWI_cilician_garrisons } WWI_evacuate_cilician_garrisons = yes }
		}
	}
}
