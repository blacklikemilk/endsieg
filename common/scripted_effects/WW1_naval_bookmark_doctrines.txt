# Replace inherited WWII doctrine even where the country starts without ships.
endsieg_initialize_ww1_naval_doctrine = {
	if = {
		limit = {
			OR = { has_navy_size = { size > 0 } has_any_grand_doctrine = naval }
		}
		set_grand_doctrine = endsieg_great_war_naval
	}
	if = {
		limit = {
			OR = {
				has_navy_size = { unit = battleship size > 0 }
				has_navy_size = { unit = battle_cruiser size > 0 }
				has_navy_size = { unit = heavy_cruiser size > 0 }
			}
		}
		if = {
			limit = { OR = { has_tech = trade_interdiction2 has_tech = indirect_intervention } }
			set_sub_doctrine = endsieg_ww1_naval_raiders
		}
		else = { set_sub_doctrine = endsieg_ww1_naval_line }
		if = {
			limit = { endsieg_ww1_bookmark_service_veteran = yes }
			if = {
				limit = { endsieg_ww1_bookmark_long_service = yes }
				add_mastery = { amount = 300 track = endsieg_ww1_naval_capitals }
			}
			else = { add_mastery = { amount = 150 track = endsieg_ww1_naval_capitals } }
		}
	}
	if = {
		limit = {
			OR = {
				has_navy_size = { unit = destroyer size > 0 }
				has_navy_size = { unit = light_cruiser size > 0 }
			}
		}
		if = {
			limit = { OR = { has_tech = escort_sub original_tag = ENG original_tag = FRA } }
			set_sub_doctrine = endsieg_ww1_naval_escort
		}
		else = { set_sub_doctrine = endsieg_ww1_naval_torpedo }
		if = {
			limit = { endsieg_ww1_bookmark_service_veteran = yes }
			if = {
				limit = { endsieg_ww1_bookmark_long_service = yes }
				add_mastery = { amount = 275 track = endsieg_ww1_naval_screens }
			}
			else = { add_mastery = { amount = 125 track = endsieg_ww1_naval_screens } }
		}
	}
	if = {
		limit = { has_navy_size = { unit = submarine size > 0 } }
		if = {
			limit = { OR = { has_tech = convoy_raiding original_tag = GER } }
			set_sub_doctrine = endsieg_ww1_naval_undersea
		}
		else = { set_sub_doctrine = endsieg_ww1_naval_coastal }
		if = {
			limit = { endsieg_ww1_bookmark_service_veteran = yes }
			if = {
				limit = { endsieg_ww1_bookmark_long_service = yes }
				add_mastery = { amount = 225 track = endsieg_ww1_naval_submarines }
			}
			else = { add_mastery = { amount = 100 track = endsieg_ww1_naval_submarines } }
		}
	}
}
