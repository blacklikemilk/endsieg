MEX_WW1_offer_central_convention = {
 if = { limit = { has_country_flag = MEX_WW1_ZT_received } MEX_WW1_ZT_request_guarantees = yes }
 else = {

 if = {
  limit = {
   MEX_WW1_diplomacy_allowed = yes
   NOT = { has_completed_focus = MEX_WW1_armed_neutrality }
   NOT = { OR = { has_completed_focus = MEX_WW1_ratify_entente has_country_flag = MEX_WW1_ratifying_entente_focus } }
   NOT = { has_country_flag = MEX_WW1_central_offer_pending }
  }
  set_country_flag = MEX_WW1_central_offer_pending
  clr_country_flag = MEX_WW1_central_consent
  clr_country_flag = MEX_WW1_central_refused
  clr_country_flag = MEX_WW1_central_withdrawn
  if = {
   limit = { MEX_WW1_central_host_available = yes }
   GER = { country_event = { id = mex_ww1_diplomacy.1 days = 1 } }
   country_event = { id = mex_ww1_diplomacy.101 days = 30 }
  }
  else = {
   clr_country_flag = MEX_WW1_central_offer_pending
   set_country_flag = MEX_WW1_central_withdrawn
   add_political_power = 10
   country_event = { id = mex_ww1_diplomacy.11 days = 1 }
  }
 }

 }
}

MEX_WW1_ratify_central_convention = {
 if = {
  limit = {
  has_country_flag = MEX_WW1_ZT_terms_accepted

   MEX_WW1_central_host_available = yes
   OR = { has_completed_focus = MEX_WW1_ratify_central has_country_flag = MEX_WW1_ratifying_central_focus }
   has_country_flag = MEX_WW1_central_consent
  }
  GER = { add_to_faction = ROOT }
  if = {
   limit = { is_in_faction_with = GER }
   set_country_flag = MEX_WW1_central_alignment
   clr_country_flag = MEX_WW1_central_ratification_pending
   add_opinion_modifier = { target = GER modifier = germany_friend }
  }
  else = {
   set_country_flag = MEX_WW1_central_ratification_pending
   country_event = { id = mex_ww1_diplomacy.13 days = 1 }
  }
 }
 else = {
  if = {
   limit = { MEX_WW1_diplomacy_allowed = yes OR = { has_completed_focus = MEX_WW1_ratify_central has_country_flag = MEX_WW1_ratifying_central_focus } }
   set_country_flag = MEX_WW1_central_ratification_pending
   country_event = { id = mex_ww1_diplomacy.13 days = 1 }
  }
 }
}
