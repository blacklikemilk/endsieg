# Existing saves can still have legacy leaders instead of registered characters.
MOR_update_succession = {
	if = {
		limit = { has_government = neutrality has_start_date < 1936.1.1 date < 1936.1.1 }
		if = {
			limit = {
				date < 1912.8.13
				has_country_leader = { name = "Mohammed V" ruling_only = yes }
			}
			if = {
				limit = { has_character = MOR_abd_al_hafid }
				promote_character = MOR_abd_al_hafid
			}
			else = {
				create_country_leader = {
					name = "Abd al Hafid"
					picture = "gfx/leaders/MOR/Portrait_Morocco_Abd_al_Hafid.dds"
					ideology = despotism
				}
			}
		}
		else_if = {
			limit = {
				NOT = { date < 1912.8.13 }
				date < 1927.11.18
				OR = {
					has_country_leader = { name = "Abd al Hafid" ruling_only = yes }
					has_country_leader = { name = "Mohammed V" ruling_only = yes }
				}
			}
			if = {
				limit = { has_character = MOR_moulay_youssef }
				promote_character = MOR_moulay_youssef
			}
			else = {
				create_country_leader = {
					name = "Moulay Youssef"
					picture = "gfx/leaders/MOR/Portrait_Morocco_Moulay_Youssef.dds"
					ideology = despotism
				}
			}
		}
		else_if = {
			limit = {
				NOT = { date < 1927.11.18 }
				has_country_leader = { name = "Moulay Youssef" ruling_only = yes }
			}
			if = {
				limit = { has_character = MOR_mohammed_v }
				promote_character = MOR_mohammed_v
			}
			else = {
				create_country_leader = {
					name = "Mohammed V"
					picture = "GFX_portrait_Mohammed_V"
					ideology = despotism
				}
			}
		}
	}
}
