# Public offer effects run in GER scope from the corresponding focus reward.
# Each focus may start its contact once; accepted agreements are finalized in
# Germany after a real partner reply. No faction, war or territorial effects.

GW_D_offer_mexican_reply = {
    if = {
        limit = {
            tag = GER
            has_start_date < 1936.1.1
            date < 1936.1.1
            NOT = { has_country_flag = GW_D_mexican_reply_started }
        }
        set_country_flag = GW_D_mexican_reply_started
        set_country_flag = GW_D_mexican_reply_pending
        if = {
            limit = {
                has_start_date < 1936.1.1
                date < 1936.1.1
                NOT = { has_capitulated = yes }
                country_exists = MEX
                NOT = { has_war_with = MEX }
                MEX = { NOT = { has_capitulated = yes } NOT = { has_war_with = GER } }
            }
            MEX = { country_event = { id = ww1_world_contacts.1 days = 1 } }
            country_event = { id = ww1_world_contacts.101 days = 30 }
        }
        else = {
            set_country_flag = GW_D_mexican_reply_answered
            set_country_flag = GW_D_mexican_reply_withdrawn
            country_event = { id = ww1_world_contacts.2 hours = 1 }
        }
    }
}
