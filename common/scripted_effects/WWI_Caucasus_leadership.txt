# The split may activate dormant tags without promoting their republican leaders.
# Run in each republic after the transfer and once for eligible existing saves.
WW1_Caucasus_initialize_georgia = {
	if = {
		limit = {
			tag = GEO
			date > 1918.5.25 date < 1921.3.18
			owns_state = 231 controls_province = 1599
			OR = { has_country_flag = transcauc_georgia_separated has_country_flag = WW1_Caucasus_republic_from_transcaucasia }
			NOT = { has_country_flag = WW1_GEO_independence_army_initialized }
		}
		set_country_flag = WW1_GEO_independence_army_initialized
		if = { limit = { num_divisions < 1 } load_oob = GEO_1918_independence }
	}
	if = {
		limit = {
			tag = GEO
			exists = yes
			date > 1918.5.25
			date < 1921.3.18
			has_government = democratic
			owns_state = 231
			controls_state = 231
			NOT = { has_country_flag = WW1_Caucasus_republican_leader_installed }
			OR = {
				AND = {
					is_subject = no
					OR = {
						has_country_flag = transcauc_georgia_separated
						has_country_flag = WW1_Caucasus_republic_from_transcaucasia
						has_global_flag = treaty_of_brest_litovsk
						NOT = { has_start_date < 1918.5.26 }
					}
				}
				AND = {
					is_subject_of = GER
					has_country_flag = transcauc_georgia_separated
				}
			}
		}
		if = {
			limit = { has_character = GEO_noe_zhordania }
			promote_character = { character = GEO_noe_zhordania ideology = liberalism }
		}
		else = {
			create_country_leader = {
				name = "Noe Zhordania"
				picture = "gfx/leaders/GEO/portrait_GEO_noe_zhordania.dds"
				ideology = liberalism
				expire = "1953.1.11"
			}
		}
		set_country_flag = WW1_Caucasus_republican_leader_installed
	}
}

# Government succession for independent republics created by the 1918 split.
WW1_Caucasus_update_georgian_cabinet = {
	if = {
		limit = {
			has_country_flag = WW1_Caucasus_republican_leader_installed
			has_government = democratic
			is_subject = no
			date > 1918.5.25
			date < 1918.6.24
			NOT = { has_country_flag = WW1_GEO_ramishvili_installed }
		}
		set_country_flag = WW1_GEO_ramishvili_installed
		create_country_leader = {
			name = "Noe Ramishvili"
			picture = "gfx/leaders/GEO/Portrait_Georgia_Noe_Ramishvili.dds"
			ideology = liberalism
			expire = "1930.12.7"
		}
	}
	if = {
		limit = {
			has_country_flag = WW1_GEO_ramishvili_installed
			has_government = democratic
			is_subject = no
			date > 1918.6.23
			date < 1921.3.18
		}
		clr_country_flag = WW1_GEO_ramishvili_installed
		clr_country_flag = WW1_Caucasus_republican_leader_installed
		WW1_Caucasus_initialize_georgia = yes
	}
}

WW1_Caucasus_initialize_baku_commune = {
	set_capital = { state = 1016 remember_old_capital = no }
	set_popularities = { fascism = 0 neutrality = 0 democratic = 0 communism = 100 }
	set_politics = { ruling_party = communism elections_allowed = no }
	WW1_Caucasus_update_baku_government = yes
}

WW1_Caucasus_update_baku_government = {
	if = {
		limit = {
			tag = BAK
			exists = yes
			has_start_date < 1919.1.1
			date > 1918.4.24
			date < 1922.1.1
			1016 = { is_owned_by = BAK is_fully_controlled_by = BAK }
		}
		if = {
			limit = {
				has_government = communism
				NOT = { has_country_flag = WW1_BAK_directorate_installed }
				NOT = { has_country_leader = { character = BAK_stepan_shahumyan ruling_only = yes } }
				OR = {
					date < 1918.8.1
					NOT = { has_country_flag = WW1_BAK_shaumyan_installed }
				}
			}
			if = {
				limit = { has_character = BAK_stepan_shahumyan }
				promote_character = { character = BAK_stepan_shahumyan ideology = leninism }
			}
			else_if = {
				limit = { NOT = { has_country_leader = { name = "Stepan Shaumyan" ruling_only = yes } } }
				create_country_leader = {
					name = "Stepan Shaumyan"
					picture = "gfx/leaders/BAK/BAK_stepan_shaumianv.png"
					ideology = leninism
					expire = "1965.1.1"
				}
			}
			set_country_flag = WW1_BAK_shaumyan_installed
		}
		if = {
			limit = {
				has_government = neutrality
				has_country_flag = WW1_BAK_directorate_installed
				NOT = { has_country_flag = WW1_BAK_directorate_cabinet_checked }
				NOT = { has_country_leader = { character = BAK_ww1_directorate ruling_only = yes } }
			}
			if = {
				limit = { has_character = BAK_ww1_directorate }
				promote_character = { character = BAK_ww1_directorate ideology = despotism }
			}
			else_if = {
				limit = { NOT = { has_country_leader = { name = "Central Caspian Directorate" ruling_only = yes } } }
				create_country_leader = {
					name = "Central Caspian Directorate"
					desc = BAK_ww1_directorate_desc
					picture = "gfx/leaders/BAK/Portrait_BAK_Central_Caspian_Directorate.dds"
					ideology = despotism
					expire = "1965.1.1"
				}
			}
			set_party_name = { ideology = neutrality name = BAK_ww1_directorate_party long_name = BAK_ww1_directorate_party }
			set_country_flag = WW1_BAK_directorate_cabinet_checked
		}
	}
}

# The provisional government fell in September; inherit only the idle historical city-state.
WW1_Caucasus_finish_baku_transition = {
	if = {
		limit = {
			tag = BAK
			exists = yes
			is_ai = yes
			has_start_date < 1919.1.1
			date > 1918.9.16
			date < 1922.1.1
			has_global_flag = welcome_event_WW1_active
			OR = { is_historical_focus_on = yes has_global_flag = compiegne }
			NOT = { has_global_flag = central_powers_victory }
			NOT = { has_global_flag = player_wants_custom_peacedeals }
			has_government = neutrality
			has_country_flag = WW1_BAK_directorate_installed
			NOT = { has_country_flag = WW1_BAK_challenged_transcaucasia }
			is_subject = no
			is_in_faction = no
			has_war = no
			NOT = { any_owned_state = { NOT = { state = 1016 } } }
			1016 = { is_owned_by = BAK is_fully_controlled_by = BAK }
			AZR = {
				exists = yes
				is_ai = yes
				is_subject = no
				is_in_faction = no
				has_war = no
				has_government = democratic
				has_country_flag = WW1_Caucasus_republic_from_transcaucasia
			}
			229 = { is_owned_by = AZR is_fully_controlled_by = AZR }
		}
		1016 = { transfer_state_to = AZR }
		AZR = { set_capital = { state = 1016 remember_old_capital = no } }
	}
}
