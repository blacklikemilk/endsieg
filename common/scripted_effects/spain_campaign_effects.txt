# Overview: the Spanish campaign of 1910-1945.
# add_power_balance_value rejects script constants, so the balance nudges use file macros.
@bop_crown_small = -0.05
@bop_crown = -0.1
@bop_crown_large = -0.2
@bop_cortes_small = 0.05
@bop_cortes = 0.1
@bop_cortes_large = 0.2
# Chapter initialisation and transitions, the officer corps, coups, regional control, the Nationalist faction struggle,
# persistent development, wartime diplomacy and reconstruction all live here.
# Every value comes from constant:spain_campaign. Event ids belong to namespace spain_campaign.

# =====================================================================================
# Initialisation
# =====================================================================================

spain_campaign_initialize = {
	spain_ensure_campaign_state = yes
	set_country_flag = spain_campaign_initialized

	if = {
		limit = { date < 1923.9.13 }
		spain_open_restoration_chapter = yes
	}
	else_if = {
		limit = { date < 1931.4.14 }
		spain_open_rivera_chapter = yes
	}
	else_if = {
		limit = { date < 1936.1.1 }
		spain_open_republic_chapter = yes
	}
	else_if = {
		limit = { date < 1939.4.1 }
		spain_open_civil_war_chapter = yes
	}
	else = {
		spain_open_postwar_chapter = yes
	}
}

# Any Spanish country that gains the campaign - a civil-war revolter, or a country created
# by a scripted tag change - gets the same variable set without repeating the chapter choice.
spain_ensure_campaign_state = {
	if = {
		limit = { NOT = { has_variable = spain_military_loyalty } }
		set_variable = { var = spain_military_loyalty value = constant:spain_campaign.loyalty_open }
		set_variable = { var = spain_coup_pressure value = constant:spain_campaign.coup_open }
		set_variable = { var = spain_industry value = constant:spain_campaign.industry_open }
		set_variable = { var = spain_agriculture value = constant:spain_campaign.agriculture_open }
		set_variable = { var = spain_infrastructure value = constant:spain_campaign.infrastructure_open }
		set_variable = { var = spain_education value = constant:spain_campaign.education_open }
		set_variable = { var = spain_catalan_autonomy value = constant:spain_campaign.catalan_autonomy_open }
		set_variable = { var = spain_basque_autonomy value = constant:spain_campaign.basque_autonomy_open }
		set_variable = { var = spain_falange value = constant:spain_campaign.faction_open }
		set_variable = { var = spain_carlists value = constant:spain_campaign.faction_open }
		set_variable = { var = spain_monarchists value = constant:spain_campaign.faction_open }
		set_variable = { var = spain_army_faction value = constant:spain_campaign.faction_open }
		set_variable = { var = spain_nationalist_tension value = 0 }
		# Earned tension is kept apart from the weekly recomputation, so a legacy nudge is not
		# discarded by the next faction refresh.
		set_variable = { var = spain_nationalist_tension_carried value = 0 }
		set_variable = { var = spain_crown_patronage value = constant:spain_campaign.patronage_open }
		set_variable = { var = spain_axis_leverage value = constant:spain_campaign.leverage_open }
		set_variable = { var = spain_allied_leverage value = constant:spain_campaign.leverage_open }
		set_variable = { var = spain_rif_commitment value = constant:spain_campaign.rif_open }
		set_variable = { var = spain_reconstruction value = constant:spain_campaign.reconstruction_open }
		set_variable = { var = spain_tick value = 0 }
	}
	set_country_flag = spain_campaign_initialized
}

# =====================================================================================
# Chapter openers. These only establish state; the milestone events are fired by the
# transitions below, so a bookmark start does not open on a popup.
# =====================================================================================

spain_open_restoration_chapter = {
	set_country_flag = spain_chapter_restoration
	spain_cortes_initialize = yes
	load_focus_tree = { tree = spain_restoration_focus keep_completed = yes }
	mark_focus_tree_layout_dirty = yes
	spain_carry_over_choices = yes
	set_country_flag = spain_bop_pending
	set_country_flag = spain_focus_restore_pending
	
	if = {
		limit = { NOT = { has_country_flag = spain_restoration_settled } }
		set_country_flag = spain_restoration_settled
		add_ideas = spain_caciquismo_system
	}
	spain_refresh_campaign_ideas = yes
}

spain_open_rivera_chapter = {
	set_country_flag = spain_chapter_rivera
	spain_cortes_initialize = yes
	load_focus_tree = { tree = spain_rivera_focus keep_completed = yes }
	mark_focus_tree_layout_dirty = yes
	spain_carry_over_choices = yes
	set_country_flag = spain_bop_pending
	set_country_flag = spain_focus_restore_pending
	
	if = {
		limit = { NOT = { has_country_flag = spain_rivera_settled } }
		set_country_flag = spain_rivera_settled
		if = { limit = { has_idea = spain_caciquismo_system } remove_ideas = spain_caciquismo_system }
		if = { limit = { has_idea = spain_parliamentary_government } remove_ideas = spain_parliamentary_government }
		add_ideas = spain_military_directorate
	}
	spain_refresh_campaign_ideas = yes
}

spain_open_republic_chapter = {
	set_country_flag = spain_chapter_republic
	spain_cortes_initialize = yes
	load_focus_tree = { tree = spain_republic_focus keep_completed = yes }
	mark_focus_tree_layout_dirty = yes
	spain_carry_over_choices = yes
	set_country_flag = spain_bop_pending
	set_country_flag = spain_focus_restore_pending
	
	if = {
		limit = { NOT = { has_country_flag = spain_republic_settled } }
		set_country_flag = spain_republic_settled
		if = { limit = { has_idea = spain_caciquismo_system } remove_ideas = spain_caciquismo_system }
		if = { limit = { has_idea = spain_crown_government } remove_ideas = spain_crown_government }
		if = { limit = { has_idea = spain_military_directorate } remove_ideas = spain_military_directorate }
		add_ideas = spain_parliamentary_government
	}
	spain_refresh_campaign_ideas = yes
}

spain_open_civil_war_chapter = {
	set_country_flag = spain_chapter_civil_war
	load_focus_tree = { tree = spanish_focus keep_completed = yes }
	mark_focus_tree_layout_dirty = yes
	spain_carry_over_choices = yes
	set_country_flag = spain_bop_pending
	if = {
		limit = { has_country_flag = spain_legacy_armed_militias }
		add_equipment_to_stockpile = { type = infantry_equipment amount = constant:spain_campaign.legacy_arms }
	}
	if = {
		limit = { has_country_flag = spain_legacy_parliamentary }
		add_stability = constant:spain_campaign.stability_medium
	}
	if = {
		limit = { has_country_flag = spain_legacy_corporatism }
		add_war_support = constant:spain_campaign.war_support_small
	}
	if = {
		limit = { NOT = { has_country_flag = spain_factions_opened } }
		set_country_flag = spain_factions_opened
		set_variable = { var = spain_falange value = 30 }
		set_variable = { var = spain_carlists value = 20 }
		set_variable = { var = spain_monarchists value = 25 }
		set_variable = { var = spain_army_faction value = 25 }
	}
	spain_refresh_campaign_ideas = yes
}

spain_open_postwar_chapter = {
	set_country_flag = spain_bop_pending
	set_country_flag = spain_chapter_postwar
	spain_carry_over_choices = yes
	spain_cortes_initialize = yes
	if = { limit = { has_power_balance = { id = SPR_crown_and_cortes } } remove_power_balance = { id = SPR_crown_and_cortes } }
	if = {
		limit = { NOT = { has_country_flag = spain_postwar_settled } }
		set_country_flag = spain_postwar_settled
		if = { limit = { NOT = { has_idea = spain_civil_war_reconstruction } } add_ideas = spain_civil_war_reconstruction }
		# A gain earned before 1939 is carried over, not thrown away.
		if = { limit = { NOT = { has_variable = spain_reconstruction } } set_variable = { var = spain_reconstruction value = constant:spain_campaign.reconstruction_open } }
	}
	spain_refresh_campaign_ideas = yes
}

spain_open_republican_postwar_chapter = {
	set_country_flag = spain_chapter_republican_postwar
	if = { limit = { has_power_balance = { id = SPR_crown_and_cortes } } remove_power_balance = { id = SPR_crown_and_cortes } }
	load_focus_tree = { tree = spain_republic_postwar_focus keep_completed = yes }
	mark_focus_tree_layout_dirty = yes
	# The Republic's chamber outlives the war: its seats are what the post-war government legislates with.
	spain_cortes_initialize = yes
	if = {
		limit = { NOT = { has_country_flag = spain_republican_postwar_settled } }
		set_country_flag = spain_republican_postwar_settled
		if = { limit = { NOT = { has_idea = spain_civil_war_reconstruction } } add_ideas = spain_civil_war_reconstruction }
		if = { limit = { NOT = { has_variable = spain_reconstruction } } set_variable = { var = spain_reconstruction value = constant:spain_campaign.reconstruction_open } }
	}
	spain_refresh_campaign_ideas = yes
}

}

spain_enter_postwar_chapter = {
	if = {
		limit = { OR = { has_government = democratic has_government = communism } }
		spain_open_republican_postwar_chapter = yes
	}
	else = {
		spain_open_postwar_chapter = yes
	}
	country_event = { id = spain_campaign.61 days = 1 }
}

spain_campaign_tick = {
	# The civil-war revolter carries the same systems. It is driven from the original
	# country because there is no daily on_action for a tag that does not exist at load.
	if = {
		limit = { country_exists = SPA }
		SPA = {
			if = {
				limit = { NOT = { has_country_flag = spain_campaign_initialized } }
				spain_ensure_campaign_state = yes
			}
		}
	}

	if = {
		limit = { has_country_flag = spain_campaign_initialized }
		if = {
			limit = { has_country_flag = spain_focus_restore_pending }
			clr_country_flag = spain_focus_restore_pending
			spain_campaign_bootstrap = yes
		}
		spain_ensure_chapter_tree = yes
		spain_campaign_transition_check = yes
		add_to_variable = { var = spain_tick value = 1 }
		if = {
			limit = { check_variable = { var = spain_tick value = 7 compare = greater_than_or_equals } }
			set_variable = { var = spain_tick value = 0 }
			spain_campaign_weekly = yes
			if = {
				limit = { country_exists = SPA }
				SPA = { spain_campaign_weekly = yes }
			}
		}
	}
}

spain_campaign_transition_check = {
	# The pronunciamiento. The Restoration cannot survive the year of Annual and the
	# collapse of the turno.
	if = {
		limit = {
			spain_in_restoration_chapter = yes
			date > 1923.9.12
			NOT = { has_country_flag = spain_rivera_crisis_fired }
		}
		set_country_flag = spain_rivera_crisis_fired
		country_event = { id = spain_campaign.10 days = 1 }
	}

	# The dictatorship exhausts itself in 1930.
	if = {
		limit = {
			spain_in_rivera_chapter = yes
			date > 1930.1.27
			NOT = { has_country_flag = spain_rivera_fall_fired }
		}
		set_country_flag = spain_rivera_fall_fired
		country_event = { id = spain_campaign.16 days = 1 }
	}

	# If the monarchy tries to outlast the dictatorship, the Republic arrives anyway.
	if = {
		limit = {
			spain_in_rivera_chapter = yes
			date > 1931.4.13
			NOT = { has_country_flag = spain_republic_proclaimed_fired }
		}
		set_country_flag = spain_republic_proclaimed_fired
		country_event = { id = spain_campaign.20 days = 1 }
	}

	# A campaign played from 1910 reaches the republican decade through its own tree,
	# and switches to the civil-war chapter at the start of 1936.
	if = {
		limit = {
			spain_in_republic_chapter = yes
			date > 1936.1.1
			NOT = { has_country_flag = spain_civil_war_chapter_fired }
		}
		set_country_flag = spain_civil_war_chapter_fired
		country_event = { id = spain_campaign.25 days = 1 }
	}

	# A campaign that never had a civil war still has to enter the post-war decade.
	if = {
		limit = {
			OR = {
				has_country_flag = spain_chapter_civil_war
				has_country_flag = spain_chapter_republic
				has_country_flag = spain_chapter_rivera
				has_country_flag = spain_chapter_restoration
			}
			OR = {
				has_global_flag = scw_over
				date > 1939.3.31
			}
			NOT = { has_country_flag = spain_postwar_chapter_fired }
		}
		set_country_flag = spain_postwar_chapter_fired
		country_event = { id = spain_campaign.60 days = 1 }
	}
}

spain_refresh_campaign_ideas = {
	spain_sync_officer_ideas = yes
	spain_sync_development_tiers = yes
	spain_sync_autonomy_ideas = yes
	spain_sync_rif_ideas = yes
}

spain_sync_officer_ideas = {
	if = {
		limit = { spain_loyalty_high = yes }
		if = { limit = { has_idea = spain_officer_corps_restive } remove_ideas = spain_officer_corps_restive }
		if = { limit = { has_idea = spain_officer_corps_estranged } remove_ideas = spain_officer_corps_estranged }
		add_ideas = spain_officer_corps_loyal
	}
	else_if = {
		limit = { spain_loyalty_low = yes }
		if = { limit = { has_idea = spain_officer_corps_loyal } remove_ideas = spain_officer_corps_loyal }
		if = { limit = { has_idea = spain_officer_corps_restive } remove_ideas = spain_officer_corps_restive }
		add_ideas = spain_officer_corps_estranged
	}
	else = {
		if = { limit = { has_idea = spain_officer_corps_loyal } remove_ideas = spain_officer_corps_loyal }
		if = { limit = { has_idea = spain_officer_corps_estranged } remove_ideas = spain_officer_corps_estranged }
		add_ideas = spain_officer_corps_restive
	}
}

	if = { limit = { has_idea = spain_falangist_state } remove_ideas = spain_falangist_state }
	if = { limit = { has_idea = spain_carlist_state } remove_ideas = spain_carlist_state }

	if = { limit = { has_idea = spain_military_directorate } remove_ideas = spain_military_directorate }

	set_variable = { var = spain_regime_leader value = 4 }

	if = {
		limit = {
			check_variable = { spain_falange > spain_carlists }
			check_variable = { spain_falange > spain_monarchists }
			check_variable = { spain_falange > spain_army_faction }
		}
		set_variable = { var = spain_regime_leader value = 1 }
	}
	else_if = {
		limit = {
			check_variable = { spain_carlists > spain_falange }
			check_variable = { spain_carlists > spain_monarchists }
			check_variable = { spain_carlists > spain_army_faction }
		}
		set_variable = { var = spain_regime_leader value = 2 }
	}
	else_if = {
		limit = {
			check_variable = { spain_monarchists > spain_falange }
			check_variable = { spain_monarchists > spain_carlists }
			check_variable = { spain_monarchists > spain_army_faction }
		}
		set_variable = { var = spain_regime_leader value = 3 }
	}

	if = { limit = { check_variable = { var = spain_regime_leader value = 1 compare = equals } } add_ideas = spain_falangist_state }
	else_if = { limit = { check_variable = { var = spain_regime_leader value = 2 compare = equals } } add_ideas = spain_carlist_state }
	else_if = { limit = { check_variable = { var = spain_regime_leader value = 3 compare = equals } } add_ideas = spain_monarchist_state }
	else = { add_ideas = spain_military_directorate }
}

		if = {
			limit = { has_idea = spain_armed_neutrality }
			add_to_variable = { var = spain_axis_leverage value = constant:spain_campaign.leverage_loss }
			add_to_variable = { var = spain_allied_leverage value = constant:spain_campaign.leverage_loss }
		}
		# A Spain that has aligned with nobody drifts towards whichever bloc is at war next door.
		# Armed neutrality declared as policy (the post-war focus) and strict neutrality declared in
		# 1914 both freeze that drift; leverage can still be bought by decision and by focus.
		if = {
			limit = {
				NOT = { has_idea = spain_axis_alignment }
				NOT = { has_idea = spain_allied_alignment }
				NOT = { has_idea = spain_armed_neutrality }
				NOT = { has_country_flag = spain_armed_neutral }
				NOT = { has_country_flag = spain_strict_neutrality }
			}
			if = {
				limit = { country_exists = GER GER = { has_war = yes } }
				add_to_variable = { var = spain_axis_leverage value = constant:spain_campaign.leverage_gain }
			}
			if = {
				limit = { country_exists = ENG ENG = { has_war = yes } }
				add_to_variable = { var = spain_allied_leverage value = constant:spain_campaign.leverage_gain }
			}
		}
		# A Spain that settled with the powers in 1910-23 is courted by both blocs rather than
		# ignored, and that is the ground the 1939-45 competition starts from.
		if = {
			limit = { has_country_flag = spain_powers_settled }
			add_to_variable = { var = spain_allied_leverage value = constant:spain_campaign.leverage_gain_small }
			add_to_variable = { var = spain_axis_leverage value = constant:spain_campaign.leverage_gain_small }
		}
	}
}

# Overview: brings a late start up to date. A player who begins in 1914, 1918 or 1936
# inherits the focuses the country would plausibly have finished by then. Dates are promoted so a
# focus is never completed before its prerequisites, and the emission order is topological.
spain_restore_past_focuses = {
	# Generated by tools/build_spain_restore_schedule.py. A late start completes the
	# milestones history had already reached, in prerequisite order, so no entry is
	# skipped by the engine. Never hand-edit: change the seeds in the tool.
	# A 1910 start completes nothing, because no date gate has passed.
	if = {
		limit = { has_country_flag = spain_chapter_restoration }

# Overview: engine systems that used to be created during on_startup. They now run on the first
# daily tick, so the balance of power and the chapter history are built while the game is live.
spain_campaign_bootstrap = {
	if = {
		limit = { has_country_flag = spain_bop_pending }
		clr_country_flag = spain_bop_pending
		if = {
			limit = { has_country_flag = spain_chapter_republic }
			set_power_balance = {
				id = SPR_crown_and_cortes
				left_side = SPR_crown_side
				right_side = SPR_cortes_side
				set_value = constant:spain_campaign.bop_republic_start
			}
		}
		else = {
			set_power_balance = {
				id = SPR_crown_and_cortes
				left_side = SPR_crown_side
				right_side = SPR_cortes_side
				set_default = yes
			}
		}
	}
	spain_cortes_initialize = yes
	spain_refresh_campaign_ideas = yes
	spain_restore_past_focuses = yes
}

# Overview: keeps the chapter tree on the country. Other on_actions in the mod assign
# generic trees by tag, so a chapter re-asserts its own tree if something replaced it.
spain_ensure_chapter_tree = {
	if = {
		limit = { has_country_flag = spain_chapter_restoration NOT = { has_focus_tree = spain_restoration_focus } }
		load_focus_tree = { tree = spain_restoration_focus keep_completed = yes }
		mark_focus_tree_layout_dirty = yes
	}
	else_if = {
		limit = { has_country_flag = spain_chapter_rivera NOT = { has_focus_tree = spain_rivera_focus } }
		load_focus_tree = { tree = spain_rivera_focus keep_completed = yes }
		mark_focus_tree_layout_dirty = yes
	}
	else_if = {
		limit = { has_country_flag = spain_chapter_republic NOT = { has_focus_tree = spain_republic_focus } }
		load_focus_tree = { tree = spain_republic_focus keep_completed = yes }
		mark_focus_tree_layout_dirty = yes
	}
	else_if = {
		limit = { has_country_flag = spain_chapter_republican_postwar NOT = { has_focus_tree = spain_republic_postwar_focus } }
		load_focus_tree = { tree = spain_republic_postwar_focus keep_completed = yes }
		mark_focus_tree_layout_dirty = yes
	}
}

# Overview: era handover for the Spanish campaign. Each chapter opener calls this once, so

# Overview: what happens when the Republic reaches the question of the Cortes itself.
# Upholding it keeps the balance of power and the chamber; dissolving it is the only way to
# remove the balance, and it costs the government the room it governs from.
spain_uphold_the_cortes = {
	set_country_flag = spain_cortes_defended
	clr_country_flag = spain_cortes_dissolved
	clr_country_flag = spain_elections_suspended
	if = { limit = { has_idea = spain_cortes_suspended } remove_ideas = spain_cortes_suspended }
	add_stability = constant:spain_campaign.stability_medium
	add_political_power = constant:spain_campaign.pp_medium
	subtract_from_variable = { var = spain_nationalist_tension_carried value = 5 }
	add_to_variable = { var = spain_military_loyalty value = 5 }
}

spain_dissolve_the_cortes = {
	set_country_flag = spain_cortes_dissolved
	clr_country_flag = spain_cortes_defended
	# The chamber is not merely prorogued: no election is held again under this order, which is
	# what common/decisions/spain_parties.txt reads before it offers a general election.
	set_country_flag = spain_elections_suspended
	if = {
		limit = { has_power_balance = { id = SPR_crown_and_cortes } }
		remove_power_balance = { id = SPR_crown_and_cortes }
	}
	add_political_power = constant:spain_campaign.pp_dissolution_cost
	add_stability = constant:spain_campaign.stability_collapse
	add_war_support = constant:spain_campaign.war_support_medium
	add_to_variable = { var = spain_nationalist_tension_carried value = 20 }
	subtract_from_variable = { var = spain_military_loyalty value = 10 }
	add_to_variable = { var = spain_coup_pressure value = 15 }
	if = { limit = { NOT = { has_idea = spain_cortes_suspended } } add_ideas = spain_cortes_suspended }
}
