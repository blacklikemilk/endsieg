WW1_clear_political_support_spirits = {
 hidden_effect = {
  if = { limit = { has_idea = WW1_parliamentary_reform_movement } remove_ideas = WW1_parliamentary_reform_movement }
  if = { limit = { has_idea = WW1_constitutional_government_support } remove_ideas = WW1_constitutional_government_support }
  if = { limit = { has_idea = WW1_conservative_mobilisation } remove_ideas = WW1_conservative_mobilisation }
  if = { limit = { has_idea = WW1_state_authority_support } remove_ideas = WW1_state_authority_support }
  if = { limit = { has_idea = WW1_workers_political_movement } remove_ideas = WW1_workers_political_movement }
  if = { limit = { has_idea = WW1_revolutionary_government_support } remove_ideas = WW1_revolutionary_government_support }
 }
}

WW1_support_democratic_programme = {
 if = { limit = { has_start_date < 1936.1.1 date < 1936.1.1 }
  WW1_clear_political_support_spirits = yes
  if = { limit = { has_government = democratic } add_timed_idea = { idea = WW1_constitutional_government_support days = 365 } }
  else = { add_timed_idea = { idea = WW1_parliamentary_reform_movement days = 720 } }
 }
}
