# Seed experience accrued before the bookmark. After startup, the game awards mastery through service activity.
# Each regular track has five 100-point rewards. Later-adopted tracks start further behind.
endsieg_initialize_land_bookmark_doctrine = {
	if = {
		limit = { NOT = { has_any_grand_doctrine = land } }
		if = {
			limit = { tag = GER }
			set_grand_doctrine = new_mobile_warfare
		}
		else_if = {
			limit = { OR = { tag = SOV tag = CHI tag = PRC } }
			set_grand_doctrine = mass_assault
		}
		else_if = {
			limit = { tag = USA }
			set_grand_doctrine = superior_firepower
		}
		else = {
			set_grand_doctrine = grand_battleplan
		}
	}
	if = {
		limit = { has_doctrine = new_mobile_warfare }
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = infantry }
			}
			set_sub_doctrine = mobile_infantry
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = operations }
				has_start_date > 1942.1.1
			}
			set_sub_doctrine = mission_type_tactics
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = combat_support }
				has_start_date > 1943.1.1
			}
			if = {
				limit = { has_dlc = "La Resistance" }
				set_sub_doctrine = mobile_recon_and_assault
			}
			else = { set_sub_doctrine = mobile_recon_and_assault_no_lar }
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = armor }
				has_start_date > 1944.1.1
			}
			if = {
				limit = { has_dlc = "La Resistance" }
				set_sub_doctrine = armored_spearhead
			}
			else = { set_sub_doctrine = armored_spearhead_no_lar }
		}
	}
	else_if = {
		limit = { has_doctrine = mass_assault }
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = infantry }
			}
			set_sub_doctrine = large_unit_tactics
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = operations }
				has_start_date > 1942.1.1
			}
			if = {
				limit = { has_dlc = "La Resistance" }
				set_sub_doctrine = deep_battle
			}
			else = { set_sub_doctrine = deep_battle_no_lar }
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = combat_support }
				has_start_date > 1943.1.1
			}
			set_sub_doctrine = fire_concentration
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = armor }
				has_start_date > 1944.1.1
			}
			set_sub_doctrine = streamlined_deployment
		}
	}
	else_if = {
		limit = { has_doctrine = superior_firepower }
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = infantry }
			}
			set_sub_doctrine = assault_infantry
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = operations }
				has_start_date > 1942.1.1
			}
			set_sub_doctrine = rapid_domination
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = combat_support }
				has_start_date > 1943.1.1
			}
			set_sub_doctrine = fire_concentration
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = armor }
				has_start_date > 1944.1.1
			}
			set_sub_doctrine = armored_infantry_support
		}
	}
	else_if = {
		limit = { has_doctrine = grand_battleplan }
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = infantry }
			}
			set_sub_doctrine = defensive_postures
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = operations }
				has_start_date > 1942.1.1
			}
			if = {
				limit = { tag = JAP }
				set_sub_doctrine = infiltration_tactics
			}
			else = { set_sub_doctrine = grand_assault }
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = combat_support }
				has_start_date > 1943.1.1
			}
			set_sub_doctrine = field_engineering
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = armor }
				has_start_date > 1944.1.1
			}
			set_sub_doctrine = armored_infantry_support
		}
	}
	if = {
		limit = {
			has_subdoctrine_in_track = infantry
			NOT = { has_mastery = { amount = 1 track = infantry } }
		}
		if = {
			limit = { has_start_date > 1944.1.1 }
			add_mastery = { amount = 500 track = infantry }
		}
		else_if = {
			limit = { has_start_date > 1943.1.1 }
			add_mastery = { amount = 450 track = infantry }
		}
		else_if = {
			limit = { has_start_date > 1942.1.1 }
			add_mastery = { amount = 325 track = infantry }
		}
		else = { add_mastery = { amount = 225 track = infantry } }
	}
	if = {
		limit = {
			has_subdoctrine_in_track = operations
			NOT = { has_mastery = { amount = 1 track = operations } }
		}
		if = {
			limit = { has_start_date > 1945.1.1 }
			add_mastery = { amount = 500 track = operations }
		}
		else_if = {
			limit = { has_start_date > 1944.12.1 }
			add_mastery = { amount = 475 track = operations }
		}
		else_if = {
			limit = { has_start_date > 1944.1.1 }
			add_mastery = { amount = 400 track = operations }
		}
		else_if = {
			limit = { has_start_date > 1943.1.1 }
			add_mastery = { amount = 275 track = operations }
		}
		else_if = {
			limit = { has_start_date > 1942.1.1 }
			add_mastery = { amount = 125 track = operations }
		}
	}
	if = {
		limit = {
			has_subdoctrine_in_track = combat_support
			NOT = { has_mastery = { amount = 1 track = combat_support } }
		}
		if = {
			limit = { has_start_date > 1945.1.1 }
			add_mastery = { amount = 375 track = combat_support }
		}
		else_if = {
			limit = { has_start_date > 1944.12.1 }
			add_mastery = { amount = 325 track = combat_support }
		}
		else_if = {
			limit = { has_start_date > 1944.1.1 }
			add_mastery = { amount = 250 track = combat_support }
		}
		else_if = {
			limit = { has_start_date > 1943.1.1 }
			add_mastery = { amount = 125 track = combat_support }
		}
	}
	if = {
		limit = {
			has_subdoctrine_in_track = armor
			NOT = { has_mastery = { amount = 1 track = armor } }
		}
		if = {
			limit = { has_start_date > 1945.1.1 }
			add_mastery = { amount = 225 track = armor }
		}
		else_if = {
			limit = { has_start_date > 1944.12.1 }
			add_mastery = { amount = 175 track = armor }
		}
		else_if = {
			limit = { has_start_date > 1944.1.1 }
			add_mastery = { amount = 100 track = armor }
		}
	}
}

endsieg_initialize_naval_bookmark_doctrine = {
	if = {
		limit = { NOT = { has_any_grand_doctrine = naval } }
		if = {
			limit = { tag = GER }
			set_grand_doctrine = new_convoy_raiding
		}
		else_if = {
			limit = { OR = { tag = USA tag = JAP } }
			set_grand_doctrine = new_base_strike
		}
		else = {
			set_grand_doctrine = new_fleet_in_being
		}
	}
	if = {
		limit = { has_doctrine = new_convoy_raiding }
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = submarines }
				has_navy_size = { size > 0 unit = submarine }
			}
			set_sub_doctrine = wolfpacks
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = screens }
				has_start_date > 1942.1.1
			}
			set_sub_doctrine = support_integrated_operations
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = capital_ships }
				has_start_date > 1943.1.1
				OR = { has_navy_size = { size > 0 unit = battleship } has_navy_size = { size > 0 unit = battle_cruiser } has_navy_size = { size > 0 unit = heavy_cruiser } }
			}
			set_sub_doctrine = armored_raiders
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = carriers }
				has_start_date > 1944.1.1
				has_navy_size = { size > 0 unit = carrier }
			}
			set_sub_doctrine = carrier_battlegroups
		}
	}
	else_if = {
		limit = { has_doctrine = new_base_strike }
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = screens }
			}
			set_sub_doctrine = support_integrated_operations
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = submarines }
				has_start_date > 1942.1.1
				has_navy_size = { size > 0 unit = submarine }
			}
			set_sub_doctrine = capital_hunters
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = capital_ships }
				has_start_date > 1943.1.1
				OR = { has_navy_size = { size > 0 unit = battleship } has_navy_size = { size > 0 unit = battle_cruiser } has_navy_size = { size > 0 unit = heavy_cruiser } }
			}
			set_sub_doctrine = line_of_battle
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = carriers }
				has_start_date > 1944.1.1
				has_navy_size = { size > 0 unit = carrier }
			}
			if = {
				limit = { tag = USA }
				set_sub_doctrine = carrier_battlegroups
			}
			else = { set_sub_doctrine = massed_carrier_fleet }
		}
	}
	else_if = {
		limit = { has_doctrine = new_fleet_in_being }
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = screens }
			}
			set_sub_doctrine = convoy_escort
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = submarines }
				has_start_date > 1942.1.1
				has_navy_size = { size > 0 unit = submarine }
			}
			set_sub_doctrine = submarine_fleet_operations
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = capital_ships }
				has_start_date > 1943.1.1
				OR = { has_navy_size = { size > 0 unit = battleship } has_navy_size = { size > 0 unit = battle_cruiser } has_navy_size = { size > 0 unit = heavy_cruiser } }
			}
			set_sub_doctrine = line_of_battle
		}
		if = {
			limit = {
				NOT = { has_subdoctrine_in_track = carriers }
				has_start_date > 1944.1.1
				has_navy_size = { size > 0 unit = carrier }
			}
			set_sub_doctrine = floating_airfields
		}
	}
	if = {
		limit = {
			has_subdoctrine_in_track = submarines
			NOT = { has_mastery = { amount = 1 track = submarines } }
		}
		if = {
			limit = { has_doctrine = new_convoy_raiding }
			if = {
				limit = { has_start_date > 1944.1.1 }
				add_mastery = { amount = 500 track = submarines }
			}
			else_if = {
				limit = { has_start_date > 1943.1.1 }
				add_mastery = { amount = 450 track = submarines }
			}
			else_if = {
				limit = { has_start_date > 1942.1.1 }
				add_mastery = { amount = 325 track = submarines }
			}
			else = { add_mastery = { amount = 225 track = submarines } }
		}
		else = {
			if = {
				limit = { has_start_date > 1945.1.1 }
				add_mastery = { amount = 500 track = submarines }
			}
			else_if = {
				limit = { has_start_date > 1944.12.1 }
				add_mastery = { amount = 475 track = submarines }
			}
			else_if = {

			}
			else_if = {
				limit = { has_start_date > 1942.1.1 }
				add_mastery = { amount = 125 track = submarines }
			}
		}
	}
	if = {

			has_subdoctrine_in_track = screens

		}

			limit = { has_doctrine = new_convoy_raiding }
			if = {
				limit = { has_start_date > 1945.1.1 }
				add_mastery = { amount = 500 track = screens }
			}
			else_if = {
				limit = { has_start_date > 1944.12.1 }
				add_mastery = { amount = 475 track = screens }
			}
			else_if = {
				limit = { has_start_date > 1944.1.1 }
				add_mastery = { amount = 400 track = screens }
			}
			else_if = {
				limit = { has_start_date > 1943.1.1 }
				add_mastery = { amount = 275 track = screens }
			}
			else_if = {
				limit = { has_start_date > 1942.1.1 }
				add_mastery = { amount = 125 track = screens }
			}
		}
		else = {
			if = {
				limit = { has_start_date > 1944.1.1 }
				add_mastery = { amount = 500 track = screens }
			}
			else_if = {
				limit = { has_start_date > 1943.1.1 }
				add_mastery = { amount = 450 track = screens }

			}

		}

			NOT = { has_mastery = { amount = 1 track = capital_ships } }
		}
		if = {

			add_mastery = { amount = 375 track = capital_ships }
		}
		else_if = {
			limit = { has_start_date > 1944.12.1 }
			add_mastery = { amount = 325 track = capital_ships }
		}
		else_if = {
			limit = { has_start_date > 1944.1.1 }
			add_mastery = { amount = 250 track = capital_ships }
		}
		else_if = {
			limit = { has_start_date > 1943.1.1 }
			add_mastery = { amount = 125 track = capital_ships }

		}
		if = {
			limit = { has_start_date > 1945.1.1 }
			add_mastery = { amount = 225 track = carriers }
		}
		else_if = {
			limit = { has_start_date > 1944.12.1 }
			add_mastery = { amount = 175 track = carriers }

			add_mastery = { amount = 100 track = carriers }
