# A campaign can keep its Great War doctrine after the WWII trees unlock.
endsieg_add_era_mastery_bonus = {
	if = {
		limit = { OR = { date < 1936.1.1 has_doctrine = $ww1_doctrine$ } }
		add_mastery_bonus = { name = $name$ bonus = $bonus$ days = $days$ track = $ww1_track$ }
	}
	else = {
		add_mastery_bonus = { name = $name$ bonus = $bonus$ days = $days$ track = $track$ }
	}
}
