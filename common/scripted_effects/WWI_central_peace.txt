# A signatory withdraws individually. Other Entente wars are never white-peaced here.
WW1_CP_detach_current_signatory = {
	save_event_target_as = WW1_CP_withdrawing_country
	if = {
		limit = { is_faction_leader = yes }
		# A living Russia takes the alliance first; Britain is the next historical host.
		if = { limit = { RIE = { exists = yes is_subject = no NOT = { has_capitulated = yes } NOT = { tag = PREV } is_in_faction_with = PREV } } RIE = { set_faction_leader = yes } }
		else_if = { limit = { ENG = { exists = yes is_subject = no NOT = { has_capitulated = yes } NOT = { tag = PREV } is_in_faction_with = PREV } } ENG = { set_faction_leader = yes } }
		else = { random_other_country = { limit = { is_in_faction_with = event_target:WW1_CP_withdrawing_country is_subject = no NOT = { has_capitulated = yes } } set_faction_leader = yes } }
	}
	if = { limit = { is_in_faction = yes is_faction_leader = no } leave_faction = yes }
	else_if = { limit = { is_faction_leader = yes } dismantle_faction = yes }
	every_country = {
		limit = { OR = { tag = GER is_in_faction_with = GER } has_war_with = event_target:WW1_CP_withdrawing_country }
		white_peace = event_target:WW1_CP_withdrawing_country
	}
}
WW1_CP_accept_separate_armistice = {
	if = {
		limit = { WW1_CP_is_active_enemy = yes NOT = { has_country_flag = WW1_CP_separate_armistice_signed } }
		set_country_flag = WW1_CP_separate_armistice_signed
		set_country_flag = WW1_CP_fought_germany
		save_event_target_as = WW1_CP_armistice_parent
		WW1_CP_detach_current_signatory = yes
		every_country = {
			limit = { is_subject_of = event_target:WW1_CP_armistice_parent }
			set_country_flag = WW1_CP_separate_armistice_signed
			WW1_CP_detach_current_signatory = yes
		}
	}
}
