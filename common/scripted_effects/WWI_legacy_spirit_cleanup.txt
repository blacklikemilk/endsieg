	if = {
		limit = { has_country_flag = WW1_CP_legacy_spirits_cleanup_pending has_war = no }
		if = { limit = { has_idea = unrestricted_sub_warfare } remove_ideas = unrestricted_sub_warfare }
		if = { limit = { has_idea = shorter_training_GE1 } remove_ideas = shorter_training_GE1 }
		if = { limit = { has_idea = shadow_kaiser } remove_ideas = shadow_kaiser }
		if = { limit = { has_idea = desertion } remove_ideas = desertion }
		if = { limit = { has_idea = americans_are_here } remove_ideas = americans_are_here }
		clr_country_flag = WW1_CP_legacy_spirits_cleanup_pending
	}
}
