# Dated national turning points. Each flag is set before the event is queued;
# a later scenario cannot replay a milestone whose historical day has passed.
GER_INT_schedule_historical_milestones = {
	if = {
		limit = { has_government = democratic NOT = { has_completed_focus = INT_GER_defend_democracy } NOT = { has_completed_focus = INT_GER_appoint_hitler } }
		if = {
			limit = { date > 1919.1.4 date < 1919.1.6 NOT = { has_country_flag = GER_INT_milestone_spartacist } }
			set_country_flag = GER_INT_milestone_spartacist
			country_event = { id = INT_ger_milestone.13 days = 0 }
		}
		if = {
			limit = { date > 1919.4.6 date < 1919.4.8 NOT = { has_country_flag = GER_INT_milestone_bavarian_council } }
			set_country_flag = GER_INT_milestone_bavarian_council
			country_event = { id = INT_ger_milestone.14 days = 0 }
		}
		if = {
			limit = { date > 1919.5.2 date < 1919.5.4 NOT = { has_country_flag = GER_INT_milestone_bavarian_end } }
			set_country_flag = GER_INT_milestone_bavarian_end
			country_event = { id = INT_ger_milestone.15 days = 0 }
		}
		if = {
			limit = { date > 1921.5.10 date < 1921.5.12 NOT = { has_country_flag = GER_INT_milestone_london } }
			set_country_flag = GER_INT_milestone_london

		if = {
			limit = { date > 1922.4.15 date < 1922.4.17 NOT = { has_country_flag = GER_INT_milestone_rapallo } country_exists = SOV }
			set_country_flag = GER_INT_milestone_rapallo

		}
		if = {

# Each decree is dispatched only on its historical calendar day.
GER_INT_schedule_dissolutions = {
	if = {
		limit = {
			date > 1924.3.12
			date < 1924.3.14
			NOT = { has_country_flag = GER_INT_dissolution_1 }
		}
		set_country_flag = GER_INT_dissolution_1
		country_event = { id = INT_weimar.1 days = 0 }
	}
	if = {
		limit = {
			date > 1924.10.19
