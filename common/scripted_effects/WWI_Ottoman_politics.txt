  remove_ideas = OT1_P_contested_restoration
  if = {
   limit = { has_country_leader = { ruling_only = yes character = TUR_abdulhamid_ii } }
   if = {
    limit = { date < 1918.7.3 has_character = TUR_mehmed_v }
    TUR_mehmed_v = {
     add_country_leader_role = {
      promote_leader = yes
      country_leader = { ideology = despotism expire = "1936.1.1.1" traits = { cornered_fox } }
     }
    }
   }
   else = { endsieg_character_activate_tur_mehmed_vi_0177e887 = yes TUR_mehmed_vi = { promote_character = yes } }
   retire_character = TUR_abdulhamid_ii
  }
  else_if = { limit = { has_character = TUR_abdulhamid_ii } retire_character = TUR_abdulhamid_ii }
 }
}
