# Overview: the generic release of one Chinese warlord tag from the state that owns its seat.
# Shared by every pre-1936 warlord slice: a later tag adds a wrapper, not a rewrite.
#
# Why this is a body plus one wrapper per tag, and not a single meta_effect:
# meta_effect parameters are definition-internal (installed effects_documentation.md, "meta_effect"),
# so a scripted-effect caller cannot hand a tag, a seat or a character token to a meta_effect body.
# The split below carries the parameterisation instead:
#
#   china_release_warlord_receive  the shared body. Runs in the RECEIVING tag's scope, takes the
#                                  seat from the variable `china_release_seat`, and holds no tag,
#                                  seat, character token or per-tag flag. Every warlord runs this.
#   china_release_warlord_<tag>    one wrapper per tag. Runs in the seat's current owner's scope and
#                                  is the only place that tag's literals live: its owner guard flag,
#                                  its seat, and its leading character.
#
# Adding a warlord means adding one wrapper of the same shape and one guarded line in the owner's
# on_action. The body is not edited.
#
# Each wrapper sets up its own government after the shared state transfer.
china_release_warlord_receive = {
	if = {
		limit = { NOT = { has_country_flag = china_warlord_released } }
		set_country_flag = china_warlord_released
		# Vanilla resolves a variable seat here (events/GOE_Raj.txt: transfer_state = var:ROOT…), so
		# the body does not need to know which province it is releasing.
		transfer_state = var:china_release_seat
	}
}

# Shanxi: seat 615, the province Yan Xishan has governed since 1911, and therefore not a 1936-only
# identity. Scope: the seat's current owner (the Republic, RCH).
china_release_warlord_shx = {
	if = {
		limit = {
			NOT = { has_country_flag = china_released_shx }
			owns_state = 615
			SHX = { NOT = { owns_state = 615 } }
		}
		set_country_flag = china_released_shx
		SHX = {
			set_variable = { china_release_seat = 615 }
			china_release_warlord_receive = yes
			# add_state_core takes a database token rather than a variable, so
			# the core stays beside its seat literal.
			add_state_core = 615
			china_initialize_shanxi_government = yes
		}
	}
}

# Recruitment alone does not install Yan's non-aligned leader role on release.
china_initialize_shanxi_government = {
	if = {
		limit = {
			has_start_date < 1936.1.1
			date < 1936.1.1
			has_country_flag = china_warlord_released
			NOT = { has_country_flag = china_shanxi_government_initialized }
			owns_state = 615
		}
		set_popularities = { neutrality = 75 democratic = 20 communism = 5 fascism = 0 }
		china_install_shanxi_government = yes
		set_country_flag = china_shanxi_government_initialized
	}
}

china_install_shanxi_government = {
	set_politics = { ruling_party = neutrality elections_allowed = no }
	set_party_name = {
		ideology = neutrality
		name = SHX_ww1_clique_party
		long_name = SHX_ww1_clique_party
	}
	if = {
		limit = { NOT = { has_country_leader = { name = SHX_yan_xishan ruling_only = yes } } }
		if = {
			limit = { has_character = SHX_yan_xishan }
			promote_character = { character = SHX_yan_xishan ideology = despotism }
		}
		else = {
			create_country_leader = {
				name = SHX_yan_xishan
				picture = "gfx/leaders/SHX/portrait_shx_yan_xishan.dds"
				ideology = despotism
				expire = "1965.1.1"
			}
		}
	}
}

# Subject creation can copy Beijing's ruling ideology. Restore Yan while the
# historical provincial agreement is in force, including in existing saves.
china_restore_shanxi_provincial_government = {
	if = {
		limit = {
			is_ai = yes
			is_historical_focus_on = yes
			has_start_date < 1920.7.14
			date < 1920.7.14
			615 = { is_owned_and_controlled_by = SHX }
			has_country_flag = china_shanxi_provincial_affiliation
			is_subject_of = RCH
			has_autonomy_state = autonomy_ww1_shanxi_province
			has_government = democratic
		}
		china_install_shanxi_government = yes
	}
}

# The Beijing government recognized Yan's provincial administration. He could
# still withhold troops. The release receipts also identify already-running saves.
# Scope: RCH.
china_attach_shanxi_province = {
	if = {
		limit = {
			date > 1916.6.5
			date < 1920.7.14
			is_ai = yes
			is_historical_focus_on = yes
			is_subject = no
			has_government = democratic
			has_country_flag = china_released_shx
			608 = { is_owned_and_controlled_by = RCH }
			SHX = {
				exists = yes
				is_ai = yes
				is_subject = no
				is_in_faction = no
				has_war = no
				has_country_flag = china_warlord_released
				NOT = { has_country_flag = china_shanxi_provincial_affiliation }
				615 = { is_owned_and_controlled_by = SHX }
			}
		}
		set_autonomy = {
			target = SHX
			autonomy_state = autonomy_ww1_shanxi_province
			end_wars = no
			end_civil_wars = no
		}
		SHX = {
			if = {
				limit = { is_subject_of = RCH }
				set_country_flag = china_shanxi_provincial_affiliation
				china_restore_shanxi_provincial_government = yes
			}
		}
	}
}

# Restore independent diplomacy before the existing clique-war tree opens.
# Scope: SHX.
china_end_shanxi_provincial_affiliation = {
	if = {
		limit = {
			NOT = { date < 1920.7.14 }
			has_country_flag = china_shanxi_provincial_affiliation
			is_subject_of = RCH
			has_autonomy_state = autonomy_ww1_shanxi_province
		}
		RCH = {
			set_autonomy = {
				target = SHX
				autonomy_state = autonomy_free
				end_wars = no
				end_civil_wars = no
			}
		}
	}
}
