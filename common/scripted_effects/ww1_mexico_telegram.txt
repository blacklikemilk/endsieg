MEX_WW1_ZT_initialize = {
	if = {
		limit = { MEX_WW1_ZT_recipient_available = yes has_start_date > 1917.12.31 }
		set_country_flag = MEX_WW1_ZT_received
		set_country_flag = MEX_WW1_ZT_assessed
		set_country_flag = MEX_WW1_ZT_report_considered
		if = {
			limit = { has_completed_focus = MEX_WW1_ratify_central is_in_faction_with = GER }
			set_country_flag = MEX_WW1_ZT_terms_accepted
			set_country_flag = MEX_WW1_central_consent
		}
		else = { set_country_flag = MEX_WW1_ZT_rejected }
	}
}
MEX_WW1_ZT_receive_offer = {
	if = {
		limit = { MEX_WW1_ZT_recipient_available = yes NOT = { has_country_flag = MEX_WW1_ZT_received } OR = { MEX_WW1_ZT_sender_available = yes AND = { has_completed_focus = MEX_WW1_ratify_central is_in_faction_with = GER } } }
		set_global_flag = zimmerman_telegram
		set_country_flag = MEX_WW1_ZT_received
		set_country_flag = MEX_WW1_zimmermann_offer
		if = {
			limit = { has_completed_focus = MEX_WW1_ratify_central is_in_faction_with = GER }
			set_country_flag = MEX_WW1_ZT_assessed
			set_country_flag = MEX_WW1_ZT_report_considered
			set_country_flag = MEX_WW1_ZT_terms_accepted
			set_country_flag = MEX_WW1_central_consent
		}
		else = { country_event = { id = mex_ww1_telegram.1 days = 1 } }
	}
}
MEX_WW1_ZT_authorize_assessment = {
	if = {
		limit = { MEX_WW1_ZT_recipient_available = yes has_country_flag = MEX_WW1_ZT_received NOT = { has_country_flag = MEX_WW1_ZT_assessed } NOT = { has_country_flag = MEX_WW1_ZT_assessment_active } }
		if = {
			limit = { MEX_WW1_ZT_sender_available = yes }
			set_country_flag = MEX_WW1_ZT_assessment_authorized
			set_country_flag = MEX_WW1_ZT_assessment_active
		}
		else = { MEX_WW1_ZT_withdraw = yes }
	}
}
MEX_WW1_ZT_finish_assessment = {
	if = {
		limit = { has_country_flag = MEX_WW1_ZT_assessment_active MEX_WW1_ZT_recipient_available = yes }
		clr_country_flag = MEX_WW1_ZT_assessment_active
		clr_country_flag = MEX_WW1_ZT_assessment_authorized
		set_country_flag = MEX_WW1_ZT_assessed
		if = { limit = { has_country_flag = MEX_WW1_ZT_report_waiting } MEX_WW1_ZT_acknowledge_report = yes }
	}
}
MEX_WW1_ZT_acknowledge_report = {
	if = {
		limit = { MEX_WW1_ZT_recipient_available = yes has_country_flag = MEX_WW1_ZT_received }
		if = {
			limit = { OR = { has_country_flag = MEX_WW1_ZT_assessed has_country_flag = MEX_WW1_ZT_withdrawn } }
			clr_country_flag = MEX_WW1_ZT_report_waiting
			set_country_flag = MEX_WW1_ZT_report_considered
			if = { limit = { NOT = { has_country_flag = MEX_WW1_ZT_report_seen } } set_country_flag = MEX_WW1_ZT_report_seen country_event = { id = mex_ww1_telegram.2 days = 1 } }
		}
		else = { set_country_flag = MEX_WW1_ZT_report_waiting }
	}
}
MEX_WW1_ZT_request_guarantees = {
	if = {
		limit = { MEX_WW1_ZT_can_counteroffer = yes }
		set_country_flag = MEX_WW1_ZT_terms_pending
		clr_country_flag = MEX_WW1_ZT_terms_refused
		clr_country_flag = MEX_WW1_ZT_withdrawn
		clr_country_flag = MEX_WW1_central_consent
		GER = { country_event = { id = mex_ww1_telegram.3 days = 7 } }
		country_event = { id = mex_ww1_telegram.103 days = 45 }
	}
	else = {
		if = {
			limit = { MEX_WW1_ZT_recipient_available = yes has_country_flag = MEX_WW1_ZT_received NOT = { has_country_flag = MEX_WW1_ZT_terms_pending } NOT = { has_country_flag = MEX_WW1_ZT_terms_accepted } NOT = { has_country_flag = MEX_WW1_ZT_rejected } }
			MEX_WW1_ZT_withdraw = yes
		}
	}
}
MEX_WW1_ZT_withdraw = {
	clr_country_flag = MEX_WW1_ZT_assessment_active
	clr_country_flag = MEX_WW1_ZT_assessment_authorized
	clr_country_flag = MEX_WW1_ZT_terms_pending
	clr_country_flag = MEX_WW1_central_offer_pending
	clr_country_flag = MEX_WW1_central_consent
	if = {
		limit = { has_country_flag = MEX_WW1_ZT_received NOT = { has_country_flag = MEX_WW1_ZT_terms_accepted } NOT = { has_country_flag = MEX_WW1_ZT_rejected } }
		set_country_flag = MEX_WW1_ZT_withdrawn
		set_country_flag = MEX_WW1_ZT_assessed
		set_country_flag = MEX_WW1_ZT_report_considered
		if = { limit = { MEX_WW1_ZT_recipient_available = yes NOT = { has_country_flag = MEX_WW1_ZT_withdrawal_seen } } set_country_flag = MEX_WW1_ZT_withdrawal_seen country_event = { id = mex_ww1_telegram.6 days = 1 } }
	}
}
MEX_WW1_ZT_reject = {
	if = {
		limit = { MEX_WW1_ZT_recipient_available = yes has_country_flag = MEX_WW1_ZT_received NOT = { has_country_flag = MEX_WW1_ZT_rejected } }
		set_country_flag = MEX_WW1_ZT_rejected
		set_country_flag = MEX_WW1_ZT_assessed
		set_country_flag = MEX_WW1_ZT_report_considered
		set_global_flag = mexico_says_no
		clr_country_flag = MEX_WW1_ZT_assessment_active
		clr_country_flag = MEX_WW1_ZT_assessment_authorized
		clr_country_flag = MEX_WW1_ZT_terms_pending
		clr_country_flag = MEX_WW1_ZT_terms_accepted
		clr_country_flag = MEX_WW1_central_consent
		clr_country_flag = MEX_WW1_central_offer_pending
		country_event = { id = mex_ww1_telegram.5 days = 1 }
	}
}
MEX_WW1_ZT_publish = {
	if = {
		limit = { has_country_flag = MEX_WW1_ZT_received NOT = { has_global_flag = MEX_WW1_ZT_published } }
		set_global_flag = MEX_WW1_ZT_published
		news_event = { id = mex_ww1_telegram.200 hours = 6 }
		if = {
			limit = { country_exists = USA USA = { NOT = { is_in_faction_with = GER } NOT = { has_country_flag = US2_R_german_agreement } } }
			USA = {
				if = { limit = { NOT = { has_global_flag = USA_zimmerman_condemned } } country_event = { id = ww1_usa.7 days = 1 } }
				country_event = { id = mex_ww1_telegram.10 days = 2 }
			}
		}
	}
}
