# Country-scoped cabinet choice. Legacy focus callbacks only queue the offer.
WW1_british_prepare_war_entry = {
 if = {
  limit = {
   tag = ENG has_focus_tree = british_focus_ww1
   has_start_date < 1936.1.1 date < 1936.1.1
   NOT = { has_war_with = GER }
  }
  # The engine guarantee otherwise enters Britain as a defender before the
  # cabinet event can offer neutrality. The treaty itself remains recorded.
  if = {
   limit = { country_exists = BEL BEL = { is_guaranteed_by = ENG } }
   diplomatic_relation = { country = BEL relation = guarantee active = no }
   set_country_flag = EN1_R_belgian_treaty_pledged
  }
  if = {
   limit = { has_country_flag = WW1_ENG_war_entry_pending NOT = { WW1_british_war_entry_context = yes } }
   clr_country_flag = WW1_ENG_war_entry_pending
  }
 }
}

WW1_british_queue_war_entry = {
 if = {
  limit = { WW1_british_may_offer_war_entry = yes }
  set_country_flag = { flag = WW1_ENG_war_entry_pending days = 7 }
  country_event = { id = ww1_britain.5 hours = 1 }
 }
}

WW1_british_accept_war_entry = {
 if = {
  limit = { WW1_british_war_entry_context = yes NOT = { has_country_flag = WW1_ENG_war_entry_resolved } }
  set_country_flag = WW1_ENG_war_entry_resolved
  set_country_flag = WW1_ENG_war_entry_accepted
  clr_country_flag = WW1_ENG_war_entry_pending
  # The cabinet's choice records the actual focus path before any faction entry.
  WW1_crisis_ENG_belgian_response = yes
  set_country_flag = EN1_R_entente_policy
  set_global_flag = back_france

  # Britain is the declarant. Joining a faction first can turn the intervention
  # into an engine-generated defensive call with Germany shown as declarant.
  if = {
   limit = { has_completed_focus = EN1_join_war NOT = { has_war_with = GER } NOT = { is_in_faction_with = GER } }
   declare_war_on = { target = GER type = annex_everything }
  }
  if = {
   limit = { has_war_with = GER has_completed_focus = EN1_join_war WW1_british_french_host_available = yes }
   FRA = {
    if = {
     limit = { is_in_faction = no }
     set_rule = { can_create_factions = yes }
     create_faction_from_template = { template = faction_template_ww1_entente name = Entente }
    }
    if = {
     limit = { ENG = { is_in_faction = no WW1_focus_authorizes_faction_entry_from_previous = yes } }
     add_to_faction = ENG
    }
   }
  }
  else_if = {
   # A separate French armistice leaves the surviving Russian-led Entente intact.
   limit = { has_war_with = GER has_completed_focus = EN1_join_war WW1_british_russian_host_available = yes }
   RIE = {
    faction_leader = {
     if = {
      limit = { ENG = { is_in_faction = no WW1_focus_authorizes_faction_entry_from_previous = yes } }
      add_to_faction = ENG
     }
    }
   }
  }
  if = {
   limit = { has_war_with = GER }
   set_global_flag = britain_joins_war
   if = { limit = { OR = { is_in_faction_with = FRA is_in_faction_with = RIE } } set_global_flag = britain_joins_entente }
   news_event = { id = ww1_british_entry.1 hours = 6 }
  }
 }
}

WW1_british_refuse_war_entry = {
 if = {
  limit = { WW1_british_entry_route_allowed = yes NOT = { has_war_with = GER } NOT = { has_country_flag = WW1_ENG_war_entry_resolved } }
  set_country_flag = WW1_ENG_war_entry_resolved
  set_country_flag = WW1_ENG_war_entry_refused
  clr_country_flag = WW1_ENG_war_entry_pending
  set_country_flag = EN1_R_neutral_policy
  clr_country_flag = EN1_R_entente_policy
  clr_global_flag = back_france
  set_global_flag = british_neutrality
  add_timed_idea = { idea = EN1_R_neutral_shipping days = 180 }
  news_event = { id = ww1_british_entry.2 hours = 6 }
 }
}
