# Reconfigure a ratified coalition in place. Completed objectives and
# membership are preserved. No initiative is awarded by these helpers.
WW1_configure_anglo_german_faction = {
 if = {
  limit = { has_start_date < 1936.1.1 date < 1936.1.1 is_faction_leader = yes }
  set_faction_manifest = faction_manifest_ww1_mutual_defence
  set_faction_rule = joining_rule_ww1_focus_commitment
  set_faction_rule = faction_peace_take_claimed_focus
  set_faction_rule = change_leader_rule_industry
  if = { limit = { has_faction_goal = faction_goal_a_military_base NOT = { has_completed_faction_goal = faction_goal_a_military_base } } remove_faction_goal = faction_goal_a_military_base }
  if = { limit = { has_faction_goal = faction_goal_defeating_our_enemies NOT = { has_completed_faction_goal = faction_goal_defeating_our_enemies } } remove_faction_goal = faction_goal_defeating_our_enemies }
  if = { limit = { has_faction_goal = faction_goal_the_atomic_race NOT = { has_completed_faction_goal = faction_goal_the_atomic_race } } remove_faction_goal = faction_goal_the_atomic_race }
  if = { limit = { NOT = { has_faction_goal = faction_goal_an_infantry_army } } add_faction_goal = faction_goal_an_infantry_army }
  if = { limit = { NOT = { has_faction_goal = faction_goal_convoy_escort_force } } add_faction_goal = faction_goal_convoy_escort_force }
  if = { limit = { NOT = { has_faction_goal = faction_goal_rule_the_waves } } add_faction_goal = faction_goal_rule_the_waves }
 }
}
