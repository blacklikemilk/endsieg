		}

			RIE = {
				NOT = { controls_state = 808 }
				NOT = { controls_state = 206 }
				NOT = { controls_state = 202 }
			}
			# The February Revolution began on 23 February 1917 in the
			# Julian-calendar convention used by this WW1 event chain.
			date > 1917.2.22
		}
	}

	mean_time_to_happen = { days = 1 }

	option = {
		name = ww1_russia.24.a
		news_event = { hours = 6 id = endsieg_news.217 }
		set_global_flag = february_revolution1
		set_politics = {
			ruling_party = democratic
			elections_allowed = no    }
		# Keep the Imperial Russia gameplay tag for the civil-war chain, while
		# displaying the provisional Russian Republic identity.
		set_cosmetic_tag = RRE

        set_popularities = {
            fascism = 0
            neutrality = 20
            democratic = 45
            communism = 35
		}
		retire_country_leader = yes
		endsieg_character_activate_rie_georgy_lvov_f80568e6 = yes
		RIE_Georgy_Lvov = { promote_character = yes }
		add_war_support = 0.15
		add_stability = 0.15
		add_political_power = 50

		hidden_effect = {
			country_event = { id = ww1_russia.25 days = 14 }
			country_event = { id = ww1_russia.32 days = 130 }
			country_event = { id = ww1_russia.33 days = 160 }

	id = ww1_russia.44
	title = ww1_russia.44.t
	desc = ww1_russia.44.d
	picture = GFX_report_event_russian_mob

	fire_only_once = yes

	trigger = {
		has_global_flag = welcome_event_WW1_active
		tag = RIE
		NOT = { date > 1915.1.1 }
		OR = {
			has_global_flag = schlieffen_plan
			has_global_flag = rupprecht_plan
			has_global_flag = kronprinz_plan
		}
	}

	mean_time_to_happen = { days = 1 }

	option = {
		name = ww1_russia.45.a
		news_event = { hours = 6 id = endsieg_news.151 }
		set_global_flag = russian_mobilization_begins
		RIE = {
			load_oob = RUS_Mobilization
		}
	}
}
