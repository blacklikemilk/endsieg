# Overview: Iran's Great Game negotiations - the demand events, the Majlis votes and the escalations.
# Namespace iran_n. Ids: 1-3 the northern railway, 4-6 the Gulf and the telegraph, 7-9 the Caspian
# fisheries, 10-11 the German economic offer. Each demand is one event with up to seven visible options
# (accept or submit to the Majlis, sign by decree, refuse, compromise, delay, seek a rival's support),
# every option gated by its own trigger so only coherent answers appear. Nothing here is fired by
# date: iran_negotiation_tick opens the demand and the options resolve it.
# Option keys skip .d, which is the description key: a=accept, b=submit, c=decree, e=refuse,
# f=compromise, g=delay, h=seek support.

# Durations are file macros because day fields reject constants; keep them equal across the three negotiation files.
@negotiation_delay_days = 120
@negotiation_followup_days = 75
# How long the chamber remembers a treaty signed over its head. The flag vetoes the yearly Budget
# Vote mission while it lasts, so it has to expire: no decree may disable the core loop for good.
@decree_treaty_days = 365

add_namespace = iran_n

# ---------------------------------------------------------------- demand 1: the northern railway, 1912-1914
country_event = {
	id = iran_n.1
	title = iran_n.1.t
	desc = iran_n.1.d
	picture = GFX_report_event_russo_persian
	is_triggered_only = yes

	immediate = { set_country_flag = iran_negotiation_open }

	option = {
		name = iran_n.1.a
		trigger = {
			OR = {
				NOT = { has_power_balance = { id = PER_crown_and_majlis } }
				NOT = { power_balance_value = { id = PER_crown_and_majlis value > 0.10 } }
			}
		}
		ai_chance = {
			factor = 40
			modifier = { factor = 2 is_historical_focus_on = yes }
			modifier = { factor = 2 has_country_flag = PER_bop_crown_side }
			modifier = { factor = 0.5 has_country_flag = iran_nationalist_front }
		}
		iran_demand_russian_railway_accept = yes
	}
	option = {
		name = iran_n.1.b
		trigger = {
			has_power_balance = { id = PER_crown_and_majlis }
			power_balance_value = { id = PER_crown_and_majlis value > 0.10 }
		}
		ai_chance = {
			factor = 45
			modifier = { factor = 2 has_country_flag = PER_bop_majlis_side }
		}
		set_country_flag = iran_demand_russian_railway_submitted
		country_event = { id = iran_n.2 days = @negotiation_followup_days }
	}
	option = {
		name = iran_n.1.c
		trigger = {
			has_power_balance = { id = PER_crown_and_majlis }
			power_balance_value = { id = PER_crown_and_majlis value < -0.10 }
		}
		ai_chance = {
			factor = 30
			modifier = { factor = 3 has_country_flag = PER_bop_crown_side }
		}
		# the Court signs it and presents the chamber with a fait accompli
		iran_demand_russian_railway_accept = yes
		PER_bop_push_crown_small = yes
		set_country_flag = { flag = iran_decree_treaty value = 1 days = @decree_treaty_days }
		add_stability = constant:iran_story.stability_cost
	}
	option = {
		name = iran_n.1.e
		ai_chance = {
			factor = 30
			modifier = { factor = 2 has_country_flag = iran_nationalist_front }
			modifier = { factor = 2 has_country_flag = iran_republic }
		}
		iran_demand_russian_railway_reject = yes
	}
	option = {
		name = iran_n.1.f
		trigger = { NOT = { has_country_flag = iran_demand_russian_railway_delayed } }
		ai_chance = {
			factor = 35
			modifier = { factor = 2 iran_path_majlis = yes }
		}
		iran_demand_russian_railway_compromise = yes
	}
	option = {
		name = iran_n.1.g
		trigger = {
			NOT = { has_country_flag = iran_demand_russian_railway_delayed }
			check_variable = { var = iran_authority value = 20 compare = greater_than_or_equals }
		}
		ai_chance = {
			factor = 20
			modifier = { factor = 2 is_historical_focus_on = no }
		}
		# one postponement only: it comes back in four months with the compromise off the table
		set_country_flag = iran_demand_russian_railway_delayed
		add_to_variable = { var = iran_authority value = constant:iran_story.authority_loss }
		add_stability = constant:iran_story.stability_cost
		country_event = { id = iran_n.1 days = @negotiation_delay_days }
	}
	option = {
		name = iran_n.1.h
		trigger = { iran_sponsor_britain_valid = yes }
		ai_chance = {
			factor = 25
			modifier = { factor = 2 has_idea = iran_balancing_powers }
		}
		iran_demand_russian_railway_support = yes
	}
}
