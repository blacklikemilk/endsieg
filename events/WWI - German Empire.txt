		has_global_flag = welcome_event_WW1_active

		date > 1917.1.1

		NOT = { has_global_flag = zimmerman_telegram }
	}
	option = {
		name = ww1_germany.13.a
		set_global_flag = zimmerman_telegram
		# The telegram's discovery immediately provoked the United States.
		# Previously this flag only unlocked a U.S. focus, leaving the event

		if = {
			limit = { country_exists = USA }
			USA = { country_event = { id = ww1_usa.7 days = 1 } }
		}
	}
}

# The visible programme callback is called by the Kronprinz focus. After all
# responses arrive, the same event is called again and launches the war.
country_event = {
	id = ww1_germany.258
	title = ww1_germany.258.t
	desc = ww1_germany.258.d
	picture = GFX_report_event_kronprinz_plan
	is_triggered_only = yes
	trigger = {
		tag = GER
		has_global_flag = GE1_anglo_german_alliance
		has_global_flag = GE1_continental_alliance_active
		has_global_flag = kronprinz_plan
		has_start_date < 1914.7.1
		date > 1914.6.1
		country_exists = RIE
		RIE = { NOT = { has_capitulated = yes } }
		NOT = { has_war_with = ENG }
		NOT = { has_war_with = FRA }
		NOT = { has_war_with = RIE }
		NOT = { has_global_flag = GE1_D_anglo_german_war_refused }
		NOT = { has_global_flag = GE1_D_continental_war_refused }
		NOT = { has_global_flag = GE1_D_italian_war_refused }
		OR = {
			AND = {
				NOT = { has_country_flag = GE1_continental_war_programme }
				NOT = { has_global_flag = GE1_D_anglo_german_war_pending }
			}
			AND = {
				has_country_flag = GE1_continental_war_programme
				has_global_flag = GE1_D_anglo_german_war_pending
				has_global_flag = GE1_D_anglo_german_war_accepted
				has_global_flag = GE1_D_continental_war_accepted
				has_global_flag = GE1_anglo_german_war_entry_agreed
				OR = {
					NOT = { ITA = { is_in_faction_with = FRA } }
					has_global_flag = GE1_D_italian_war_accepted
				}
			}
		}
	}
	option = {
		name = ww1_germany.258.a
		trigger = {
			NOT = { has_country_flag = GE1_continental_war_programme }
			NOT = { has_global_flag = GE1_D_anglo_german_war_pending }
		}
		set_country_flag = GE1_continental_war_programme
		country_event = { id = ww1_germany.249 days = 1 }
	}
	option = {
		name = ww1_germany.258.b
		trigger = {
			has_country_flag = GE1_continental_war_programme
			has_global_flag = GE1_D_anglo_german_war_pending
			has_global_flag = GE1_D_anglo_german_war_accepted
			has_global_flag = GE1_D_continental_war_accepted
			has_global_flag = GE1_anglo_german_war_entry_agreed
			OR = {
				NOT = { ITA = { is_in_faction_with = FRA } }
				has_global_flag = GE1_D_italian_war_accepted
			}
		}
		GE1_D_launch_continental_war = yes
	}
}
