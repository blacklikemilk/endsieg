# Borders in Eastern Europe consolidate in stages after the Central Powers'
# defeat. The 1919 bookmark uses the same events as a continuing WWI campaign.
add_namespace = INT_eastern

# passed to Estonia and Latvia.
country_event = {
	id = INT_eastern.3
	hidden = yes
	is_triggered_only = yes
	immediate = { INT_eastern_arm_successors = yes }
}

# A Ukrainian surrender to the Soviets must absorb the whole country,
# including custom states, without waiting for its other allies to surrender.
country_event = {
	id = INT_eastern.4
	hidden = yes
	is_triggered_only = yes
	trigger = {
		tag = SOV
		date > 1917.12.31
		date < 1936.1.1
		OR = {
			has_country_flag = INT_ukraine_integrated
			AND = {
				country_exists = UKR
				UKR = {
					NOT = { is_subject_of = SOV }
					has_country_flag = INT_ukraine_defeated_by_soviets
					INT_eastern_ukraine_defeat_confirmed = yes
				}
			}
			AND = {
				NOT = { country_exists = UKR }
				any_owned_state = { is_core_of = UKR }
			}
		}
	}
	immediate = {
		if = {
			limit = {
				country_exists = UKR
				UKR = {
					NOT = { is_subject_of = SOV }
					has_country_flag = INT_ukraine_defeated_by_soviets
					INT_eastern_ukraine_defeat_confirmed = yes
				}
			}
			UKR = { clr_country_flag = INT_ukraine_defeated_by_soviets }
			annex_country = { target = UKR transfer_troops = no }
			if = {
				limit = { NOT = { country_exists = UKR } }
				set_country_flag = INT_ukraine_integrated
			}
		}
		INT_eastern_absorb_moldavian_subjects = yes
	}
}

# The capitulation callback records which government surrendered to the Red Army.
country_event = {
	id = INT_eastern.5
	hidden = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = {
				INT_eastern_historical_soviet_campaign = yes
				event_target:INT_eastern_defeated_government = {
					exists = yes
					is_ai = yes
					OR = { tag = EST tag = LAT tag = LIT tag = POL }
					OR = {
						AND = { has_capitulated = yes has_war_with = SOV is_subject = no }
						is_subject_of = SOV
					}
				}
			}
			annex_country = { target = event_target:INT_eastern_defeated_government transfer_troops = no }
		}
	}
}

# If a generic peace conference eliminated historical Lithuania, Kaunas can
# still sign the Moscow settlement. Only territory held by the signatories moves.
country_event = {
	id = INT_eastern.13
	hidden = yes
	fire_only_once = yes
	trigger = {
		tag = SOV INT_eastern_historical_soviet_campaign = yes
		has_country_flag = INT_historical_lit_war
		date > 1920.7.11 date < 1920.10.7
		NOT = { has_global_flag = moscow_lit_treaty }
		OR = { NOT = { country_exists = LIT } LIT = { is_ai = yes is_subject_of = SOV } }
		11 = { OR = { is_owned_by = SOV is_owned_by = LIT } OR = { is_fully_controlled_by = SOV is_fully_controlled_by = LIT } }
	}
	mean_time_to_happen = { days = 1 }
	immediate = {
		if = { limit = { LIT = { is_subject_of = SOV } }
			set_autonomy = { target = LIT autonomy_state = autonomy_free end_wars = no end_civil_wars = no }
		}
		11 = { if = { limit = { is_owned_by = SOV } transfer_state_to = LIT } }
		LIT = {
			if = { limit = { is_in_faction_with = SOV } leave_faction = yes }
			if = { limit = { has_war_with = POL } white_peace = POL }
			set_country_flag = INT_LIT_restored_at_moscow
			drop_cosmetic_tag = yes
			set_politics = { ruling_party = democratic elections_allowed = yes }
			create_country_leader = { name = "Aleksandras Stulginskis" ideology = conservatism expire = "1940.6.15" }
			set_capital = { state = 11 }
			INT_eastern_settle_moscow_frontier = yes
			news_event = { id = endsieg_news.67 hours = 6 }
		}
	}
}

# The second regional government survived the German withdrawal in Crimea.
# Establish it before a Soviet-Ukrainian peace conference can release Crimea.
country_event = {
	id = INT_eastern.14
	title = INT_eastern.14.t
	desc = INT_eastern.14.d
	picture = GFX_report_event_regional
	fire_only_once = yes
	trigger = {
		tag = SOV is_ai = yes is_historical_focus_on = yes
		INT_eastern_local_wars_context = yes
		date > 1918.11.14 date < 1919.4.29
		137 = {
			owner = { is_ai = yes }
			OR = {
				AND = { is_owned_by = UKR is_fully_controlled_by = UKR UKR = { is_subject = no } }
				AND = { is_owned_by = CRI is_fully_controlled_by = CRI CRI = { OR = { is_subject = no is_subject_of = SOV } } }
			}
		}
	}
	mean_time_to_happen = { days = 1 }
	immediate = {
		if = { limit = { CRI = { is_subject_of = SOV } }
			set_autonomy = { target = CRI autonomy_state = autonomy_free end_wars = yes end_civil_wars = no }
		}
		137 = { if = { limit = { is_owned_by = UKR } transfer_state_to = CRI } }
		CRI = {
			if = { limit = { is_in_faction_with = SOV } leave_faction = yes }
			set_politics = { ruling_party = neutrality elections_allowed = no }
			set_cosmetic_tag = INT_CRI_regional
			create_country_leader = { name = "Solomon Krym" ideology = despotism picture = "gfx/leaders/TSS/Portrait_Solomon_Krym.dds" expire = "1936.9.9" }
		}
	}
	option = { name = INT_eastern.14.a }
}
