add_namespace = ww1_netherlands

# Keep the 1910 campaign's cabinet in step with the historical handover.
country_event = {
	id = ww1_netherlands.1
	is_triggered_only = no
	hidden = yes
	fire_only_once = yes

	trigger = {
		tag = HOL
		has_global_flag = welcome_event_WW1_active
		has_start_date < 1913.8.30
		date > 1913.8.28
		date < 1936.1.1
		has_country_leader = { character = HOL_theo_heemskerk ruling_only = yes }
	}

	mean_time_to_happen = { days = 1 }

	immediate = {
		retire_character = HOL_theo_heemskerk
		recruit_character = HOL_pieter_linden
		promote_character = HOL_pieter_linden
	}
}
