# Overview: Republic of China events, 1912-1936.
# Every event is fired from the RCH daily tick, so none of them can fire for the Qing.
add_namespace = rch_campaign

# The republic is proclaimed
country_event = {
	id = rch_campaign.1
	title = rch_campaign.1.t
	desc = rch_campaign.1.d
	picture = GFX_report_event_puyi

	is_triggered_only = yes

	option = {
		name = rch_campaign.1.a
		add_political_power = constant:rch_story.political_capital
		add_to_variable = { rch_legitimacy = constant:rch_story.legitimacy_gain }
	}
}
