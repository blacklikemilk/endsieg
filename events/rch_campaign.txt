# Overview: Republic of China events, 1912-1936.
# Every event is fired from the RCH daily tick, so none of them can fire for the Qing.
add_namespace = rch_campaign

# The republic is proclaimed
country_event = {
	id = rch_campaign.1
	title = rch_campaign.1.t
	desc = rch_campaign.1.d
	picture = GFX_report_event_puyi

	is_triggered_only = yes

	option = {
		name = rch_campaign.1.a
		add_political_power = constant:rch_story.political_capital
		add_to_variable = { rch_legitimacy = constant:rch_story.legitimacy_gain }
	}
}

# The Second Revolution
country_event = {
	id = rch_campaign.3
	title = rch_campaign.3.t
	desc = rch_campaign.3.d
	picture = GFX_report_event_kmt_rebellion

	is_triggered_only = yes

	option = {
		name = rch_campaign.3.a
		ai_chance = { factor = 3 modifier = { factor = 0 is_historical_focus_on = yes } }
		rch_settle_parliamentary = yes
	}
	option = {
		name = rch_campaign.3.b
		ai_chance = { factor = 4 }
		if = {
			limit = { has_country_leader = { name = "Yuan Shikai" ruling_only = yes } }
			rch_settle_beiyang = yes
			WW1_RCH_promote_yuan_emperor = yes
		}
		else = { rch_settle_beiyang = yes }
	}
	option = {
		name = rch_campaign.3.c
		ai_chance = { factor = 2 modifier = { factor = 0 is_historical_focus_on = yes } }
		rch_settle_kmt = yes
	}
}

# Yuan Shikai dies
country_event = {
	id = rch_campaign.5
	title = rch_campaign.5.t
	desc = { text = rch_campaign.5.d trigger = { has_country_flag = rch_succession_yuan_died NOT = { has_country_leader = { name = "Yuan Shikai" ruling_only = yes } } } }
	desc = { text = rch_campaign.5.alternative.d trigger = { OR = { NOT = { has_country_flag = rch_succession_yuan_died } has_country_leader = { name = "Yuan Shikai" ruling_only = yes } } } }
	picture = GFX_report_event_yuan_shikai_dead

	is_triggered_only = yes

	option = {
		name = rch_campaign.5.a
		trigger = {
			check_variable = { var = rch_succession_final_tier value = 0 compare = equals }
		}
	}

		name = rch_campaign.5.c
		trigger = {
			check_variable = { var = rch_succession_final_tier value = 1 compare = equals }

# The Hongxian choice, offered once in the last three weeks of 1915.
country_event = {
	id = rch_campaign.42
	title = rch_campaign.42.t
	desc = { text = rch_campaign.42.already.d trigger = { has_global_flag = yuan_emperor } }
	desc = { text = rch_campaign.42.d trigger = { NOT = { has_global_flag = yuan_emperor } } }
	picture = GFX_report_event_monarchy_revival

	is_triggered_only = yes

	option = {
		name = rch_campaign.42.b
		trigger = { NOT = { has_global_flag = yuan_emperor } WW1_RCH_can_proclaim_yuan_emperor = yes }
		ai_chance = { factor = 4 modifier = { factor = 100 is_historical_focus_on = yes } }
		if = { limit = { WW1_RCH_can_proclaim_yuan_emperor = yes } WW1_RCH_install_yuan_emperor = yes }
		add_to_variable = { rch_legitimacy = constant:rch_story.throne_legitimacy_loss }
		add_to_variable = { rch_military_influence = constant:rch_story.throne_military_step }
		set_country_flag = rch_hongxian
	}
	option = {
		name = rch_campaign.42.c
		trigger = { has_global_flag = yuan_emperor }
		ai_chance = { factor = 1 }
		add_to_variable = { rch_legitimacy = constant:rch_story.throne_legitimacy_loss }
		add_to_variable = { rch_military_influence = constant:rch_story.throne_military_step }
		set_country_flag = rch_hongxian
	}
	option = {
		name = rch_campaign.42.a
		trigger = { NOT = { has_global_flag = yuan_emperor } }
		ai_chance = { factor = 1 modifier = { factor = 0 is_historical_focus_on = yes WW1_RCH_can_proclaim_yuan_emperor = yes } }
		add_to_variable = { rch_legitimacy = constant:rch_story.throne_legitimacy_gain }
		add_to_variable = { rch_nationalist_support = constant:rch_story.throne_nationalist_gain }
		clr_country_flag = rch_gate_throne_refused
		set_country_flag = rch_gate_throne_refused
	}
}
