# Greece Joins Entente
country_event = {
trigger = { has_start_date < 1936.1.1 date < 1936.1.1 tag = FRA country_exists = GRE is_faction_leader = yes FR3_R_may_lead_actual_alliance = yes NOT = { has_war_with = GRE } GRE = { OR = { is_in_faction = no is_in_faction_with = FRA } } }

	id = ww1_france.6
	title = ww1_france.6.t
	desc = ww1_france.6.d
	picture = GFX_report_event_greece_joins_entente

	is_triggered_only = yes

	option = {
		name = ww1_france.6.a
		set_global_flag = greece_joins_entente
		FRA = {
			if = {
				limit = { NOT = { GRE = { is_in_faction_with = FRA } } }
				if = { limit = { GRE = { WW1_focus_authorizes_faction_entry_from_previous = yes } } add_to_faction = GRE }
			}
		}
		GRE = {
			if = {
				limit = { FRA = { has_war_with = GER } NOT = { has_war_with = GER } }
				add_to_war = { targeted_alliance = FRA enemy = GER }
			}
			if = {
				limit = { country_exists = BUL FRA = { has_war_with = BUL } NOT = { has_war_with = BUL } }
				add_to_war = { targeted_alliance = FRA enemy = BUL }
			}
		}
	}
}
