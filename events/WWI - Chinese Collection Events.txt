# Abdication of Puyi
country_event = {
	id = ww1_china.19
	title = ww1_china.19.t
	desc = ww1_china.19.d
	picture = GFX_report_event_puyi

	fire_only_once = yes

	trigger = {
		NOT = { has_global_flag = qing_xinhai_settled }
		NOT = { has_country_flag = WW1_QIN_abdication_pending }
		has_global_flag = welcome_event_WW1_active
		tag = QIN
		OR = {
			AND = {
				qing_historical_collapse = yes
				OR = { NOT = { has_country_flag = WW1_QIN_xinhai_war_started } has_country_flag = { flag = WW1_QIN_xinhai_war_started days > 110 } }
				date > 1912.2.11
				has_global_flag = xinhai_revolution
				has_war_with = RCH
				RCH = {
					exists = yes
					NOT = { has_capitulated = yes }
					any_owned_state = { always = yes }
				}
			}
			AND = {
				surrender_progress > 0.45
				is_ai = yes
				NOT = { qing_historical_collapse = yes }
			}
			AND = {
				surrender_progress > 0.80
				NOT = { controls_state = 608 } #beijing
				is_ai = no
			}
		}
	}

	mean_time_to_happen = {
		days = 1
	}

	option = {
		custom_effect_tooltip = WW1_historical_option_tt
		name = ww1_china.19.a
		if = {
			limit = {
				# The same route must still be valid when the option is chosen.
				AND = {
					NOT = { has_global_flag = qing_xinhai_settled }
					has_global_flag = welcome_event_WW1_active
					tag = QIN
					OR = {
						AND = {
							qing_historical_collapse = yes
							OR = { NOT = { has_country_flag = WW1_QIN_xinhai_war_started } has_country_flag = { flag = WW1_QIN_xinhai_war_started days > 110 } }
							date > 1912.2.11
							has_global_flag = xinhai_revolution
							has_war_with = RCH
							RCH = {
								exists = yes
								NOT = { has_capitulated = yes }
								any_owned_state = { always = yes }
							}
						}
						AND = {
							surrender_progress > 0.45
							is_ai = yes
							NOT = { qing_historical_collapse = yes }
						}
						AND = {
							surrender_progress > 0.80
							NOT = { controls_state = 608 } #beijing
							is_ai = no
						}
					}
				}
				exists = yes
				has_war_with = RCH
				RCH = { exists = yes NOT = { has_capitulated = yes } }
			}
			# Resolve from the surviving Republic after the event owner is annexed.
			set_country_flag = WW1_QIN_abdication_pending
			RCH = { country_event = { id = ww1_china.27 hours = 1 } }
		}
	}
}

# Intermission QIN to RCH
country_event = {
	id = ww1_china.27
	title = ww1_china.27.t
	desc = ww1_china.27.d
	picture = GFX_report_event_puyi

	is_triggered_only = yes

	immediate = {
		if = {
			limit = { country_exists = QIN has_war_with = QIN QIN = { has_country_flag = WW1_QIN_abdication_pending } }
			if = { limit = { is_ai = yes QIN = { is_ai = no } } change_tag_from = QIN }
			set_global_flag = end_of_qing
			set_global_flag = qing_xinhai_settled
			QIN = { set_country_flag = qing_collapsed remove_ideas = qing_divided_army_loyalties }
			annex_country = { target = QIN transfer_troops = yes }
			TIB = { WW1_TIB_settle_independence = yes }
		}
		else_if = { limit = { country_exists = QIN } QIN = { clr_country_flag = WW1_QIN_abdication_pending } }
	}

	option = {
		name = ww1_china.27.a
		rch_initialize_campaign = yes
	}
}
