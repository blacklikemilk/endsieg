# Royal Family Dead
country_event = {
	id = interwar_soviet.37
	title = interwar_soviet.37.t
	desc = interwar_soviet.37.d
	picture = GFX_report_event_royal_family

	fire_only_once = yes

	trigger = {
		NOT = { date > 1919.1.1 }
		date > 1918.8.30
		tag = SOV
		NOT = {
			WHR = {
				controls_state = 653
			}
		}
	}

	mean_time_to_happen = { days = 1 }

	option = {
		name = interwar_soviet.37.a
		set_global_flag = royal_family_dead
	}
	option = {
		name = interwar_soviet.37.b
		set_global_flag = royal_family_kept_alive
	}
}

	}
}

# Historical fallback for an unresolved Red-White war. The daily scheduler
# fires this on 25 October 1922, or at the next tick in an older ongoing save.
country_event = {
	id = interwar_soviet.62
	title = interwar_soviet.62.t
	desc = interwar_soviet.62.d
	picture = GFX_report_event_russian_civil_war
	is_triggered_only = yes
	fire_only_once = yes
	trigger = {
		tag = SOV
		date > 1922.10.24
		date < 1936.1.1
		has_global_flag = russian_civil_war
		NOT = { has_global_flag = RUS_CW_Over }
		has_government = communism
		has_capitulated = no
		is_subject = no
		has_war_with = WHR
	}
	immediate = { INT_SOV_finish_russian_civil_war = yes }
	option = { name = interwar_soviet.62.a }
}

# A victory earned earlier uses the same settlement after engine callbacks end.

		NOT = { has_global_flag = RUS_CW_Over }

		has_capitulated = no
		is_subject = no
	}
	immediate = { INT_SOV_finish_russian_civil_war = yes }
}
