# Germans are in Paris!
country_event = {
	id = kaiserschlachtvictory.1
	title = kaiserschlachtvictory.1.t
	desc = kaiserschlachtvictory.1.d
	picture = GFX_report_event_kaiserschlacht_victory

	fire_only_once = yes

	trigger = {
		has_global_flag = welcome_event_WW1_active
		NOT = { has_global_flag = player_wants_custom_peacedeals }
		tag = GER
		NOT = { date > 1936.1.1 }
		NOT = {has_global_flag = compiegne}
		FRA = { has_war_with = GER }
		AND = {
			GER = {
				controls_state = 16 # Ille de France
			}
		}

		NOT = { has_global_flag = WW1_french_armistice_signed }
		FRA = { OR = { has_capitulated = yes surrender_progress > 0.75 } }
	}

	mean_time_to_happen = { days = 1 }

	option = {
		name = kaiserschlachtvictory.1.a
		ai_chance = { factor = 100}
		FRA = {
			country_event = { id = kaiserschlachtvictory.2 hours = 12}
		}
	}
	option = {
		name = kaiserschlachtvictory.1.b
		ai_chance = {factor = 0}
		add_war_support = -0.20
		add_stability = -0.20
		add_political_power = -150
		set_global_flag = totalwarww1
		custom_effect_tooltip = no_content_ahead_tt
		if = { limit = { country_has_war_exhaustion = yes } add_to_variable = { war_exhaustion_variable = 300} }
		hidden_effect = {
			hidden_effect = {sound_effect = diplomatic_failure}
		}
	}
}
