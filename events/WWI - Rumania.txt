# The Romanian daily on-action checks dates, war participants and state control.
country_event = {
    id = eg_romania.1
    title = eg_romania.1.t
    desc = eg_romania.1.d
    picture = GFX_report_event_treaty_of_bucharest
    is_triggered_only = yes
    fire_only_once = yes

    immediate = {
        set_global_flag = bucharestreaty
        clr_country_flag = WW1_ROM_focsani_armistice
        WW1_ROM_armistice_with_central_powers = yes
        every_state = {
            limit = {
                OR = { is_owned_by = ROM AND = { has_state_flag = WW1_ROM_bucharest_occupation OR = { is_owned_by = GER is_owned_by = AUH is_owned_by = BUL } } }
                OR = { state = 77 state = 1015 state = 1030 state = 1018 }
                OR = { is_fully_controlled_by = ROM is_fully_controlled_by = GER is_fully_controlled_by = AUH is_fully_controlled_by = BUL }
            }
            clr_state_flag = WW1_ROM_bucharest_occupation
            set_state_flag = WW1_ROM_bucharest_cession
            if = { limit = { state = 77 } transfer_state_to = BUL }
            else_if = { limit = { state = 1018 } transfer_state_to = AUH }
            # Germany represents the four-power condominium on the map.
            else = { transfer_state_to = GER }
        }
        if = {
            limit = { owns_state = 79 controls_state = 79 46 = { has_state_flag = WW1_ROM_bucharest_occupation } }
            set_capital = { state = 79 }
        }
        WW1_ROM_apply_bucharest_oil_concession = yes
        news_event = { hours = 6 id = endsieg_news.5 }
    }
    option = { name = eg_romania.1.a }
}
