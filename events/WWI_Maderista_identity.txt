add_namespace = endsieg_maderista_identity

# Repair names in old Mexican saves without changing the Madagascar tag globally.
country_event = {
 id = endsieg_maderista_identity.1
 hidden = yes
 trigger = {
  tag = MEX
  has_start_date < 1936.1.1
  date < 1936.1.1
  has_global_flag = maderista_revolution
  NOT = { has_global_flag = maderista_victory }
  OR = {
   MDS = { exists = yes NOT = { has_cosmetic_tag = MDS } }
   MAD = { has_war_with = MEX owns_state = 482 NOT = { owns_state = 543 } NOT = { has_cosmetic_tag = MDS } }
  }
 }
 mean_time_to_happen = { days = 1 }
 immediate = { endsieg_ww1_restore_maderista_names = yes }
}
