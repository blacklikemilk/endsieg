#Georgia Declares Independence
country_event = {
	id = transcauc.1
	title = transcauc.1.t
	desc = transcauc.1.d
	picture = GFX_report_event_georgiasecedes

	fire_only_once = yes

	trigger = {
		has_global_flag = welcome_event_WW1_active
		has_start_date < 1918.5.27
		NOT = { date > 1932.12.1 }
		tag = TRC
		NOT = { is_subject_of = GER }
		date > 1918.5.25
		owns_state = 231
		controls_state = 231
	}

	mean_time_to_happen = {
		days = 1
	}

	option = {
		custom_effect_tooltip = WW1_historical_option_tt
		name = transcauc.1.a
		ai_chance = { factor = 100}
		GEO = {
			transfer_state = 231
			set_country_flag = transcauc_georgia_separated
			WW1_Caucasus_initialize_georgia = yes
		}
	}
	option = {
		name = transcauc.1.b
		trigger = { NOT = { country_exists = GEO } }
		ai_chance = { factor = 0}
		GEO = {
			transfer_state = 231 #Georgia
			set_country_flag = WW1_Caucasus_republic_from_transcaucasia
			WW1_Caucasus_initialize_georgia = yes
		}
		TRC = { declare_war_on = { target = GEO type = annex_everything } }
	}
	option = {
		name = transcauc.1.c
		trigger = { NOT = { country_exists = GEO } }
		ai_chance = { factor = 0}
		GEO = {
			transfer_state = 231
			change_tag_from = TRC
			set_country_flag = transcauc_georgia_separated
			WW1_Caucasus_initialize_georgia = yes
		}
	}
}

#Armenia and Azerbaijan Declare Independence
country_event = {
	id = transcauc.2
	title = transcauc.2.t
	desc = transcauc.2.d
	picture = GFX_report_event_transcaucdead

	fire_only_once = yes

	trigger = {
		has_global_flag = welcome_event_WW1_active
		has_start_date < 1918.5.29
		NOT = { date > 1932.12.1 }
		tag = TRC
		NOT = { is_subject_of = GER }
		date > 1918.5.27
		is_ai = no
		OR = { AND = { NOT = { country_exists = ARM } OR = { 230 = { is_owned_by = TRC is_controlled_by = TRC } 990 = { is_owned_by = TRC is_controlled_by = TRC } } } AND = { NOT = { country_exists = AZR } NOT = { country_exists = ARM } 229 = { is_owned_by = TRC is_controlled_by = TRC } OR = { 230 = { is_owned_by = TRC is_controlled_by = TRC } 990 = { is_owned_by = TRC is_controlled_by = TRC } } } }
		NOT = { has_war_with = GEO }
	}

	mean_time_to_happen = {
		days = 1
	}

	option = {
		name = transcauc.2.a # For Armenia!
		trigger = { NOT = { country_exists = ARM } OR = { 230 = { is_owned_by = TRC is_controlled_by = TRC } 990 = { is_owned_by = TRC is_controlled_by = TRC } } }
		ai_chance = {factor = 50}
		GEO = {
			if = {
				limit = { 231 = { is_owned_by = TRC is_controlled_by = TRC } }
				transfer_state = 231
				set_country_flag = WW1_Caucasus_republic_from_transcaucasia
				WW1_Caucasus_initialize_georgia = yes
			}
		}
		AZR = {
			if = { limit = { 229 = { is_owned_by = TRC is_controlled_by = TRC } NOT = { country_exists = AZR } } transfer_state = 229 load_oob = AZR_1918_independence }
			else_if = { limit = { 229 = { is_owned_by = TRC is_controlled_by = TRC } } transfer_state = 229 }
			set_country_flag = WW1_Caucasus_republic_from_transcaucasia
			WW1_Caucasus_initialize_azerbaijan = yes
		}
		ARM = {
			# Continue as Armenia instead of annexing the Transcaucasian
			# Republic. The remaining states are transferred below.
			change_tag_from = TRC
			if = { limit = { 230 = { is_owned_by = TRC is_controlled_by = TRC } } transfer_state = 230 }
			if = { limit = { 990 = { is_owned_by = TRC is_controlled_by = TRC } } transfer_state = 990 }
			if = { limit = { 230 = { is_owned_by = ARM is_controlled_by = ARM } } load_oob = "ARM_1918" }
			set_country_flag = WW1_Caucasus_republic_from_transcaucasia
			WW1_Caucasus_initialize_armenia = yes
			if = {
				limit = {
					country_exists = BAK
					NOT = { has_war_with = BAK }
				}
				declare_war_on = {
					target = BAK
					type = annex_everything
				}
			}
		}
		if = { limit = { country_exists = AZR country_exists = ARM AZR = { NOT = { has_war_with = ARM } } }
		AZR = {
			declare_war_on = {
				target = ARM
				type = take_state
				generator = { 230 990 }
			}
			if = {
				limit = {
					country_exists = BAK
					NOT = { has_war_with = BAK }
				}
				declare_war_on = {
					target = BAK
					type = annex_everything
				}
			}
		} }
	}
	option = {
		ai_chance = { factor = 50 }
		name = transcauc.2.b # For Azerbaijan!
		trigger = { NOT = { country_exists = AZR } NOT = { country_exists = ARM } 229 = { is_owned_by = TRC is_controlled_by = TRC } OR = { 230 = { is_owned_by = TRC is_controlled_by = TRC } 990 = { is_owned_by = TRC is_controlled_by = TRC } } }
		GEO = {
			if = {
				limit = { 231 = { is_owned_by = TRC is_controlled_by = TRC } }
				transfer_state = 231
				set_country_flag = WW1_Caucasus_republic_from_transcaucasia
				WW1_Caucasus_initialize_georgia = yes
			}
		}
		ARM = {
			if = { limit = { 230 = { is_owned_by = TRC is_controlled_by = TRC } } transfer_state = 230 }
			if = { limit = { 990 = { is_owned_by = TRC is_controlled_by = TRC } } transfer_state = 990 }
			if = { limit = { 230 = { is_owned_by = ARM is_controlled_by = ARM } } load_oob = "ARM_1918" }
			set_country_flag = WW1_Caucasus_republic_from_transcaucasia
			WW1_Caucasus_initialize_armenia = yes
		}
		AZR = {
			transfer_state = 229
			load_oob = AZR_1918_independence
			if = {
				limit = {
					ROOT = {is_ai = no}
				}
				change_tag_from = TRC
			}
			set_country_flag = WW1_Caucasus_republic_from_transcaucasia
			WW1_Caucasus_initialize_azerbaijan = yes
		}
		if = { limit = { country_exists = AZR country_exists = ARM AZR = { NOT = { has_war_with = ARM } } }
		AZR = {
			declare_war_on = {
				target = ARM
				type = take_state
				generator = { 230 990 }
			}
			if = {
				limit = {
					country_exists = BAK
					NOT = { has_war_with = BAK }
				}
				declare_war_on = {
					target = BAK
					type = annex_everything
				}
			}
		} }
	}
}

# Armenia and Azerbaijan Declare Independence
country_event = {
	id = transcauc.10
	title = transcauc.10.t
	desc = transcauc.10.d
	picture = GFX_report_event_transcaucdead

	fire_only_once = yes

	trigger = {
		has_global_flag = welcome_event_WW1_active
		has_start_date < 1918.5.30
		NOT = { date > 1932.12.1 }
		tag = TRC
		NOT = { is_subject_of = GER }
		date > 1918.5.28
		is_ai = yes
		NOT = { owns_state = 231 }
		OR = { owns_state = 229 owns_state = 230 owns_state = 990 }
	}

	mean_time_to_happen = {
		days = 1
	}

	option = {
		custom_effect_tooltip = WW1_historical_option_tt
		name = transcauc.10.a
		if = { limit = { NOT = { country_exists = AZR } } release = AZR }
		else = { AZR = { if = { limit = { 229 = { is_owned_by = TRC is_controlled_by = TRC } } transfer_state = 229 } } }
		if = { limit = { NOT = { country_exists = ARM } } release = ARM }
		else = { ARM = {
			if = { limit = { 230 = { is_owned_by = TRC is_controlled_by = TRC } } transfer_state = 230 }
			if = { limit = { 990 = { is_owned_by = TRC is_controlled_by = TRC } } transfer_state = 990 }
		} }
		AZR = {
			set_country_flag = WW1_Caucasus_republic_from_transcaucasia
			WW1_Caucasus_initialize_azerbaijan = yes
		}
		ARM = {
			set_country_flag = WW1_Caucasus_republic_from_transcaucasia
			WW1_Caucasus_initialize_armenia = yes
		}
		every_state = {
			limit = { is_core_of = TRC }
			remove_core_of = TRC
		}
	}
}

# A Soviet conference can leave the Azerbaijani SSR with Baku alone.
country_event = {
	id = transcauc.11
	title = transcauc.11.t
	desc = transcauc.11.d
	picture = GFX_report_event_generic_parliament
	is_triggered_only = yes
	fire_only_once = yes
	immediate = {
		set_country_flag = WW1_AZR_revolutionary_committee
		if = { limit = { NOT = { has_country_leader = { name = "Nariman Narimanov" ruling_only = yes } } }
			create_country_leader = { name = "Nariman Narimanov" picture = "gfx/leaders/AZR/Portrait_Nariman_Narimanov.dds" ideology = marxism expire = "1925.3.19" }
			set_country_flag = WW1_AZR_narimanov_portrait_installed
		}
		229 = { if = { limit = { is_owned_by = SOV is_fully_controlled_by = SOV } transfer_state_to = AZR } }
		WW1_AZR_end_inherited_wars = yes
		if = { limit = { controls_state = 1016 } set_capital = { state = 1016 remember_old_capital = no } }
	}
	option = { name = transcauc.11.a }
}
