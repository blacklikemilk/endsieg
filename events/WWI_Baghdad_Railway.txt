				limit = { NOT = { OR = { has_war_with = GER has_war_with = TUR } } }
				endsieg_fixed_effect_414e13638fbbf09b = yes
				GER = { set_country_flag = GE1_baghdad_railway_british_accommodation }
			}
			else = {
				GER = {
					set_country_flag = GE1_baghdad_railway_british_wartime_opposition
					set_country_flag = GE1_baghdad_railway_british_protest
				}
			}
			GER = {
				set_country_flag = GE1_baghdad_railway_british_response_received
				GE1_baghdad_railway_reconcile_british_response = yes
			}
		}
	}
	option = {
		name = ww1_baghdad.1.b
		trigger = { GE1_baghdad_railway_britain_can_declare_war = yes }
		ai_chance = {
			factor = 20
			modifier = { factor = 0 is_historical_focus_on = yes }
			modifier = { factor = 0 has_war = yes }
			modifier = { factor = 0 NOT = { has_war_support > 0.40 } }
		}
		if = {
			limit = { GE1_baghdad_railway_britain_can_declare_war = yes }
			declare_war_on = { target = GER type = annex_everything }
			if = {
				limit = { has_war_with = GER }
				GER = {
					set_country_flag = GE1_baghdad_railway_british_protest
					set_country_flag = GE1_baghdad_railway_british_war_response
					set_country_flag = GE1_baghdad_railway_british_response_received
					GE1_baghdad_railway_reconcile_british_response = yes
					custom_effect_tooltip = GE1_baghdad_railway_bond_required_tt
				}
			}
		}
	}
}
