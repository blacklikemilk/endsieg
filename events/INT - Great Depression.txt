add_namespace = INT_depression

# Fired once in late October 1929 for every contemporary major power.
country_event = {
	id = INT_depression.1
	title = INT_depression.1.t
	desc = INT_depression.1.d
	picture = GFX_report_event_generic_factory
	is_triggered_only = yes

	option = {
		name = INT_depression.1.a
		if = {
			limit = { tag = GER }
			add_ideas = INT_GER_great_depression
			INT_GER_industry_crash_effect = yes
			if = {
				limit = { has_global_flag = versailles date > 1929.10.23 date < 1930.1.1 }
				news_event = { id = INT_ger_news.9 days = 1 }
			}
		}
		else_if = { limit = { tag = USA } add_ideas = INT_USA_great_depression }
		else_if = { limit = { tag = ENG } add_ideas = INT_ENG_great_depression }
		else_if = { limit = { tag = FRA } add_ideas = INT_FRA_great_depression }
		else_if = { limit = { tag = ITA } add_ideas = INT_ITA_great_depression }
		else_if = { limit = { tag = JAP } add_ideas = INT_JAP_great_depression }
		else_if = { limit = { tag = SOV } add_ideas = INT_SOV_great_depression }
		else = { add_ideas = INT_great_depression }
		set_country_flag = INT_depression_hit
	}
}
