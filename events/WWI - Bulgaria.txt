# Treaty of Bucharest/Constantinople 1913
country_event = {
	id = ww1_bulgaria.2
	title = ww1_bulgaria.2.t
	desc = ww1_bulgaria.2.d
	picture = GFX_report_event_treaty_of_london_1913

	fire_only_once = yes

	trigger = {
		has_global_flag = welcome_event_WW1_active
		NOT = { date > 1914.1.1 }
		tag = BUL
		has_global_flag = second_balkan_war
		# Give the delayed invitations time to resolve without requiring both countries to join.
		OR = {
			AND = {
				has_global_flag = second_balkan_war_romania_joined
				has_global_flag = second_balkan_war_ottomans_joined
			}
			has_global_flag = { flag = second_balkan_war days > 24 }
		}
		OR = {
			AND = {
				is_ai = yes
				OR = {
					surrender_progress > 0.10
					NOT = { 1021 = { is_owned_and_controlled_by = BUL } }
					NOT = { 184 = { is_owned_and_controlled_by = BUL } }
				}
			}
			AND = {
				is_ai = no
				OR = {
					surrender_progress > 0.25 #should trigger if Varna falls
				}
			}
		}
	}

	mean_time_to_happen = { days = 1 }

	option = {
		custom_effect_tooltip = WW1_historical_option_tt
		name = ww1_serbia.3.a
		set_global_flag = treaty_of_bucharest_1913
		news_event = { id = ww1_postwar_news.27 hours = 6 }

		if = {
			limit = { country_exists = SER owns_state = 1021 }
			SER = { transfer_state = 1021 }
		}
		if = {
			limit = { country_exists = ROM has_war_with = ROM owns_state = 77 }
			ROM = { transfer_state = 77 }
		}
		if = {
			limit = { country_exists = TUR has_war_with = TUR owns_state = 1036 }
			TUR = { transfer_state = 1036 }
		}
		if = {
			limit = { country_exists = GRE has_war_with = GRE owns_state = 184 }
			GRE = { transfer_state = 184 }
		}
		if = {
			limit = { country_exists = ALB 934 = { is_owned_by = BUL } }
			ALB = { transfer_state = 934 }
		}
		every_other_country = {
			limit = {
				OR = { tag = SER tag = ROM tag = TUR tag = GRE tag = MNT }
				has_war_with = BUL
			}
			white_peace = BUL
		}
		clr_global_flag = second_balkan_war
		clr_global_flag = second_balkan_war_romania_joined
		clr_global_flag = second_balkan_war_ottomans_joined
		hidden_effect = {
			SER = { if = { limit = { WW1_is_balkan_league_member = yes } dismantle_faction = yes } }
		}
	}
}

# The French-led Allied administration replaces Bulgaria in Western Thrace.
country_event = {
    id = ww1_bulgaria.42
    hidden = yes
    is_triggered_only = no
    trigger = {
        tag = FRA
        country_exists = BUL
        has_start_date < 1920.5.14
        NOT = { date < 1919.10.14 }
        date < 1920.5.14
        has_global_flag = armistice_of_salonica
        NOT = {
            OR = {
                has_war_with = BUL
                has_war_with = GRE
                has_global_flag = central_powers_victory
                has_global_flag = player_wants_custom_peacedeals
            }
        }
        997 = {
            OR = {
                is_owned_and_controlled_by = BUL
                AND = {
                    is_owned_and_controlled_by = GRE
                    BUL = { has_country_flag = WW1_BUL_armistice_applied }
                }
            }
        }
    }
    mean_time_to_happen = { days = 1 }
    immediate = {
        transfer_state = 997
        if = {
            limit = { date < 1919.10.15 }
            GRE = { country_event = { id = ww1_bulgaria.43 days = 213 } }
        }
    }
}
