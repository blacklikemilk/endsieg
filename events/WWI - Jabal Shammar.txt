add_namespace = shammar

# Assasination of the Emir
country_event = {
	id = shammar.1
	title = shammar.1.t
	desc = shammar.1.d
	picture = GFX_report_event_emir_dead

	fire_only_once = yes

	trigger = {
		has_global_flag = welcome_event_WW1_active
		has_start_date < 1920.3.30
		date < 1921.4.1
		has_country_leader = { character = SHA_saud_abdulaziz ruling_only = yes }
		tag = SHA
		date > 1920.3.28
	}

	mean_time_to_happen = {
		days = 1
	}

	option = {
		name = shammar.1.a
		add_stability = -0.05
		retire_character = SHA_saud_abdulaziz
		if = {
			limit = { has_character = SHA_abdullah_bin_mutaib }
			promote_character = SHA_abdullah_bin_mutaib
		}
		set_politics = { ruling_party = neutrality elections_allowed = no }
	}
}

# Fall of Central Powers
country_event = {
	id = shammar.2
	title = shammar.2.t
	desc = shammar.2.d
	picture = GFX_report_event_mudros

	is_triggered_only = yes

	option = {
		name = shammar.2.a
		if = {
			limit = { has_global_flag = armistice_of_mudros NOT = { has_global_flag = central_powers_victory } }
			every_other_country = {
				limit = { has_war_with = SHA OR = { tag = ENG tag = FRA tag = HEJ tag = EN3 is_in_faction_with = ENG is_subject_of = ENG } NOT = { tag = NEJ } }
				white_peace = SHA
			}
			if = { limit = { is_in_faction_with = GER is_faction_leader = no } leave_faction = yes }
		}
	}
}

# Use the spring campaign to date Muhammad's contested accession. Abdullah was reported in Riyadh by August.
country_event = {
	id = shammar.3
	title = shammar.3.t
	desc = shammar.3.d
	picture = GFX_report_event_shammar_war
	fire_only_once = yes
	trigger = {
		tag = SHA owns_state = 1014 has_government = neutrality
		has_start_date < 1921.4.1 date > 1921.3.31 date < 1922.1.1
		OR = {
			has_country_leader = { character = SHA_abdullah_bin_mutaib ruling_only = yes }
			has_country_leader = { name = "Abdullah bin Mutʿib" ruling_only = yes }
		}
	}
	mean_time_to_happen = { days = 1 }
	immediate = {
		if = {
			limit = { has_character = SHA_muhammad_bin_talal }
			promote_character = SHA_muhammad_bin_talal
		}
		set_politics = { ruling_party = neutrality elections_allowed = no }
	}
	option = { name = shammar.3.a }
}
