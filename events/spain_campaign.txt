# Overview: Spain's chapter milestones, crises and news for 1910-1945.
# Namespace spain_campaign. Chapter transitions are only ever triggered from these options,
# so a tree change is always visible to the player as a decision.
# The civil war itself is not fired here: events/LAR_Spain.txt and common/decisions/SPR.txt still own it.

add_namespace = spain_campaign

# =====================================================================================
# 1923 - the pronunciamiento
# =====================================================================================

country_event = {
	id = spain_campaign.10
	title = spain_campaign.10.t
	desc = spain_campaign.10.d
	picture = GFX_report_event_army_border
	is_triggered_only = yes

	option = {
		name = spain_campaign.10.a
		ai_chance = { base = 70 }
		spain_enter_rivera_chapter = yes
		spain_loyalty_gain = yes
		add_political_power = constant:spain_campaign.pp_medium
	}

	option = {
		name = spain_campaign.10.b
		ai_chance = { base = 30 }
		spain_enter_rivera_chapter = yes
		spain_push_crown_large = yes
		add_ideas = spain_crown_government
		add_stability = constant:spain_campaign.stability_small
	}
}

# =====================================================================================
# A coup attempt
# =====================================================================================

country_event = {
	id = spain_campaign.40
	title = spain_campaign.40.t
	desc = spain_campaign.40.d
	picture = GFX_report_event_army_border
	is_triggered_only = yes

	option = {
		name = spain_campaign.40.a
		ai_chance = { base = 60 }
		if = {
			limit = { spain_loyalty_low = yes }
			country_event = { id = spain_campaign.41 days = 1 }
		}
		else = {
			spain_coup_outcome_failed = yes
			add_stability = constant:spain_campaign.stability_small
			country_event = { id = spain_campaign.42 days = 1 }
		}
	}

	option = {
		name = spain_campaign.40.b
		ai_chance = { base = 25 }
		spain_coup_outcome_conceded = yes
		add_stability = constant:spain_campaign.stability_cost
		spain_push_crown = yes
	}

	option = {
		name = spain_campaign.40.c
		ai_chance = { base = 15 }
		trigger = { has_political_power > 49 }
		add_political_power = constant:spain_campaign.pp_medium
		spain_coup_outcome_dictatorship = yes
		spain_push_crown_small = yes
	}
}
