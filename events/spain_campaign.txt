# Overview: Spain's chapter milestones, crises and news for 1910-1945.
# Namespace spain_campaign. Chapter transitions are only ever triggered from these options,
# so a tree change is always visible to the player as a decision.
# The civil war itself is not fired here: events/LAR_Spain.txt and common/decisions/SPR.txt still own it.

# The Rif outgrows the state
# =====================================================================================

# Fired once by spain_rif_exhaustion_check when spain_rif_commitment reaches the exhaustion
# band. The three options are the three ways a cabinet can answer a colonial war it has paid for
# too long: send more, buy the tribes off, or dig in and admit the cost.
country_event = {
	id = spain_campaign.13
	title = spain_campaign.13.t
	desc = spain_campaign.13.d
	picture = GFX_report_event_army_border
	is_triggered_only = yes

	option = {
		name = spain_campaign.13.a
		ai_chance = { base = 50 }
		add_to_variable = { var = spain_rif_commitment value = constant:spain_campaign.rif_gain }
		add_war_support = constant:spain_campaign.war_support_small
		add_manpower = constant:spain_campaign.manpower_cost
		spain_loyalty_gain_small = yes
		spain_refresh_campaign_ideas = yes
	}

	option = {
		name = spain_campaign.13.b
		ai_chance = { base = 25 }
		add_to_variable = { var = spain_rif_commitment value = constant:spain_campaign.rif_loss }
		add_stability = constant:spain_campaign.stability_small
		add_war_support = constant:spain_campaign.war_support_cost
		add_political_power = constant:spain_campaign.pp_medium
		spain_loyalty_loss_small = yes
		set_country_flag = spain_rif_bought_off
	}

	option = {
		name = spain_campaign.13.c
		ai_chance = { base = 25 }
		if = { limit = { NOT = { has_idea = spain_rif_quagmire } } add_ideas = spain_rif_quagmire }
		spain_loyalty_gain = yes
		add_stability = constant:spain_campaign.stability_cost
		spain_refresh_campaign_ideas = yes
	}
}

# =====================================================================================
# The Crown ends the ministerial crisis itself
# =====================================================================================

# Fired by SPR_ministerial_crisis when its deadline runs out. The King does not wait for a
# cabinet that cannot count: the lists are drawn up and the machine takes another draw, which is
# what an ignored crisis costs. Once the donors are exhausted there is nothing left to
# manufacture, and the country simply watches the government lose its authority.
country_event = {
	id = spain_campaign.32
	title = spain_campaign.32.t
	desc = spain_campaign.32.d
	picture = GFX_report_event_kuncrowd
	is_triggered_only = yes

	option = {
		name = spain_campaign.32.a
		ai_chance = { base = 100 }
		hidden_effect = {
			# clear first: the appointment the turno makes may legitimately open a new crisis
			clr_country_flag = spain_ministry_crisis_open
			if = {
				limit = { spain_encasillado_available = yes }
				spain_encasillado = yes
			}
			else = {
				spain_loyalty_loss_small = yes
				add_stability = constant:spain_campaign.stability_cost
				spain_government_recount = yes
			}
			spain_cortes_refresh_display = yes
		}
		custom_effect_tooltip = spain_campaign.32.a.tt
	}
}
