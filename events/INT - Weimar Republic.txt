# The first republican advisers are recruited; late transitions catch up.
country_event = {
	id = INT_weimar.9
	title = INT_weimar.9.t
	desc = INT_weimar.9.d
	picture = GFX_INT_photo_INT_weimar_9
	is_triggered_only = yes
	option = {
		name = INT_weimar.9.a
		set_politics = { ruling_party = democratic elections_allowed = yes }
		set_popularities = { democratic = 65 neutrality = 25 communism = 10 fascism = 0 }
		if = {
			limit = { date < 1925.4.26 }
			if = { limit = { NOT = { has_character = GER_friedrich_ebert } } recruit_character = GER_friedrich_ebert }
			if = {
				limit = { NOT = { has_country_leader = { character = GER_friedrich_ebert ruling_only = yes } } }
				retire_country_leader = yes
				GER_friedrich_ebert = { promote_character = yes }
			}
		}
		else = {
			if = { limit = { NOT = { has_character = GER_paul_von_hindenburg } } recruit_character = GER_paul_von_hindenburg }
			if = {
				limit = { NOT = { has_country_leader = { character = GER_paul_von_hindenburg ruling_only = yes } } }
				retire_country_leader = yes
				GER_paul_von_hindenburg = { promote_character = yes }
			}
		}
		if = { limit = { date < 1920.3.23 NOT = { has_character = GER_INT_gustav_noske } } recruit_character = GER_INT_gustav_noske }
		if = { limit = { date < 1921.8.27 NOT = { has_character = GER_INT_matthias_erzberger } } recruit_character = GER_INT_matthias_erzberger }
		if = { limit = { date < 1933.3.24 NOT = { has_character = GER_INT_otto_wels } } recruit_character = GER_INT_otto_wels }
		if = { limit = { date > 1920.3.22 date < 1932.7.21 NOT = { has_character = GER_INT_carl_severing } } recruit_character = GER_INT_carl_severing }
		if = { limit = { date > 1922.1.31 date < 1922.6.25 NOT = { has_character = GER_INT_walther_rathenau } } recruit_character = GER_INT_walther_rathenau }
		if = { limit = { date > 1923.12.21 date < 1930.3.7 NOT = { has_character = GER_INT_hjalmar_schacht } } recruit_character = GER_INT_hjalmar_schacht }
		if = { limit = { date > 1923.11.19 has_character = havenstein } retire_character = havenstein }
		set_country_flag = GER_INT_cabinet_ready
		GER_INT_update_chancellor = yes
	}
}
