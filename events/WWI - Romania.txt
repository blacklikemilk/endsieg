add_namespace = ww1_romania

# Carol I Dies
country_event = {
	id = ww1_romania.1
	title = ww1_romania.1.t
	desc = ww1_romania.1.d
	picture = GFX_report_event_carol

	fire_only_once = yes

	trigger = {
		has_global_flag = welcome_event_WW1_active
		has_start_date < 1914.10.10
		date > 1914.10.9
		has_country_leader = { character = ROM_Carol_I ruling_only = yes }
		date < 1915.1.1
		tag = ROM
	}

	mean_time_to_happen = { days = 1 }

	option = {
		custom_effect_tooltip = WW1_historical_option_tt
		name = ww1_romania.1.a
		news_event = { hours = 6 id = endsieg_news.246 }
		retire_character = ROM_Carol_I
		promote_character = ROM_Ferdinand_I
	}
}

# Parliamentary alternation under Ferdinand's constitutional monarchy.
country_event = {
	id = ww1_romania.1922
	title = ww1_romania.1922.t
	desc = ww1_romania.1922.d
	picture = GFX_report_event_constituent_assembly
	fire_only_once = yes
	trigger = {
		tag = ROM is_historical_focus_on = yes is_ai = yes
		has_start_date < 1919.1.1 date > 1922.1.18 date < 1926.3.27
		has_global_flag = compiegne is_subject = no has_civil_war = no
		NOT = { has_global_flag = central_powers_victory }
		NOT = { has_global_flag = player_wants_custom_peacedeals }
		has_country_leader = { character = ROM_Ferdinand_I ruling_only = yes }
	}
	mean_time_to_happen = { days = 1 }
	immediate = {
		add_country_leader_role = {
			character = ROM_Ion_I_C_Bratianu
			country_leader = { ideology = liberalism expire = "1927.11.24" }
			promote_leader = yes
		}
		set_politics = { ruling_party = democratic elections_allowed = yes election_frequency = 48 last_election = "1920.5.25" }
		set_country_flag = WWI_ROM_bratianu_cabinet
	}
	option = { name = ww1_romania.1922.a }
}
