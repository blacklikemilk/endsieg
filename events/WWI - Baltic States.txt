# Dissolution of United Baltic Duchy
country_event = {
	id = ww1_baltic.1
	title = ww1_baltic.1.t
	desc = ww1_baltic.1.d
	picture = GFX_report_event_balticdissolution

	is_triggered_only = no
	trigger = {
		tag = UBD
		has_global_flag = compiegne
		date > 1918.11.10
		date < 1923.1.1
		any_owned_state = { always = yes }
		NOT = {
			OR = {
				has_global_flag = ubd_dead
				has_global_flag = central_powers_victory
				has_global_flag = player_wants_custom_peacedeals
				has_global_flag = ww1_compiegne_armistice_failed
			}
		}
	}
	fire_only_once = yes
	mean_time_to_happen = { days = 1 }

	option = {
		custom_effect_tooltip = WW1_historical_option_tt
		name = ww1_baltic.1.a
		if = {
			limit = { has_global_flag = compiegne NOT = { OR = { has_global_flag = ubd_dead has_global_flag = ww1_compiegne_armistice_failed has_global_flag = central_powers_victory has_global_flag = player_wants_custom_peacedeals } } }
			INT_eastern_dissolve_ubd = yes
			if = { limit = { date < 1919.1.1 } news_event = { hours = 6 id = endsieg_news.64 } }
		}
	}
}

			expire = "1920.6.19"

			traits = {
				#
			}
		}
	}
}
