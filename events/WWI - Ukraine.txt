# Ukranian Hetmanate
country_event = {
	id = ww1_ukraine.1
	title = ww1_ukraine.1.t
	desc = ww1_ukraine.1.d
	picture = GFX_report_event_hetmanate

	fire_only_once = yes

	trigger = {
		WW1_UKR_german_supported_coup_context = yes
	}

	mean_time_to_happen = { days = 1 }

	option = {
		custom_effect_tooltip = WW1_historical_option_tt
		name = ww1_ukraine.1.a
		if = {
			limit = { WW1_UKR_german_supported_coup_context = yes }
		set_global_flag = hetmanate
		news_event = { hours = 6 id = endsieg_news.13 }

		set_country_flag = skoropadskytakesover
		set_cosmetic_tag = UKR_HET
		set_politics = {
			ruling_party = neutrality
			elections_allowed = no    }

        set_popularities = {
            fascism = 0
            neutrality = 45
            democratic = 40
            communism = 15
		}
		retire_country_leader = yes
		endsieg_character_activate_ukr_pavlo_skoropadsky_3aea4adf = yes
		UKR_pavlo_skoropadsky = { promote_character = yes }
		UKR = {
			load_oob = "HET_1918"
		}
		if = {
			limit = { country_exists = GER }
			if = { limit = { is_subject = no NOT = { has_global_flag = RUS_ww1_brest_limited_treaty } } GER = { puppet = UKR } }
			give_military_access = GER
		}
		}
	}
}

# The Act of Union preserves the separate West Ukrainian administration.
country_event = {
	id = ww1_ukraine.4
	title = ww1_ukraine.4.t
	desc = ww1_ukraine.4.d
	picture = GFX_report_event_act_zluky
	fire_only_once = yes
	trigger = {
		INT_UKR_directory_active = yes
		has_start_date < 1919.1.22
		date > 1919.1.21
		date < 1919.2.5
		NOT = { has_global_flag = INT_UKR_act_of_union }
		WUR = { exists = yes is_subject = no OR = { owns_state = 89 owns_state = 91 } }
	}
	mean_time_to_happen = { days = 1 }
	option = {
		custom_effect_tooltip = WW1_historical_option_tt
		name = ww1_ukraine.4.a
		if = {
			limit = {
				INT_UKR_directory_active = yes
				WUR = { exists = yes is_subject = no OR = { owns_state = 89 owns_state = 91 } }
				NOT = { has_war_with = WUR }
			}
			set_global_flag = INT_UKR_act_of_union
			give_military_access = WUR
			WUR = { give_military_access = UKR }
			news_event = { id = ww1_ukraine.16 hours = 6 }
		}
	}
}
