# First Balkan War
country_event = {
	id = ww1_serbia.2
	title = ww1_serbia.2.t
	desc = ww1_serbia.2.d
	picture = GFX_report_event_first_balkan_war

	fire_only_once = yes

	trigger = {
		has_global_flag = welcome_event_WW1_active
		has_start_date < 1912.10.8
		NOT = { date > 1912.10.15 }
		NOT = { has_global_flag = let_go_balkans }
		tag = SER
		date > 1912.10.7
		country_exists = TUR
		#has_global_flag = treaty_of_ouchy #The Italians are accounted for now.
	}

	mean_time_to_happen = { days = 1 }

	option = {
		name = ww1_serbia.2.a
		add_threat = 5

		# Serbia must enter the war itself: only the three Balkan League
		# participants are invited into this declaration.
		declare_war_on = { target = TUR type = annex_everything }
		every_other_country = {
			limit = {
				OR = {
					tag = BUL
					tag = GRE
					tag = MNT
				}
				NOT = { has_war_with = TUR }
				NOT = { is_in_faction_with = TUR }
				NOT = { has_non_aggression_pact_with = TUR }
			}
			declare_war_on = { target = TUR type = annex_everything }
		}
		TUR = {
			# Retain the existing early-scenario penalty even if another effect
			# removed it before the Balkan War begins.
			TUR_stab_ensure_initialized = yes
			# The Ottoman response was hampered through the First Balkan War,
			# which concluded with the Treaty of London in May 1913.
			add_timed_idea = { idea = surprise_attack days = 240 }
		}
		# Give the three principal Balkan League armies a short-lived mobilization
		# boost for the campaign without carrying it into the Second Balkan War.
		BUL = {
			add_timed_idea = { idea = patriotic_fervor days = 240 }
		}
		GRE = {
			add_timed_idea = { idea = patriotic_fervor days = 240 }
		}
		add_timed_idea = { idea = patriotic_fervor days = 240 }

		set_global_flag = first_balkan_war
	}
}

#Balkan War 1 - Peace?
country_event = {
	id = ww1_serbia.301
	title = ww1_serbia.301.t
	desc = ww1_serbia.301.d
	picture = GFX_report_event_balkan_league

	fire_only_once = yes

	trigger = {
		has_global_flag = welcome_event_WW1_active
		NOT = { has_global_flag = franz_dead }  #WW1 hasn't started.
		has_global_flag = welcome_event_1910
		has_global_flag = first_balkan_war
		has_war_with = TUR
		tag = SER

		OR = {
			# The Treaty of London ended the First Balkan War on 30 May 1913.
			# Do not let an AI front-line stalemate suppress the settlement.
			date > 1913.5.29
			TUR = {
				OR = {
					AND = {
						surrender_progress > 0.20
						is_ai = yes
						SER = { is_ai = yes }
					}
					AND = {
						NOT = {
							controls_state = 731 	#Salonika
							controls_state = 1062 	#Central Epirus
							controls_state = 805 	#Northern Epirus
							controls_state = 44 	#Albania
							controls_state = 1037 	#Outer Montenegro
							controls_state = 1063 	#Eastern Rumelia no ban plz
							controls_state = 1021 	#East Macedonia
							controls_state = 1038 	#South Macedonia
							controls_state = 106 	#North Macedonia
							controls_state = 802 	#Kosovo
							controls_state = 1036 	#Edirne
						}
					}
				}
			}
		}
	}

	mean_time_to_happen = { days = 1 }

	#Option only gets taken by ai if serbia is ai and a different country isnt
	option = {
		name = ww1_serbia.301.a

		ai_chance = {
			factor = 100
			modifier = {
				factor = 0
				any_country = {
					is_in_faction_with = SER
					is_ai = no
				}
			}
		}
		TUR = { country_event = { id =  ww1_serbia.302 hours = 6 } }
	}

	option = {
		name = ww1_serbia.301.b #How about we push for Constantinople instead?
		ai_chance = { factor = 0 }
		add_political_power = -150
		add_stability = -0.05
		add_war_support = -0.05
	}

	option = {
		name = ww1_serbia.301.w #AI Serbia passes peace decision on to Human Balkans. In MP, each have an equal chance.
		ai_chance = {
			factor = 0 #Because 100 is a neat number
			modifier = {
				any_country ={
					is_in_faction_with = SER
				}
				factor = 100
			}
		}
		trigger = {
			SER = { is_ai = yes }
			any_country = {
				is_in_faction_with = SER
				is_ai = no
			}
		}
		random_country = {
			limit = {
				is_in_faction_with = SER
				is_ai = no
			}
			country_event = { id = ww1_serbia.301 }
		}
	}
}

# Treaty of London
country_event = {
	id = ww1_serbia.3
	title = ww1_serbia.3.t
	desc = ww1_serbia.3.d
	picture = GFX_report_event_treaty_of_london_1913

	is_triggered_only = yes

	option = {
		name = ww1_serbia.3.a
		WW1_SCUTARI_record_london_control = yes
		set_global_flag = treaty_of_london_1913 #For Balkan War 2
		SER = {
			diplomatic_relation = {
				country = TUR
				relation = military_access
				active = yes
			}
			white_peace = TUR
			transfer_state = 802 #Kosovo
			transfer_state = 106 #Macedonia
			transfer_state = 1038 #Debar
		}
		ALB = { # Albanian territory recognized by the London settlement.
			if = { limit = { has_war_with = TUR } white_peace = TUR }
			diplomatic_relation = {
				country = TUR
				relation = military_access
				active = yes
			}
			if = {
				limit = {
					OR = {
						AND = {
							TUR = { owns_state = 44 }
							OR = { TUR = { controls_state = 44 } SER = { controls_state = 44 } MNT = { controls_state = 44 } ALB = { controls_state = 44 } }
						}
						AND = { SER = { has_country_flag = WW1_SER_albanian_legacy_coast owns_state = 44 controls_state = 44 } }
					}
				}
				transfer_state = 44
			}
		}
		MNT = {
			diplomatic_relation = {
				country = TUR
				relation = military_access
				active = yes
			}
			white_peace = TUR
			transfer_state = 1037
		}
		BUL = {
			diplomatic_relation = {
				country = TUR
				relation = military_access
				active = yes
			}
			white_peace = TUR
			transfer_state = 1063
			transfer_state = 1036
			transfer_state = 997
			transfer_state = 184
			transfer_state = 1021
		}
		GRE = {
			diplomatic_relation = {
				country = TUR
				relation = military_access
				active = yes
			}
			white_peace = TUR
			transfer_state = 1062
			transfer_state = 731
			transfer_state = 1042
			transfer_state = 973
			transfer_state = 805
			transfer_state = 1024
		}
		TUR = {
			white_peace = BUL
			white_peace = MNT
			white_peace = SER
		}
		every_state = {
			limit = {
				AND ={
					is_core_of = TUR
					OR = {
						is_core_of = BUL
						is_core_of = GRE
						is_core_of = ALB
						is_core_of = SER
					}
				}
			}
			remove_core_of = TUR #removing balkan cores
			1036 = {add_core_of = TUR} #Except Adrianople
		}
		SER = { if = { limit = { WW1_is_balkan_league_member = yes is_faction_leader = yes } dismantle_faction = yes } }
		hidden_effect = { SER = { WW1_SCUTARI_apply_london_frontier = yes } }
	}
}
