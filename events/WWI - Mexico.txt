# Huerta Falls From Power
country_event = {
	id = ww1_mexico.20
	title = ww1_mexico.20.t
	desc = ww1_mexico.20.d
	picture = GFX_report_event_huerta

	fire_only_once = yes

	trigger = {
		has_global_flag = welcome_event_WW1_active
		has_start_date < 1914.7.16
		NOT = { date > 1932.1.1 }
		tag = MEX
		date > 1914.1.1
		NOT = { has_global_flag = fall_of_huerta }
		OR = {
			has_country_leader = { ruling_only = yes character = MEX_victoriano_huerta }
			# Older saves may have run the imported 1940 election during the revolution.
			AND = { has_global_flag = huerta_takes_power has_global_flag = anti_huerta_revolution }
		}
		OR = {
			MDS = { controls_state = 277 }
			MAD = { has_war_with = MEX owns_state = 482 controls_state = 277 }
			ZAP = { controls_state = 277 }
			CNT = { controls_state = 277 }
			VIL = { controls_state = 277 }
			# Huerta resigned on 15 July 1914 even if the AI fronts are still static.
			date > 1914.7.14
		}
	}

	mean_time_to_happen = {
		days = 1
	}

	option = {
		name = ww1_mexico.20.a
		news_event = { hours = 6 id = endsieg_news.206 }
		set_global_flag = fall_of_huerta
		MEX = {
			endsieg_ww1_absorb_revolutionaries = yes
			remove_ideas = unpopular_dictatorship_1
			remove_ideas = unpopular_dictatorship_2
			retire_country_leader = yes
			endsieg_character_retire_mex_victoriano_huerta_ff0e42d0 = yes
			if = {
				limit = { NOT = { endsieg_character_active_mex_venustiano_carranza_89bf253f = yes } }
				endsieg_character_activate_mex_venustiano_carranza_89bf253f = yes
			}
			set_politics = { ruling_party = democratic elections_allowed = yes }
			set_popularities = { fascism = 0 neutrality = 20 democratic = 70 communism = 10 }
			promote_character = { character = MEX_venustiano_carranza ideology = conservatism }
		}
	}
}

# Repair existing WW1 saves as well as history-loaded rebel tags. No repeated OOB load.
country_event = {
	id = ww1_mexico.21
	hidden = yes
	fire_only_once = yes
	trigger = {
		tag = MEX
		has_global_flag = welcome_event_WW1_active
		date < 1936.1.1
		NOT = { has_global_flag = fall_of_huerta }
		OR = { ZAP = { has_war_with = MEX } CNT = { has_war_with = MEX } VIL = { has_war_with = MEX } }
	}
	mean_time_to_happen = { days = 1 }
	immediate = {
		if = { limit = { ZAP = { has_war_with = MEX } } ZAP = { endsieg_ww1_restore_zapata = yes } }
		if = { limit = { CNT = { has_war_with = MEX } } CNT = { endsieg_ww1_restore_carranza = yes } }
		if = { limit = { VIL = { has_war_with = MEX } } VIL = { endsieg_ww1_restore_villa = yes } }
		endsieg_ww1_anti_huerta_coalition = yes
	}
}
